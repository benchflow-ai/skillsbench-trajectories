"""
Function to analyze Wyckoff position multiplicities and coordinates from CIF files.
"""

from fractions import Fraction
from typing import Dict, Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def fraction_to_rational_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a rational string with denominator <= max_denominator.

    Args:
        value: The float value to convert
        max_denominator: Maximum denominator to use (default 12)

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8")
    """
    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the first atom
    of each Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with:
        - 'wyckoff_multiplicity_dict': Mapping of Wyckoff positions to multiplicities
        - 'wyckoff_coordinates_dict': Mapping of Wyckoff positions to coordinates of first atom
    """
    # Use pymatgen to parse and analyze the structure
    structure = Structure.from_file(filepath)

    # Use SpacegroupAnalyzer to get Wyckoff information
    sga = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure
    symmetrized_structure = sga.get_symmetrized_structure()

    # Group atoms by Wyckoff position
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Get Wyckoff letters for each site
    wyckoff_letters = symmetrized_structure.wyckoff_letters

    # Create a mapping from Wyckoff letter to list of site indices
    wyckoff_groups = {}
    for idx, wyck_letter in enumerate(wyckoff_letters):
        if wyck_letter not in wyckoff_groups:
            wyckoff_groups[wyck_letter] = []
        wyckoff_groups[wyck_letter].append(idx)

    # For each unique Wyckoff position, calculate multiplicity and get first atom coordinates
    for wyck_letter, site_indices in wyckoff_groups.items():
        # Get the multiplicity
        multiplicity = len(site_indices)

        # Get coordinates of the first atom in this Wyckoff position
        first_site_idx = site_indices[0]
        first_site = symmetrized_structure[first_site_idx]

        coords = [
            fraction_to_rational_string(first_site.frac_coords[0]),
            fraction_to_rational_string(first_site.frac_coords[1]),
            fraction_to_rational_string(first_site.frac_coords[2])
        ]

        wyckoff_multiplicity_dict[wyck_letter] = multiplicity
        wyckoff_coordinates_dict[wyck_letter] = coords

    # Sort the results by Wyckoff letter for consistent output
    sorted_wyckoffs = sorted(wyckoff_multiplicity_dict.keys())

    sorted_multiplicity_dict = {k: wyckoff_multiplicity_dict[k] for k in sorted_wyckoffs}
    sorted_coordinates_dict = {k: wyckoff_coordinates_dict[k] for k in sorted_wyckoffs}

    return {
        'wyckoff_multiplicity_dict': sorted_multiplicity_dict,
        'wyckoff_coordinates_dict': sorted_coordinates_dict
    }
