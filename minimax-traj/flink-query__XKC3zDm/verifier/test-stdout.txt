package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.RichFlatMapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.util.Collector;
import org.apache.flink.api.java.utils.ParameterTool;

import java.io.FileWriter;
import java.io.PrintWriter;
import java.util.HashMap;
import java.util.Map;

public class LongestSessionPerJob extends AppBase {

    // Event type constants
    private static final int SUBMIT = 0;

    // 10 minutes in microseconds
    private static final long STAGE_GAP_MICROS = 10 * 60 * 1000000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment (local mode for testing)
        StreamExecutionEnvironment env = StreamExecutionEnvironment.createLocalEnvironment();
        env.setParallelism(1);

        // Read task events and parse them
        DataStream<TaskEvent> taskEvents = env
            .readTextFile(taskInput)
            .flatMap(new TaskEventParser());

        // Filter for SUBMIT events only
        SingleOutputStreamOperator<TaskEvent> submitEvents = taskEvents
            .filter(event -> event.getEventType() == SUBMIT);

        // Assign timestamps and watermarks
        SingleOutputStreamOperator<TaskEvent> timedEvents = submitEvents
            .assignTimestampsAndWatermarks(
                WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                    .withTimestampAssigner((event, timestamp) -> event.getTimestamp())
            );

        // Key by job ID and process to find longest stage per job
        SingleOutputStreamOperator<Tuple2<Long, Integer>> results = timedEvents
            .keyBy(event -> event.getJobId())
            .process(new LongestSessionPerJobFunction());

        // Use a custom sink to output only the final maximum per job
        results.addSink(new FinalResultSink(outputPath));

        env.execute("LongestSessionPerJob");
    }

    /**
     * Custom sink that collects results and outputs only the final maximum per job.
     */
    public static class FinalResultSink extends org.apache.flink.streaming.api.functions.sink.RichSinkFunction<Tuple2<Long, Integer>> {
        private final String outputPath;
        private Map<Long, Integer> jobMaxMap;
        private transient PrintWriter writer;

        public FinalResultSink(String outputPath) {
            this.outputPath = outputPath;
            this.jobMaxMap = new HashMap<>();
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            super.open(parameters);
            // Create parent directory if needed
            new java.io.File(outputPath).getParentFile().mkdirs();
            writer = new PrintWriter(new FileWriter(outputPath));
        }

        @Override
        public void invoke(Tuple2<Long, Integer> value, Context context) throws Exception {
            Long jobId = value.f0;
            Integer count = value.f1;
            if (!jobMaxMap.containsKey(jobId) || count > jobMaxMap.get(jobId)) {
                jobMaxMap.put(jobId, count);
            }
        }

        @Override
        public void close() throws Exception {
            // Output all final results
            for (Map.Entry<Long, Integer> entry : jobMaxMap.entrySet()) {
                writer.println("(" + entry.getKey() + "," + entry.getValue() + ")");
            }
            writer.close();
            super.close();
        }
    }

    /**
     * FlatMap function to parse task events from CSV lines.
     */
    public static class TaskEventParser extends RichFlatMapFunction<String, TaskEvent> {
        @Override
        public void flatMap(String line, Collector<TaskEvent> out) throws Exception {
            TaskEvent event = TaskEvent.fromString(line);
            if (event != null) {
                out.collect(event);
            }
        }
    }

    /**
     * Process function that identifies stages and finds the longest stage per job.
     * A stage ends when there's 10 minutes of inactivity (no SUBMIT events).
     */
    public static class LongestSessionPerJobFunction extends KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Integer>> {

        private ValueState<JobSessionState> jobState;

        @Override
        public void open(Configuration parameters) {
            jobState = getRuntimeContext().getState(
                new ValueStateDescriptor<>("jobSessionState", JobSessionState.class)
            );
        }

        @Override
        public void processElement(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            JobSessionState state = jobState.value();
            if (state == null) {
                state = new JobSessionState();
            }

            long currentTimestamp = event.getTimestamp();
            long jobId = event.getJobId();

            // Check if this starts a new stage (gap > 10 minutes)
            if (state.lastTimestamp > 0 && (currentTimestamp - state.lastTimestamp) > STAGE_GAP_MICROS) {
                // Start a new stage with 1 task
                state.currentStageTasks = 1;
            } else {
                // Continue current stage or start first stage
                state.currentStageTasks++;
            }

            // Track the longest stage for this job
            if (state.currentStageTasks > state.maxStageTasks) {
                state.maxStageTasks = state.currentStageTasks;
            }

            state.lastTimestamp = currentTimestamp;
            jobState.update(state);

            // Output the current max for this job
            out.collect(Tuple2.of(jobId, state.maxStageTasks));
        }
    }

    /**
     * State to track session/stage information per job.
     */
    public static class JobSessionState {
        long lastTimestamp = 0;
        int currentStageTasks = 0;
        int maxStageTasks = 0;
    }
}
