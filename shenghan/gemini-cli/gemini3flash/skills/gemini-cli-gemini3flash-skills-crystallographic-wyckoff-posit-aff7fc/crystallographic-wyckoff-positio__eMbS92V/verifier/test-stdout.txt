import math
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate rational 
    coordinates for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing wyckoff_multiplicity_dict and wyckoff_coordinates_dict.
    """
    try:
        # Load the structure from the CIF file
        struct = Structure.from_file(filepath)
    except Exception:
        return {}

    # Initialize SpacegroupAnalyzer to find the symmetry of the crystal
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    
    # Get the symmetry dataset using spglib via pymatgen
    ds = sga.get_symmetry_dataset()
    if not ds:
        return {}

    # Use attribute access to avoid deprecation warnings if possible
    # Note: Some versions might require dict-like access, but modern ones prefer attributes.
    try:
        wyckoffs = ds.wyckoffs
        equivalent_atoms = ds.equivalent_atoms
    except AttributeError:
        wyckoffs = ds['wyckoffs']
        equivalent_atoms = ds['equivalent_atoms']
    
    # Determine the multiplicity scaling factor for the conventional unit cell
    try:
        conv_struct = sga.get_conventional_standard_structure()
        ratio = conv_struct.volume / struct.volume
    except Exception:
        ratio = 1.0

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    seen_orbit_ids = set()
    unique_orbits = []
    
    # Convert equivalent_atoms to list once for efficient counting
    equiv_list = list(equivalent_atoms)
    
    for i in range(len(struct)):
        orbit_id = equivalent_atoms[i]
        if orbit_id not in seen_orbit_ids:
            seen_orbit_ids.add(orbit_id)
            
            letter = wyckoffs[i]
            # Multiplicity in the provided unit cell
            orbit_count = equiv_list.count(orbit_id)
            # Scale the count to get the multiplicity in the conventional cell
            orbit_mult_conv = int(round(orbit_count * ratio))
            
            unique_orbits.append({
                'letter': letter,
                'mult': orbit_mult_conv,
                'coords': struct[i].frac_coords
            })
            
    # Group results by Wyckoff letter
    for orbit in unique_orbits:
        letter = orbit['letter']
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = orbit['mult']
            
            formatted_coords = []
            for c in orbit['coords']:
                c_norm = c % 1
                f = Fraction(round(c_norm, 10)).limit_denominator(12)
                if f == 1:
                    formatted_coords.append("0")
                else:
                    formatted_coords.append(str(f))
            wyckoff_coordinates_dict[letter] = formatted_coords
        else:
            wyckoff_multiplicity_dict[letter] += orbit['mult']
            
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
