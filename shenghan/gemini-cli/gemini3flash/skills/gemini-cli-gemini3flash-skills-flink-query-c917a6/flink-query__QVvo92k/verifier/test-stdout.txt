    public static class StageProcessor extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, String> {
        private ValueState<Integer> currentStageCount;
        private ValueState<Long> lastSubmitTimestamp;
        private ValueState<Integer> maxStageCount;
        private final int sesssize;

        public StageProcessor(int sesssize) {
            this.sesssize = sesssize;
        }

        @Override
        public void open(Configuration parameters) {
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class, 0));
            lastSubmitTimestamp = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTimestamp", Long.class, 0L));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class, 0));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<String> out) throws Exception {
            if (event.eventType != 0) return;

            long lastTimestamp = lastSubmitTimestamp.value();
            int currentCount = currentStageCount.value();
            int maxCount = maxStageCount.value();

            long gap = sesssize * 1000L; // sesssize in seconds to ms
            long eventTimeMs = event.timestamp / 1000;

            if (lastTimestamp > 0 && (eventTimeMs - lastTimestamp) >= gap) {
                if (currentCount > maxCount) {
                    maxCount = currentCount;
                    maxStageCount.update(maxCount);
                }
                currentCount = 1;
            } else {
                currentCount++;
            }

            currentStageCount.update(currentCount);
            lastSubmitTimestamp.update(eventTimeMs);
            ctx.timerService().registerEventTimeTimer(eventTimeMs + gap);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<String> out) throws Exception {
            if (event.eventType >= 3 && event.eventType <= 5) { // FAIL, FINISH, or KILL
                int currentCount = currentStageCount.value();
                int maxCount = maxStageCount.value();
                if (currentCount > maxCount) {
                    maxCount = currentCount;
                }
                if (maxCount > 0) {
                    out.collect(String.format("(%d,%d)", event.jobId, maxCount));
                }
                currentStageCount.clear();
                lastSubmitTimestamp.clear();
                maxStageCount.clear();
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            Long lastTimestamp = lastSubmitTimestamp.value();
            if (lastTimestamp != null && timestamp == lastTimestamp + sesssize * 1000L) {
                int currentCount = currentStageCount.value();
                int maxCount = maxStageCount.value();
                if (currentCount > maxCount) {
                    maxStageCount.update(currentCount);
                }
                currentStageCount.update(0);
            }
        }
    }

    public static class PrintSink implements org.apache.flink.streaming.api.functions.sink.SinkFunction<String> {
        private final String outputPath;
        private transient PrintWriter writer;

        public PrintSink(String outputPath) {
            this.outputPath = outputPath;

        @Override
        public void invoke(String value, Context context) throws Exception {
            if (writer == null) {
                if (outputPath == null) return;
                writer = new PrintWriter(new FileWriter(outputPath, true));
            }
            writer.println(value);
            writer.flush();
        }
