I created a Node.js script `export_obj.js` that loads the Three.js object from `/root/data/object.js`, expands the `InstancedMesh` components (which are not natively supported by the standard `OBJExporter`), applies the requested -90 degree X-rotation to align with Blender's Z-up coordinate system, and exports the result to `/root/output/object.obj`.

```javascript
import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

function expandInstancedMesh(root) {
  const instancedMeshes = [];
  root.traverse((child) => {
    if (child.isInstancedMesh) {
      instancedMeshes.push(child);
    }
  });

  for (const imesh of instancedMeshes) {
    const parent = imesh.parent;
    const group = new THREE.Group();
    group.name = imesh.name + '_expanded';
    group.copy(imesh, false);

    for (let i = 0; i < imesh.count; i++) {
      const mesh = new THREE.Mesh(imesh.geometry, imesh.material);
      const matrix = new THREE.Matrix4();
      imesh.getMatrixAt(i, matrix);
      mesh.applyMatrix4(matrix);
      mesh.name = `${imesh.name}_instance_${i}`;
      group.add(mesh);
    }

    if (parent) {
      parent.add(group);
      parent.remove(imesh);
    }
  }
}

async function run() {
  const root = createScene();
  expandInstancedMesh(root);

  // Apply -90 degrees X-rotation as requested
  root.rotation.x = -Math.PI / 2;
  root.updateMatrixWorld(true);

  const exporter = new OBJExporter();
  const result = exporter.parse(root);

  const outputPath = '/root/output/object.obj';
  const outputDir = path.dirname(outputPath);
  if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
  }

  fs.writeFileSync(outputPath, result);
  console.log(`Exported to ${outputPath}`);
}

run().catch(console.error);
```
