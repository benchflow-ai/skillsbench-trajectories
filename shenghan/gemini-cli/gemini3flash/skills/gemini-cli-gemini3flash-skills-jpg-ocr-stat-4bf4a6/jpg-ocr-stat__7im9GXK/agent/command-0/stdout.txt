import pytesseract
from PIL import Image
import os
import re
from datetime import datetime
from openpyxl import Workbook

def clean_amount(raw_val):
    mapping = {'O': '0', 'I': '1', 'S': '5', 'Z': '2', 'B': '8', 'G': '9', 'U': '0', 'u': '0', 'o': '0'}
    cleaned_chars = []
    for c in raw_val:
        if c.isdigit() or c in '.,': cleaned_chars.append(c)
        elif c in mapping: cleaned_chars.append(mapping[c])
    cleaned = "".join(cleaned_chars)
    dot_idx = cleaned.rfind('.')
    comma_idx = cleaned.rfind(',')
    sep_idx = max(dot_idx, comma_idx)
    if sep_idx != -1:
        main = re.sub(r'[^\d]', '', cleaned[:sep_idx])
        dec = re.sub(r'[^\d]', '', cleaned[sep_idx+1:])
    else:
        main = re.sub(r'[^\d]', '', cleaned[:-2])
        dec = re.sub(r'[^\d]', '', cleaned[-2:])
    if not main: return None
    if len(dec) > 2: dec = dec[:2]
    elif len(dec) == 1: dec += "0"
    elif len(dec) == 0: dec = "00"
    return f"{main}.{dec}"

def extract_date(text):
    patterns = [
        r'(\d{1,2})[/-](\d{1,2})[/-](\d{4})',
        r'(\d{4})[/-](\d{1,2})[/-](\d{1,2})',
        r'(\d{1,2})[/-](\d{1,2})[/-](\d{2})\b',
        r'\b(\d{2})(\d{2})(\d{4})\b',
        r'Date\s*:\s*(\d{1,2})(\d{1,2})(\d{4})',
    ]
    lines = text.split('\n')
    for line in lines:
        for pattern in patterns:
            match = re.search(pattern, line)
            if match:
                g = match.groups()
                try:
                    if len(g) == 3:
                        if len(g[2]) == 4: d, m, y = int(g[0]), int(g[1]), int(g[2])
                        elif len(g[0]) == 4: y, m, d = int(g[0]), int(g[1]), int(g[2])
                        elif len(g[2]) == 2: d, m, y = int(g[0]), int(g[1]), 2000 + int(g[2])
                        else: continue
                    else: continue
                    if m > 12 and str(m).startswith('6'): m = int('0' + str(m)[1:])
                    if 1 <= m <= 12 and 1 <= d <= 31 and 2010 <= y <= 2026:
                        return f"{y:04d}-{m:02d}-{d:02d}"
                    if d > 12 and m <= 12:
                        if 1 <= m <= 12 and 1 <= d <= 31 and 2010 <= y <= 2026:
                             return f"{y:04d}-{m:02d}-{d:02d}"
                except: continue
    return None

def extract_total(text):
    lines = text.split('\n')
    keywords = ['GRAND TOTAL', 'TOTAL RM', 'TOTAL AMOUNT', 'TOTAL DUE', 'AMOUNT DUE', 'BALANCE DUE', 'NETT TOTAL', 'NET TOTAL', 'TOTAL', 'AMOUNT']
    exclusion = ['SUBTOTAL', 'SUB TOTAL', 'TAX', 'GST', 'SST', 'DISCOUNT', 'CHANGE', 'CASH TENDERED']
    amount_pattern = r'(\d[\d\s,\.]*[\.,]\s?[0-9A-Za-z]{2,3})\b'
    for kw in keywords:
        for i, line in enumerate(lines):
            line_upper = line.upper()
            if kw in line_upper:
                if any(ex in line_upper for ex in exclusion) and not any(ex in kw for ex in exclusion):
                    if "INCL" in line_upper: pass
                    else: continue
                matches = re.findall(amount_pattern, line)
                if matches:
                    val = clean_amount(matches[-1])
                    if val: return val
                if i + 1 < len(lines):
                    next_line = lines[i+1]
                    if not any(ex in next_line.upper() for ex in exclusion):
                        matches = re.findall(amount_pattern, next_line)
                        if matches:
                            val = clean_amount(matches[-1])
                            if val: return val
    return None

def main():
    img_dir = 'dataset/img'
    files = sorted([f for f in os.listdir(img_dir) if f.endswith('.jpg')])
    results = []
    for filename in files:
        path = os.path.join(img_dir, filename)
        text = pytesseract.image_to_string(Image.open(path))
        date = extract_date(text)
        total = extract_total(text)
        results.append((filename, date, total))
    wb = Workbook()
    ws = wb.active
    ws.title = "results"
    ws.append(["filename", "date", "total_amount"])
    for row in results: ws.append(row)
    wb.save('stat_ocr.xlsx')

if __name__ == "__main__": main()
