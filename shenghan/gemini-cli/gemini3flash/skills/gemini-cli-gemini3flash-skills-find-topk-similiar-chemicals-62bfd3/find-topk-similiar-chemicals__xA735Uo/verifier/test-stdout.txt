import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import os
import time
import warnings

# Suppress PubChemPy deprecation warnings if any
warnings.filterwarnings("ignore", category=DeprecationWarning, module="pubchempy")

_smiles_cache = {}

def get_smiles_from_name(name):
    """
    Retrieves the SMILES string for a given chemical name using PubChem.
    """
    if name in _smiles_cache:
        return _smiles_cache[name]
    try:
        # Respect PubChem's rate limits
        time.sleep(0.2)
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            # Using .smiles attribute as it's the current recommended way in PubChemPy
            # which usually defaults to isomeric SMILES
            smiles = getattr(compounds[0], 'smiles', None) or compounds[0].isomeric_smiles or compounds[0].canonical_smiles
            _smiles_cache[name] = smiles
            return smiles
    except Exception:
        pass
    _smiles_cache[name] = None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target molecule.
    
    Args:
        target_molecule_name (str): Name of the target chemical.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: A list of (name, similarity) tuples sorted by similarity (desc) and name (asc).
    """
    # 1. Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        return []

    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Morgan fingerprints with radius 2 and chirality included
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 2. Extract names from PDF
    names = []
    if os.path.exists(molecule_pool_filepath):
        try:
            with pdfplumber.open(molecule_pool_filepath) as pdf:
                for page in pdf.pages:
                    text = page.extract_text()
                    if text:
                        page_names = [n.strip() for n in text.split('\n') if n.strip()]
                        names.extend(page_names)
        except Exception:
            return []
    
    # 3. Get SMILES and compute similarity for each molecule in the pool
    results = []
    unique_names = sorted(list(set(names)))
    
    for name in unique_names:
        smiles = get_smiles_from_name(name)
        if not smiles:
            continue
        
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))
    
    # 4. Sort results: similarity (descending), then name (alphabetical)
    results.sort(key=lambda x: x[0])
    results.sort(key=lambda x: x[1], reverse=True)
    
    return results[:top_k]

if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1:
        target = sys.argv[1]
        k = int(sys.argv[2]) if len(sys.argv) > 2 else 5
        results = topk_tanimoto_similarity_molecules(target, "/root/molecules.pdf", k)
        for name, score in results:
            print(f"{name}: {score:.4f}")
