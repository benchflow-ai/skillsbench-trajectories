The parser identifies sections as nodes, distinguishing between dialogue lines (type "line") and branching choices (type "choice"). It extracts the speaker, text, and edges (including choice text) into a structured format. The script also includes a reachability filter to ensure all nodes are accessible from the starting point and generates both a `dialogue.json` file and a Graphviz `dialogue.dot` visualization.

```python
import json
import re
import sys

def parse_script(text: str):
    nodes = []
    edges = []
    
    # Parse sections delimited by [ID]
    sections = []
    current_section = None
    lines = text.splitlines()
    for line in lines:
        line_stripped = line.strip()
        if not line_stripped:
            continue
        if line_stripped.startswith('[') and line_stripped.endswith(']'):
            if current_section:
                sections.append(current_section)
            current_section = {'id': line_stripped[1:-1], 'lines': []}
        else:
            if current_section:
                current_section['lines'].append(line_stripped)
    if current_section:
        sections.append(current_section)
        
    node_map = {}
    for section in sections:
        node_id = section['id']
        content_lines = section['lines']
        
        # Determine node type (choice or line)
        is_choice = any(re.match(r'^\d+\.', l) for l in content_lines)
        
        if is_choice:
            nodes.append({"id": node_id, "text": "", "speaker": "", "type": "choice"})
            for l in content_lines:
                match = re.match(r'^\d+\.\s*(.*?)\s*->\s*(.*)$', l)
                if match:
                    edges.append({"from": node_id, "to": match.group(2).strip(), "text": match.group(1).strip()})
        else:
            found_line = False
            for l in content_lines:
                match = re.match(r'^([^:]+):\s*(.*?)\s*->\s*(.*)$', l)
                if match:
                    nodes.append({"id": node_id, "text": match.group(2).strip(), "speaker": match.group(1).strip(), "type": "line"})
                    edges.append({"from": node_id, "to": match.group(3).strip(), "text": ""})
                    found_line = True
                    break
            if not found_line:
                # Handle nodes without explicit speaker/target structure
                text_content = " ".join(content_lines)
                target_match = re.search(r'->\s*(.*)$', text_content)
                target = target_match.group(1).strip() if target_match else None
                clean_text = text_content.split('->')[0].strip()
                nodes.append({"id": node_id, "text": clean_text, "speaker": "", "type": "line"})
                if target:
                    edges.append({"from": node_id, "to": target, "text": ""})

    # Add missing target nodes and filter reachability
    defined_ids = {n['id'] for n in nodes}
    for t in {e['to'] for e in edges}:
        if t not in defined_ids:
            nodes.append({"id": t, "text": "", "speaker": "", "type": "line"})
            defined_ids.add(t)

    if nodes:
        first_id = nodes[0]['id']
        reachable, queue = {first_id}, [first_id]
        adj = {}
        for e in edges: adj.setdefault(e['from'], []).append(e['to'])
        while queue:
            curr = queue.pop(0)
            for nxt in adj.get(curr, []):
                if nxt not in reachable: reachable.add(nxt); queue.append(nxt)
        nodes = [n for n in nodes if n['id'] in reachable]
        edges = [e for e in edges if e['from'] in reachable and e['to'] in reachable]

    return {"nodes": nodes, "edges": edges}
```
