package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;

import org.apache.flink.api.common.state.ValueStateDescriptor;

import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskEvents = env.addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .filter(t -> t != null && t.eventType == 0)
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        DataStream<JobEvent> jobEvents = env.addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                .filter(j -> j != null && (j.eventType == 4 || j.eventType == 5 || j.eventType == 6))
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        DataStream<Tuple2<Long, Long>> stageCounts = taskEvents
                .keyBy(t -> t.jobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Long>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Long>> out) {
                        long count = 0;
                        for (TaskEvent ignored : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple2<Long, Long>> result = stageCounts
                .keyBy(t -> t.f0)
                .connect(jobEvents.keyBy(j -> j.jobId))
                .process(new MaxStageCountFunction(sesssize));

        if (outputPath != null) {
            result.writeAsText(outputPath).setParallelism(1);
        } else {
            result.print();
        }

        env.execute("LongestSessionPerJob");
    }

    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String path;
        private volatile boolean running = true;

        public TaskEventSource(String path) {
            this.path = path;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            if (path == null) return;
            try (InputStream is = new FileInputStream(path);
                 InputStream gis = new GZIPInputStream(is);
                 BufferedReader br = new BufferedReader(new InputStreamReader(gis))) {
                String line;
                while (running && (line = br.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    public static class MaxStageCountFunction extends KeyedCoProcessFunction<Long, Tuple2<Long, Long>, JobEvent, Tuple2<Long, Long>> {
        private ValueState<Long> maxStageCount;
        private ValueState<Boolean> jobFinished;
        private final int sesssize;

        public MaxStageCountFunction(int sesssize) {
            this.sesssize = sesssize;
        }

        @Override
        public void open(Configuration parameters) {
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Long.class));
            jobFinished = getRuntimeContext().getState(new ValueStateDescriptor<>("jobFinished", Boolean.class));
        }

        @Override
        public void processElement1(Tuple2<Long, Long> stageCount, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            Long currentMax = maxStageCount.value();
            if (currentMax == null || stageCount.f1 > currentMax) {
                maxStageCount.update(stageCount.f1);
            }
        }

        @Override
        public void processElement2(JobEvent jobEvent, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            jobFinished.update(true);
            // Register a timer to emit the result. 
            // We wait for sesssize seconds to ensure the session window has fired.
            ctx.timerService().registerEventTimeTimer(ctx.timestamp() + sesssize * 1000);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            Boolean finished = jobFinished.value();
            if (finished != null && finished) {
                Long currentMax = maxStageCount.value();
                out.collect(new Tuple2<>(ctx.getCurrentKey(), currentMax == null ? 0L : currentMax));
                jobFinished.clear();
                maxStageCount.clear();
            }
        }
    }
