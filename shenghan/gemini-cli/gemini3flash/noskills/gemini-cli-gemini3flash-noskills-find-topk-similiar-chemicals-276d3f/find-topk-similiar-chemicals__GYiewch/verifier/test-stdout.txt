import fitz
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import time

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF pool to a target molecule name.
    
    Args:
        target_molecule_name (str): The name of the target chemical.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: A list of (name, similarity) tuples, sorted by similarity (desc) and name (asc).
    """
    
    smiles_cache = {}

    def get_smiles(name):
        if name in smiles_cache:
            return smiles_cache[name]
        try:
            # Use pubchempy to fetch the compound and get its SMILES
            # Note: smiles property in pubchempy typically provides isomeric SMILES
            results = pcp.get_compounds(name, 'name')
            if results:
                smiles = results[0].smiles
                smiles_cache[name] = smiles
                return smiles
        except Exception:
            pass
        smiles_cache[name] = None
        return None

    # 1. Extract names from PDF
    all_names = []
    try:
        doc = fitz.open(molecule_pool_filepath)
        for page in doc:
            text = page.get_text()
            # Split by lines and clean up whitespace
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            all_names.extend(lines)
        doc.close()
    except Exception:
        return []
    
    # 2. Get unique names to avoid redundant API calls and handle duplicates in the pool
    unique_pool_names = list(dict.fromkeys(all_names))
    
    # 3. Fetch target molecule SMILES and compute its fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Morgan fingerprint: radius=2 (ECFP4), include chirality as requested
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 4. Fetch SMILES and compute similarity for each unique chemical in the pool
    results = []
    for name in unique_pool_names:
        # Avoid redundant call if target is in pool
        smiles = get_smiles(name)
        # Add a small delay to respect PubChem rate limits
        time.sleep(0.2)
        
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 5. Sort: primary key similarity (descending), secondary key name (alphabetical)
    # Using -x[1] for descending similarity and x[0] for ascending name
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 6. Return top k results
    return results[:top_k]
