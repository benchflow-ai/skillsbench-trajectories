For association:
Since I have $v_p$ and $v_s$, and station locations $(\text{lon}, \text{lat})$.
Distance $d_{ij}$ between station $i$ and source $j$.
Travel time $t_{p, ij} = d_{ij} / v_p$
Travel time $t_{s, ij} = d_{ij} / v_s$
Origin time $T_j = t_{p, pick, ij} - d_{ij} / v_p = t_{s, pick, ij} - d_{ij} / v_s$.
Also $t_{s, pick} - t_{p, pick} = d_{ij} (1/v_s - 1/v_p) = d_{ij} \frac{v_p - v_s}{v_p v_s}$.
So $d_{ij} = (t_{s, pick} - t_{p, pick}) \frac{v_p v_s}{v_p - v_s}$.

If I have both P and S picks at the same station, I can estimate the distance and then the origin time.
$v_p = 6.0$, $v_s = 6.0 / 1.75 \approx 3.428$.
$\frac{v_p v_s}{v_p - v_s} = \frac{6 \cdot 3.428}{6 - 3.428} = \frac{20.568}{2.572} \approx 8.0$.
So $d \approx 8.0 \cdot (t_s - t_p)$.
Then $T = t_p - d/6.0$.

```python
import numpy as np
def cluster_times(times, tolerance=2.0):
    if len(times) == 0: return []
    times = sorted(times)
    clusters = []
    if not times: return []
    curr_cluster = [times[0]]
    for i in range(1, len(times)):
        if times[i] - curr_cluster[-1] < tolerance:
            curr_cluster.append(times[i])
        else:
            clusters.append(curr_cluster)
            curr_cluster = [times[i]]
    clusters.append(curr_cluster)
    return [np.median(c) for c in clusters if len(c) >= 3] # at least 3 stations
```

```python
import obspy
import pandas as pd
import numpy as np
import seisbench.models as sbm
from collections import defaultdict
import torch
try:
    from scipy.cluster.hierarchy import fclusterdata
except ImportError:
    # Fallback to a simple greedy clustering if scipy is missing (unlikely)
    def fclusterdata(X, t, criterion='distance'):
        X = X.flatten()
        X.sort()
        clusters = np.zeros(len(X), dtype=int)
        if len(X) == 0: return clusters
        curr_id = 1
        clusters[0] = curr_id
        for i in range(1, len(X)):
            if X[i] - X[i-1] > t:
                curr_id += 1
            clusters[i] = curr_id
        return clusters

def filter_stream(st):
    groups = defaultdict(list)
    for tr in st:
        groups[(tr.stats.network, tr.stats.station, tr.stats.location)].append(tr)
    new_st = obspy.Stream()
    for key, traces in groups.items():
        found = False
        for pref in ['HH', 'BH', 'HN', 'EH']:
            selected = [tr for tr in traces if tr.stats.channel.startswith(pref)]
            if len(selected) >= 3:
                # Check for Z, N, E or Z, 1, 2
                has_z = any(tr.stats.channel.endswith('Z') for tr in selected)
                has_n = any(tr.stats.channel.endswith('N') or tr.stats.channel.endswith('1') for tr in selected)
                has_e = any(tr.stats.channel.endswith('E') or tr.stats.channel.endswith('2') for tr in selected)
                if has_z and has_n and has_e:
                    z = [tr for tr in selected if tr.stats.channel.endswith('Z')][0]
                    n = [tr for tr in selected if tr.stats.channel.endswith('N') or tr.stats.channel.endswith('1')][0]
                    e = [tr for tr in selected if tr.stats.channel.endswith('E') or tr.stats.channel.endswith('2')][0]
                    new_st.extend([z, n, e])
                    found = True
                    break
        if not found:
            new_st.extend(traces)
    return new_st

def main():
    device = "cuda" if torch.cuda.is_available() else "cpu"
    print(f"Using device: {device}")

    try:
        print("Loading waveforms...")
        st = obspy.read("/root/data/wave.mseed")
        st = filter_stream(st)
        print(f"Using {len(st)} traces after filtering.")
        
        # PhaseNet prefers 100Hz

        if sr != 100.0:
            print(f"Resampling from {sr} to 100Hz")
            st.resample(100.0)
        
        # Picking
        print("Picking phases with PhaseNet...")
        # Use socal model if available
        try:
            model = sbm.PhaseNet.from_pretrained("socal")
        except:
            print("Socal model not found, using 'pretrained'...")
            model = sbm.PhaseNet.from_pretrained("pretrained")
            
        model.to(device)
        
        # Lower threshold for better recall
        picks = model.classify(st, batch_size=256, threshold=0.15)
        df_picks = picks.to_df()
        print(f"Found {len(df_picks)} picks.")

        if len(df_picks) == 0:
            pd.DataFrame(columns=['time']).to_csv("/root/results.csv", index=False)
            return

        # Association
        # vp=6, vs=vp/1.75 => vp/vs = 1.75
        # (ts-tp) = d/vs - d/vp = d(vp-vs)/(vp*vs) = tp_travel * (vp/vs - 1)
        # tp_travel = (ts-tp) / (1.75 - 1) = (ts-tp) / 0.75
        ratio = 1.75 - 1.0 # 0.75

        station_picks = defaultdict(lambda: {'P': [], 'S': []})
        for _, row in df_picks.iterrows():
            station_picks[row['station']][row['phase']].append(row['peak_time'])

        estimates = []
        for sta, pks in station_picks.items():
            ps = sorted(pks['P'])
            ss = sorted(pks['S'])
            for p in ps:

                        if dt < 30: # 30s max S-P is plenty for local
                            t_origin = p.timestamp() - dt / ratio

        if len(est_times) >= 1:

                if len(unique_stations) >= 3:

            origins.sort()
            print(f"Found {len(origins)} associated events.")

            # Convert to ISO format

            results = pd.DataFrame({'time': times})
```
