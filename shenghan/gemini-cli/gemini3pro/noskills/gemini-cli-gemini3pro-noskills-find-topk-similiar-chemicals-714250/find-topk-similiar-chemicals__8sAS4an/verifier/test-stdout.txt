import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
import os
import time

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k) -> list:
    """
    Finds the top k similar chemicals in a PDF pool to a target molecule.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing the pool of molecules.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: List of tuples (molecule_name, similarity_score).
    """
    
    # 1. Extract chemical names from PDF
    pool_names = []
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                lines = text.split('\n')
                for line in lines:
                    name = line.strip()
                    if name:
                        pool_names.append(name)
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []

    # 2. Get Target Molecule SMILES and Fingerprint
    target_smiles = None
    try:
        comps = pcp.get_compounds(target_molecule_name, 'name')
        if comps:
            target_smiles = comps[0].canonical_smiles
    except Exception as e:
        print(f"Error fetching target molecule {target_molecule_name}: {e}")
        return []
        
    if not target_smiles:
        print(f"Target molecule '{target_molecule_name}' not found.")
        return []
        
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        print(f"Could not parse SMILES for target molecule '{target_molecule_name}'")
        return []
        
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 3. Process Pool Molecules
    results = []
    
    # Deduplicate names just in case
    pool_names = sorted(list(set(pool_names)))
    
    for name in pool_names:
        # Skip if it's the target itself (optional, but usually desired in similarity search unless specified otherwise)
        # The prompt doesn't explicitly say to exclude the target if it's in the pool, 
        # but typically "find similar chemicals to X" implies other chemicals. 
        # However, checking strictly: "Find the top k similar chemicals... to any chemicals you are given."
        # If the target is in the list, it would be 1.0 similarity. I'll leave it in if it's there, 
        # as it's technically correct that it's the most similar (identical).
        
        try:
            time.sleep(0.5) # Avoid rate limiting
            comps = pcp.get_compounds(name, 'name')
            if not comps:
                # Try handling simple failures or just skip
                # print(f"Warning: {name} not found in PubChem")
                continue
            
            smiles = comps[0].canonical_smiles
            mol = Chem.MolFromSmiles(smiles)
            if not mol:
                continue
                
            fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
            similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
            results.append((name, similarity))
            
        except Exception as e:
            print(f"Error processing {name}: {e}")
            continue

    # 4. Sort results
    # Sort by similarity (descending), then by name (ascending)
    # Python's sort is stable, so we can sort by name first, then by similarity
    results.sort(key=lambda x: x[0]) # Ascending name
    results.sort(key=lambda x: x[1], reverse=True) # Descending similarity
    
    return results[:top_k]

if __name__ == "__main__":
    # Example usage for testing
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5
    if os.path.exists(pdf_path):
        print(f"Finding top {k} similar to {target} in {pdf_path}...")
        top_mols = topk_tanimoto_similarity_molecules(target, pdf_path, k)
        for name, score in top_mols:
            print(f"{name}: {score:.4f}")
    else:
        print("PDF not found for testing.")
