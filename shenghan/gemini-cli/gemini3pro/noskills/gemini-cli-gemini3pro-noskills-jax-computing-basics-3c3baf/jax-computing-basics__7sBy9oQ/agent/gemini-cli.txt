import jax
import jax.numpy as jnp
import numpy as np
import json
import os

def solve_tasks():
    # Load problem definition
    with open("problem.json", "r") as f:
        problems = json.load(f)

    for problem in problems:
        pid = problem["id"]
        desc = problem["description"]
        input_path = problem["input"]
        output_path = problem["output"]
        
        print(f"Processing task: {pid} - {desc}")
        
        # Load input
        if input_path.endswith('.npz'):
            data = np.load(input_path)
            # We pass the data object to the specific handler
            if pid == "grad_logistic":
                run_grad_logistic(data, output_path)
            elif pid == "scan_rnn":
                run_scan_rnn(data, output_path)
            elif pid == "jit_mlp":
                run_jit_mlp(data, output_path)
            else:
                print(f"Unknown npz task: {pid}")
        elif input_path.endswith('.npy'):
            data = np.load(input_path)
            if pid == "basic_reduce":
                run_basic_reduce(data, output_path)
            elif pid == "map_square":
                run_map_square(data, output_path)
            else:
                print(f"Unknown npy task: {pid}")
        else:
            print(f"Unknown input format for {pid}")

def run_basic_reduce(x, output_path):
    # Description: Given array x, compute mean of each row.
    # x shape: (N, M). Result: (N,)
    x_jax = jnp.array(x)
    result = jnp.mean(x_jax, axis=1)
    np.save(output_path, result)
    print(f"Saved {output_path}")

def run_map_square(x, output_path):
    # Description: Apply square function to each element using vectorization.
    x_jax = jnp.array(x)
    result = jnp.square(x_jax) 
    np.save(output_path, result)
    print(f"Saved {output_path}")

def run_grad_logistic(data, output_path):
    # Description: Compute gradient of logistic loss.
    x = jnp.array(data['x'])
    y = jnp.array(data['y'])
    w = jnp.array(data['w'])
    
    def loss_fn(w, x, y):
        # Logistic loss for y in {-1, 1}
        # L = log(1 + exp(-y * w.x))
        margins = y * jnp.dot(x, w)
        loss = jnp.mean(jnp.logaddexp(0, -margins))
        return loss

    grad_fn = jax.grad(loss_fn)
    grads = grad_fn(w, x, y)
    
    np.save(output_path, grads)
    print(f"Saved {output_path}")

def run_scan_rnn(data, output_path):
    # Description: Implement RNN forward pass using scan.
    seq = jnp.array(data['seq']) # (T, D)
    init = jnp.array(data['init']) # (H,)
    Wx = jnp.array(data['Wx'])
    Wh = jnp.array(data['Wh'])
    b = jnp.array(data['b'])
    
    def step_fn(carry, x):
        h_prev = carry
        # Assuming weights are stored as (Out, In) from math notation
        # So we need to transpose them for x @ W
        h_next = jnp.tanh(jnp.dot(x, Wx.T) + jnp.dot(h_prev, Wh.T) + b)
        return h_next, h_next

    final_carry, stacked_outputs = jax.lax.scan(step_fn, init, seq)
    
    np.save(output_path, stacked_outputs)
    print(f"Saved {output_path}")

def run_jit_mlp(data, output_path):
    # Description: Implement and JIT compile a 2-layer MLP.
    X = jnp.array(data['X'])
    W1 = jnp.array(data['W1'])
    b1 = jnp.array(data['b1'])
    W2 = jnp.array(data['W2'])
    b2 = jnp.array(data['b2'])
    
    def mlp_fn(X, W1, b1, W2, b2):
        # Layer 1
        h = jnp.dot(X, W1) + b1
        h = jax.nn.relu(h)
        # Layer 2
        out = jnp.dot(h, W2) + b2
        return out
    
    # JIT compile
    jitted_mlp = jax.jit(mlp_fn)
    
    # Run
    result = jitted_mlp(X, W1, b1, W2, b2)
    
    np.save(output_path, result)
    print(f"Saved {output_path}")

if __name__ == "__main__":
    solve_tasks()
