import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

// Mock browser environment if necessary (OBJExporter generally doesn't need much, 
// but sometimes creates canvas for textures or uses self)
// Just in case, we can define a minimal global window/document if it crashes.
// But usually for geometry it's fine.

async function run() {
    try {
        const root = createScene();
        
        // Apply -90 degrees X rotation for Blender Z-up
        root.rotation.x = -Math.PI / 2;
        
        // Update matrices to ensure the rotation is propagated
        root.updateMatrixWorld(true);
        
        const exporter = new OBJExporter();
        const result = exporter.parse(root);
        
        const outputPath = path.join(process.cwd(), 'output', 'object.obj');
        fs.writeFileSync(outputPath, result);
        console.log(`Successfully exported to ${outputPath}`);
        
    } catch (error) {
        console.error("Error exporting OBJ:", error);
        process.exit(1);
    }
}

run();
