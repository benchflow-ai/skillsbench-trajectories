from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict

def float_to_rational(x: float, max_denom: int = 12) -> str:
    """
    Converts a float to a rational string with a maximum denominator.
    """
    f = Fraction(x).limit_denominator(max_denom)
    return str(f)

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes a CIF file to extract Wyckoff position multiplicities and 
    approximate coordinates for the first atom of each Wyckoff position.
    """
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)
    
    # Analyze symmetry
    sga = SpacegroupAnalyzer(structure)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    wyckoff_symbols = sym_struct.wyckoff_symbols
    equivalent_indices = sym_struct.equivalent_indices
    
    # Iterate through each group of equivalent sites
    for i, indices in enumerate(equivalent_indices):
        symbol_str = wyckoff_symbols[i]
        
        # Calculate multiplicity (number of sites in the group)
        multiplicity = len(indices)
        
        # Extract Wyckoff letter (remove digits)
        # e.g., "4a" -> "a", "8c" -> "c"
        wyckoff_letter = "".join([char for char in symbol_str if not char.isdigit()])
        
        # Get coordinates of the first atom in the group
        first_idx = indices[0]
        first_site = sym_struct.sites[first_idx]
        coords = first_site.frac_coords
        
        # Convert coordinates to rational strings with denominator <= 12
        rational_coords = [float_to_rational(c) for c in coords]
        
        # Store results
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
