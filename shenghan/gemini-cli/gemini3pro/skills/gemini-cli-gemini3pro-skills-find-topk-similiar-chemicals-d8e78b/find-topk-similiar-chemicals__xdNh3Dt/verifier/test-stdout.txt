import os
import time
import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs

def get_smiles_from_name(name):
    """
    Retrieves the isomeric SMILES for a given chemical name using PubChemPy.
    Returns None if the chemical cannot be found.
    """
    retries = 3
    for i in range(retries):
        try:
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                # Prefer isomeric_smiles to capture chirality
                return compounds[0].isomeric_smiles
            else:
                return None
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                if i < retries - 1:
                    time.sleep(1 * (i + 1)) # Backoff
                    continue
            print(f"Error retrieving SMILES for {name}: {e}")
            return None
    return None

def extract_names_from_pdf(filepath):
    """
    Extracts chemical names from a PDF file.
    Assumes one name per line.
    """
    names = []
    try:
        reader = pypdf.PdfReader(filepath)
        text = ""
        for page in reader.pages:
            text += page.extract_text() + "\n"
        
        lines = text.split('\n')
        for line in lines:
            cleaned_line = line.strip()
            if cleaned_line:
                names.append(cleaned_line)
    except Exception as e:
        print(f"Error reading PDF {filepath}: {e}")
    return names

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k) -> list:
    """
    Finds the top k similar chemicals in `molecule_pool_filepath` to `target_molecule_name`.
    
    Args:
        target_molecule_name (str): The name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: A list of the top k similar chemical names, sorted by similarity (desc) then name (asc).
    """
    
    # 1. Get Target SMILES
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    # 2. Extract Candidate Names from PDF
    candidate_names = extract_names_from_pdf(molecule_pool_filepath)
    candidate_names = sorted(list(set(candidate_names))) # Remove duplicates and initial sort
    
    # 3. Prepare Target Molecule for RDKit
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
         raise ValueError(f"RDKit could not parse SMILES for target molecule: {target_molecule_name}")
    
    # Generate Target Fingerprint (Morgan, radius=2, include chirality)
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, radius=2, useChirality=True)
    
    results = []
    
    # 4. Process Candidates
    for name in candidate_names:
        time.sleep(0.2) # Rate limiting for PubChem

        # Skip if it's the target itself (optional, but usually good practice in similarity search, 
        # though the prompt doesn't explicitly say to exclude it. 
        # "Find similar chemicals... to any chemicals you are given." 
        # Usually implies finding *other* chemicals. But if the list contains the target, 
        # it would be similarity 1.0. Let's keep it if it's in the list, unless it's strictly distinct.
        # I'll keep it to be safe, standard behavior is to include everything in the pool.)
        
        # Optimize: If name is same string, we know similarity is 1.0 (if resolved same)
        
        smiles = get_smiles_from_name(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        
        results.append((name, similarity))
        
    # 5. Sort Results
    # Primary key: similarity (descending)
    # Secondary key: name (ascending) - alphabetical ordering when ties happen
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 6. Return Top K Names
    top_results = [name for name, score in results[:top_k]]
    
    return top_results

if __name__ == "__main__":
    # Example usage for testing
    try:
        # You can create a dummy molecules.pdf here or rely on the existing one
        target = "Aspirin"

            for m in similar_mols:

    except Exception as e:
        print(f"An error occurred: {e}")
