from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    try:
        s = Structure.from_file(filepath)
    except Exception as e:
        return {"error": str(e)}

    # Analyze symmetry
    sga = SpacegroupAnalyzer(s)
    
    try:
        sym_struct = sga.get_symmetrized_structure()
    except Exception as e:
        return {"error": str(e)}

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # sym_struct.equivalent_indices is a list of lists of atom indices
    # sym_struct.wyckoff_symbols is a list of wyckoff symbols for each group (aligned with equivalent_indices)
    
    for i, indices in enumerate(sym_struct.equivalent_indices):
        # Get Wyckoff symbol for this group
        full_symbol = sym_struct.wyckoff_symbols[i]
        
        # Extract just the letter. 
        # Example: '4a', '8c' -> 'a', 'c'
        letter = "".join([c for c in full_symbol if c.isalpha()])
        
        multiplicity = len(indices)
        
        # Get coordinates of the representative site (first index in the group)
        representative_idx = indices[0]
        site = sym_struct[representative_idx]
        coords = site.frac_coords
        
        formatted_coords = []
        for c in coords:
            # Normalize to [0, 1)
            val = c % 1.0
            
            # Round to closest rational with denominator <= 12
            f = Fraction(val).limit_denominator(12)
            
            # Handle the case where rounding pushes it to 1 (e.g. 0.999 -> 1)
            if f == 1:
                f = Fraction(0, 1)
            
            formatted_coords.append(str(f))
            
        wyckoff_multiplicity_dict[letter] = multiplicity
        wyckoff_coordinates_dict[letter] = formatted_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
