import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
import time
import sys

# Suppress warnings for cleaner output
from rdkit import RDLogger
RDLogger.DisableLog('rdApp.*')

def get_smiles(name, retries=5, delay=1):
    for attempt in range(retries):
        try:
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                return compounds[0].isomeric_smiles
            return None
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                if attempt < retries - 1:
                    # print(f"Server busy for {name}, retrying in {delay}s...", file=sys.stderr)
                    time.sleep(delay)
                    delay *= 2
                    continue
            print(f"Error fetching {name}: {e}", file=sys.stderr)
            return None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Read molecules from PDF
    pool_names = []
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.split('\n')
                for line in lines:
                    cleaned = line.strip()
                    if cleaned:
                        pool_names.append(cleaned)
    
    # Get target molecule info
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target: {target_molecule_name}")
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Could not create molecule from SMILES for target: {target_molecule_name}")
        
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    results = []
    
    # Process pool molecules
    # Remove duplicates if any to avoid redundant API calls
    unique_pool_names = sorted(list(set(pool_names)))
    
    for name in unique_pool_names:
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))
        
    # Sort: Descending similarity, then Alphabetical name
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k names
    return [x[0] for x in results[:top_k]]

if __name__ == "__main__":
    try:

        top_matches = topk_tanimoto_similarity_molecules("Aspirin", "molecules.pdf", 5)
        print("Top 5 similar to Aspirin:", top_matches)
    except Exception as e:
        print(f"Execution failed: {e}")
