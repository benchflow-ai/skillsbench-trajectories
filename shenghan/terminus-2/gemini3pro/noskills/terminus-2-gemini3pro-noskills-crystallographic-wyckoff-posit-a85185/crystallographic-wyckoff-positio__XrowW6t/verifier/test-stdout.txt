from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
import re
from typing import Any, Dict

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    # Load structure from CIF file
    s = Structure.from_file(filepath)
    
    # Analyze symmetry to identify equivalent sites and Wyckoff positions
    # We use SpacegroupAnalyzer to get the symmetrized structure
    sga = SpacegroupAnalyzer(s)
    sym = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Iterate over equivalent site groups
    for i, indices in enumerate(sym.equivalent_indices):
        # Get the Wyckoff symbol (e.g., '4a', '8c')
        symbol = sym.wyckoff_symbols[i]
        
        # Extract the letter part of the Wyckoff symbol (e.g., 'a', 'c')
        # We remove all digits from the symbol string
        letter = re.sub(r'[0-9]', '', symbol)
        
        # Multiplicity is the number of atoms in this equivalent group
        multiplicity = len(indices)
        
        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get coordinates of the first atom in the group
        first_atom_index = indices[0]
        first_atom = sym[first_atom_index]
        coords = first_atom.frac_coords
        
        # Convert coordinates to rational number strings with max denominator 12
        coords_str = [str(Fraction(x).limit_denominator(12)) for x in coords]
        
        # Store coordinates
        wyckoff_coordinates_dict[letter] = coords_str
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
