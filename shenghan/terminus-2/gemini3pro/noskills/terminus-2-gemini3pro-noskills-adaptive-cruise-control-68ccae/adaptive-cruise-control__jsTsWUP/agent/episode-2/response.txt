```python
# pid_controller.py
class PIDController:
    def __init__(self, kp, ki, kd):
        self.kp = kp
        self.ki = ki
        self.kd = kd
        self.integral = 0.0
        self.prev_error = 0.0

    def reset(self):
        self.integral = 0.0
        self.prev_error = 0.0

    def compute(self, error, dt):
        self.integral += error * dt
        derivative = (error - self.prev_error) / dt if dt > 0 else 0.0
        output = (self.kp * error) + (self.ki * self.integral) + (self.kd * derivative)
        self.prev_error = error
        return output
```

```python
# acc_system.py
from pid_controller import PIDController

class AdaptiveCruiseControl:
    def __init__(self, config):
        self.set_speed = config['acc_settings']['set_speed']
        self.time_headway = config['acc_settings']['time_headway']
        self.min_distance = config['acc_settings']['min_distance']
        self.emergency_ttc_threshold = config['acc_settings']['emergency_ttc_threshold']
        self.max_accel = config['vehicle']['max_acceleration']
        self.max_decel = config['vehicle']['max_deceleration']
        
        # Initialize PIDs with values from config (which will come from tuning_results.yaml later)
        # Note: The prompt says simulation.py reads tuning_results.yaml. 
        # But acc_system needs to be initialized. I will assume config passed here has the PID gains.
        
        self.pid_speed = PIDController(
            config['pid_speed']['kp'],
            config['pid_speed']['ki'],
            config['pid_speed']['kd']
        )
        self.pid_distance = PIDController(
            config['pid_distance']['kp'],
            config['pid_distance']['ki'],
            config['pid_distance']['kd']
        )

    def compute(self, ego_speed, lead_speed, distance, dt):
        mode = 'cruise'
        acceleration_cmd = 0.0
        distance_error = None
        
        # Check for lead vehicle
        if distance is not None and lead_speed is not None:
            # Calculate TTC
            rel_speed = ego_speed - lead_speed
            ttc = float('inf')
            if rel_speed > 0:
                ttc = distance / rel_speed
            
            if ttc < self.emergency_ttc_threshold:
                mode = 'emergency'
                acceleration_cmd = self.max_decel # Full braking
                # In emergency, we might still track distance error for logging
                safe_distance = self.min_distance + self.time_headway * ego_speed
                distance_error = distance - safe_distance
            else:
                mode = 'follow'
                # Distance Control
                safe_distance = self.min_distance + self.time_headway * ego_speed
                distance_error = distance - safe_distance
                # We want distance >= safe_distance. 
                # If distance > safe_distance, error > 0. We want to speed up?
                # Usually PID minimizes error. Target is 0.
                # If distance is large, error is large positive. We want positive accel.
                # If distance is small, error is negative. We want negative accel.
                acceleration_cmd = self.pid_distance.compute(distance_error, dt)
        else:
            # Cruise Control
            mode = 'cruise'
            speed_error = self.set_speed - ego_speed
            acceleration_cmd = self.pid_speed.compute(speed_error, dt)
            distance_error = None

        # Clamp acceleration
        acceleration_cmd = max(self.max_decel, min(self.max_accel, acceleration_cmd))
        
        return acceleration_cmd, mode, distance_error
```
