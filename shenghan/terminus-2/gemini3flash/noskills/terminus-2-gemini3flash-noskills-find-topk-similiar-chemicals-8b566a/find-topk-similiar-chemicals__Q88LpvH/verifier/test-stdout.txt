import fitz
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem
import os

def get_smiles(name, cache):
    if name in cache:
        return cache[name]
    try:
        # Using pubchempy to get SMILES efficiently
        props = pcp.get_properties('IsomericSMILES', name, 'name')
        if props:
            smiles = props[0].get('IsomericSMILES')
            cache[name] = smiles
            return smiles
    except Exception:
        pass
    cache[name] = None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    cache = {}
    
    # 1. Get SMILES for target
    target_smiles = get_smiles(target_molecule_name, cache)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    # Morgan fingerprint: radius=2, include chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 2. Extract names from PDF
    if not os.path.exists(molecule_pool_filepath):
        return []
        
    all_names = []
    try:
        doc = fitz.open(molecule_pool_filepath)
        for page in doc:
            text = page.get_text()
            # Split by newline and strip. Filter out empty strings.
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            all_names.extend(lines)
        doc.close()
    except Exception:
        return []
    
    # 3. Get SMILES and compute similarity for each unique candidate name
    unique_names = sorted(list(set(all_names)))
    results = []
    
    for name in unique_names:
        smiles = get_smiles(name, cache)
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 4. Sort results: similarity descending, then name ascending
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k names
    return [item[0] for item in results[:top_k]]
