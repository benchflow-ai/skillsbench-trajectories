package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.RuntimeExecutionMode;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.typeinfo.TypeHint;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.tuple.Tuple3;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.assigners.GlobalWindows;
import org.apache.flink.streaming.api.windowing.triggers.EventTimeTrigger;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.functions.windowing.WindowFunction;
import org.apache.flink.streaming.api.windowing.windows.GlobalWindow;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;
import java.util.Objects;

public class LongestSessionPerJob extends AppBase {
    public static void main(String[] args) throws Exception {
        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input");
        String jobInput = params.get("job_input");
        String outputPath = params.get("output");
        int sesssize = params.getInt("sesssize", 600);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setRuntimeMode(RuntimeExecutionMode.BATCH);
        env.setParallelism(1);

        DataStream<TaskEvent> taskEvents = env.readTextFile(taskInput)
                .map(TaskEvent::fromString)
                .filter(Objects::nonNull)
                .filter(e -> e.eventType == 0) // SUBMIT
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        DataStream<Long> finishedJobs = env.readTextFile(jobInput)
                .map(JobEvent::fromString)
                .filter(Objects::nonNull)
                .filter(e -> e.eventType == 4) // FINISH
                .map(e -> e.jobId)
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<Long>forMonotonousTimestamps()
                                .withTimestampAssigner((id, timestamp) -> 0L)
                );

        DataStream<Tuple2<Long, Integer>> stageCounts = taskEvents
                .keyBy(e -> e.jobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent e : input) count++;
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple3<Long, Integer, Boolean>> taskResults = stageCounts
                .map(t -> new Tuple3<>(t.f0, t.f1, false))
                .returns(new TypeHint<Tuple3<Long, Integer, Boolean>>(){});
        
        DataStream<Tuple3<Long, Integer, Boolean>> jobResults = finishedJobs
                .map(id -> new Tuple3<>(id, 0, true))
                .returns(new TypeHint<Tuple3<Long, Integer, Boolean>>(){});

        taskResults.union(jobResults)
                .keyBy(t -> t.f0)
                .window(GlobalWindows.create())
                .trigger(EventTimeTrigger.create())
                .apply(new WindowFunction<Tuple3<Long, Integer, Boolean>, String, Long, GlobalWindow>() {
                    @Override
                    public void apply(Long jobId, GlobalWindow window, Iterable<Tuple3<Long, Integer, Boolean>> input, Collector<String> out) {
                        int maxCount = 0;
                        boolean finished = false;
                        for (Tuple3<Long, Integer, Boolean> t : input) {
                            if (t.f2) finished = true;
                            else if (t.f1 > maxCount) maxCount = t.f1;
                        }
                        if (finished && maxCount > 0) {
                            out.collect("(" + jobId + "," + maxCount + ")");
                        }
                    }
                })
                .writeAsText(outputPath).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }
}
