import re
from fractions import Fraction
from typing import Any, Dict
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def to_fraction_str(val: float) -> str:
    # Normalize to [0, 1)
    val = val % 1
    f = Fraction(val).limit_denominator(12)
    # After limit_denominator, it might become 1/1
    if f.numerator == 0 or f.numerator == f.denominator:
        return "0"
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    sga = SpacegroupAnalyzer(structure)
    dataset = sga.get_symmetry_dataset()
    
    # dataset.wyckoffs gives the Wyckoff letter for each atom in the input structure
    # dataset.equivalent_atoms gives an integer ID for each set of equivalent atoms
    wyckoffs = dataset.wyckoffs
    equivalent_atoms = dataset.equivalent_atoms
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    seen_letters = set()
    seen_equivalent_ids = set()
    
    # We iterate through atoms to find unique Wyckoff positions (represented by letters)
    for i, (letter, eq_id) in enumerate(zip(wyckoffs, equivalent_atoms)):
        if letter not in seen_letters:
            # This is the first atom encountered for this Wyckoff letter
            seen_letters.add(letter)
            
            # Calculate multiplicity: how many atoms in the structure are equivalent to this one
            # Note: In some cases, multiple orbits might have the same Wyckoff letter.
            # Based on the example, we take the first one encountered.
            multiplicity = list(equivalent_atoms).count(eq_id)
            wyckoff_multiplicity_dict[letter] = multiplicity
            
            # Get the fractional coordinates of the first atom of this Wyckoff position
            coords = structure[i].frac_coords
            wyckoff_coordinates_dict[letter] = [to_fraction_str(c) for c in coords]
    
    # Sort by letter for a consistent result
    sorted_keys = sorted(wyckoff_multiplicity_dict.keys())
    
    return {
        "wyckoff_multiplicity_dict": {k: wyckoff_multiplicity_dict[k] for k in sorted_keys},
        "wyckoff_coordinates_dict": {k: wyckoff_coordinates_dict[k] for k in sorted_keys}
    }
