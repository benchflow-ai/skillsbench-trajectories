public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String outputPath = params.get("output", null);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setStreamTimeCharacteristic(TimeCharacteristic.EventTime);
        env.setParallelism(1);

        DataStream<String> taskLines = env.readTextFile(taskInput);

        DataStream<TaskEvent> taskEvents = taskLines
                .map(new MapFunction<String, TaskEvent>() {
                    @Override
                    public TaskEvent map(String s) throws Exception {
                        return TaskEvent.fromString(s);
                    }
                })
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent event) throws Exception {
                        return event.eventType == 0; // SUBMIT
                    }
                })
                .assignTimestampsAndWatermarks(new BoundedOutOfOrdernessTimestampExtractor<TaskEvent>(Time.seconds(0)) {
                    @Override
                    public long extractTimestamp(TaskEvent event) {
                        return event.timestamp / 1000; // convert microseconds to milliseconds
                    }
                });

        DataStream<Tuple2<Long, Integer>> sessionCounts = taskEvents
                .keyBy(event -> event.jobId)
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent event : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple2<Long, Integer>> maxSessionCounts = sessionCounts
                .keyBy(t -> t.f0)
                .process(new KeyedProcessFunction<Long, Tuple2<Long, Integer>, Tuple2<Long, Integer>>() {
                    private ValueState<Integer> maxState;

                    @Override
                    public void open(Configuration parameters) {
                        maxState = getRuntimeContext().getState(new ValueStateDescriptor<>("max", Integer.class));
                    }

                    @Override
                    public void processElement(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Integer currentMax = maxState.value();
                        if (currentMax == null || value.f1 > currentMax) {
                            maxState.update(value.f1);
                        }
                        ctx.timerService().registerEventTimeTimer(ctx.timerService().currentWatermark() + 1000000); // register a timer in the future
                        // Actually, a better way to emit at the end of a finite stream is to use a very large timer
                        ctx.timerService().registerEventTimeTimer(Long.MAX_VALUE - 1);
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        if (timestamp == Long.MAX_VALUE - 1) {
                            out.collect(new Tuple2<>(ctx.getCurrentKey(), maxState.value()));
                        }
                    }
                });

        maxSessionCounts
                .map(new MapFunction<Tuple2<Long, Integer>, String>() {
                    @Override
                    public String map(Tuple2<Long, Integer> t) {
                        return "(" + t.f0 + "," + t.f1 + ")";
                    }
                })
                .writeAsText(outputPath).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

        env.setStreamTimeCharacteristic(TimeCharacteristic.EventTime);
        env.setParallelism(1);

        DataStream<String> taskLines = env.readTextFile(taskInput);

        DataStream<TaskEvent> taskEvents = taskLines
                .map(new MapFunction<String, TaskEvent>() {
                    @Override
                    public TaskEvent map(String s) throws Exception {
                        return TaskEvent.fromString(s);
                    }
                })
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent event) throws Exception {
                        return event.eventType == 0; // SUBMIT
                    }
                })
                .assignTimestampsAndWatermarks(new BoundedOutOfOrdernessTimestampExtractor<TaskEvent>(Time.seconds(0)) {
                    @Override
                    public long extractTimestamp(TaskEvent event) {
                        return event.timestamp / 1000; // convert microseconds to milliseconds
                    }
                });

        DataStream<Tuple2<Long, Integer>> sessionCounts = taskEvents
                .keyBy(event -> event.jobId)
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent event : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple2<Long, Integer>> maxSessionCounts = sessionCounts
                .keyBy(t -> t.f0)
                .process(new KeyedProcessFunction<Long, Tuple2<Long, Integer>, Tuple2<Long, Integer>>() {
                    private ValueState<Integer> maxState;

                    @Override
                    public void open(Configuration parameters) {
                        maxState = getRuntimeContext().getState(new ValueStateDescriptor<>("max", Integer.class));
                    }

                    @Override
                    public void processElement(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Integer currentMax = maxState.value();
                        if (currentMax == null || value.f1 > currentMax) {
                            maxState.update(value.f1);
                        }
                        ctx.timerService().registerEventTimeTimer(ctx.timerService().currentWatermark() + 1000000); // register a timer in the future
                        // Actually, a better way to emit at the end of a finite stream is to use a very large timer
                        ctx.timerService().registerEventTimeTimer(Long.MAX_VALUE - 1);
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        if (timestamp == Long.MAX_VALUE - 1) {
                            out.collect(new Tuple2<>(ctx.getCurrentKey(), maxState.value()));
                        }
                    }
                });

        maxSessionCounts
                .map(new MapFunction<Tuple2<Long, Integer>, String>() {
                    @Override
                    public String map(Tuple2<Long, Integer> t) {
                        return "(" + t.f0 + "," + t.f1 + ")";
                    }
                })
                .writeAsText(outputPath).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }
