import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import warnings

# Suppress deprecation warnings
warnings.filterwarnings("ignore", category=DeprecationWarning)

def get_smiles_from_name(name):
    try:
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            # Use isomeric_smiles if available to better support chirality
            return compounds[0].isomeric_smiles or compounds[0].canonical_smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # 1. Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 2. Extract names from PDF
    molecule_names = []
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    molecule_names.extend(text.split('\n'))
    except Exception:
        return []
    
    # Filter out empty strings and strip whitespace
    molecule_names = [name.strip() for name in molecule_names if name.strip()]
    
    # 3. Get SMILES and compute similarity for each molecule in the pool
    results = []
    name_to_smiles = {}
    
    for name in molecule_names:
        if name not in name_to_smiles:
            smiles = get_smiles_from_name(name)
            name_to_smiles[name] = smiles
        
        smiles = name_to_smiles[name]
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 4. Sort results: descending by similarity, then alphabetical by name
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k names (unique names if they appear multiple times in the PDF? 
    # The task says "top k similar chemicals in molecules.pdf". 
    # Usually this means if a name appears twice, it might be listed twice or we take unique.
    # Given the output of the test, it returned both 'Acetylsalicylic acid' and 'Aspirin'.
    # If the PDF has duplicate names, they should be treated as separate entries or we can deduplicate.
    # Looking at the list I extracted, 'Aspirin' and 'Acetylsalicylic acid' are both there.
    # I will keep the names as they appear in the PDF.
    
    return [name for name, sim in results[:top_k]]
