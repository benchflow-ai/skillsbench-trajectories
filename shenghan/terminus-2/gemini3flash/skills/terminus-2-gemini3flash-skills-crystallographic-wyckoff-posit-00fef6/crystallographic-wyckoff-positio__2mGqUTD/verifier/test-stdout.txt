import pymatgen.core as mg
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, Any]:
    # Load structure from CIF file
    struct = mg.Structure.from_file(filepath)
    
    # Analyze symmetry using SpacegroupAnalyzer
    # symprec is set to 0.01 by default, which is generally suitable for CIF files
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    dataset = sga.get_symmetry_dataset()
    
    if dataset is None:
        return {"wyckoff_multiplicity_dict": {}, "wyckoff_coordinates_dict": {}}
    
    wyckoff_letters = dataset['wyckoffs']
    # Fractional coordinates of the sites in the input structure
    frac_coords = struct.frac_coords
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # To store the first encountered coordinate for each Wyckoff letter
    seen_letters = set()
    
    for i, letter in enumerate(wyckoff_letters):
        # Update multiplicity count
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = 0
        wyckoff_multiplicity_dict[letter] += 1
        
        # If this is the first time we see this Wyckoff letter, store its rationalized coordinates
        if letter not in seen_letters:
            seen_letters.add(letter)
            coords = frac_coords[i]
            rat_coords = []
            for c in coords:
                # Normalize coordinate to [0, 1)
                val = float(c) % 1
                # Use Fraction.limit_denominator to find the closest rational with denominator <= 12
                f = Fraction(val).limit_denominator(12)
                
                # Handle the case where f is 1/1 (due to val being very close to 1)
                if f.numerator == f.denominator:
                    rat_coords.append("0")
                else:
                    rat_coords.append(str(f))
            wyckoff_coordinates_dict[letter] = rat_coords
            
    # Sort dictionaries by Wyckoff letter for consistent output
    sorted_letters = sorted(wyckoff_multiplicity_dict.keys())
    
    return {
        "wyckoff_multiplicity_dict": {l: wyckoff_multiplicity_dict[l] for l in sorted_letters},
        "wyckoff_coordinates_dict": {l: wyckoff_coordinates_dict[l] for l in sorted_letters}
    }

if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1:
        import json
        result = analyze_wyckoff_position_multiplicities_and_coordinates(sys.argv[1])
        print(json.dumps(result, indent=4))
