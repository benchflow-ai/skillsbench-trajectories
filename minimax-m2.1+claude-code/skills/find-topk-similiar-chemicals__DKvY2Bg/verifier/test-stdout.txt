"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import re
import warnings
import requests
from typing import List, Tuple
from rdkit import Chem
from rdkit.Chem import rdMolDescriptors
import PyPDF2

# Suppress RDKit deprecation warnings
warnings.filterwarnings('ignore', category=DeprecationWarning)


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem PUG REST API.

    Args:
        chemical_name: Name of the chemical compound

    Returns:
        SMILES string representation of the chemical
    """
    # Search PubChem for the compound
    base_url = "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name"
    url = f"{base_url}/{chemical_name}/property/SMILES/JSON"

    try:
        response = requests.get(url, timeout=10)
        if response.status_code == 200:
            data = response.json()
            if 'PropertyTable' in data and 'Properties' in data['PropertyTable']:
                properties = data['PropertyTable']['Properties']
                if properties and 'SMILES' in properties[0]:
                    return properties[0]['SMILES']
    except Exception as e:
        print(f"Error fetching SMILES for {chemical_name}: {e}")

    return None


def read_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Read chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of chemical names found in the PDF
    """
    molecules = []

    try:
        with open(pdf_filepath, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            for page in pdf_reader.pages:
                text = page.extract_text()
                # Simple extraction - each line might be a chemical name
                # You can customize this based on PDF structure
                lines = text.split('\n')
                for line in lines:
                    line = line.strip()
                    if line and len(line) > 2:  # Filter out short/empty lines
                        molecules.append(line)
    except FileNotFoundError:
        print(f"File {pdf_filepath} not found")
    except Exception as e:
        print(f"Error reading PDF: {e}")

    return molecules


def get_morgan_fingerprint(smiles: str, radius: int = 2) -> object:
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES representation of the molecule
        radius: Radius for Morgan fingerprint (default: 2)

    Returns:
        RDKit Morgan fingerprint object
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    # Generate Morgan fingerprint with radius=2 and include chirality
    fingerprint = rdMolDescriptors.GetMorganFingerprintAsBitVect(
        mol, radius=2, nBits=2048, useChirality=True
    )
    return fingerprint


def tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0.0 to 1.0)
    """
    if fp1 is None or fp2 is None:
        return 0.0

    return Chem.DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(target_molecule_name: str,
                                      molecule_pool_filepath: str,
                                      top_k: int) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to a target molecule.

    Args:
        target_molecule_name: Name of the target chemical compound
        molecule_pool_filepath: Path to the PDF file containing the molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity
        (descending), then alphabetically for ties
    """
    # Convert target molecule to SMILES
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []

    # Generate fingerprint for target molecule
    target_fp = get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        print(f"Could not generate fingerprint for target molecule: {target_molecule_name}")
        return []

    # Read molecules from PDF
    molecule_names = read_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for molecule_name in molecule_names:
        # Skip if it's the same as target
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES for this molecule
        smiles = get_smiles_from_pubchem(molecule_name)
        if smiles is None:
            continue

        # Generate fingerprint
        fp = get_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = tanimoto_similarity(target_fp, fp)

        similarities.append((molecule_name, similarity))

    # Sort by similarity (descending), then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    # Note: This requires molecules.pdf to exist in the current directory
    target = "aspirin"
    pdf_path = "molecules.pdf"
    k = 5

    print(f"Finding top {k} molecules similar to {target}...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for molecule, similarity in results:
        print(f"{molecule}: {similarity:.4f}")
