"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import logging
import re
import warnings
from typing import List, Tuple

import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs

# Suppress RDKit deprecation warnings
warnings.filterwarnings('ignore', category=DeprecationWarning)

# Disable RDKit logger
rdkit_logger = logging.getLogger('rdkit')
rdkit_logger.setLevel(logging.ERROR)


def get_molecule_from_name(chemical_name: str) -> Chem.Mol:
    """
    Convert a chemical name to an RDKit molecule using PubChem.

    Args:
        chemical_name: The name of the chemical

    Returns:
        RDKit molecule object

    Raises:
        ValueError: If the chemical name cannot be found
    """
    compounds = pcp.get_compounds(chemical_name, 'name')
    if not compounds:
        raise ValueError(f"Could not find chemical: {chemical_name}")

    # Get the first (most likely) compound
    compound = compounds[0]

    # Convert to SMILES and then to RDKit molecule
    smiles = compound.smiles
    mol = Chem.MolFromSmiles(smiles)

    if mol is None:
        raise ValueError(f"Could not convert {chemical_name} to molecule")

    return mol


def get_morgan_fingerprint(mol, radius: int = 2, include_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        mol: RDKit molecule
        radius: Radius for Morgan fingerprint (default 2)
        include_chirality: Whether to include chirality (default True)

    Returns:
        Morgan fingerprint as bit vector
    """
    fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius, useChirality=include_chirality)
    return fp


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity coefficient (0-1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def parse_pdf_chemicals(filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    import pypdf

    chemicals = []
    reader = pypdf.PdfReader(filepath)

    for page in reader.pages:
        text = page.extract_text()
        if text:
            # Find chemical-like names (words that are likely chemical names)
            # Chemical names typically don't contain digits only and are multi-word
            lines = text.split('\n')
            for line in lines:
                line = line.strip()
                # Skip empty lines and headers
                if not line or len(line) < 2:
                    continue
                # Add the line as a chemical name
                chemicals.append(line)

    return chemicals


def topk_tanimoto_similarity_molecules(target_molecule_name: str, molecule_pool_filepath: str, top_k: int) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to the target molecule from a pool of molecules.

    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to the PDF file containing the molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of (chemical_name, similarity_score) tuples sorted by similarity (descending),
        with alphabetical ordering for ties
    """
    # Get target molecule and its fingerprint
    target_mol = get_molecule_from_name(target_molecule_name)
    target_fp = get_morgan_fingerprint(target_mol)

    # Parse chemicals from PDF
    pool_chemicals = parse_pdf_chemicals(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for chem_name in pool_chemicals:
        try:
            mol = get_molecule_from_name(chem_name)
            fp = get_morgan_fingerprint(mol)
            similarity = compute_tanimoto_similarity(target_fp, fp)
            similarities.append((chem_name, similarity))
        except (ValueError, Exception):
            # Skip chemicals that can't be processed
            continue

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <k>")
        print("Example: python solution.py caffeine molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"Top {k} molecules similar to {target}:")
    for i, (name, sim) in enumerate(results, 1):
        print(f"{i}. {name}: {sim:.4f}")
