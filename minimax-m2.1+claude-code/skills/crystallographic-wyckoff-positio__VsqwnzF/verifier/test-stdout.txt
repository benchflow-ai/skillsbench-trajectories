"""
CIF File Wyckoff Position Analyzer

This module provides functionality to analyze CIF files and extract Wyckoff position
multiplicities and coordinates.
"""

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Dict, Any, Tuple
import re


def rational_approx(coord: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a rational approximation.

    Args:
        coord: The coordinate value (0-1)
        max_denominator: Maximum denominator for rational approximation

    Returns:
        String representation of the rational number
    """
    frac = Fraction(coord).limit_denominator(max_denominator)
    if frac.numerator == 0:
        return "0"
    elif frac.numerator == frac.denominator:
        return "1"
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing wyckoff_multiplicity_dict and wyckoff_coordinates_dict
    """
    # Read the CIF file
    structure = Structure.from_file(filepath)

    # Analyze space group symmetry
    sga = SpacegroupAnalyzer(structure)

    # Get symmetrized structure to identify equivalent positions
    symmetrized = sga.get_symmetrized_structure()

    # Initialize output dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Process each inequivalent site
    for site_set in symmetrized.equivalent_sites:
        # Get the first site from the set
        first_site = site_set[0]

        # Get the Wyckoff position letter (using the label if available)
        # The multiplicity is the number of equivalent sites
        multiplicity = len(site_set)

        # For labeling, we'll use letters in alphabetical order
        # Start with 'a' for the first position, 'b' for the second, etc.
        letter_index = len(wyckoff_multiplicity_dict)
        wyckoff_letter = chr(ord('a') + letter_index)

        # Store the multiplicity
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get the coordinates and convert to rational format
        coords = first_site.frac_coords
        rational_coords = [
            rational_approx(coord) for coord in coords
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
