from pymatgen.core import Structure
import spglib
from fractions import Fraction


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Mapping of Wyckoff position to multiplicity
        - wyckoff_coordinates_dict: Mapping of Wyckoff position to coordinates of first atom
    """
    # Read the CIF file
    structure = Structure.from_file(filepath)

    # Convert structure to spglib cell format
    cell = (
        structure.lattice.matrix,
        structure.frac_coords,
        structure.atomic_numbers
    )

    # Get symmetry dataset with Wyckoff positions
    dataset = spglib.get_symmetry_dataset(cell, symprec=1e-5)

    # Get Wyckoff positions (letters)
    wyckoff_letters = dataset.wyckoffs

    # Get symmetry operations
    rotations = dataset.rotations
    translations = dataset.translations

    # Group atoms by Wyckoff position
    wyckoff_dict = {}

    for i, (wyckoff_letter, coord, atom_type) in enumerate(zip(wyckoff_letters, structure.frac_coords, structure.species)):

        if wyckoff_letter not in wyckoff_dict:
            # First occurrence of this Wyckoff position
            wyckoff_dict[wyckoff_letter] = {
                'sites': [(coord, atom_type)],
                'representative': coord
            }
        else:
            # Add to existing Wyckoff position
            wyckoff_dict[wyckoff_letter]['sites'].append((coord, atom_type))

    # Count multiplicity for each Wyckoff position
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for wyckoff_letter, info in wyckoff_dict.items():
        # Count unique sites (some sites might be duplicated due to symmetry)
        multiplicity = len(info['sites'])

        # Store multiplicity
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get the representative coordinates
        coords = info['representative']

        # Convert to rational fractions with denominator <= 12
        rational_coords = []
        for coord in coords:
            frac = Fraction(coord).limit_denominator(12)
            rational_coords.append(str(frac))

        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
