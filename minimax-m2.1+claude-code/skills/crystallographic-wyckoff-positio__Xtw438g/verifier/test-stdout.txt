from typing import Dict, Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def decimal_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal to a fraction string with a maximum denominator.

    Args:
        value: Decimal value to convert
        max_denominator: Maximum denominator allowed

    Returns:
        String representation of the fraction
    """
    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
            - wyckoff_multiplicity_dict: Mapping of Wyckoff labels to multiplicities
            - wyckoff_coordinates_dict: Mapping of Wyckoff labels to coordinate strings
    """
    # Read the CIF file using pymatgen
    structure = Structure.from_file(filepath)

    # Use SpacegroupAnalyzer to get symmetry information
    sga = SpacegroupAnalyzer(structure)

    # Get the symmetry dataset which contains Wyckoff positions
    dataset = sga.get_symmetry_dataset()
    wyckoff_labels = dataset.wyckoffs

    # Dictionary to store Wyckoff position information
    wyckoff_info = {}

    # Iterate through the structure sites and Wyckoff labels
    for site_idx, (site, wyckoff_label) in enumerate(zip(structure, wyckoff_labels)):
        if wyckoff_label not in wyckoff_info:
            # Count multiplicity by counting occurrences of this Wyckoff label
            multiplicity = wyckoff_labels.count(wyckoff_label)

            # Convert fractional coordinates to fraction strings
            coord_strings = [
                decimal_to_fraction_string(site.frac_coords[0]),
                decimal_to_fraction_string(site.frac_coords[1]),
                decimal_to_fraction_string(site.frac_coords[2])
            ]

            # Store information for this Wyckoff position
            wyckoff_info[wyckoff_label] = {
                'multiplicity': multiplicity,
                'coordinates': coord_strings
            }

    # Build the output dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for label, info in wyckoff_info.items():
        wyckoff_multiplicity_dict[label] = info['multiplicity']
        wyckoff_coordinates_dict[label] = info['coordinates']

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
