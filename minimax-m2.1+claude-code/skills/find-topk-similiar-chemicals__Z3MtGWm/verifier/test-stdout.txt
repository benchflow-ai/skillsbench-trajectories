"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import rdMolDescriptors, DataStructs
import pubchempy as pcp
from typing import List, Tuple, Optional
import warnings
import time

# Suppress warnings for cleaner output
warnings.filterwarnings('ignore')


def extract_molecules_from_pdf(filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []

    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by lines and filter out empty lines
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)

    return molecules


def get_molecule_smiles(name: str) -> Optional[str]:
    """
    Get SMILES representation of a molecule by name using PubChem.

    Args:
        name: Chemical name

    Returns:
        SMILES string or None if not found
    """
    try:
        # Try to get the compound from PubChem
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            # Use smiles property (not deprecated like isomeric_smiles)
            return compounds[0].smiles
    except Exception as e:
        pass  # Silent error handling to reduce noise

    return None


def calculate_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True) -> Optional[bytes]:
    """
    Calculate Morgan fingerprint for a SMILES string.

    Args:
        smiles: SMILES representation of the molecule
        radius: Radius for Morgan fingerprint (default: 2)
        include_chirality: Whether to include chirality (default: True)

    Returns:
        Binary fingerprint representation or None if error
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None

        # Calculate Morgan fingerprint
        fingerprint = rdMolDescriptors.GetMorganFingerprintAsBitVect(
            mol,
            radius=radius,
            useChirality=include_chirality,
            nBits=2048  # Standard bit vector size
        )

        return fingerprint
    except Exception as e:
        print(f"Error calculating fingerprint for {smiles}: {e}")
        return None

def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top-k most similar molecules to a target molecule from a PDF pool.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity
    """
    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    target_smiles = get_molecule_smiles(target_molecule_name)
    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []

    # Calculate target fingerprint
    target_fp = calculate_morgan_fingerprint(target_smiles, radius=2, include_chirality=True)
    if target_fp is None:
        print(f"Could not calculate fingerprint for target: {target_molecule_name}")
        return []

    # Calculate similarities for all molecules in pool
    similarities = []

    for mol_name in pool_molecules:
        # Skip if same as target
        if mol_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES for molecule
        mol_smiles = get_molecule_smiles(mol_name)
        if mol_smiles is None:
            continue

        # Calculate fingerprint
        mol_fp = calculate_morgan_fingerprint(mol_smiles, radius=2, include_chirality=True)
        if mol_fp is None:
            continue

        # Calculate Tanimoto similarity
        # We need to use the actual DataStructs comparison
        from rdkit.Chem import DataStructs
        similarity = DataStructs.TanimotoSimilarity(target_fp, mol_fp)

        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    return similarities[:top_k]

    """

    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule

    # Calculate target fingerprint using Morgan fingerprint (radius=2, include chirality)
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        print(f"Could not parse SMILES for target: {target_molecule_name}")
        return []

    target_fp = rdMolDescriptors.GetMorganFingerprintAsBitVect(
        target_mol,
        radius=2,
        useChirality=True,
        nBits=2048
    )

    # Calculate similarities for all molecules in pool
    similarities = []

    for mol_name in pool_molecules:
        # Skip if same as target (check both name and SMILES if possible)
        if mol_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES for molecule
        mol_smiles = get_molecule_smiles(mol_name)
        if mol_smiles is None:
            continue

        # Calculate fingerprint
        mol_mol = Chem.MolFromSmiles(mol_smiles)
        if mol_mol is None:
            continue

        mol_fp = rdMolDescriptors.GetMorganFingerprintAsBitVect(
            mol_mol,
            radius=2,
            useChirality=True,
            nBits=2048
        )

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, mol_fp)

        similarities.append((mol_name, similarity))

        # Small delay to avoid overwhelming PubChem server
        time.sleep(0.01)

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    return similarities[:top_k]
