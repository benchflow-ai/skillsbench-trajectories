"""
Materials Science CIF Analyzer

This module provides functionality to analyze Wyckoff position multiplicities and
calculate approximate coordinates for the first atom of each Wyckoff position from
CIF (Crystallographic Information Framework) files generated by SHELX through X-ray
Diffraction.

The main function analyzes crystal structures to extract:
1. Wyckoff position multiplicities (how many symmetry-equivalent positions)
2. Fractional coordinates for representative atoms in each Wyckoff position

Coordinates are converted to rational numbers with denominators ≤ 12 for simplified
representation.

Example usage:
    result = analyze_wyckoff_position_multiplicities_and_coordinates("/path/to/file.cif")
    print(result)
    # Output:
    # {
    #     "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
    #     "wyckoff_coordinates_dict": {
    #         "a": ["0", "1/2", "1/2"],
    #         "c": ["3/8", "1/9", "8/9"]
    #     }
    # }
"""

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Dict, Any, List


def fraction_to_rational_string(frac_value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point number to a rational string with limited denominator.

    Args:
        frac_value: The fractional value to convert
        max_denominator: Maximum denominator allowed (default: 12)

    Returns:
        String representation of the rational number
    """
    frac = Fraction(frac_value).limit_denominator(max_denominator)
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Mapping of Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Mapping of Wyckoff letter to coordinate list
    """
    # Load the structure from CIF file
    struct = Structure.from_file(filepath)

    # Analyze the symmetry and get Wyckoff positions
    sga = SpacegroupAnalyzer(struct)
    symmetrized_structure = sga.get_symmetrized_structure()

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Collect unique Wyckoff positions
    seen_wyckoff_positions = set()

    for i, eq_site_set in enumerate(symmetrized_structure.equivalent_sites):
        if len(eq_site_set) > 0:
            wyckoff_symbol = symmetrized_structure.wyckoff_symbols[i]
            letter = wyckoff_symbol[-1]  # Extract Wyckoff letter
            multiplicity = int(wyckoff_symbol[:-1])  # Extract multiplicity

            # Only process each unique Wyckoff position once
            if letter not in seen_wyckoff_positions:
                seen_wyckoff_positions.add(letter)

                # Get the first atom's coordinates
                first_site = eq_site_set[0]
                coords = first_site.frac_coords

                # Store multiplicity
                wyckoff_multiplicity_dict[letter] = multiplicity

                # Convert coordinates to rational strings
                coord_strings = [
                    fraction_to_rational_string(coords[0]),
                    fraction_to_rational_string(coords[1]),
                    fraction_to_rational_string(coords[2])
                ]
                wyckoff_coordinates_dict[letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
