"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import rdMolDescriptors
import pubchempy as pcp
from typing import List, Tuple
from rdkit.DataStructs.cDataStructs import ExplicitBitVect
import time
import warnings


def extract_molecules_from_pdf(filepath: str) -> List[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file containing molecule names

    Returns:
        List of molecule names
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by lines and filter out empty lines
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)
    return molecules


def name_to_smiles(molecule_name: str) -> str:
    """
    Convert a molecule name to SMILES using PubChem.

    Args:
        molecule_name: Name of the molecule

    Returns:
        SMILES representation of the molecule
    """
    try:
        # Suppress deprecation warnings temporarily
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            # Search PubChem for the compound
            compounds = pcp.get_compounds(molecule_name, 'name')
        if compounds:
            # Use connectivity_smiles instead of deprecated canonical_smiles
            return compounds[0].connectivity_smiles
    except Exception as e:
        print(f"Error converting {molecule_name} to SMILES: {e}")
    return None


def compute_morgan_fingerprint(smiles: str) -> ExplicitBitVect:
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES representation of the molecule

    Returns:
        Morgan fingerprint (Bitvect)
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    # Suppress deprecation warnings for MorganGenerator
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        # radius=2, include chirality=True
        fp = rdMolDescriptors.GetMorganFingerprintAsBitVect(mol, radius=2, nBits=2048, useChirality=True)
    return fp


def tanimoto_similarity(fp1: ExplicitBitVect, fp2: ExplicitBitVect) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score
    """
    return Chem.DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(target_molecule_name: str, molecule_pool_filepath: str, top_k: int) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to a target molecule from a PDF.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF containing the molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity in descending order
    """
    # Extract molecules from PDF
    molecule_pool = extract_molecules_from_pdf(molecule_pool_filepath)

    # Convert target molecule to SMILES and compute fingerprint
    target_smiles = name_to_smiles(target_molecule_name)
    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []

    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        print(f"Could not compute fingerprint for target molecule: {target_molecule_name}")
        return []

    # Compute similarities with all molecules in the pool
    similarities = []
    for i, molecule in enumerate(molecule_pool):
        # Skip if it's the same as the target
        if molecule.lower() == target_molecule_name.lower():
            continue

        smiles = name_to_smiles(molecule)
        if smiles is None:
            continue

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Compute Tanimoto similarity
        similarity = tanimoto_similarity(target_fp, fp)
        similarities.append((molecule, similarity))

        # Add a small delay every 10 requests to avoid overwhelming PubChem
        if i % 10 == 0:
            time.sleep(0.1)

    # Sort by similarity in descending order, then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]
