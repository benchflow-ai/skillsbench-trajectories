package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FlatMapFunction;
import org.apache.flink.api.common.state.ListState;
import org.apache.flink.api.common.state.ListStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.streaming.api.functions.sink.RichSinkFunction;
import org.apache.flink.util.Collector;
import org.apache.flink.util.OutputTag;

import java.io.FileWriter;
import java.io.IOException;
import java.io.PrintWriter;
import java.time.Duration;
import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.List;

public class LongestSessionPerJob extends AppBase {

    private static final long SESSION_GAP_MICROS = 10 * 60 * 1000 * 1000L; // 10 minutes in microseconds
    private static final OutputTag<String> OUTPUT_TAG = new OutputTag<String>("output") {};

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        System.out.println("output  " + outputPath);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1); // For deterministic results

        // Read task events from the gzipped CSV file
        DataStream<String> textLines = env.readTextFile(taskInput);

        DataStream<TaskEvent> taskEvents = textLines
                .flatMap(new TaskEventParser());

        // Filter for SUBMIT events (event type = 0)
        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(event -> event.getEventType() == 0);

        // Use process function to compute longest stage per job
        SingleOutputStreamOperator<String> result = submitEvents
                .keyBy(TaskEvent::getJobId)
                .process(new StageComputer());

        // Get side output
        DataStream<String> output = result.getSideOutput(OUTPUT_TAG);

        // Use test sink if available, otherwise write to file
        if (out != null) {
            output.addSink(out);
        } else if (outputPath != null) {
            output.addSink(new FileSink(outputPath));
        } else {
            output.print();
        }

        env.execute("LongestSessionPerJob");
    }

    /**
     * File sink for writing output to a file
     */
    public static class FileSink extends RichSinkFunction<String> {
        private final String path;
        private transient PrintWriter writer;

        public FileSink(String path) {
            this.path = path;
        }

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) throws Exception {
            super.open(parameters);
            writer = new PrintWriter(new FileWriter(path, false)); // overwrite, not append
            writer.print(""); // Clear file
            writer.close();
            writer = new PrintWriter(new FileWriter(path, true));
        }

        @Override
        public void invoke(String value, Context context) throws Exception {
            writer.println(value);
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.close();
            }
        }
    }

    /**
     * Parser for gzipped task events CSV
     */
    public static class TaskEventParser implements FlatMapFunction<String, TaskEvent> {
        @Override
        public void flatMap(String line, Collector<TaskEvent> out) throws Exception {
            String[] fields = line.split(",");

            if (fields.length < 9) {
                return;
            }

            try {
                long timestamp = Long.parseLong(fields[0]);
                long jobId = Long.parseLong(fields[2]);
                int taskIndex = Integer.parseInt(fields[3]);
                int machineId = fields[4].isEmpty() ? -1 : Integer.parseInt(fields[4]);
                int eventType = Integer.parseInt(fields[5]);
                int priority = Integer.parseInt(fields[8]);

                out.collect(new TaskEvent(timestamp, jobId, taskIndex, machineId, eventType, priority));
            } catch (NumberFormatException e) {
                // Skip malformed lines
            }
        }
    }

    /**
     * Process function to find longest stage per job
     */
    public static class StageComputer extends KeyedProcessFunction<Long, TaskEvent, String> {

        private transient ListState<TaskEvent> eventsState;

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) throws Exception {
            eventsState = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("events", TaskEvent.class));
        }

        @Override
        public void processElement(TaskEvent event, Context ctx, Collector<String> out) throws Exception {
            // Store event in state
            eventsState.add(event);

            // Register a processing time timer that will fire
            // In batch mode, this will fire after all events are processed
            ctx.timerService().registerProcessingTimeTimer(System.currentTimeMillis() + 1000);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            // Only emit through side output
            // Collect all events from state
            List<TaskEvent> events = new ArrayList<>();
            for (TaskEvent event : eventsState.get()) {
                events.add(event);
            }

            if (events.isEmpty()) {
                return;
            }

            // Sort events by timestamp
            Collections.sort(events, Comparator.comparingLong(TaskEvent::getTimestamp));

            long jobId = events.get(0).getJobId();
            long currentStageStart = -1;
            long lastSubmitTime = -1;
            long maxStageTasks = 0;
            int currentStageTasks = 0;

            for (TaskEvent event : events) {
                long eventTimestamp = event.getTimestamp();

                if (currentStageStart == -1) {
                    // First event
                    currentStageStart = eventTimestamp;
                    currentStageTasks = 1;
                } else if (eventTimestamp - lastSubmitTime > SESSION_GAP_MICROS) {
                    // New stage - finalize current stage
                    if (currentStageTasks > maxStageTasks) {
                        maxStageTasks = currentStageTasks;
                    }
                    // Start new stage
                    currentStageStart = eventTimestamp;
                    currentStageTasks = 1;
                } else {
                    // Same stage - add task
                    currentStageTasks++;
                }

                lastSubmitTime = eventTimestamp;
            }

            // Finalize last stage
            if (currentStageTasks > maxStageTasks) {
                maxStageTasks = currentStageTasks;
            }

            // Emit through side output
            ctx.output(OUTPUT_TAG, "(" + jobId + "," + maxStageTasks + ")");
        }
    }
}
