"""Analyze Wyckoff position multiplicities and coordinates from CIF files."""

from collections import defaultdict
from fractions import Fraction
from typing import Any, Dict

from pymatgen.core import Structure
from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def rationalize_coordinate(coord: float, max_denom: int = 12) -> str:
    """Convert a coordinate to a rational string with denominator <= max_denom."""
    frac = Fraction(coord).limit_denominator(max_denom)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'
    """
    # Parse the CIF file using pymatgen's CifParser
    parser = CifParser(filepath)
    structures = parser.parse_structures(primitive=True)

    if not structures:
        raise ValueError(f"Could not parse structure from CIF file: {filepath}")

    structure = structures[0]

    # Get space group information
    analyzer = SpacegroupAnalyzer(structure)
    spg_symbol = analyzer.get_space_group_symbol()
    spg_number = analyzer.get_space_group_number()

    # Get the symmetrized structure to find symmetry-equivalent sites
    symmetrized = analyzer.get_symmetrized_structure()

    # Get equivalent indices and Wyckoff letters
    equivalent_indices = symmetrized.equivalent_indices
    wyckoff_letters = symmetrized.wyckoff_letters

    # Group atoms by Wyckoff position
    wyckoff_groups = defaultdict(lambda: {"indices": [], "multiplicity": 0, "first_coords": None})

    for i, letter in enumerate(wyckoff_letters):
        if letter not in wyckoff_groups:
            wyckoff_groups[letter]["indices"] = []
            wyckoff_groups[letter]["first_coords"] = structure[i].frac_coords
        wyckoff_groups[letter]["indices"].append(i)
        wyckoff_groups[letter]["multiplicity"] = len(wyckoff_groups[letter]["indices"])

    # Build output dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for letter in sorted(wyckoff_groups.keys()):
        data = wyckoff_groups[letter]
        wyckoff_multiplicity_dict[letter] = data["multiplicity"]
        coords = data["first_coords"]
        wyckoff_coordinates_dict[letter] = [
            rationalize_coordinate(coords[0]),
            rationalize_coordinate(coords[1]),
            rationalize_coordinate(coords[2])
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
