"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.
Uses pymatgen for crystallographic analysis.
"""
from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with:
        - "wyckoff_multiplicity_dict": mapping of Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": mapping of Wyckoff letter to [x, y, z] coordinates
          as simplified fractions (denominator <= 12)
    """
    # Load the structure from CIF
    structure = Structure.from_file(filepath)

    # Get space group information
    spg_analyzer = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure which groups atoms by Wyckoff positions
    symmetrized = spg_analyzer.get_symmetrized_structure()

    # Get the symmetry dataset which contains Wyckoff labels
    sym_data = spg_analyzer.get_symmetry_dataset()

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Build mapping from site index to wyckoff letter
    # The wyckoffs list in sym_data is indexed by the original structure atom indices
    wyckoff_letters = sym_data.get('wyckoffs', [])

    # Iterate over the Wyckoff positions in the symmetrized structure
    for wyckoffs in symmetrized.equivalent_sites:
        if len(wyckoffs) == 0:
            continue

        # Get the Wyckoff letter and multiplicity
        # The wyckoffs list contains all symmetry-equivalent sites
        multiplicity = len(wyckoffs)

        # Get representative site - first site in the Wyckoff set
        rep_site = wyckoffs[0]

        # Find the Wyckoff letter by matching the representative site with original atoms
        wyckoff_letter = None
        for atom_idx, site in enumerate(structure):
            if (abs(site.x - rep_site.x) < 1e-6 and
                abs(site.y - rep_site.y) < 1e-6 and
                abs(site.z - rep_site.z) < 1e-6):
                if atom_idx < len(wyckoff_letters):
                    wyckoff_letter = wyckoff_letters[atom_idx]
                break

        if wyckoff_letter is None:
            # Fallback: use sequential letters
            existing_letters = list(wyckoff_multiplicity_dict.keys())
            wyckoff_letter = chr(ord('a') + len(existing_letters))

        # Convert Cartesian coordinates to fractional coordinates
        frac_coords = structure.lattice.get_fractional_coords(
            [rep_site.x, rep_site.y, rep_site.z]
        )

        # Normalize to [0, 1) range
        frac_coords = [_normalize_fractional_coord(c) for c in frac_coords]

        # Round coordinates to fractions with denominator <= 12
        coords = [
            _fraction_to_string(_round_to_fraction(frac_coords[0])),
            _fraction_to_string(_round_to_fraction(frac_coords[1])),
            _fraction_to_string(_round_to_fraction(frac_coords[2]))
        ]

        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def _normalize_fractional_coord(coord: float) -> float:
    """
    Normalize a fractional coordinate to the [0, 1) range.
    Handles negative values and values >= 1.
    """
    # Handle negative coordinates
    while coord < 0:
        coord += 1
    # Handle coordinates >= 1
    while coord >= 1:
        coord -= 1
    return coord


def _round_to_fraction(value: float, max_denominator: int = 12) -> Fraction:
    """
    Round a floating point value to the nearest fraction with maximum denominator.
    """
    frac = Fraction(value).limit_denominator(max_denominator)
    return frac


def _fraction_to_string(frac: Fraction) -> str:
    """
    Convert a Fraction to string representation.
    Handles special cases like 0, 1, -1, etc.
    """
    if frac.numerator == 0:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"
