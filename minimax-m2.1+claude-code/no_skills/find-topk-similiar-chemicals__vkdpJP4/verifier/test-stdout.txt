"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

from typing import List, Tuple, Optional
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import pubchempy as pcp
from pypdf import PdfReader
import warnings
warnings.filterwarnings('ignore')
import logging
logging.getLogger('rdkit').setLevel(logging.ERROR)


def get_smiles_from_pubchem(chemical_name: str) -> Optional[str]:
    """Get SMILES string for a chemical name using PubChem."""
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].connectivity_smiles
    except Exception:
        pass
    return None


def smiles_to_morgan_fp(smiles: str, radius: int = 2, include_chirality: bool = True) -> Optional[DataStructs.ExplicitBitVect]:
    """Convert SMILES to Morgan fingerprint."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    fp = AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=radius, useChirality=include_chirality
    )
    return fp


def load_molecule_names_from_pdf(filepath: str) -> List[str]:
    """Load molecule names from PDF file (one name per line)."""
    reader = PdfReader(filepath)
    names = []

    for page in reader.pages:
        text = page.extract_text()
        if text:
            lines = text.split('\n')
            for line in lines:
                name = line.strip()
                if name:
                    names.append(name)

    return names


def tanimoto_similarity(fp1: DataStructs.ExplicitBitVect, fp2: DataStructs.ExplicitBitVect) -> float:
    """Compute Tanimoto similarity between two fingerprints."""
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to the target molecule.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing the molecule pool
        top_k: Number of similar molecules to return

    Returns:
        List of (molecule_name, similarity_score) tuples, sorted by similarity (descending),
        with alphabetical ordering for ties.
    """
    # Get SMILES for target molecule from PubChem
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for molecule: {target_molecule_name}")

    # Convert target to Morgan fingerprint
    target_fp = smiles_to_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Load molecule names from PDF
    molecule_names = load_molecule_names_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for name in molecule_names:
        # Skip the target molecule itself if present
        if name.lower() == target_molecule_name.lower():
            continue

        smiles = get_smiles_from_pubchem(name)
        if smiles:
            mol_fp = smiles_to_morgan_fp(smiles)
            if mol_fp is not None:
                similarity = tanimoto_similarity(target_fp, mol_fp)
                similarities.append((name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules("Caffeine", "/root/molecules.pdf", 5)
    print("Top 5 molecules similar to Caffeine:")
    for name, score in result:
        print(f"  {name}: {score:.4f}")
