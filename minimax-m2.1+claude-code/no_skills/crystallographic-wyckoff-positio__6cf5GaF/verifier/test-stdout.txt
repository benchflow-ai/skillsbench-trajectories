"""
Analyze Wyckoff position multiplicities and calculate approximate coordinates
for the first atom of each Wyckoff position from CIF files.
"""

from fractions import Fraction
from typing import Any, Dict


def parse_fraction(value: str) -> float:
    """Parse a fraction string to float."""
    if '/' in value:
        return float(Fraction(value))
    return float(value)


def fraction_to_string(frac: Fraction, max_denominator: int = 12) -> str:
    """Convert a fraction to string representation."""
    f = Fraction(frac).limit_denominator(max_denominator)
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'
    """
    try:
        from pymatgen.core import Lattice, Structure
        from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
    except ImportError:
        # Fallback implementation if pymatgen is not available
        return analyze_wyckoff_simple(filepath)

    # Parse CIF file using pymatgen
    structure = Structure.from_file(filepath)

    # Use SpacegroupAnalyzer to get space group and symmetry info
    spg_analyzer = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure with Wyckoff information
    symmetrized_structure = spg_analyzer.get_symmetrized_structure()

    # Extract Wyckoff positions
    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list] = {}

    # Get equivalent sites (groups of symmetry-equivalent atoms)
    equivalent_sites = symmetrized_structure.equivalent_sites
    wyckoff_letters = symmetrized_structure.wyckoff_letters

    # Get Wyckoff letters for each unique position
    # wyckoff_letters is indexed by site index in the original structure
    # We need to map each equivalent_sites group to its Wyckoff letter
    for i, site_group in enumerate(equivalent_sites):
        site = site_group[0]
        # Find the Wyckoff letter for this site's index
        # Get the first site's index in the structure
        site_idx = structure.sites.index(site)
        letter = wyckoff_letters[site_idx] if site_idx < len(wyckoff_letters) else 'a'
        multiplicity = len(site_group)

        # Store multiplicity if new
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = multiplicity
            # Convert coordinates to fraction strings
            coords = [
                fraction_to_string(Fraction(site.frac_coords[0]).limit_denominator(12)),
                fraction_to_string(Fraction(site.frac_coords[1]).limit_denominator(12)),
                fraction_to_string(Fraction(site.frac_coords[2]).limit_denominator(12))
            ]
            wyckoff_coordinates_dict[letter] = coords

    # Sort by Wyckoff letter for consistent output
    wyckoff_multiplicity_dict = dict(sorted(wyckoff_multiplicity_dict.items()))
    wyckoff_coordinates_dict = dict(sorted(wyckoff_coordinates_dict.items()))

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def analyze_wyckoff_simple(filepath: str) -> Dict[str, Any]:
    """
    Simple fallback implementation using basic CIF parsing.
    Groups atoms by symmetry to determine Wyckoff positions.
    """
    with open(filepath, 'r') as f:
        content = f.read()

    # Parse atom positions from CIF
    atom_sites = []
    in_atom_site_loop = False

    for line in content.split('\n'):
        if '_atom_site_type_symbol' in line:
            in_atom_site_loop = True
            continue

        if in_atom_site_loop:
            if line.strip().startswith('#') or line.strip().startswith('_') or not line.strip():
                if line.strip() and not line.strip().startswith('_'):
                    in_atom_site_loop = False
                continue

            parts = line.split()
            if len(parts) >= 8:
                atom_type = parts[0]
                label = parts[1]
                multiplicity = int(parts[2])
                x = parse_fraction(parts[3])
                y = parse_fraction(parts[4])
                z = parse_fraction(parts[5])
                atom_sites.append({
                    'type': atom_type,
                    'label': label,
                    'multiplicity': multiplicity,
                    'coords': (x, y, z)
                })

    # Group atoms by type and find equivalent positions
    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list] = {}

    # Group by atom type
    atoms_by_type: Dict[str, list] = {}
    for site in atom_sites:
        atom_type = site['type'].replace('+', '').replace('-', '')
        if atom_type not in atoms_by_type:
            atoms_by_type[atom_type] = []
        atoms_by_type[atom_type].append(site)

    # Find unique Wyckoff positions based on coordinates
    visited = set()
    wyckoff_id = 0
    wyckoff_letters = 'abcdefghijklmnopqrstuvwxyz'

    for atom_type, sites in atoms_by_type.items():
        for site in sites:
            site_key = tuple([round(c, 4) for c in site['coords']])
            if site_key in visited:
                continue

            # Find all equivalent sites (within tolerance)
            equivalent = [site]
            for other in sites:
                if other['coords'] != site['coords']:
                    other_key = tuple([round(c, 4) for c in other['coords']])
                    if other_key == site_key:
                        equivalent.append(other)

            multiplicity = len(equivalent)
            if atom_type not in wyckoff_multiplicity_dict.values():
                wyckoff_letter = wyckoff_letters[wyckoff_id] if wyckoff_id < len(wyckoff_letters) else 'a'
                wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
                wyckoff_coordinates_dict[wyckoff_letter] = [
                    fraction_to_string(Fraction(site['coords'][0]).limit_denominator(12)),
                    fraction_to_string(Fraction(site['coords'][1]).limit_denominator(12)),
                    fraction_to_string(Fraction(site['coords'][2]).limit_denominator(12))
                ]
                wyckoff_id += 1

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
