package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FlatMapFunction;
import org.apache.flink.api.common.functions.RuntimeContext;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.util.Collector;

import java.time.Duration;

public class LongestSessionPerJob extends AppBase {

    private static final long SESSION_TIMEOUT_MICROS = 10 * 60 * 1000 * 1000L; // 10 minutes in microseconds
    private static final int SUBMIT_EVENT_TYPE = 0; // SUBMIT event type

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  "+taskInput);
        System.out.println("job_input  "+jobInput);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events as strings and parse them
        org.apache.flink.streaming.api.datastream.DataStream<String> taskLines = env
            .readTextFile(taskInput)
            .name("TaskEvents");

        // Parse task events and filter for SUBMIT events
        org.apache.flink.streaming.api.datastream.DataStream<TaskEvent> submitEvents = taskLines
            .flatMap(new FlatMapFunction<String, TaskEvent>() {
                @Override
                public void flatMap(String line, Collector<TaskEvent> out) throws Exception {
                    if (line == null || line.isEmpty()) {
                        return;
                    }
                    try {
                        TaskEvent event = parseTaskEvent(line);
                        if (event != null && event.eventType == SUBMIT_EVENT_TYPE) {
                            out.collect(event);
                        }
                    } catch (Exception e) {
                        // Skip malformed lines
                    }
                }
            })
            .name("ParseAndFilterSubmitEvents")
            .assignTimestampsAndWatermarks(
                WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(1))
                    .withTimestampAssigner((event, timestamp) -> event.timestamp)
            );

        // Key by jobId and process to find stage boundaries and task counts
        org.apache.flink.streaming.api.datastream.DataStream<Tuple2<Long, Integer>> result = submitEvents
            .keyBy(event -> event.jobId)
            .process(new StageDetector())
            .name("StageDetection");

        // Write output to file - one line per tuple (jobId, maxTasksInStage)
        result.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    /**
     * Process function to detect stages based on inactivity timeout.
     * A stage ends when there is no SUBMIT event for 10 minutes.
     * Each task submission (even for the same task that was resubmitted) counts.
     */
    private static class StageDetector extends KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Integer>> {

        // State to track the current stage's task count
        private transient org.apache.flink.api.common.state.ValueState<Integer> currentStageTaskCount;

        // State to track the timestamp of the last SUBMIT event in current stage
        private transient org.apache.flink.api.common.state.ValueState<Long> lastSubmitTime;

        // State to track the maximum task count across all stages for this job
        private transient org.apache.flink.api.common.state.ValueState<Integer> maxStageTaskCount;

        // State to track if we've already output the result for this job
        private transient org.apache.flink.api.common.state.ValueState<Boolean> outputGenerated;

        public void open(RuntimeContext context) {
            currentStageTaskCount = context.getState(
                new org.apache.flink.api.common.state.ValueStateDescriptor<>("currentStageTaskCount", Integer.class));
            lastSubmitTime = context.getState(
                new org.apache.flink.api.common.state.ValueStateDescriptor<>("lastSubmitTime", Long.class));
            maxStageTaskCount = context.getState(
                new org.apache.flink.api.common.state.ValueStateDescriptor<>("maxStageTaskCount", Integer.class));
            outputGenerated = context.getState(
                new org.apache.flink.api.common.state.ValueStateDescriptor<>("outputGenerated", Boolean.class));
        }

        @Override
        public void processElement(TaskEvent event,
                                   KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Integer>>.Context ctx,
                                   Collector<Tuple2<Long, Integer>> out) throws Exception {

            long currentTime = event.timestamp;
            Long lastTime = lastSubmitTime.value();

            // If there's an existing timer and last submit time, check if this event should extend or start a new stage
            if (lastTime != null && currentTime - lastTime >= SESSION_TIMEOUT_MICROS) {
                // Previous stage ended due to timeout, update max if needed
                Integer stageCount = currentStageTaskCount.value();
                if (stageCount != null && stageCount > 0) {
                    Integer maxCount = maxStageTaskCount.value();
                    if (maxCount == null || stageCount > maxCount) {
                        maxStageTaskCount.update(stageCount);
                    }
                }
                // Start new stage with this event
                currentStageTaskCount.update(1);
            } else if (lastTime == null) {
                // First event in a new stage
                currentStageTaskCount.update(1);
            } else {
                // Continue current stage - increment task count
                Integer count = currentStageTaskCount.value();
                currentStageTaskCount.update(count + 1);
            }

            // Update last submit time
            lastSubmitTime.update(currentTime);

            // Register a timer for when this stage should end (10 minutes from now)
            long timerTs = currentTime + SESSION_TIMEOUT_MICROS;
            ctx.timerService().registerEventTimeTimer(timerTs);
        }

        @Override
        public void onTimer(long timestamp,
                            KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Integer>>.OnTimerContext ctx,
                            Collector<Tuple2<Long, Integer>> out) throws Exception {

            Long lastTime = lastSubmitTime.value();
            if (lastTime == null) {
                return;
            }

            // Check if the timer has expired (meaning no new events arrived before the timeout)
            if (timestamp >= lastTime + SESSION_TIMEOUT_MICROS) {
                // Stage has ended, update max count
                Integer stageCount = currentStageTaskCount.value();
                if (stageCount != null && stageCount > 0) {
                    Integer maxCount = maxStageTaskCount.value();
                    if (maxCount == null || stageCount > maxCount) {
                        maxStageTaskCount.update(stageCount);
                    }
                }

                // Clear current stage state
                currentStageTaskCount.clear();
                lastSubmitTime.clear();

                // Output the result if we haven't already
                Boolean alreadyOutput = outputGenerated.value();
                if (alreadyOutput == null || !alreadyOutput) {
                    Integer finalMax = maxStageTaskCount.value();
                    if (finalMax != null && finalMax > 0) {
                        out.collect(new Tuple2<>(ctx.getCurrentKey(), finalMax));
                        outputGenerated.update(true);
                    }
                }
            }
        }

        @Override
        public void close() throws Exception {
            // Nothing to do here
        }
    }

    /**
     * Parse a task event line from CSV format.
     * Format: timestamp,missing_info,job_id,task_index,machine_id,event_type,user,scheduling_class,priority,cpu_request,memory_request,disk_space,diff_machine
     */
    private static TaskEvent parseTaskEvent(String line) {
        String[] fields = line.split(",");
        if (fields.length < 13) {
            return null;
        }

        try {
            TaskEvent event = new TaskEvent();
            event.timestamp = Long.parseLong(fields[0]);
            event.missingInfo = fields[1].isEmpty() ? null : fields[1];
            event.jobId = Long.parseLong(fields[2]);
            event.taskIndex = Integer.parseInt(fields[3]);
            event.machineId = Long.parseLong(fields[4]);
            event.eventType = Integer.parseInt(fields[5]);
            event.user = fields[6].isEmpty() ? null : fields[6];
            event.schedulingClass = Integer.parseInt(fields[7]);
            event.priority = Integer.parseInt(fields[8]);
            event.cpuRequest = Double.parseDouble(fields[9]);
            event.memoryRequest = Double.parseDouble(fields[10]);
            event.diskSpace = Double.parseDouble(fields[11]);
            event.differentMachines = Integer.parseInt(fields[12]);
            return event;
        } catch (NumberFormatException e) {
            return null;
        }
    }
}
