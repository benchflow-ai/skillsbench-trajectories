"""
Find top k similar chemicals using Tanimoto similarity with Morgan fingerprints.
"""

import re
from typing import List, Tuple

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import RDLogger
# Suppress RDKit logging
RDLogger.DisableLog('rdApp.*')

try:
    import pubchempy as pcp
except ImportError:
    pcp = None


def extract_chemical_names_from_pdf(filepath: str) -> List[str]:
    """Extract chemical names from a PDF file."""
    chemical_names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by lines and clean up
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    # Skip empty lines and headers
                    if line and not line.lower().startswith('chemical'):
                        # Extract chemical name (first part before any numbers)
                        chemical_names.append(line)
    return chemical_names


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """Get SMILES string for a chemical name using PubChem."""
    if pcp is None:
        raise ImportError("pubchempy is required. Install with: pip install pubchempy")

    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def name_to_smiles(chemical_name: str) -> str:
    """Convert a chemical name to SMILES using PubChem."""
    # Try direct PubChem lookup
    smiles = get_smiles_from_pubchem(chemical_name)
    if smiles:
        return smiles

    # Try with "chemical" prefix if not found
    if not chemical_name.lower().startswith('chemical'):
        smiles = get_smiles_from_pubchem(f"chemical {chemical_name}")
        if smiles:
            return smiles

    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True) -> tuple:
    """Compute Morgan fingerprint from SMILES string.

    Returns a tuple of (mol, fp) where mol is the RDKit molecule and fp is the fingerprint.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None, None

    fp = AllChem.GetMorganFingerprintAsBitVect(
        mol,
        radius=radius,
        useChirality=include_chirality,
        nBits=2048
    )
    return mol, fp


def tanimoto_similarity(fp1, fp2) -> float:
    """Calculate Tanimoto similarity between two fingerprints."""
    if fp1 is None or fp2 is None:
        return 0.0
    return DataStructs.TanimotoSimilarity(fp1, fp2)

from rdkit import DataStructs


def get_molecule_similarity(target_smiles: str, candidate_smiles: str) -> float:
    """Calculate Tanimoto similarity between two molecules."""
    target_mol, target_fp = compute_morgan_fingerprint(target_smiles)
    candidate_mol, candidate_fp = compute_morgan_fingerprint(candidate_smiles)

    if target_mol is None or candidate_mol is None:
        return 0.0

    return tanimoto_similarity(target_fp, candidate_fp)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to a target molecule.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of (molecule_name, similarity_score) tuples, sorted by
        descending similarity, alphabetical order for ties.
    """
    # Get SMILES for target molecule
    target_smiles = name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute target fingerprint
    target_mol, target_fp = compute_morgan_fingerprint(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract chemical names from PDF
    candidate_names = extract_chemical_names_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for name in candidate_names:
        if name == target_molecule_name:
            continue  # Skip the target itself

        candidate_smiles = name_to_smiles(name)
        if candidate_smiles is None:
            continue

        candidate_mol, candidate_fp = compute_morgan_fingerprint(candidate_smiles)
        if candidate_mol is None:
            continue

        similarity = tanimoto_similarity(target_fp, candidate_fp)
        similarities.append((name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        target_molecule_name="caffeine",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5
    )
    print("Top 5 similar molecules to caffeine:")
    for name, score in result:
        print(f"  {name}: {score:.4f}")
