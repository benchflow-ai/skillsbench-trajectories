"""
Analyze Wyckoff position multiplicities and calculate approximate coordinates
for the first atom of each Wyckoff position from CIF files.
"""
from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict | Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with:
            - wyckoff_multiplicity_dict: Maps Wyckoff letter to multiplicity
            - wyckoff_coordinates_dict: Maps Wyckoff letter to coordinates of first atom
    """
    # Load structure from CIF
    structure = Structure.from_file(filepath)

    # Get space group analyzer
    sga = SpacegroupAnalyzer(structure)

    # Get symmetry dataset with Wyckoff info
    symmetry_dataset = sga.get_symmetry_dataset()

    # Extract Wyckoff information (use attribute interface to avoid deprecation warning)
    wyckoff_letters = symmetry_dataset.wyckoffs

    # Build dictionary mapping Wyckoff letter to multiplicity and coordinates
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    seen_wyckoff = {}  # Track first occurrence of each Wyckoff letter

    for idx, (site, wyckoff_letter) in enumerate(zip(structure, wyckoff_letters)):
        # Count multiplicity for this Wyckoff position
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff_letter] = wyckoff_letters.count(wyckoff_letter)

        # Store first atom coordinates for this Wyckoff position
        if wyckoff_letter not in seen_wyckoff:
            coords = site.frac_coords
            wyckoff_coordinates_dict[wyckoff_letter] = [
                _frac_to_rational_string(coord) for coord in coords
            ]
            seen_wyckoff[wyckoff_letter] = True

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def _frac_to_rational_string(value: float, max_denom: int = 12) -> str:
    """
    Convert a float to a rational string representation.

    Args:
        value: Float value to convert
        max_denom: Maximum denominator for rational approximation

    Returns:
        String representation of the rational number
    """
    # Handle exact integers
    if abs(value - round(value)) < 1e-10:
        return str(int(round(value)))

    # Convert to fraction with bounded denominator
    frac = Fraction(value).limit_denominator(max_denom)

    # Handle the case where the fraction is exactly 0 or 1
    if frac.numerator == 0:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)

    # Return as "numerator/denominator" string
    return f"{frac.numerator}/{frac.denominator}"
