"""
Tanimoto similarity search for molecules using Morgan fingerprints.
"""

import re
import warnings
from typing import List, Tuple, Optional

try:
    from rdkit import Chem
    from rdkit.Chem import AllChem
except ImportError:
    raise ImportError("RDKit is required. Install with: pip install rdkit")


def extract_chemical_names_from_pdf(pdf_path: str) -> List[str]:
    """
    Extract chemical names from a PDF file (one name per line).

    Args:
        pdf_path: Path to the PDF file containing chemical names.

    Returns:
        List of chemical names.
    """
    import pdfplumber

    names = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        names.append(line)
    return names


def get_smiles_from_name(chemical_name: str) -> Optional[str]:
    """
    Get SMILES string for a chemical name using PubChem.

    Args:
        chemical_name: The name of the chemical.

    Returns:
        SMILES string for the chemical, or None if not found.
    """
    try:
        import pubchempy as pcp
        with warnings.catch_warnings():
            warnings.simplefilter("ignore", category=DeprecationWarning)
            compounds = pcp.get_compounds(chemical_name, 'name')
            if compounds:
                return compounds[0].canonical_smiles
    except Exception:
        pass

    # Try searching with a slight variation
    try:
        import pubchempy as pcp
        with warnings.catch_warnings():
            warnings.simplefilter("ignore", category=DeprecationWarning)
            # Try with different name variations
            search_name = chemical_name.replace('-', ' ')
            compounds = pcp.get_compounds(search_name, 'name')
            if compounds:
                return compounds[0].canonical_smiles
    except Exception:
        pass

    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string of the molecule.
        radius: Radius for Morgan fingerprint (default 2).
        include_chirality: Whether to include chirality (default True).

    Returns:
        RDKit fingerprint object, or None if conversion fails.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    fp = AllChem.GetMorganFingerprintAsBitVect(
        mol, radius, useChirality=include_chirality
    )
    return fp


def tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint.
        fp2: Second fingerprint.

    Returns:
        Tanimoto similarity coefficient.
    """
    from rdkit import DataStructs
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to the target molecule.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to the PDF file containing the molecule pool.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score), sorted by similarity
        in descending order. Alphabetical ordering for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)

    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    target_fp = compute_morgan_fingerprint(target_smiles)

    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecule names from PDF
    molecule_names = extract_chemical_names_from_pdf(molecule_pool_filepath)

    # Cache SMILES and fingerprints for the pool
    cache = {}
    for name in molecule_names:
        if name.lower() == target_molecule_name.lower():
            continue  # Skip the target itself

        smiles = get_smiles_from_name(name)
        if smiles is None:
            continue  # Skip molecules we can't look up

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue  # Skip molecules with invalid SMILES

        cache[name] = (smiles, fp)

    # Compute similarities
    similarities = []
    for name, (smiles, fp) in cache.items():
        similarity = tanimoto_similarity(target_fp, fp)
        similarities.append((name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    return similarities[:top_k]


def main():
    """Main function for testing."""
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <molecules_pdf> <k>")
        print("Example: python solution.py aspirin /root/molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"Top {k} molecules similar to '{target}':")
    for i, (name, score) in enumerate(results, 1):
        print(f"{i}. {name}: {score:.4f}")


if __name__ == "__main__":
    main()
