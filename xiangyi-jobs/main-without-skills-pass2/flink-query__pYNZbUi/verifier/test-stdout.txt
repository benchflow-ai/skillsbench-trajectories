package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.state.ListState;
import org.apache.flink.api.common.state.ListStateDescriptor;
import org.apache.flink.api.common.typeinfo.Types;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;
import org.apache.flink.util.Collector;

import java.io.BufferedReader;
import java.io.FileInputStream;
import java.io.InputStreamReader;
import java.io.Serializable;
import java.time.Duration;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    // 10 minutes in milliseconds
    private static final long SESSION_GAP_MS = 10 * 60 * 1000;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events
        DataStream<TaskEvent> taskEvents = env
                .addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .name("TaskEvents");

        // Read job events
        DataStream<JobEvent> jobEvents = env
                .addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                .name("JobEvents");

        // Filter for SUBMIT events only and convert to unified event type
        DataStream<UnifiedEvent> submitEvents = taskEvents
                .filter(event -> event != null && event.getEventType() == TaskEvent.SUBMIT)
                .map(event -> new UnifiedEvent(event.getJobId(), event.getTimestamp() / 1000, false));

        // Convert job finish events to unified event type
        DataStream<UnifiedEvent> finishEvents = jobEvents
                .filter(event -> event != null && event.isFinished())
                .map(event -> new UnifiedEvent(event.getJobId(), event.getTimestamp() / 1000, true));

        // Union both streams and assign timestamps
        DataStream<UnifiedEvent> allEvents = submitEvents.union(finishEvents)
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<UnifiedEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.timestamp)
                );

        // Process by job ID
        DataStream<String> result = allEvents
                .keyBy(event -> event.jobId)
                .process(new SessionProcessor());

        // Write output
        result.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    public static class UnifiedEvent implements Serializable, Comparable<UnifiedEvent> {
        public long jobId;
        public long timestamp;
        public boolean isJobFinish;

        public UnifiedEvent() {}

        public UnifiedEvent(long jobId, long timestamp, boolean isJobFinish) {
            this.jobId = jobId;
            this.timestamp = timestamp;
            this.isJobFinish = isJobFinish;
        }

        @Override
        public int compareTo(UnifiedEvent other) {
            return Long.compare(this.timestamp, other.timestamp);
        }
    }

    public static class SessionProcessor extends KeyedProcessFunction<Long, UnifiedEvent, String> {

        private ListState<Long> submitTimestamps;

        @Override
        public void open(Configuration parameters) {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Types.LONG));
        }

        @Override
        public void processElement(UnifiedEvent event, Context ctx, Collector<String> out) throws Exception {
            if (event.isJobFinish) {
                // Job finished - compute sessions and output max
                outputResult(event.jobId, out);
            } else {
                // Task SUBMIT event - store timestamp
                submitTimestamps.add(event.timestamp);
            }
        }

        private void outputResult(long jobId, Collector<String> out) throws Exception {
            // Collect all timestamps
            List<Long> timestamps = new ArrayList<>();
            for (Long ts : submitTimestamps.get()) {
                timestamps.add(ts);
            }

            if (timestamps.isEmpty()) {
                // No submit events, nothing to output
                return;
            }

            // Sort by timestamp
            Collections.sort(timestamps);

            // Compute sessions
            long maxSessionCount = 0;
            long currentSessionCount = 1;
            long lastTimestamp = timestamps.get(0);

            for (int i = 1; i < timestamps.size(); i++) {
                long currentTimestamp = timestamps.get(i);
                if (currentTimestamp - lastTimestamp > SESSION_GAP_MS) {
                    // New session
                    if (currentSessionCount > maxSessionCount) {
                        maxSessionCount = currentSessionCount;
                    }
                    currentSessionCount = 1;
                } else {
                    currentSessionCount++;
                }
                lastTimestamp = currentTimestamp;
            }

            // Check the last session
            if (currentSessionCount > maxSessionCount) {
                maxSessionCount = currentSessionCount;
            }

            // Output result
            out.collect("(" + jobId + "," + maxSessionCount + ")");

            // Clear state
            submitTimestamps.clear();
        }
    }

    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String filePath;
        private volatile boolean isRunning = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(new GZIPInputStream(new FileInputStream(filePath))))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    public static class JobEventSource implements SourceFunction<JobEvent> {

        private volatile boolean isRunning = true;

        }

                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }
}
