"""
Compute top-k Tanimoto similarity between a target molecule and molecules listed in a PDF.

Requirements implemented:
- Use external chemistry resource (PubChem) to convert names -> SMILES
- Morgan fingerprints (radius=2, include chirality) and Tanimoto similarity
- Sort by descending similarity, then alphabetical for ties
"""

from __future__ import annotations

import json
import time
from dataclasses import dataclass
from typing import Dict, Iterable, List, Optional, Tuple
from urllib import error, parse, request

from pypdf import PdfReader
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem

PUBCHEM_PUG_URL = (
    "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{name}/property/IsomericSMILES/JSON"
)

# Simple in-memory cache to avoid repeated network calls in a single process
_SMILES_CACHE: Dict[str, Optional[str]] = {}


@dataclass(frozen=True)
class SimilarityResult:
    name: str
    similarity: float


def _read_molecule_names_from_pdf(filepath: str) -> List[str]:
    reader = PdfReader(filepath)
    names: List[str] = []
    for page in reader.pages:
        text = page.extract_text() or ""
        for line in text.splitlines():
            line = line.strip()
            if line:
                names.append(line)
    # De-duplicate while preserving order
    seen = set()
    unique: List[str] = []
    for name in names:
        if name not in seen:
            seen.add(name)
            unique.append(name)
    return unique


def _fetch_isomeric_smiles_pubchem(name: str, retries: int = 2, timeout: int = 15) -> Optional[str]:
    if name in _SMILES_CACHE:
        return _SMILES_CACHE[name]

    encoded = parse.quote(name)
    url = PUBCHEM_PUG_URL.format(name=encoded)

    last_error: Optional[Exception] = None
    for attempt in range(retries + 1):
        try:
            with request.urlopen(url, timeout=timeout) as resp:
                if resp.status != 200:
                    raise error.HTTPError(url, resp.status, resp.reason, resp.headers, None)
                data = json.loads(resp.read().decode("utf-8"))
            props = data.get("PropertyTable", {}).get("Properties", [])
            smiles = None
            if props:
                smiles = (
                    props[0].get("IsomericSMILES")
                    or props[0].get("CanonicalSMILES")
                    or props[0].get("SMILES")
                )
            _SMILES_CACHE[name] = smiles
            return smiles
        except error.HTTPError as exc:
            last_error = exc
            if exc.code == 404:
                _SMILES_CACHE[name] = None
                return None
        except Exception as exc:  # noqa: BLE001 - keep broad to retry transient errors
            last_error = exc
        if attempt < retries:
            time.sleep(0.5 * (attempt + 1))

    # Cache failures to avoid repeated lookups in the same run
    _SMILES_CACHE[name] = None
    if last_error:
        return None
    return None


def _mol_from_name(name: str) -> Optional[Chem.Mol]:
    smiles = _fetch_isomeric_smiles_pubchem(name)
    if not smiles:
        return None
    return Chem.MolFromSmiles(smiles)


def _morgan_fp(mol: Chem.Mol):
    return AllChem.GetMorganFingerprintAsBitVect(mol, 2, nBits=2048, useChirality=True)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Tuple[str, float]]:
    """
    Return the top-k most similar molecules to the target using Tanimoto similarity.

    Returns a list of (molecule_name, similarity) sorted by:
    1) similarity descending
    2) name alphabetical for ties
    """

    if top_k <= 0:
        return []

    pool = _read_molecule_names_from_pdf(molecule_pool_filepath)

    target_mol = _mol_from_name(target_molecule_name)
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")

    target_fp = _morgan_fp(target_mol)

    results: List[Tuple[str, float]] = []
    for name in pool:
        mol = _mol_from_name(name)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        sim = float(DataStructs.TanimotoSimilarity(target_fp, fp))
        results.append((name, sim))

    results.sort(key=lambda item: (-item[1], item[0]))
    return results[:top_k]


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Compute top-k Tanimoto similarity to molecules in a PDF.")
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument("pdf", help="Path to molecules PDF")
    parser.add_argument("k", type=int, help="Top-k results")
    args = parser.parse_args()

    for name, sim in topk_tanimoto_similarity_molecules(args.target, args.pdf, args.k):
        print(f"{name}\t{sim:.4f}")
