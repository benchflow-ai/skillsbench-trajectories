package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FilterFunction;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.connector.file.src.FileSource;
import org.apache.flink.connector.file.src.reader.TextLineInputFormat;
import org.apache.flink.core.fs.Path;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.sink.SinkFunction;
import org.apache.flink.streaming.api.functions.sink.filesystem.StreamingFileSink;
import org.apache.flink.streaming.api.functions.sink.filesystem.rollingpolicies.OnCheckpointRollingPolicy;
import org.apache.flink.util.Collector;

public class LongestSessionPerJob extends AppBase {

    // Event types
    private static final int EVENT_SUBMIT = 0;
    private static final int EVENT_FINISH = 4;

    // Session gap in microseconds: 10 minutes
    private static final long SESSION_GAP_MICROS = 10L * 60 * 1000 * 1000;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Create file sources for task events
        FileSource<String> taskFileSource = FileSource
                .forRecordStreamFormat(new TextLineInputFormat(), new Path(taskInput))
                .build();

        // Create file sources for job events
        FileSource<String> jobFileSource = FileSource
                .forRecordStreamFormat(new TextLineInputFormat(), new Path(jobInput))
                .build();

        // Read and parse task events
        DataStream<TaskEvent> taskEvents = env
                .fromSource(taskFileSource, WatermarkStrategy.noWatermarks(), "TaskEventSource")
                .map(new MapFunction<String, TaskEvent>() {
                    @Override
                    public TaskEvent map(String line) throws Exception {
                        return TaskEvent.fromString(line);
                    }
                })
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent event) throws Exception {
                        return event != null;
                    }
                });

        // Read and parse job events
        DataStream<JobEvent> jobEvents = env
                .fromSource(jobFileSource, WatermarkStrategy.noWatermarks(), "JobEventSource")
                .map(new MapFunction<String, JobEvent>() {
                    @Override
                    public JobEvent map(String line) throws Exception {
                        return JobEvent.fromString(line);
                    }
                })
                .filter(new FilterFunction<JobEvent>() {
                    @Override
                    public boolean filter(JobEvent event) throws Exception {
                        return event != null;
                    }
                });

        // Filter task events to only SUBMIT events
        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent event) throws Exception {
                        return event.eventType == EVENT_SUBMIT;
                    }
                })
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000) // Convert to milliseconds
                );

        // Filter job events to only FINISH events
        DataStream<JobEvent> finishEvents = jobEvents
                .filter(new FilterFunction<JobEvent>() {
                    @Override
                    public boolean filter(JobEvent event) throws Exception {
                        return event.eventType == EVENT_FINISH;
                    }
                })
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000) // Convert to milliseconds
                );

        // Map submit events to (jobId, timestamp) tuples
        DataStream<Tuple2<Long, Long>> submitSignals = submitEvents
                .keyBy(event -> event.jobId)
                .map(new MapFunction<TaskEvent, Tuple2<Long, Long>>() {
                    @Override
                    public Tuple2<Long, Long> map(TaskEvent event) throws Exception {
                        return new Tuple2<>(event.jobId, event.timestamp);
                    }
                });

        // Map job finish events to (jobId, timestamp) tuples
        DataStream<Tuple2<Long, Long>> finishSignals = finishEvents
                .keyBy(event -> event.jobId)
                .map(new MapFunction<JobEvent, Tuple2<Long, Long>>() {
                    @Override
                    public Tuple2<Long, Long> map(JobEvent event) throws Exception {
                        return new Tuple2<>(event.jobId, event.timestamp);
                    }
                });

        // Connect submit events with finish events and process
        DataStream<String> results = submitSignals
                .keyBy(t -> t.f0)
                .connect(finishSignals.keyBy(t -> t.f0))
                .process(new JobStageProcessor());

        // Write results to file
        StreamingFileSink<String> sink = StreamingFileSink
                .forRowFormat(new Path(outputPath), new org.apache.flink.api.common.serialization.SimpleStringEncoder<String>("UTF-8"))
                .withRollingPolicy(OnCheckpointRollingPolicy.build())
                .build();

        results.addSink((SinkFunction<String>) sinkOrTest(sink));

        env.execute("LongestSessionPerJob");
    }

    /**
     * Co-process function to track sessions and find the longest stage when job finishes
     */
    public static class JobStageProcessor extends KeyedCoProcessFunction<Long, Tuple2<Long, Long>, Tuple2<Long, Long>, String> {

        private transient ValueState<Long> lastSubmitTime;
        private transient ValueState<Integer> currentSessionCount;
        private transient ValueState<Integer> maxSessionCount;

        @Override
        public void open(Configuration parameters) throws Exception {
            lastSubmitTime = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("lastSubmitTime", Long.class));
            currentSessionCount = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("currentSessionCount", Integer.class));
            maxSessionCount = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("maxSessionCount", Integer.class));
        }

        @Override
        public void processElement1(Tuple2<Long, Long> submitSignal, Context ctx, Collector<String> out) throws Exception {
            long jobId = submitSignal.f0;
            long submitTime = submitSignal.f1;

            Long lastTime = lastSubmitTime.value();
            Integer currentCount = currentSessionCount.value();
            Integer maxCount = maxSessionCount.value();

            if (currentCount == null) {
                currentCount = 0;
            }
            if (maxCount == null) {
                maxCount = 0;
            }

            if (lastTime == null) {
                // First submit event for this job
                currentSessionCount.update(1);
                lastSubmitTime.update(submitTime);
            } else {
                long gap = submitTime - lastTime;

                if (gap > SESSION_GAP_MICROS) {
                    // Session ended, check if it was the longest
                    if (currentCount > maxCount) {
                        maxSessionCount.update(currentCount);
                    }
                    // Start new session
                    currentSessionCount.update(1);
                } else {
                    // Continue current session
                    currentSessionCount.update(currentCount + 1);
                }

                lastSubmitTime.update(submitTime);
            }
        }

        @Override
        public void processElement2(Tuple2<Long, Long> finishSignal, Context ctx, Collector<String> out) throws Exception {
            long jobId = finishSignal.f0;

            // Job finished, need to emit the result
            Integer currentCount = currentSessionCount.value();
            Integer maxCount = maxSessionCount.value();

            if (currentCount == null) {
                currentCount = 0;
            }
            if (maxCount == null) {
                maxCount = 0;
            }

            // The final session might still be ongoing, so compare with maxCount
            int finalMax = Math.max(currentCount, maxCount);

            // Emit the result
            out.collect("(" + jobId + "," + finalMax + ")");

            // Clear state
            lastSubmitTime.clear();
            currentSessionCount.clear();
            maxSessionCount.clear();
        }
    }
}
