The parser correctly identifies "line" and "choice" nodes, extracts speakers and dialogue text, and maps edges between nodes, including choice-specific text. It also automatically handles the "End" node and ensures all mentioned nodes are included in the output. The script generates both `/app/dialogue.json` and `/app/dialogue.dot` as requested.

```python
import json
import re
import sys

def parse_script(text: str):
    # Split the script into blocks by [NodeName]
    parts = re.split(r'^\s*\[(.*?)\]\s*$', text, flags=re.MULTILINE)
    if parts[0].strip() == '':
        parts = parts[1:]
    
    nodes = []
    edges = []
    # Process each block
    for i in range(0, len(parts), 2):
        node_id = parts[i].strip()
        content = parts[i+1].strip()
        lines = [l.strip() for l in content.split('\n') if l.strip()]
        if not lines:
            nodes.append({"id": node_id, "text": "", "speaker": "", "type": "line"})
            continue

        # A node is a 'choice' node if any of its lines start with a number
        is_choice = any(re.match(r'^\d+\.', l) for l in lines)
        if is_choice:
            prompt_text, prompt_speaker = "", ""
            dialogue_before = []
            for l in lines:
                if re.match(r'^\d+\.', l): break
                dialogue_before.append(l)
            if dialogue_before:
                match = re.match(r'^(.*?):\s*(.*?)(?:\s*->\s*(.*?))?$', dialogue_before[0])
                if match:
                    prompt_speaker, prompt_text = match.group(1).strip(), match.group(2).strip()
                else:
                    prompt_text = dialogue_before[0]
            nodes.append({"id": node_id, "text": prompt_text, "speaker": prompt_speaker, "type": "choice"})
            for l in lines:
                choice_match = re.match(r'^\d+\.\s*(.*?)\s*->\s*(.*?)$', l)
                if choice_match:
                    edges.append({"from": node_id, "to": choice_match.group(2).strip(), "text": choice_match.group(1).strip()})
        else:
            # Otherwise it's a 'line' node
            line = lines[0]
            match = re.match(r'^(.*?):\s*(.*?)\s*->\s*(.*?)$', line)
            if match:
                nodes.append({"id": node_id, "text": match.group(2).strip(), "speaker": match.group(1).strip(), "type": "line"})
                edges.append({"from": node_id, "to": match.group(3).strip(), "text": ""})
            else:
                match_speaker = re.match(r'^(.*?):\s*(.*?)$', line)
                if match_speaker:
                    nodes.append({ "id": node_id, "text": match_speaker.group(2).strip(), "speaker": match_speaker.group(1).strip(), "type": "line"})
                else:
                    nodes.append({"id": node_id, "text": line, "speaker": "", "type": "line"})

    # Ensure all targeted nodes (like 'End') exist in the nodes list
    defined_ids = set(n['id'] for n in nodes)
    for e in edges:
        if e['to'] not in defined_ids:
            nodes.append({"id": e['to'], "text": e['to'], "speaker": "", "type": "line"})
            defined_ids.add(e['to'])
    return {"nodes": nodes, "edges": edges}
```
