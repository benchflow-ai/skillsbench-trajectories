import functools
import time
from typing import Dict, List, Optional

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator


def _extract_molecule_names_from_pdf(filepath: str) -> List[str]:
    """Extract molecule names from a PDF where each name appears on its own line."""
    names: List[str] = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if name:
                    names.append(name)
    return names


@functools.lru_cache(maxsize=1024)
def _pubchem_name_to_smiles(name: str) -> Optional[str]:
    """Resolve a chemical name to an isomeric SMILES via PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
    except Exception:
        return None
    if not compounds:
        return None
    compound = compounds[0]
    # Prefer isomeric SMILES (PubChemPy exposes it as .smiles).
    smiles = compound.smiles or compound.canonical_smiles
    return smiles


def _name_to_mol(name: str) -> Optional[Chem.Mol]:
    smiles = _pubchem_name_to_smiles(name)
    if not smiles:
        return None
    mol = Chem.MolFromSmiles(smiles)
    return mol


_MORGAN_GEN = rdFingerprintGenerator.GetMorganGenerator(
    radius=2,
    fpSize=2048,
    includeChirality=True,
)


def _morgan_fp(mol: Chem.Mol) -> DataStructs.cDataStructs.ExplicitBitVect:
    return _MORGAN_GEN.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Dict[str, object]]:
    """
    Compute top-k most similar molecules from a PDF list using Morgan fingerprints
    (radius=2, include chirality) with Tanimoto similarity.

    Returns a list of dicts: {"name": str, "similarity": float, "smiles": str}
    sorted by descending similarity, then alphabetical name for ties.
    """
    if top_k <= 0:
        return []

    target_mol = _name_to_mol(target_molecule_name)
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule name: {target_molecule_name}")

    target_fp = _morgan_fp(target_mol)
    names = _extract_molecule_names_from_pdf(molecule_pool_filepath)

    results: List[Dict[str, object]] = []
    for name in names:
        mol = _name_to_mol(name)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        similarity = float(DataStructs.TanimotoSimilarity(target_fp, fp))
        smiles = _pubchem_name_to_smiles(name)
        if smiles is None:
            continue
        results.append({"name": name, "similarity": similarity, "smiles": smiles})
        # Be gentle to PubChem in case of cache misses.
        time.sleep(0.05)

    results.sort(key=lambda r: (-r["similarity"], str(r["name"]).casefold()))
    return results[:top_k]


if __name__ == "__main__":
    # Example usage
    top = topk_tanimoto_similarity_molecules(
        target_molecule_name="Aspirin",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5,
    )
    for item in top:
        print(f"{item['name']}: {item['similarity']:.3f} ({item['smiles']})")
