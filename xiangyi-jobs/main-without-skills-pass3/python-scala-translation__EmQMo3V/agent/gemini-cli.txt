trait BaseTokenizer[T] {
  /** Convert value to token. */
  def tokenize(value: T): Token
  
  /** Lazy tokenization of multiple values. */
  def tokenizeBatch(values: Iterable[T]): Iterator[Token] = values.iterator.map(tokenize)
}

class StringTokenizer(
    encoding: String = "UTF-8",
    normalizer: String => String = identity
) extends BaseTokenizer[Either[String, Array[Byte]]] {
  
  override def tokenize(value: Either[String, Array[Byte]]): Token = {
    val strValue = value match {
      case Left(s)  => s
      case Right(b) => new String(b, encoding)
    }
    Token(normalizer(strValue), TokenType.STRING)
  }

  def tokenize(value: String): Token = tokenize(Left(value))
  def tokenize(value: Array[Byte]): Token = tokenize(Right(value))
}

class NumericTokenizer(
    precision: Int = 6,
    formatOptions: Map[String, Any] = Map.empty
) extends BaseTokenizer[Any] {
  
  override def tokenize(value: Any): Token = {
    val strValue = value match {
      case d: BigDecimal => s"%.${precision}f".format(d)
      case d: Double     => s"%.${precision}f".format(d)
      case f: Float      => s"%.${precision}f".format(f)
      case i: Int        => i.toString
      case l: Long       => l.toString
      case _             => value.toString
    }
    Token(strValue, TokenType.NUMERIC, Map("original_type" -> value.getClass.getSimpleName))
  }
}

class TemporalTokenizer(formatStr: Option[String] = None) extends BaseTokenizer[Any] {
  private val isoFormatter = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss")
  private val dateFormatter = DateTimeFormatter.ISO_LOCAL_DATE

  override def tokenize(value: Any): Token = {
    val strValue = value match {
      case dt: LocalDateTime =>
        formatStr.map(DateTimeFormatter.ofPattern).getOrElse(isoFormatter).format(dt)
      case d: LocalDate =>
        formatStr.map(DateTimeFormatter.ofPattern).getOrElse(dateFormatter).format(d)
      case _ => value.toString
    }
    Token(strValue, TokenType.TEMPORAL)
  }
}

class UniversalTokenizer {
  private val stringTokenizer = new StringTokenizer()
  private val numericTokenizer = new NumericTokenizer()
  private val temporalTokenizer = new TemporalTokenizer()

  def tokenize(value: String): Token = stringTokenizer.tokenize(value)
  def tokenize(value: Array[Byte]): Token = stringTokenizer.tokenize(value)
  def tokenize(value: Int): Token = numericTokenizer.tokenize(value)
  def tokenize(value: Long): Token = numericTokenizer.tokenize(value)
  def tokenize(value: Double): Token = numericTokenizer.tokenize(value)
  def tokenize(value: BigDecimal): Token = numericTokenizer.tokenize(value)
  def tokenize(value: LocalDateTime): Token = temporalTokenizer.tokenize(value)
  def tokenize(value: LocalDate): Token = temporalTokenizer.tokenize(value)
  def tokenize(value: Tokenizable): Token = Token(value.toToken, TokenType.STRUCTURED)
  def tokenizeNull: Token = Token("NULL", TokenType.NULL)

  /** Dispatch to appropriate tokenizer based on runtime type. */
  def tokenizeAny(value: Any): Token = value match {
    case null               => tokenizeNull
    case v: Tokenizable     => tokenize(v)
    case s: String          => tokenize(s)
    case b: Array[Byte]     => tokenize(b)
    case i: Int             => tokenize(i)
    case l: Long            => tokenize(l)
    case d: Double          => tokenize(d)
    case bd: BigDecimal     => tokenize(bd)
    case dt: LocalDateTime  => tokenize(dt)
    case d: LocalDate       => tokenize(d)

  }

class TokenRegistry[T] {
  private var _registry: Map[String, TokenContainer[T]] = Map.empty
  private var _handlers: List[T => Option[Token]] = Nil

  def register(key: String, container: TokenContainer[T]): Unit = {
    _registry += (key -> container)
  }

  def addHandler(handler: T => Option[Token]): Unit = {
    _handlers = _handlers :+ handler
  }

  def process(key: String): List[Option[Token]] = {
    _registry.get(key) match {
      case Some(container) =>
        container.getAll.toList.map { item =>
          _handlers.view.flatMap(h => h(item)).headOption
        }
      case None => Nil
    }
  }
}

class JsonTokenizer(pretty: Boolean = false) {
  def tokenize(value: Json): Token = {
    val jsonStr = if (pretty) value.spaces2 else value.noSpaces
    Token(jsonStr, TokenType.STRUCTURED, Map("json" -> true))
  }

  def tokenizePath(value: Json, path: String): Option[Token] = {
    val parts = path.split('.')
    val found = parts.foldLeft(Option(value)) { (curr, part) =>
      curr.flatMap { j =>
        j.asObject.flatMap(_.apply(part)) orElse
          Try(part.toInt).toOption.flatMap(idx => j.asArray.flatMap(_.lift(idx)))
      }
    }
    found.map(tokenize)
  }
}

class WhitespaceTokenizer(
    lowercase: Boolean = false,
    minLength: Int = 0,
    maxLength: Option[Int] = None,
    stripPunctuation: Boolean = false
) {
  private val punctuation = ".,!?;:'\"()[]{}".toSet

  private def processToken(word: String): Option[String] = {
    var w = word
    if (stripPunctuation) {
      w = w.dropWhile(punctuation.contains).reverse.dropWhile(punctuation.contains).reverse
    }

    if (lowercase) {
      w = w.toLowerCase
    }

    if (w.length < minLength) return None

    maxLength.foreach { max =>
      if (w.length > max) w = w.substring(0, max)
    }

    if (w.isEmpty) None else Some(w)
  }

  def tokenize(text: String): List[Token] = {
    text.split("\\s+").filter(_.nonEmpty).toList.zipWithIndex.flatMap { case (word, i) =>
      processToken(word).map { processed =>
        Token(processed, TokenType.STRING, Map("position" -> i, "original" -> word))
      }
    }
  }

  def tokenizeToStrings(text: String): List[String] = tokenize(text).map(_.value)

  def tokenizeWithPositions(text: String): List[(String, Int, Int)] = {
    var currentPos = 0
    text.split("\\s+").filter(_.nonEmpty).toList.flatMap { word =>
      val start = text.indexOf(word, currentPos)
      val end = start + word.length
      currentPos = end
      processToken(word).map(processed => (processed, start, end))
    }
  }

  def countTokens(text: String): Int = tokenize(text).size
}

case class TokenizerBuilder[T](
    private val normalizers: List[String => String] = Nil,
    private val validators: List[T => Boolean] = Nil,
    private val metadata: Map[String, Any] = Map.empty
) {
  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] =
    copy(normalizers = normalizers :+ normalizer)

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] =
    copy(validators = validators :+ validator)

  def withMetadata(metadata: (String, Any)*): TokenizerBuilder[T] =
    copy(metadata = this.metadata ++ metadata.toMap)

  def build(): T => Token = { (value: T) =>
    validators.foreach { validator =>
      if (!validator(value)) throw new IllegalArgumentException(s"Validation failed for $value")
    }

    val normalizedValue = normalizers.foldLeft(value.toString)((acc, norm) => norm(acc))
    Token(normalizedValue, TokenType.STRING, metadata)
  }
}

object TokenizerBuilder {
  def apply[T](): TokenizerBuilder[T] = new TokenizerBuilder[T]()
}
