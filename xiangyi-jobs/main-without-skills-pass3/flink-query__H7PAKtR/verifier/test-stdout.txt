                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000) // Convert microseconds to milliseconds
                );

        DataStream<JobEvent> jobEvents = env.addSource(jobSourceOrTest(jobSource))
                .assignTimestampsAndWatermarks(

                );

        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(event -> event.getEventType() == TaskEvent.SUBMIT);

        SingleOutputStreamOperator<Tuple2<Long, Long>> result = submitEvents

    public static class SessionComputer extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Long>> {

        private final long sessionGapMicroseconds;

        // State to store all submit event timestamps
        private ListState<Long> submitTimestamps;

        // State to track if the job has finished
        private ValueState<Boolean> jobFinished;

        public SessionComputer(long sessionGapMicroseconds) {
            this.sessionGapMicroseconds = sessionGapMicroseconds;
        }

        @Override
        public void open(Configuration parameters) {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Types.LONG));
            jobFinished = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("jobFinished", Types.BOOLEAN));
        }

        @Override
        public void processElement1(TaskEvent taskEvent, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            // Skip if job already finished
            Boolean finished = jobFinished.value();
            if (finished != null && finished) {
                return;
            }

            // Store the submit event timestamp
            submitTimestamps.add(taskEvent.getTimestamp());
        }

        @Override
        public void processElement2(JobEvent jobEvent, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            // Job finished - compute the longest session
            Boolean finished = jobFinished.value();
            if (finished != null && finished) {
                return;
            }

            jobFinished.update(true);

            // Get all submit timestamps and sort them
            List<Long> timestamps = new ArrayList<>();
            for (Long ts : submitTimestamps.get()) {
                timestamps.add(ts);
            }

            long longestSessionCount = computeLongestSession(timestamps, sessionGapMicroseconds);

            // Output result
            long jobId = ctx.getCurrentKey();
            out.collect(new Tuple2<>(jobId, longestSessionCount));

            // Clear state
            submitTimestamps.clear();
        }

        private long computeLongestSession(List<Long> timestamps, long sessionGap) {
            if (timestamps.isEmpty()) {
                return 0L;
            }

            // Sort timestamps
            Collections.sort(timestamps);

            long longestSession = 0;
            long currentSession = 1;

            for (int i = 1; i < timestamps.size(); i++) {
                long gap = timestamps.get(i) - timestamps.get(i - 1);
                if (gap > sessionGap) {
                    // New session starts
                    if (currentSession > longestSession) {
                        longestSession = currentSession;
                    }
                    currentSession = 1;
                } else {
                    // Same session
                    currentSession++;
                }
            }

            // Check the last session
            if (currentSession > longestSession) {
                longestSession = currentSession;
            }

            return longestSession;
        }
    }

                    }
                }
            }
        }

        @Override

        }
