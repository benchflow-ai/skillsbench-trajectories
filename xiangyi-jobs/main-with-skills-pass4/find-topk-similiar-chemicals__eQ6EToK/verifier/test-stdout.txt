"""Compute top-k Tanimoto similarity for molecules listed in a PDF.

This module relies on PubChem (via pubchempy) to resolve chemical names
into SMILES, and RDKit to compute Morgan fingerprints (radius=2, chirality).
"""

from __future__ import annotations

from functools import lru_cache
from typing import List, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs


def _extract_molecule_names_from_pdf(pdf_path: str) -> List[str]:
    names: List[str] = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if name:
                    names.append(name)
    return names


@lru_cache(maxsize=None)
def _name_to_smiles(name: str) -> str | None:
    """Resolve a chemical name to an isomeric SMILES using PubChem.

    Returns None if no structure is found.
    """
    try:
        compounds = pcp.get_compounds(name, "name")
    except Exception:
        return None
    if not compounds:
        return None
    compound = compounds[0]
    return compound.isomeric_smiles or compound.canonical_smiles


def _smiles_to_mol(smiles: str) -> Chem.Mol | None:
    if not smiles:
        return None
    return Chem.MolFromSmiles(smiles)


def _morgan_fp(mol: Chem.Mol):
    return AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, nBits=2048, useChirality=True)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Tuple[str, float]]:
    """Return top-k most similar molecules to target from a PDF name list.

    Similarity: Morgan fingerprints (radius=2, chirality) with Tanimoto.
    Sorted by descending similarity, then alphabetical name for ties.
    """
    if top_k <= 0:
        return []

    target_smiles = _name_to_smiles(target_molecule_name)
    target_mol = _smiles_to_mol(target_smiles) if target_smiles else None
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")

    target_fp = _morgan_fp(target_mol)

    names = _extract_molecule_names_from_pdf(molecule_pool_filepath)

    scored: List[Tuple[str, float]] = []
    for name in names:
        smiles = _name_to_smiles(name)
        mol = _smiles_to_mol(smiles) if smiles else None
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        scored.append((name, float(sim)))

    scored.sort(key=lambda x: (-x[1], x[0].lower()))
    return scored[:top_k]


if __name__ == "__main__":
    # Example usage (not required by caller)
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_name> <molecules.pdf> <top_k>")
        raise SystemExit(2)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    for name, score in results:
        print(f"{name}\t{score:.4f}")
