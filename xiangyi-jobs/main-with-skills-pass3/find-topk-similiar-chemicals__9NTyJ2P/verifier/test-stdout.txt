import os
from functools import lru_cache
from typing import List, Optional, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


def _extract_molecule_names_from_pdf(pdf_path: str) -> List[str]:
    if not os.path.exists(pdf_path):
        raise FileNotFoundError(f"PDF not found: {pdf_path}")

    names: List[str] = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if name:
                    names.append(name)
    return names


@lru_cache(maxsize=None)
def _smiles_from_pubchem(name: str) -> Optional[str]:
    try:
        compounds = pcp.get_compounds(name, "name")
    except Exception:
        return None
    if not compounds:
        return None

    compound = compounds[0]
    smiles = compound.smiles or compound.canonical_smiles
    return smiles


_MORGAN_GEN = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _morgan_fp(mol: Chem.Mol):
    return _MORGAN_GEN.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Tuple[str, float]]:
    """
    Return the top-k most similar molecules in the pool PDF to the target name.

    Returns a list of (molecule_name, tanimoto_similarity) tuples sorted by
    descending similarity and alphabetical order for ties.
    """
    if top_k <= 0:
        return []

    pool_names = _extract_molecule_names_from_pdf(molecule_pool_filepath)

    target_smiles = _smiles_from_pubchem(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")

    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(
            f"Could not parse target SMILES from PubChem: {target_molecule_name}"
        )

    target_fp = _morgan_fp(target_mol)

    results: List[Tuple[str, float]] = []
    for name in pool_names:
        smiles = _smiles_from_pubchem(name)
        if not smiles:
            continue
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    results.sort(key=lambda item: (-item[1], item[0].lower()))
    return results[:top_k]


if __name__ == "__main__":
    # Example usage:
    # python /root/workspace/solution.py "Aspirin" /root/molecules.pdf 5
    import sys

    if len(sys.argv) == 4:
        target = sys.argv[1]
        pdf_path = sys.argv[2]
        k = int(sys.argv[3])
        top = topk_tanimoto_similarity_molecules(target, pdf_path, k)
        for name, sim in top:
            print(f"{name}\t{sim:.4f}")
    else:
        print("Usage: python solution.py <target_name> <molecules.pdf> <top_k>")
