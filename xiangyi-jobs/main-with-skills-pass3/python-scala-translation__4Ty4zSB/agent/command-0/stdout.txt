class WhitespaceTokenizer(
    lowercase: Boolean = false,
    minLength: Int = 0,
    maxLength: Option[Int] = None,
    stripPunctuation: Boolean = false
) {
  private val punctuation = ".,!?;:'\"()[]{}".toSet

  private def processToken(word: String): Option[String] = {
    var w = if (stripPunctuation) word.filterNot(punctuation.contains) else word
    if (lowercase) w = w.toLowerCase

    if (w.length < minLength) return None

    maxLength.foreach { max =>
      if (w.length > max) w = w.substring(0, max)
    }

    if (w.isEmpty) None else Some(w)
  }

  def tokenize(text: String): List[Token] = {
    text
      .split("\\s+")
      .zipWithIndex
      .flatMap { case (word, i) =>
        processToken(word).map { processed =>
          Token(
            processed,
            TokenType.STRING,
            Map("position" -> i, "original" -> word)
          )
        }
      }
      .toList
  }

  def tokenizeToStrings(text: String): List[String] = {
    tokenize(text).map(_.value)
  }

  def tokenizeWithPositions(text: String): List[(String, Int, Int)] = {
    var currentPos = 0
    text.split("\\s+").toList.flatMap { word =>
      val start = text.indexOf(word, currentPos)
      val end = start + word.length
      currentPos = end

      processToken(word).map(processed => (processed, start, end))
    }
  }

  def countTokens(text: String): Int = tokenize(text).length
}

        val entries = v.map { case (k, v) => s""""$k":${serialize(v)}""" }.mkString(",")
        s"{$entries}"
    }
  }

  /** Extract and tokenize a value at a JSON path. */
  def tokenizePath(value: JsonValue, path: String): Option[Token] = {
    val parts = path.split('.')
    var current: Option[JsonValue] = Some(value)

    for (part <- parts if current.isDefined) {
      current = current.get match {
        case JsonObject(v) => v.get(part)
        case JsonArray(v) if part.forall(_.isDigit) =>
          val idx = part.toInt
          if (idx >= 0 && idx < v.length) Some(v(idx)) else None
        case _ => None
      }
    }

    current.map(tokenize)
  }
}

/** Basic tokenizer that splits text by whitespace. */
class WhitespaceTokenizer(
    val lowercase: Boolean = false,
    val minLength: Int = 0,
    val maxLength: Option[Int] = None,
    val stripPunctuation: Boolean = false
) {
  private val punctuation = ".,!?;:'\"()[]{}".toSet

  private def processToken(word: String): Option[String] = {
    var w = if (stripPunctuation) word.filterNot(punctuation.contains) else word
    if (lowercase) w = w.toLowerCase

    if (w.length < minLength) return None

    maxLength.foreach { max =>
      if (w.length > max) w = w.substring(0, max)
    }

    if (w.isEmpty) None else Some(w)
  }

  /** Split text by whitespace and return list of tokens. */
  def tokenize(text: String): List[Token] = {
    text
      .split("\\s+")
      .zipWithIndex
      .flatMap { case (word, i) =>
        processToken(word).map { processed =>
          Token(
            processed,
            TokenType.STRING,
            Map("position" -> i, "original" -> word)
          )
        }
      }
      .toList
  }

  /** Split text by whitespace and return list of token strings. */
  def tokenizeToStrings(text: String): List[String] = {
    tokenize(text).map(_.value)
  }

  /** Tokenize text and return tokens with character positions. */
  def tokenizeWithPositions(text: String): List[(String, Int, Int)] = {
    var currentPos = 0
    text.split("\\s+").toList.flatMap { word =>
      val start = text.indexOf(word, currentPos)
      val end = start + word.length
      currentPos = end

      processToken(word).map(processed => (processed, start, end))
    }
  }

  /** Return the number of tokens in the text. */
  def countTokens(text: String): Int = tokenize(text).length
}

// ============================================================================
// Builder Pattern
// ============================================================================

/** Fluent builder for creating tokenizers. */
class TokenizerBuilder[T] {
  private val _normalizers = mutable.ListBuffer.empty[String => String]
  private val _validators = mutable.ListBuffer.empty[T => Boolean]
  private val _metadata = mutable.Map.empty[String, Any]

  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] = {
    _normalizers += normalizer
    this
  }

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] = {
    _validators += validator
    this
  }

  def withMetadata(kwargs: (String, Any)*): TokenizerBuilder[T] = {
    _metadata ++= kwargs.toMap
    this
  }

  /** Build the final tokenizer function. */
  def build(): T => Token = {
    val normalizers = _normalizers.toList
    val validators = _validators.toList
    val metadata = _metadata.toMap

    (value: T) => {
      validators.foreach { validator =>
        if (!validator(value)) throw new IllegalArgumentException(s"Validation failed for $value")
      }

      var strValue = value.toString
      normalizers.foreach { normalizer =>
        strValue = normalizer(strValue)
      }

      Token(strValue, TokenType.STRING, metadata)
    }
  }
}
