package clusterdata.query;

public class LongestSessionPerJob extends AppBase {

    private static final int EVENT_TYPE_SUBMIT = 0;
    private static final int EVENT_TYPE_FINISH = 4;
    private static final long SESSION_GAP_MICROS = 10L * 60 * 1000000; // 10 minutes in microseconds

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        final int sesssize = params.getInt("sesssize", 600);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events from gzipped CSV file
        DataStream<TaskEvent> taskEvents = env.addSource(
                taskSourceOrTest(new TaskEventSource(taskInput))
        );

        // Read job events from gzipped CSV file
        DataStream<JobEvent> jobEvents = env.addSource(
                jobSourceOrTest(new JobEventSource(jobInput))
        );

        // Get finished jobs with their finish timestamps
        DataStream<Tuple2<Long, Long>> finishedJobs = jobEvents
                .filter(event -> event.eventType == EVENT_TYPE_FINISH)
                .map(event -> new Tuple2<>(event.jobId, event.timestamp));

        // Filter task events for SUBMIT events only
        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(event -> event.eventType == EVENT_TYPE_SUBMIT);

        // Assign timestamps and watermarks
        DataStream<TaskEvent> submitEventsWithTimestamps = submitEvents
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy
                                .<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000) // Convert to milliseconds
                );

        // Group by jobId and apply session windows (10 minutes gap)
        DataStream<Tuple2<Long, Integer>> sessionStats = submitEventsWithTimestamps
                .keyBy(event -> event.jobId)
                .window(EventTimeSessionWindows.withGap(Time.milliseconds(SESSION_GAP_MICROS / 1000)))
                .process(new ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void process(Long jobId, Context context, Iterable<TaskEvent> elements,
                                        Collector<Tuple2<Long, Integer>> out) {
                        // Count all SUBMIT events in this session
                        int count = 0;
                        for (TaskEvent event : elements) {
                            count++;
                        }
                        // Emit (jobId, count) for this session
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        // Use CoFlatMap to join sessions with finished jobs and track the maximum
        DataStream<Tuple2<Long, Integer>> result = sessionStats
                .keyBy(tuple -> tuple.f0)
                .connect(finishedJobs.keyBy(tuple -> tuple.f0))
                .flatMap(new org.apache.flink.streaming.api.functions.co.RichCoFlatMapFunction<
                        Tuple2<Long, Integer>, Tuple2<Long, Long>, Tuple2<Long, Integer>>() {

                    private transient org.apache.flink.api.common.state.ValueState<Integer> maxSessionCount;
                    private transient org.apache.flink.api.common.state.ValueState<Boolean> finishedState;

                    @Override
                    public void open(org.apache.flink.configuration.Configuration parameters) {
                        maxSessionCount = getRuntimeContext().getState(
                                new org.apache.flink.api.common.state.ValueStateDescriptor<>(
                                        "maxSessionCount", Integer.class
                                )
                        );
                        finishedState = getRuntimeContext().getState(
                                new org.apache.flink.api.common.state.ValueStateDescriptor<>(
                                        "finishedState", Boolean.class
                                )
                        );
                    }

                    @Override
                    public void flatMap1(Tuple2<Long, Integer> session, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        // Update maximum session count
                        Integer currentMax = maxSessionCount.value();
                        if (currentMax == null || session.f1 > currentMax) {
                            maxSessionCount.update(session.f1);
                        }
                    }

                    @Override
                    public void flatMap2(Tuple2<Long, Long> finishedJob, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        finishedState.update(true);
                        Integer maxCount = maxSessionCount.value();
                        if (maxCount != null) {
                            out.collect(new Tuple2<>(finishedJob.f0, maxCount));
                        }
                    }
                });

        // Format output as (jobId,count)
        DataStream<String> formattedOutput = result.map(tuple -> "(" + tuple.f0 + "," + tuple.f1 + ")");

        // Write to file
        if (outputPath != null) {
            formattedOutput.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                    .setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

    static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String filePath;
        private volatile boolean running = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(
                            new GZIPInputStream(
                                    new java.io.FileInputStream(filePath))))) {
                String line;
                while (running && (line = reader.readLine()) != null) {
                    TaskEvent event = parseTaskEvent(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            } catch (IOException e) {
                throw new RuntimeException("Error reading task events", e);
            }
        }

        @Override
        public void cancel() {
            running = false;
        }

        private TaskEvent parseTaskEvent(String line) {
            try {
                String[] fields = line.split(",", -1);
                if (fields.length < 13) {
                    return null;
                }

                long timestamp = Long.parseLong(fields[0]);
                String missingInfo = fields[1];
                long jobId = Long.parseLong(fields[2]);
                long taskIndex = Long.parseLong(fields[3]);
                String machineId = fields[4];
                int eventType = Integer.parseInt(fields[5]);
                String user = fields[6];
                int schedulingClass = Integer.parseInt(fields[7]);
                int priority = Integer.parseInt(fields[8]);
                double cpuRequest = Double.parseDouble(fields[9]);
                double ramRequest = Double.parseDouble(fields[10]);
                double diskRequest = Double.parseDouble(fields[11]);
                int differentMachinesRestriction = Integer.parseInt(fields[12]);

                return new TaskEvent(timestamp, missingInfo, jobId, taskIndex, machineId,
                        eventType, user, schedulingClass, priority, cpuRequest, ramRequest,
                        diskRequest, differentMachinesRestriction);
            } catch (Exception e) {
                return null;
            }
        }
    }

    static class JobEventSource implements SourceFunction<JobEvent> {
        private final String filePath;
        private volatile boolean running = true;

        public JobEventSource(String filePath) {

        }

        @Override

                while (running && (line = reader.readLine()) != null) {

                    }

            } catch (IOException e) {

            }
        }

        @Override
        public void cancel() {
            running = false;
        }

        private JobEvent parseJobEvent(String line) {
            try {
                String[] fields = line.split(",", -1);
                if (fields.length < 8) {
                    return null;
                }

                long timestamp = Long.parseLong(fields[0]);
                String missingInfo = fields[1];
                long jobId = Long.parseLong(fields[2]);
                int eventType = Integer.parseInt(fields[3]);
                String user = fields[4];
                int schedulingClass = Integer.parseInt(fields[5]);
                String jobName = fields[6];
                String logicalJobName = fields[7];

                return new JobEvent(timestamp, missingInfo, jobId, eventType, user,
                        schedulingClass, jobName, logicalJobName);
            } catch (Exception e) {
                return null;
            }
        }
    }
