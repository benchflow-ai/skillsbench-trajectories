class StringTokenizer(
    val encoding: String = "utf-8",
    val normalizer: String => String = identity
) extends BaseTokenizer[StrOrBytes] {

  override def tokenize(value: StrOrBytes): Token = {
    val normalized = normalizer(value.asString(encoding))
    Token(normalized, TokenType.STRING)
  }

  def tokenize(value: String): Token = tokenize(StrOrBytes.Str(value))

  def tokenize(value: Array[Byte]): Token = tokenize(StrOrBytes.Bytes(value))
}

class NumericTokenizer(
    val precision: Int = 6,
    val formatOptions: Map[String, Any] = Map.empty
) extends BaseTokenizer[Any] {

  private def fixed(value: BigDecimal): String =
    String.format(Locale.US, s"%.${precision}f", value.bigDecimal)

  private def token(valueStr: String, originalType: String): Token =
    Token(valueStr, TokenType.NUMERIC, Map("original_type" -> originalType) ++ formatOptions)

  override def tokenize(value: Any): Token = value match {
    case bd: BigDecimal => token(fixed(bd), "BigDecimal")
    case d: Double      => token(fixed(BigDecimal(d)), "Double")
    case f: Float       => token(fixed(BigDecimal.decimal(f)), "Float")
    case i: Int         => token(i.toString, "Int")
    case l: Long        => token(l.toString, "Long")
    case s: Short       => token(s.toString, "Short")
    case b: Byte        => token(b.toString, "Byte")
    case other          => throw new IllegalArgumentException(s"Unsupported numeric type: ${other.getClass.getName}")
  }

  def tokenize(value: Int): Token = tokenize(value: Any)
  def tokenize(value: Double): Token = tokenize(value: Any)
  def tokenize(value: BigDecimal): Token = tokenize(value: Any)
}

class TemporalTokenizer(val formatStr: Option[String] = None) extends BaseTokenizer[Any] {
  private val isoFormatter = DateTimeFormatter.ofPattern("yyyy-MM-dd'T'HH:mm:ss")
  private val dateFormatter = DateTimeFormatter.ofPattern("yyyy-MM-dd")

  override def tokenize(value: Any): Token = {
    val formatter = formatStr.map(DateTimeFormatter.ofPattern).getOrElse {
      value match {
        case _: LocalDateTime => isoFormatter
        case _: LocalDate     => dateFormatter
        case _                => dateFormatter
      }
    }

    val formatted = value match {
      case dt: LocalDateTime => dt.format(formatter)
      case d: LocalDate      => d.format(formatter)
      case other             => throw new IllegalArgumentException(s"Unsupported temporal type: ${other.getClass.getName}")
    }

    Token(formatted, TokenType.TEMPORAL)
  }

  def tokenize(value: LocalDateTime): Token = tokenize(value: Any)
  def tokenize(value: LocalDate): Token = tokenize(value: Any)
}
