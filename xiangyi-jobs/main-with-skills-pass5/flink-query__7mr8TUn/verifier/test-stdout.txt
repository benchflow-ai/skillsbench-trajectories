package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;

import org.apache.flink.api.common.state.ListState;

import org.apache.flink.configuration.Configuration;

import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;

public class LongestSessionPerJob extends AppBase {

    // Session gap: 10 minutes in milliseconds
    private static final long SESSION_GAP_MS = 10L * 60L * 1000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events
        DataStream<TaskEvent> taskEvents = env
                .addSource(taskSourceOrTest(new GzipCsvTaskSource(taskInput)))
                .name("TaskEvents");

        // Read job events
        DataStream<JobEvent> jobEventsStream = env
                .addSource(jobSourceOrTest(new GzipCsvJobSource(jobInput)))
                .name("JobEvents");

        // Filter only SUBMIT events
        // Map to (jobId, timestamp_ms, isJobTerminal=false)
        DataStream<Tuple3<Long, Long, Boolean>> submitEventsMarked = taskEvents
                .filter(e -> e != null && e.getEventType() == TaskEvent.SUBMIT)
                .map(e -> Tuple3.of(e.getJobId(), e.getTimestamp() / 1000, false))
                .returns(Types.TUPLE(Types.LONG, Types.LONG, Types.BOOLEAN));

        // Filter only terminal job events (FAIL, FINISH, KILL)
        // Map to (jobId, timestamp_ms, isJobTerminal=true)
        DataStream<Tuple3<Long, Long, Boolean>> terminalEventsMarked = jobEventsStream
                .filter(e -> e != null && e.isTerminal())
                .map(e -> Tuple3.of(e.getJobId(), e.getTimestamp() / 1000, true))
                .returns(Types.TUPLE(Types.LONG, Types.LONG, Types.BOOLEAN));

        // Union all events
        DataStream<Tuple3<Long, Long, Boolean>> allEvents = submitEventsMarked.union(terminalEventsMarked);

        // Assign watermarks
        DataStream<Tuple3<Long, Long, Boolean>> withWatermarks = allEvents
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<Tuple3<Long, Long, Boolean>>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.f1)
                );

        // Process all events for each job - collect SUBMIT events and compute sessions when job terminates
        DataStream<String> result = withWatermarks
                .keyBy(t -> t.f0)
                .process(new SessionProcessor());

        result.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    /**
     * Collects task SUBMIT event timestamps and computes sessions when job terminates.
     * This approach ensures we have all events before computing the result.
     */
    public static class SessionProcessor extends KeyedProcessFunction<Long, Tuple3<Long, Long, Boolean>, String> {

        // List of task SUBMIT timestamps
        private ListState<Long> submitTimestamps;
        // Job termination timestamp
        private ValueState<Long> terminationTime;
        // Whether we've emitted result
        private ValueState<Boolean> emitted;

        @Override
        public void open(Configuration parameters) throws Exception {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Types.LONG));
            terminationTime = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("terminationTime", Types.LONG));
            emitted = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("emitted", Types.BOOLEAN));
        }

        @Override
        public void processElement(Tuple3<Long, Long, Boolean> event, Context ctx, Collector<String> out) throws Exception {
            long eventTime = event.f1;
            boolean isTerminal = event.f2;

            if (isTerminal) {
                // Job termination event
                Long currentTermTime = terminationTime.value();
                if (currentTermTime == null || eventTime < currentTermTime) {
                    terminationTime.update(eventTime);
                }
                // Register a timer to compute result after watermark passes all possible task events
                // We wait for session gap time to ensure all task events have been processed
                ctx.timerService().registerEventTimeTimer(eventTime + SESSION_GAP_MS + 1);
            } else {
                // Task SUBMIT event - store timestamp
                submitTimestamps.add(eventTime);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            Boolean alreadyEmitted = emitted.value();
            if (alreadyEmitted != null && alreadyEmitted) {
                return;
            }

            Long termTime = terminationTime.value();
            if (termTime == null) {
                return;
            }

            // Collect and sort all submit timestamps
            List<Long> timestamps = new ArrayList<>();
            for (Long ts : submitTimestamps.get()) {
                timestamps.add(ts);
            }

            if (timestamps.isEmpty()) {
                return;
            }

            Collections.sort(timestamps);

            // Compute sessions - a session is a group of submits within SESSION_GAP_MS of each other
            long maxSessionCount = 0;
            long currentSessionCount = 1;
            long lastTimestamp = timestamps.get(0);

            for (int i = 1; i < timestamps.size(); i++) {
                long currentTs = timestamps.get(i);
                if (currentTs - lastTimestamp < SESSION_GAP_MS) {
                    // Same session - gap is less than 10 minutes
                    currentSessionCount++;
                } else {
                    // New session - gap is >= 10 minutes (inactivity period of 10 minutes)
                    maxSessionCount = Math.max(maxSessionCount, currentSessionCount);
                    currentSessionCount = 1;
                }
                lastTimestamp = currentTs;
            }
            // Don't forget the last session
            maxSessionCount = Math.max(maxSessionCount, currentSessionCount);

            if (maxSessionCount > 0) {
                out.collect("(" + ctx.getCurrentKey() + "," + maxSessionCount + ")");
                emitted.update(true);
            }
        }
    }

    public static class GzipCsvTaskSource implements SourceFunction<TaskEvent> {
        private final String filePath;
        private volatile boolean isRunning = true;

        public GzipCsvTaskSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(
                            new GZIPInputStream(
                                    new java.io.FileInputStream(filePath))))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }
