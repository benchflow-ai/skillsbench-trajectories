"""Compute top-k Tanimoto similarity (Morgan fingerprints) against molecules in a PDF list."""

from __future__ import annotations

from dataclasses import dataclass
from functools import lru_cache
from typing import List

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs


@dataclass(frozen=True)
class SimilarityResult:
    name: str
    similarity: float


def _extract_names_from_pdf(pdf_path: str) -> List[str]:
    names: List[str] = []
    seen = set()
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if not name:
                    continue
                if name in seen:
                    continue
                seen.add(name)
                names.append(name)
    return names


@lru_cache(maxsize=2048)
def _name_to_smiles_pubchem(name: str) -> str | None:
    try:
        compounds = pcp.get_compounds(name, "name")
    except Exception:
        return None
    if not compounds:
        return None
    compound = compounds[0]
    smiles = compound.isomeric_smiles or compound.canonical_smiles
    return smiles


def _name_to_mol(name: str) -> Chem.Mol | None:
    smiles = _name_to_smiles_pubchem(name)
    if not smiles:
        return None
    return Chem.MolFromSmiles(smiles)


def _morgan_fp(mol: Chem.Mol):
    return AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=2, nBits=2048, useChirality=True
    )


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """
    Compute top-k most similar molecules from a PDF list to a target name.

    Returns a list of (name, similarity) tuples sorted by descending similarity,
    with alphabetical ordering on ties.
    """
    if top_k <= 0:
        return []

    names = _extract_names_from_pdf(molecule_pool_filepath)
    target_mol = _name_to_mol(target_molecule_name)
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")
    target_fp = _morgan_fp(target_mol)

    results: List[SimilarityResult] = []
    for name in names:
        mol = _name_to_mol(name)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        sim = float(DataStructs.TanimotoSimilarity(target_fp, fp))
        results.append(SimilarityResult(name=name, similarity=sim))

    results.sort(key=lambda r: (-r.similarity, r.name.casefold()))

    top_results = results[: min(top_k, len(results))]
    return [(r.name, r.similarity) for r in top_results]


if __name__ == "__main__":
    # Example usage: adjust target as needed.
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    top_k = 5
    print(topk_tanimoto_similarity_molecules(target, pdf_path, top_k))
