class StringTokenizer(encoding: String = "utf-8", normalizer: String => String = identity) 
  extends BaseTokenizer[Either[String, Array[Byte]]] {
  
  def tokenize(value: Either[String, Array[Byte]]): Token = {
    val strValue = value match {
      case Right(bytes) => new String(bytes, encoding)
      case Left(s) => s
    }
    val normalized = normalizer(strValue)
    Token(normalized, TokenType.STRING)
  }
  
  // Convenience overloads
  def tokenize(s: String): Token = tokenize(Left(s))
  def tokenize(b: Array[Byte]): Token = tokenize(Right(b))
}

class NumericTokenizer(precision: Int = 6, formatOptions: Map[String, Any] = Map.empty) 
  extends BaseTokenizer[Any] {
  
  def tokenize(value: Any): Token = {
    val (strValue, originalType) = value match {
      case d: BigDecimal => (f"$d%.${precision}f", "BigDecimal")
      case f: Double => (f"$f%.${precision}f", "Float")
      case f: Float => (f"$f%.${precision}f", "Float")
      case i: Int => (i.toString, "Int")
      case l: Long => (l.toString, "Long")
      case other => (other.toString, other.getClass.getSimpleName)
    }
    
    Token(strValue, TokenType.NUMERIC, Map("original_type" -> originalType))
  }
}
