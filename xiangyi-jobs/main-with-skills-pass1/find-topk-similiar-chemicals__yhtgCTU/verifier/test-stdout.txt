import os
import pypdf
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem
from rdkit import RDLogger
import warnings

# Suppress RDKit logs
RDLogger.DisableLog('rdApp.*')

def get_smiles(name):
    """
    Fetches the Isomeric SMILES for a given chemical name from PubChem.
    Returns None if not found or on error.
    """
    try:
        # Suppress PubChemPy deprecation warnings if any
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            c = pcp.get_compounds(name, 'name')
            if c:
                return c[0].isomeric_smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target chemical.
    
    Args:
        target_molecule_name (str): Name of the target chemical.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: List of chemical names sorted by similarity (desc) and name (asc).
    """
    
    # 1. Get Target SMILES and Fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Target molecule '{target_molecule_name}' not found in PubChem.")
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Target molecule '{target_molecule_name}' yielded invalid SMILES.")
        
    # Generate fingerprint for target: Morgan, radius 2, chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 2. Extract names from PDF
    names = []
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                for line in text.splitlines():
                    clean_line = line.strip()
                    if clean_line:
                        names.append(clean_line)
    except Exception as e:
        raise ValueError(f"Error reading PDF file: {e}")

    # Deduplicate names
    unique_names = sorted(list(set(names)))
    
    results = []
    
    # 3. Process each chemical
    for name in unique_names:
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))
        
    # 4. Sort: Descending similarity, then Alphabetical name
    # Python's sort is stable.
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k names
    top_names = [r[0] for r in results[:top_k]]
    return top_names
