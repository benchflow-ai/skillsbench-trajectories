from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict

def float_to_fraction(x):
    f = Fraction(x).limit_denominator(12)
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    s = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(s)
    ss = sga.get_symmetrized_structure()
    
    multiplicity_dict = {}
    coordinates_dict = {}
    
    for i, indices in enumerate(ss.equivalent_indices):
        symbol = ss.wyckoff_symbols[i]
        letter = symbol[-1]
        multiplicity = len(indices)
        
        multiplicity_dict[letter] = multiplicity
        
        first_atom = ss.sites[indices[0]]
        coords = first_atom.frac_coords
        formatted_coords = [float_to_fraction(c) for c in coords]
        coordinates_dict[letter] = formatted_coords
        
    return {
        "wyckoff_multiplicity_dict": multiplicity_dict,
        "wyckoff_coordinates_dict": coordinates_dict
    }

if __name__ == "__main__":
