import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
from pypdf import PdfReader
import os

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Helper to get SMILES
    def get_smiles(name):
        try:
            comps = pcp.get_compounds(name, 'name')
            if comps:
                return comps[0].isomeric_smiles
        except:
            return None
        return None

    # 1. Process Target
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return [] # Or raise error
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
        
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 2. Extract Pool Names
    pool_names = []
    try:
        reader = PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                for line in text.split('\n'):
                    clean_name = line.strip()
                    if clean_name:
                        pool_names.append(clean_name)
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []

    # Deduplicate while preserving ability to track
    unique_pool = sorted(list(set(pool_names)))
    
    results = []
    
    # 3. Calculate Similarities
    # Note: Fetching one by one might be slow, but ensures correct mapping.
    for name in unique_pool:
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, sim))

    # 4. Sort: Descending similarity, then Alphabetical name
    # Sort by name (ascending) first
    results.sort(key=lambda x: x[0])
    # Then sort by similarity (descending) - Python sort is stable
    results.sort(key=lambda x: x[1], reverse=True)
    
    # 5. Return Top K
    return [x[0] for x in results[:top_k]]
