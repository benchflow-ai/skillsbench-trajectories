"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
Uses PubChem API to get SMILES from chemical names and RDKit for similarity computation.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit.DataStructs import TanimotoSimilarity
import pubchempy as pcp
from typing import List, Tuple


def extract_chemicals_from_pdf(pdf_path: str) -> List[str]:
    """Extract chemical names from the molecules PDF file."""
    chemicals = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Each line is a chemical name
                lines = text.strip().split('\n')
                chemicals.extend([line.strip() for line in lines if line.strip()])
    return chemicals


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Query PubChem to get SMILES representation for a chemical name.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string or None if not found
    """
    try:
        results = pcp.get_compounds(chemical_name, 'name')
        if results:
            # Return SMILES of the first matching compound
            return results[0].smiles
    except Exception:
        pass
    return None


def get_morgan_fingerprint(smiles: str, radius: int = 2) -> object:
    """
    Generate a Morgan fingerprint for a SMILES string.

    Args:
        smiles: SMILES representation of a molecule
        radius: Radius for Morgan fingerprint (default: 2)

    Returns:
        Morgan fingerprint object or None if SMILES is invalid
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is not None:
            gen = AllChem.GetMorganGenerator(radius=radius, includeChirality=True)
            return gen.GetFingerprint(mol)
    except Exception:
        pass
    return None


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k chemicals most similar to a target molecule using Morgan fingerprints
    and Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the molecules PDF file
        top_k: Number of top similar molecules to return

    Returns:
        List of top k similar chemical names, sorted by descending similarity and
        alphabetically for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    target_fp = get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Extract all chemicals from PDF
    chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for chemical_name in chemicals:
        if chemical_name.lower() == target_molecule_name.lower():
            # Skip the target molecule itself
            continue

        smiles = get_smiles_from_pubchem(chemical_name)
        if smiles is None:
            continue

        fp = get_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = TanimotoSimilarity(target_fp, fp)
        similarities.append((chemical_name, similarity))

    # Sort by descending similarity, then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return [name for name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    try:
        result = topk_tanimoto_similarity_molecules("Aspirin", "/root/molecules.pdf", 5)
        print(f"Top 5 chemicals similar to Aspirin:")
        for i, chemical in enumerate(result, 1):
            print(f"{i}. {chemical}")
    except Exception as e:
        print(f"Error: {e}")
