"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to SMILES using PubChem
3. Compute Morgan fingerprints and Tanimoto similarity
4. Return top-k most similar molecules
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs
from functools import lru_cache


@lru_cache(maxsize=1024)
def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to SMILES using PubChem.

    Args:
        chemical_name: The name of the chemical compound

    Returns:
        The canonical SMILES string or None if not found
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of molecule names extracted from the PDF
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


# Create Morgan fingerprint generator with radius=2 and chirality
_morgan_generator = rdFingerprintGenerator.GetMorganGenerator(
    radius=2,
    fpSize=2048,
    includeChirality=True
)


def compute_morgan_fingerprint(smiles: str):
    """
    Compute Morgan fingerprint for a molecule given its SMILES.

    Args:
        smiles: SMILES string of the molecule

    Returns:
        Morgan fingerprint object or None if SMILES is invalid
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_generator.GetFingerprint(mol)


def tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0 to 1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find top-k most similar molecules to the target molecule from a PDF pool.

    This function:
    1. Extracts molecule names from the PDF file
    2. Converts the target molecule name to SMILES using PubChem
    3. Computes Morgan fingerprints (radius=2, with chirality)
    4. Calculates Tanimoto similarity between target and all pool molecules
    5. Returns top-k most similar molecules sorted by similarity (descending),
       with alphabetical ordering for ties

    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of molecule names sorted by Tanimoto similarity (descending),
        with alphabetical ordering for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities for all molecules in the pool
    similarities = []
    for mol_name in pool_molecules:
        smiles = get_smiles_from_name(mol_name)
        if smiles is None:
            continue

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = tanimoto_similarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    top_k_molecules = [mol_name for mol_name, _ in similarities[:top_k]]

    return top_k_molecules


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        target_molecule_name="Aspirin",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5
    )
    print("Top 5 similar molecules to Aspirin:")
    for i, mol in enumerate(result, 1):
        print(f"  {i}. {mol}")
