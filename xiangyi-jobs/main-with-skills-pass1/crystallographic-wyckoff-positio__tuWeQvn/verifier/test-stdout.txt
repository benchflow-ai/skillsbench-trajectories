"""
Crystal structure analysis module for analyzing Wyckoff positions and coordinates.

This module provides functionality to analyze CIF files generated by SHELX through
X-ray Diffraction, calculating Wyckoff position multiplicities and approximate
coordinates for the first atom of each Wyckoff position.
"""

from typing import Any
from fractions import Fraction
import spglib
from pymatgen.core import Structure


def _float_to_fraction(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point number to a rational number string.

    Args:
        value: The floating-point value to convert
        max_denominator: Maximum denominator to allow in fraction (default: 12)

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8")
        Returns "0" for values close to 0 and "1" for values close to 1.
    """
    # Handle edge cases
    if abs(value) < 1e-10:
        return "0"
    if abs(value - 1.0) < 1e-10:
        return "1"

    # Convert to fraction with limit on denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # If the fraction is essentially 0 or 1, return the simple form
    if frac.numerator == 0:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)

    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    This function reads a CIF file (typically generated by SHELX through X-ray
    Diffraction), determines the Wyckoff positions of atoms, calculates the
    multiplicity of each Wyckoff position, and extracts the coordinates of
    the first atom in each unique Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letters to their multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff letters to fractional coordinates
          of the first atom [x, y, z] in that position, as rational strings

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Get symmetry dataset using spglib
    # Convert structure to spglib format: (lattice, positions, atom_types)
    lattice = structure.lattice.matrix
    positions = structure.frac_coords

    # Create atom type numbers (using element index)
    atom_types = [structure.sites[i].species.elements[0].Z for i in range(len(structure))]

    # Use spglib to get symmetry information
    dataset = spglib.get_symmetry_dataset(
        (lattice, positions, atom_types),
        symprec=1e-5
    )

    # Extract Wyckoff symbols for each atom
    wyckoff_symbols = dataset.wyckoffs

    # Get standardized positions from the dataset
    std_positions = dataset.std_positions

    # Create mapping from Wyckoff letter to atoms
    wyckoff_dict = {}
    for idx, wyckoff_letter in enumerate(wyckoff_symbols):
        if wyckoff_letter not in wyckoff_dict:
            wyckoff_dict[wyckoff_letter] = {
                'first_atom_idx': idx,
                'count': 0
            }
        wyckoff_dict[wyckoff_letter]['count'] += 1

    # Build result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Sort Wyckoff letters for consistent ordering
    sorted_wyckoff_letters = sorted(wyckoff_dict.keys())

    for wyckoff_letter in sorted_wyckoff_letters:
        # Get multiplicity (count of atoms with this Wyckoff letter)
        multiplicity = wyckoff_dict[wyckoff_letter]['count']
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get coordinates of first atom with this Wyckoff letter
        first_atom_idx = wyckoff_dict[wyckoff_letter]['first_atom_idx']
        coords = std_positions[first_atom_idx]

        # Convert coordinates to fractional strings
        coord_strings = [_float_to_fraction(coord) for coord in coords]
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
