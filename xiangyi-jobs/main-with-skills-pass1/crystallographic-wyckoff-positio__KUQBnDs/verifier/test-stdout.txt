import json
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> (dict[str, dict] | dict[str, Any]):
    """
    Analyzes a CIF file to determine Wyckoff position multiplicities and 
    the approximate rational coordinates of the first atom for each Wyckoff position.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Mapping of Wyckoff letters to their total multiplicities.
        - "wyckoff_coordinates_dict": Mapping of Wyckoff letters to the rational coordinates 
                                     of the first atom found for that letter.
    """
    # Read the crystal structure from the CIF file
    try:
        struct = Structure.from_file(filepath)
    except Exception:
        # If loading fails, return empty dictionaries
        return {
            "wyckoff_multiplicity_dict": {},
            "wyckoff_coordinates_dict": {}
        }
    
    # Analyze symmetry using SpacegroupAnalyzer
    # symprec=0.01 is generally a good default for finding symmetry in CIF files
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    dataset = sga.get_symmetry_dataset()
    
    if not dataset:
        return {
            "wyckoff_multiplicity_dict": {},
            "wyckoff_coordinates_dict": {}
        }
    
    # wyckoffs is a list of Wyckoff letters for each site in the input structure
    wyckoff_letters = dataset.wyckoffs
    
    multiplicity_dict = {}
    coordinates_dict = {}
    
    def format_coord(c: float) -> str:
        """Helper to convert a coordinate to a rational string with denominator <= 12."""
        # Normalize to [0, 1) and handle periodic boundary conditions
        val = float(c) % 1
        
        # Round very small/large values to 0 to avoid precision artifacts
        if val < 1e-6 or val > 0.999999:
            val = 0.0
            
        # Find the closest rational number with denominator <= 12
        f = Fraction(val).limit_denominator(12)
        
        # Handle cases where limit_denominator returns 1/1
        if f.numerator == 0 or f.numerator == f.denominator:
            return "0"
        if f.denominator == 1:
            return str(f.numerator)
        return f"{f.numerator}/{f.denominator}"

    seen_letters = set()
    for i, letter in enumerate(wyckoff_letters):
        # Update multiplicity count
        multiplicity_dict[letter] = multiplicity_dict.get(letter, 0) + 1
        
        # Store coordinates for the first atom encountered for this Wyckoff letter
        if letter not in seen_letters:
            coords = struct[i].frac_coords
            coordinates_dict[letter] = [format_coord(x) for x in coords]
            seen_letters.add(letter)
            
    # Sort the resulting dictionaries by Wyckoff letter for consistent output
    sorted_letters = sorted(multiplicity_dict.keys())
    
    return {
        "wyckoff_multiplicity_dict": {l: multiplicity_dict[l] for l in sorted_letters},
        "wyckoff_coordinates_dict": {l: coordinates_dict[l] for l in sorted_letters}
    }
