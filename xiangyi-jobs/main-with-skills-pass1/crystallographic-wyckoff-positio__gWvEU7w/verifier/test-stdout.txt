from __future__ import annotations
from typing import Any
from fractions import Fraction
import re

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _frac_to_str(x: float, max_den: int = 12) -> str:
    """Convert a float fractional coordinate to a rational string with bounded denominator."""
    # normalize into [0, 1)
    x = float(x) % 1.0
    # handle numerical noise near boundaries
    if abs(x - 1.0) < 1e-8 or abs(x) < 1e-8:
        x = 0.0
    f = Fraction(x).limit_denominator(max_den)
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> (dict[str, dict] | dict[str, Any]):
    """
    Analyze Wyckoff position multiplicities and approximate coordinates for the first
    atom of each Wyckoff position.

    Args:
        filepath: Path to CIF file.

    Returns:
        dict with keys 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict',
        or dict with 'error' if something goes wrong.
    """
    try:
        struct = Structure.from_file(filepath)
        sga = SpacegroupAnalyzer(struct, symprec=0.1)
        sym_struct = sga.get_symmetrized_structure()

        wyckoff_multiplicity_dict: dict[str, int] = {}
        wyckoff_coordinates_dict: dict[str, list[str]] = {}

        for sym, sites in zip(sym_struct.wyckoff_symbols, sym_struct.equivalent_sites):
            m = re.match(r"(\d+)([A-Za-z]+)", sym)
            if m:
                mult = int(m.group(1))
                letter = m.group(2)
            else:
                mult = len(sites)
                letter = sym
            # Use first site of the Wyckoff position
            coords = sites[0].frac_coords
            coord_strs = [_frac_to_str(c) for c in coords]

            wyckoff_multiplicity_dict[letter] = mult
            wyckoff_coordinates_dict[letter] = coord_strs

        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
        }
    except Exception as e:
        return {"error": str(e)}
