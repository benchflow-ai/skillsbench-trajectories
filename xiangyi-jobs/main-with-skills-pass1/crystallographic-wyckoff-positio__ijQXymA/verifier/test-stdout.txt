"""Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files."""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string with denominator <= max_denominator.
    
    Args:
        value: The float value to convert (assumed to be in range [0, 1))
        max_denominator: Maximum denominator for the fraction
    
    Returns:
        String representation of the fraction (e.g., "1/2", "0", "3/8")
    """
    # Normalize to [0, 1) range
    value = value % 1.0
    
    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"
    
    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)
    
    # Handle case where fraction rounds to 1
    if frac.numerator == frac.denominator:
        return "0"
    
    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file
    
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinates of first atom
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Create SpacegroupAnalyzer with reasonable tolerance
    analyzer = SpacegroupAnalyzer(structure, symprec=0.1)
    
    # Get symmetry dataset which includes Wyckoff letters
    symmetry_dataset = analyzer.get_symmetry_dataset()
    
    # Use attribute interface instead of dict interface to avoid deprecation warning
    wyckoff_letters = symmetry_dataset.wyckoffs
    
    # Get the symmetrized structure to get proper coordinates
    symmetrized_structure = analyzer.get_symmetrized_structure()
    
    # Get equivalent atoms - list of lists of indices for equivalent sites
    equivalent_indices = symmetrized_structure.equivalent_indices
    
    # Build dictionaries for multiplicities and coordinates
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track which Wyckoff positions we've already processed
    processed_wyckoffs = set()
    
    # Iterate through equivalent site groups
    for equiv_group in equivalent_indices:
        # Get the first atom index in this group
        first_idx = equiv_group[0]
        
        # Get Wyckoff letter for this group
        wyckoff_letter = wyckoff_letters[first_idx]
        
        # Skip if we've already processed this Wyckoff position
        if wyckoff_letter in processed_wyckoffs:
            continue
        
        processed_wyckoffs.add(wyckoff_letter)
        
        # Multiplicity is the number of equivalent sites
        multiplicity = len(equiv_group)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        
        # Get fractional coordinates of the first atom in this group
        frac_coords = symmetrized_structure[first_idx].frac_coords
        
        # Convert to fraction strings
        coord_strings = [
            float_to_fraction_string(coord) for coord in frac_coords
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
