package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.functions.FilterFunction;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;

import org.apache.flink.streaming.api.functions.source.SourceFunction;

import java.io.BufferedReader;

import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    static class TaskEventSourceFunction implements SourceFunction<TaskEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public TaskEventSourceFunction(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (GZIPInputStream gzip = new GZIPInputStream(new java.io.FileInputStream(filePath));
                 BufferedReader br = new BufferedReader(new InputStreamReader(gzip))) {
                String line;
                while ((line = br.readLine()) != null && isRunning) {
                    String[] parts = line.split(",");
                    if (parts.length >= 13) {
                        try {
                            long timestamp = Long.parseLong(parts[0]);
                            String missingInfo = parts[1];
                            long jobId = Long.parseLong(parts[2]);
                            long taskIndex = Long.parseLong(parts[3]);
                            long machineId = parts[4].isEmpty() ? -1 : Long.parseLong(parts[4]);
                            int eventType = Integer.parseInt(parts[5]);
                            String user = parts[6];
                            int schedulingClass = Integer.parseInt(parts[7]);
                            int priority = Integer.parseInt(parts[8]);
                            double cpuRequest = Double.parseDouble(parts[9]);
                            double memoryRequest = Double.parseDouble(parts[10]);
                            double diskSpaceRequest = Double.parseDouble(parts[11]);
                            int differentMachinesRestriction = Integer.parseInt(parts[12]);

                            TaskEvent event = new TaskEvent(timestamp, missingInfo, jobId, taskIndex, machineId,
                                    eventType, user, schedulingClass, priority,
                                    cpuRequest, memoryRequest, diskSpaceRequest,
                                    differentMachinesRestriction);
                            ctx.collectWithTimestamp(event, timestamp);
                        } catch (Exception e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    static class JobEventSourceFunction implements SourceFunction<JobEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public JobEventSourceFunction(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<JobEvent> ctx) throws Exception {
            try (GZIPInputStream gzip = new GZIPInputStream(new java.io.FileInputStream(filePath));
                 BufferedReader br = new BufferedReader(new InputStreamReader(gzip))) {
                String line;
                while ((line = br.readLine()) != null && isRunning) {
                    String[] parts = line.split(",");
                    if (parts.length >= 8) {
                        try {
                            long timestamp = Long.parseLong(parts[0]);
                            String missingInfo = parts[1];
                            long jobId = Long.parseLong(parts[2]);
                            int eventType = Integer.parseInt(parts[3]);
                            int schedulingClass = Integer.parseInt(parts[4]);
                            String user = parts[5];
                            String jobName = parts[6];
                            String field8 = parts.length > 7 ? parts[7] : "";

                            JobEvent event = new JobEvent(timestamp, missingInfo, jobId, eventType,
                                    schedulingClass, user, jobName, field8);
                            ctx.collectWithTimestamp(event, timestamp);
                        } catch (Exception e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    static class ExtractSubmitEvents implements FilterFunction<TaskEvent> {
        @Override
        public boolean filter(TaskEvent event) {
            return event.eventType == 0; // SUBMIT events only
        }
    }

    static class StageProcessor extends ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow> {
        @Override
        public void process(Long jobId, Context context, Iterable<TaskEvent> elements,
                           Collector<Tuple2<Long, Integer>> out) {
            int taskCount = 0;
            for (TaskEvent element : elements) {
                taskCount++;
            }
            if (taskCount > 0) {
                out.collect(new Tuple2<>(jobId, taskCount));
            }
        }
    }

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        System.out.println("output  " + outputPath);
        final long sesssize = params.getLong("sesssize", 600); // in seconds, session gap

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1); // Single parallelism for correctness

        // Read task events
        SourceFunction<TaskEvent> taskEventSource = new TaskEventSourceFunction(taskInput);
        SourceFunction<TaskEvent> taskSource = taskSourceOrTest(taskEventSource);

        // Process task events to detect stages using session windows
        // A session window groups events with a gap of at most 'sesssize' seconds
        // When there's a gap larger than 'sesssize', a new window starts
        long sessionGapMs = sesssize * 1000; // Convert seconds to milliseconds

        env.addSource(taskSource)
                .assignTimestampsAndWatermarks(
                    WatermarkStrategy
                        .<TaskEvent>forBoundedOutOfOrderness(Duration.ofHours(1))
                        .withTimestampAssigner((event, context) -> event.timestamp / 1000) // Convert microseconds to milliseconds
                )
                .filter(new ExtractSubmitEvents()) // Keep only SUBMIT events
                .keyBy(e -> e.jobId) // Group by jobId
                .window(EventTimeSessionWindows.withGap(Time.milliseconds(sessionGapMs)))
                .process(new StageProcessor())
                .keyBy(e -> e.f0) // Group by jobId again
                .maxBy(1) // Get the stage with maximum task count per job
                .map(result -> "(" + result.f0 + "," + result.f1 + ")")
                .sinkTo(FileSink.forRowFormat(new Path(outputPath), new SimpleStringEncoder<String>()).build());

        env.execute("LongestSessionPerJob");
    }
