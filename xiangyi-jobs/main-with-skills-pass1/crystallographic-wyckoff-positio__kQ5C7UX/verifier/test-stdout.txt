from __future__ import annotations

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _format_fraction(value: float, max_denominator: int = 12) -> str:
    """Round a float to a simple rational with bounded denominator."""
    # Normalize into [0, 1) for fractional coordinates.
    value = value % 1.0
    frac = Fraction(value).limit_denominator(max_denominator)

    if frac.numerator == frac.denominator:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, Any]:
    """
    Analyze Wyckoff multiplicities and approximate coordinates for the first atom
    of each Wyckoff position in a CIF file.
    """
    structure = Structure.from_file(filepath)

    # Try a small range of tolerances to be robust to experimental CIF noise.
    symmetrized = None
    for symprec in (0.01, 0.1, 0.2):
        try:
            symmetrized = SpacegroupAnalyzer(structure, symprec=symprec).get_symmetrized_structure()
            break
        except Exception:
            symmetrized = None
    if symmetrized is None:
        raise ValueError(f"Unable to symmetrize structure for: {filepath}")

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    wyckoff_symbols = symmetrized.wyckoff_symbols
    eq_indices = symmetrized.equivalent_indices

    letter_counts: dict[str, int] = {}

    for i, indices in enumerate(eq_indices):
        if len(wyckoff_symbols) == len(eq_indices):
            symbol = wyckoff_symbols[i]
        else:
            symbol = wyckoff_symbols[indices[0]]

        letter = symbol.lstrip("0123456789")
        letter_counts[letter] = letter_counts.get(letter, 0) + 1
        key = letter if letter_counts[letter] == 1 else f"{letter}_{letter_counts[letter]}"

        multiplicity = len(indices)
        coords = symmetrized[indices[0]].frac_coords

        wyckoff_multiplicity_dict[key] = multiplicity
        wyckoff_coordinates_dict[key] = [_format_fraction(c) for c in coords]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
