import re
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinate lists
    """
    # Read the structure from CIF file
    struct = Structure.from_file(filepath)
    
    # Get space group analyzer
    sga = SpacegroupAnalyzer(struct)
    
    # Get the symmetrized structure
    sym_struct = sga.get_symmetrized_structure()
    
    # Get Wyckoff positions
    wyckoff_symbols = sym_struct.wyckoff_symbols
    
    # Dictionary to store multiplicities and coordinates for each Wyckoff position
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track which Wyckoff positions we've already processed
    processed_wyckoff = set()
    
    # Iterate through sites in the symmetrized structure
    for i, (site, wyckoff_symbol) in enumerate(zip(sym_struct.sites, wyckoff_symbols)):
        # Parse Wyckoff symbol to extract multiplicity and letter
        # Format is like '4a', '8c', etc.
        match = re.match(r'(\d+)([a-z])', wyckoff_symbol)
        if match:
            multiplicity_str = match.group(1)
            wyckoff_letter = match.group(2)
            multiplicity = int(multiplicity_str)
        else:
            # Fallback if format is different
            wyckoff_letter = wyckoff_symbol
            multiplicity = wyckoff_symbols.count(wyckoff_symbol)
        
        if wyckoff_letter not in processed_wyckoff:
            processed_wyckoff.add(wyckoff_letter)
            
            # Store multiplicity
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            
            # Get the fractional coordinates of the first atom with this Wyckoff letter
            frac_coords = site.frac_coords
            
            # Convert to rational fractions with denominators <= 12
            coord_fractions = []
            for coord in frac_coords:
                # Convert to fraction with max denominator of 12
                frac = Fraction(coord).limit_denominator(12)
                # Format as string
                if frac.denominator == 1:
                    coord_str = str(frac.numerator)
                else:
                    coord_str = f"{frac.numerator}/{frac.denominator}"
                coord_fractions.append(coord_str)
            
            wyckoff_coordinates_dict[wyckoff_letter] = coord_fractions
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
