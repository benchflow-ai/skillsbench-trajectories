"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.

This module provides functionality to analyze crystal structure CIF files,
extract Wyckoff position multiplicities, and calculate approximate fractional
coordinates for the first atom of each Wyckoff position.
"""

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a fractional coordinate to a rational fraction string.

    Args:
        value: Fractional coordinate value (typically in range [0, 1))
        max_denominator: Maximum denominator allowed for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-10:
        return "0"
    if abs(value - 1.0) < 1e-10:
        return "0"

    # Convert to fraction with denominator limit
    frac = Fraction(value).limit_denominator(max_denominator)

    if frac.denominator == 1:
        return str(frac.numerator % 1)  # Return "0" for integers
    else:
        return f"{frac.numerator}/{frac.denominator}"


def _coords_to_fraction_list(coords: list[float], max_denominator: int = 12) -> list[str]:
    """
    Convert a list of fractional coordinates to fraction strings.

    Args:
        coords: List of three fractional coordinates [x, y, z]
        max_denominator: Maximum denominator for fraction conversion

    Returns:
        List of three fraction strings
    """
    return [_float_to_fraction_string(c, max_denominator) for c in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    This function reads a CIF file, determines the crystal structure's space group
    symmetry, identifies unique Wyckoff positions, and calculates their multiplicities
    and representative coordinates (as rational fractions).

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        Dictionary containing:
            - "wyckoff_multiplicity_dict": Dict mapping Wyckoff letter to multiplicity
            - "wyckoff_coordinates_dict": Dict mapping Wyckoff letter to list of
              three fraction strings representing [x, y, z] coordinates

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/path/to/FeS2_mp-226.cif"
        ... )
        >>> result
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Perform symmetry analysis
    spacegroup_analyzer = SpacegroupAnalyzer(structure)

    # Get symmetrized structure with Wyckoff information
    symmetrized_structure = spacegroup_analyzer.get_symmetrized_structure()

    # Get Wyckoff symbols (e.g., ['4a', '8c'])
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Get equivalent sites (groups of symmetry-equivalent atoms)
    equivalent_sites = symmetrized_structure.equivalent_sites

    # Build result dictionaries
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for i, wyckoff_symbol in enumerate(wyckoff_symbols):
        # Parse Wyckoff symbol (e.g., "4a" -> multiplicity=4, letter="a")
        # The symbol format is "{multiplicity}{letter}"
        multiplicity_str = ""
        wyckoff_letter = ""
        for char in wyckoff_symbol:
            if char.isdigit():
                multiplicity_str += char
            else:
                wyckoff_letter = char
                break

        multiplicity = int(multiplicity_str)

        # Get the first site in this Wyckoff position for representative coordinates
        first_site = equivalent_sites[i][0]
        coords = first_site.frac_coords.tolist()

        # Convert coordinates to fraction strings
        coord_fractions = _coords_to_fraction_list(coords)

        # Store in dictionaries
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coord_fractions

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }

        "/root/cif_files/FeS2_mp-226.cif"
    )
    print(result)
