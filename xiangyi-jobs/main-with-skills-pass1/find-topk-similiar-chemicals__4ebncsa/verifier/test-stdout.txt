import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import functools
import time
import warnings

# Suppress RDKit deprecation warnings
from rdkit import RDLogger
RDLogger.DisableLog('rdApp.*')

@functools.lru_cache(maxsize=1024)
def get_smiles(name):
    retries = 3
    for i in range(retries):
        try:
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                return compounds[0].canonical_smiles
            return None
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                time.sleep(2 ** i)
                continue
            print(f"Error fetching SMILES for {name}: {e}")
            break
    return None

def get_fingerprint(smiles):
    if not smiles:
        return None
    mol = Chem.MolFromSmiles(smiles)
    if mol:
        # Morgan fingerprint with radius 2, including chirality
        return AllChem.GetMorganFingerprintAsBitVect(mol, 2, nBits=2048, useChirality=True)
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Extract names from PDF
    pool_names = []
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.split('\n')
                    for line in lines:
                        name = line.strip()
                        if name:
                            pool_names.append(name)
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []
    
    # Get target SMILES and fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []
    target_fp = get_fingerprint(target_smiles)
    if not target_fp:
        return []

    results = []
    unique_pool_names = sorted(list(set(pool_names)))
    for name in unique_pool_names:
        smiles = get_smiles(name)
        if smiles:
            fp = get_fingerprint(smiles)
            if fp:
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # Sort by similarity descending, then by name ascending
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # The task says "Find the top k similar chemicals", so we return the names
    return [item[0] for item in results[:top_k]]

if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1:
        target = sys.argv[1]
        k = int(sys.argv[2]) if len(sys.argv) > 2 else 5
        top_k_names = topk_tanimoto_similarity_molecules(target, 'molecules.pdf', k)
        for name in top_k_names:
            print(name)
