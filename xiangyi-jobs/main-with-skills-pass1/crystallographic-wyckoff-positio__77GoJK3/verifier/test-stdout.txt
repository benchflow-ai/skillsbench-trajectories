from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any


def fraction_to_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a simplified fraction string.

    Args:
        value: The float value to convert
        max_denominator: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the fraction
    """
    # Handle values very close to integers
    if abs(value - round(value)) < 1e-6:
        return str(int(round(value)))

    # Create fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # If the fraction is essentially an integer, return as integer
    if frac.denominator == 1:
        return str(frac.numerator)

    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinates of first atom
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get the symmetrized structure
    sga = SpacegroupAnalyzer(structure)
    symmetrized_structure = sga.get_symmetrized_structure()

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Get Wyckoff symbols (e.g., ['4a', '8c'])
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Get equivalent sites and their Wyckoff positions
    for i, equiv_sites in enumerate(symmetrized_structure.equivalent_sites):
        # Parse Wyckoff symbol to get letter (e.g., '4a' -> 'a')
        wyckoff_symbol = wyckoff_symbols[i]
        # Extract the letter part (last character)
        wyckoff_letter = wyckoff_symbol[-1]

        # Get multiplicity (number of equivalent sites)
        multiplicity = len(equiv_sites)

        # Get coordinates of the first atom in this Wyckoff position
        first_site = equiv_sites[0]
        coords = first_site.frac_coords

        # Convert coordinates to fraction strings
        coord_strings = [
            fraction_to_string(coord, max_denominator=12)
            for coord in coords
        ]

        # Store in dictionaries
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
