import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem
import time
import os

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF pool to a target chemical.    
    Args:
        target_molecule_name (str): The name of the chemical to compare.
        molecule_pool_filepath (str): Path to the PDF file containing the chemical pool.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: A list of the top k chemical names, sorted by similarity and then alphabetically.
    """
    # 1. Extract names from the PDF file
    names_in_pool = []
    if not os.path.exists(molecule_pool_filepath):
        return []
        
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.split('\n')
                    for line in lines:
                        name = line.strip()
                        if name:
                            names_in_pool.append(name)
    except Exception:
        # In case of PDF reading errors, return an empty list
        return []
    
    if not names_in_pool:
        return []

    # 2. Get SMILES and fingerprint for the target molecule
    try:
        target_compounds = pcp.get_compounds(target_molecule_name, 'name')
        if not target_compounds:
            return []
        
        # Prefer isomeric SMILES to include chirality if available
        target_smiles = target_compounds[0].isomeric_smiles
        if not target_smiles:
            target_smiles = target_compounds[0].canonical_smiles
    except Exception:
        return []
        
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Morgan fingerprint with radius 2 (diameter 4) and chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 3. Get SMILES and compute similarity for each chemical in the pool
    # We use a cache for unique names to minimize PubChem API requests
    smiles_cache = {}
    unique_names = sorted(list(set(names_in_pool)))
    
    for name in unique_names:
        try:
            # Respect PubChem rate limits (max 5 requests per second)
            time.sleep(0.25)
            comps = pcp.get_compounds(name, 'name')
            if comps:
                s = comps[0].isomeric_smiles
                if not s:
                    s = comps[0].canonical_smiles
                smiles_cache[name] = s
            else:
                smiles_cache[name] = None
        except Exception:
            smiles_cache[name] = None
            
    # Calculate Tanimoto similarity for all chemicals found in the pool
    results = []
    for name in names_in_pool:
        smiles = smiles_cache.get(name)
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 4. Sort results: 
    # Primary key: similarity (descending)
    # Secondary key: name (alphabetical ascending)
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Extract and return the top k chemical names
    return [item[0] for item in results[:top_k]]
