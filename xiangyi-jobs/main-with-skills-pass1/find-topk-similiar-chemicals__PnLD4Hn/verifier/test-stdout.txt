import pdfplumber
import pubchempy as pcp
from functools import lru_cache
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem


def _extract_molecule_names(pdf_path: str):
    names = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                line = line.strip()
                if line:
                    names.append(line)
    return names


@lru_cache(maxsize=None)
def _name_to_smiles(name: str):
    try:
        props = pcp.get_properties(["IsomericSMILES", "CanonicalSMILES"], name, "name")
    except Exception:
        return None
    if not props:
        return None
    smi = props[0].get("IsomericSMILES") or props[0].get("CanonicalSMILES")
    return smi


def _mol_from_name(name: str):
    smi = _name_to_smiles(name)
    if not smi:
        return None
    mol = Chem.MolFromSmiles(smi)
    return mol


def _morgan_fp(mol):
    return AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, nBits=2048, useChirality=True)


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Compute top-k most similar molecules from a PDF list to a target molecule name.

    Returns list of (molecule_name, similarity) sorted by similarity desc, then name.
    """
    names = _extract_molecule_names(molecule_pool_filepath)
    target_mol = _mol_from_name(target_molecule_name)
    if target_mol is None:
        return []
    target_fp = _morgan_fp(target_mol)

    results = []
    for name in names:
        if name.lower() == target_molecule_name.lower():
            continue
        mol = _mol_from_name(name)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, sim))

    results.sort(key=lambda x: (-x[1], x[0]))
    return results[:top_k]


if __name__ == "__main__":
    import argparse, json
    parser = argparse.ArgumentParser(description="Top-k Tanimoto similarity for molecules in a PDF list")
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument("pdf", help="PDF file with molecule names")
    parser.add_argument("k", type=int, help="Top k results")
    args = parser.parse_args()
    res = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.k)
    print(json.dumps(res, indent=2))
