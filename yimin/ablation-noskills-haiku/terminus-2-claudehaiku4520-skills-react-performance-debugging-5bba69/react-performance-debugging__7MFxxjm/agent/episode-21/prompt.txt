### 3. Bundle Size Optimization

#### Issue: Heavy Lodash Imports
- **File**: `src/app/compare/page.tsx`
- **Problem**: Imported entire lodash library for simple operations
  - `groupBy`, `sortBy`, `meanBy`, `sumBy`, `maxBy`, `minBy`
- **Impact**: ~70KB added to bundle

#### Fix Applied
Replaced with native JavaScript implementations:

```typescript
// BEFORE: import { sortBy, meanBy, sumBy } from 'lodash';

// AFTER: Native implementations
function sortBy<T>(arr: T[], key: keyof T): T[] {
  return [...arr].sort((a, b) => {
    const aVal = a[key];
    const bVal = b[key];
    if (typeof aVal === 'number' && typeof bVal === 'number') {
      return aVal - bVal;
    }
    return String(aVal).localeCompare(String(bVal));
  });
}

function meanBy<T>(arr: T[], key: keyof T): number {
  if (arr.length === 0) return 0;
  const sum = arr.reduce((acc, item) => {
    const val = item[key];
    return acc + (typeof val === 'number' ? val : 0);
  }, 0);
  return sum / arr.length;
}
```

**Expected Improvement**: ~70KB bundle size reduction

#### Issue: Heavy Mathjs Imports
- **File**: `src/app/compare/page.tsx`
- **Problem**: Imported entire mathjs library for basic statistics
  - `mean`, `std`, `median`, `quantileSeq`, `variance`
- **Impact**: ~40KB added to bundle

#### Fix Applied
Replaced with native JavaScript implementations:

```typescript
// BEFORE: import { mean, std, median, quantileSeq, variance } from 'mathjs';

// AFTER: Native implementations
function mean(arr: number[]): number {
  if (arr.length === 0) return 0;
  return arr.reduce((a, b) => a + b, 0) / arr.length;
}

function std(arr: number[]): number {
  if (arr.length === 0) return 0;
  const m = mean(arr);
  const variance = arr.reduce((acc, val) => acc + Math.pow(val - m, 2), 0) / arr.length;
  return Math.sqrt(variance);
}

function median(arr: number[]): number {
  if (arr.length === 0) return 0;
  const sorted = [...arr].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 !== 0 ? sorted[mid] : (sorted[mid - 1] + sorted[mid]) / 2;
}
```

**Expected Improvement**: ~40KB bundle size reduction
