"""Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files."""

from typing import Any
import re
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point coordinate to a fraction string.
    
    Args:
        value: The floating-point coordinate value
        max_denominator: Maximum denominator for the fraction (default: 12)
    
    Returns:
        String representation of the fraction (e.g., '1/2', '3/8', '0')
    """
    # Normalize to [0, 1) range
    value = value % 1.0
    if value < 0:
        value += 1.0
    
    # Handle values very close to 0 or 1
    if abs(value) < 1e-10 or abs(value - 1.0) < 1e-10:
        return "0"
    
    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)
    
    if frac.denominator == 1:
        return str(frac.numerator % 1)  # Handle case where it rounds to 1
    else:
        return f"{frac.numerator}/{frac.denominator}"


def coords_to_fraction_strings(coords: list, max_denominator: int = 12) -> list[str]:
    """
    Convert a list of coordinate values to fraction strings.
    
    Args:
        coords: List of floating-point coordinates
        max_denominator: Maximum denominator for fractions
    
    Returns:
        List of fraction strings
    """
    return [float_to_fraction_string(c, max_denominator) for c in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file
    
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letter to coordinate list
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Analyze symmetry
    sga = SpacegroupAnalyzer(structure)
    sym_struct = sga.get_symmetrized_structure()
    
    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Process each Wyckoff position
    for i, wyckoff_symbol in enumerate(sym_struct.wyckoff_symbols):
        # Parse Wyckoff symbol (e.g., '4a' -> multiplicity=4, letter='a')
        match = re.match(r'(\d+)([a-zA-Z]+)', wyckoff_symbol)
        if match:
            multiplicity = int(match.group(1))
            letter = match.group(2)
            
            # Store multiplicity
            wyckoff_multiplicity_dict[letter] = multiplicity
            
            # Get coordinates of the first atom in this Wyckoff position
            first_atom_idx = sym_struct.equivalent_indices[i][0]
            coords = sym_struct[first_atom_idx].frac_coords
            
            # Convert coordinates to fraction strings
            wyckoff_coordinates_dict[letter] = coords_to_fraction_strings(coords)
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
