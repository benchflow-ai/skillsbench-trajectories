from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string with denominator <= max_denominator.
    Handles values in [0, 1) range for crystallographic coordinates.
    """
    # Normalize to [0, 1) range
    value = value % 1.0
    if value < 0:
        value += 1.0
    
    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"
    
    # Convert to fraction
    frac = Fraction(value).limit_denominator(max_denominator)
    
    # Normalize fraction to be in [0, 1)
    while frac < 0:
        frac += 1
    while frac >= 1:
        frac -= 1
    
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dictionary mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Dictionary mapping Wyckoff letters to coordinates of first atom
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Create SpacegroupAnalyzer to get symmetry information
    analyzer = SpacegroupAnalyzer(structure)
    
    # Get symmetry dataset which contains Wyckoff positions
    symmetry_dataset = analyzer.get_symmetry_dataset()
    
    # Get Wyckoff letters for each atom
    wyckoff_letters = symmetry_dataset.wyckoffs
    
    # Get equivalent atoms array - atoms with same value are equivalent
    equivalent_atoms = symmetry_dataset.equivalent_atoms
    
    # Get standardized positions
    std_positions = symmetry_dataset.std_positions
    
    # Calculate multiplicities and find first atom for each Wyckoff position
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track which Wyckoff positions we've already processed
    processed_wyckoffs = set()
    
    # Group atoms by their equivalent_atoms index to find unique Wyckoff positions
    for i, (wyckoff, equiv_idx) in enumerate(zip(wyckoff_letters, equivalent_atoms)):
        if equiv_idx not in processed_wyckoffs:
            processed_wyckoffs.add(equiv_idx)
            
            # Count multiplicity - number of atoms with same equivalent_atoms value
            multiplicity = sum(1 for e in equivalent_atoms if e == equiv_idx)
            wyckoff_multiplicity_dict[wyckoff] = multiplicity
            
            # Get coordinates of the first atom in this Wyckoff position
            coords = std_positions[i]
            coord_strs = [float_to_fraction_str(c) for c in coords]
            wyckoff_coordinates_dict[wyckoff] = coord_strs
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
