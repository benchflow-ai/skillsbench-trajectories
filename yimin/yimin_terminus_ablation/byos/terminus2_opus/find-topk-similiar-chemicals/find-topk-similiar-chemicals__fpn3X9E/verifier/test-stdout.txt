import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs
from functools import lru_cache
import warnings

# Suppress deprecation warnings
warnings.filterwarnings('ignore')


@lru_cache(maxsize=1000)
def get_smiles_from_name(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem.
    Uses caching to avoid repeated API calls.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            # Use the smiles attribute (non-deprecated)
            return compounds[0].smiles
    except Exception:
        pass
    return None


def extract_chemicals_from_pdf(filepath: str) -> list:
    """
    Extract chemical names from a PDF file.
    """
    chemicals = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and filter empty strings
                names = [name.strip() for name in text.split('\n') if name.strip()]
                chemicals.extend(names)
    return chemicals


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule given its SMILES.
    Uses the newer MorganGenerator API.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    
    # Use the newer MorganGenerator API
    generator = rdFingerprintGenerator.GetMorganGenerator(
        radius=radius,
        fpSize=2048,
        includeChirality=use_chirality
    )
    return generator.GetFingerprint(mol)


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.
    """
    if fp1 is None or fp2 is None:
        return 0.0
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(target_molecule_name: str, 
                                        molecule_pool_filepath: str, 
                                        top_k: int) -> list:
    """
    Find the top k similar chemicals to the target molecule from a PDF file.
    
    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to the PDF file containing chemical names
        top_k: Number of top similar molecules to return
    
    Returns:
        List of chemical names sorted by Tanimoto similarity (descending),
        with alphabetical ordering for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        return []
    
    # Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)
    if target_fp is None:
        return []
    
    # Extract chemicals from PDF
    pool_chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)
    
    # Compute similarities for all chemicals in the pool
    similarities = []
    for chemical_name in pool_chemicals:
        # Skip if it's the same as target
        if chemical_name.lower() == target_molecule_name.lower():
            continue
            
        smiles = get_smiles_from_name(chemical_name)
        if smiles is None:
            continue
            
        fp = compute_morgan_fingerprint(smiles, radius=2, use_chirality=True)
        if fp is None:
            continue
            
        similarity = compute_tanimoto_similarity(target_fp, fp)
        similarities.append((chemical_name, similarity))
    
    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k chemical names
    return [name for name, _ in similarities[:top_k]]
