New Terminal Output:
index: TF-IDF index to search
        top_k: Number of top results per query
        num_workers: Number of worker processes
        documents: Optional list of documents for title lookup
    
    Returns:
        Tuple of (list of result lists, elapsed_time)
    """
    start_time = time.perf_counter()
    
    if num_workers is None:
        num_workers = mp.cpu_count()
    
    # Prepare document titles lookup
    doc_titles = {d.doc_id: d.title for d in documents} if documents else {}
    
    # Prepare query arguments
    query_args = [(query, top_k) for query in queries]
    
    # Process queries in parallel
    with ProcessPoolExecutor(
        max_workers=num_workers,
        initializer=_init_search_worker,
        initargs=(index, doc_titles)
    ) as executor:
        results = list(executor.map(_search_single_query, query_args))
    
    elapsed = time.perf_counter() - start_time
    
    return results, elapsed
