def zipf_choice(items: list[str], alpha: float = 1.2) -> str:
    """
    Select an item using Zipf-like distribution.
    Earlier items in the list are more likely to be selected.
    """
    n = len(items)
    weights = [1.0 / ((i + 1) ** alpha) for i in range(n)]
