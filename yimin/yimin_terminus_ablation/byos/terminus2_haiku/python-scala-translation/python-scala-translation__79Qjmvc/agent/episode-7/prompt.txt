New Terminal Output:
private val metadataMap: mutable.Map[String, Any] = mutable.Map.empty

  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] = {
    normalizers += normalizer
    this
  }

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] = {
    validators += validator
    this
  }

  def withMetadata(metadata: (String, Any)*): TokenizerBuilder[T] = {
    metadataMap ++= metadata
    this
  }

  def build(): T => Token = { value: T =>
    validators.foreach { validator =>
      if (!validator(value)) {
        throw new IllegalArgumentException(s"Validation failed for $value")
      }
    }

    var strValue = value.toString
    normalizers.foreach { normalizer =>
      strValue = normalizer(strValue)
    }

    Token(strValue, TokenType.String, metadataMap.toMap)
  }
}

object Tokenizer {
  def tokenize(value: Any): Token =
    new UniversalTokenizer().tokenize(value)

  def tokenizeBatch[T](values: Iterable[T], tokenizer: BaseTokenizer[T]): Iterator[Token] =
    tokenizer.tokenizeBatch(values)

  def toToken(value: String): Token =
    Token(value, TokenType.String)

  def withMetadata(token: Token, metadata: (String, Any)*): Token =
    token.withMetadata(metadata: _*)
}
