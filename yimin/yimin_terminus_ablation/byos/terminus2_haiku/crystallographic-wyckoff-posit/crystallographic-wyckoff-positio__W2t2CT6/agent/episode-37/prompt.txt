from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> (dict[str, dict] | dict[str, Any]):
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the first atom
    of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Maps Wyckoff letter to coordinates of first atom
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Analyze symmetry using SpacegroupAnalyzer
    sga = SpacegroupAnalyzer(structure)
    
    # Get the symmetry dataset which contains Wyckoff information
    dataset = sga.get_symmetry_dataset()
    
    # Extract Wyckoff symbols and standardized positions
    wyckoffs = list(dataset.wyckoffs)  # List of Wyckoff letters for each atom
    std_positions = dataset.std_positions  # Standardized fractional coordinates
    
    # Create dictionaries to store results
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track which Wyckoff positions we've already processed
    processed_wyckoffs = set()
    
    # Process each atom
    for i, wyckoff_letter in enumerate(wyckoffs):
        # Only process each Wyckoff position once (use the first atom)
        if wyckoff_letter not in processed_wyckoffs:
            processed_wyckoffs.add(wyckoff_letter)
            
            # Count multiplicity (number of atoms with this Wyckoff letter)
            multiplicity = sum(1 for w in wyckoffs if w == wyckoff_letter)
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            
            # Get coordinates of the first atom with this Wyckoff letter
            coords = std_positions[i]
            
            # Convert coordinates to fractions with denominators <= 12
            frac_coords = []
            for coord in coords:
                # Convert to fraction with max denominator of 12
                frac = Fraction(coord).limit_denominator(12)
                frac_coords.append(str(frac))
            
            wyckoff_coordinates_dict[wyckoff_letter] = frac_coords
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
