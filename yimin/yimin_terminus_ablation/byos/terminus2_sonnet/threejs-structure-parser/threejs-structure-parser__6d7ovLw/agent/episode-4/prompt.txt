import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import * as fs from 'fs';
import * as path from 'path';

// Create output directories
const partMeshesDir = '/root/output/part_meshes';
const linksDir = '/root/output/links';

if (!fs.existsSync(partMeshesDir)) {
    fs.mkdirSync(partMeshesDir, { recursive: true });
}
if (!fs.existsSync(linksDir)) {
    fs.mkdirSync(linksDir, { recursive: true });
}

// Initialize the OBJ exporter
const exporter = new OBJExporter();

// Create the scene
const scene = createScene();

console.log('Scene created successfully!');
console.log('Root object:', scene.name);

// Function to export a mesh or group to OBJ
function exportToOBJ(object, filepath) {
    const objData = exporter.parse(object);
    fs.writeFileSync(filepath, objData);
}

// Function to process the scene hierarchy
function processScene(rootObject) {
    const parts = [];
    
    // Traverse to find all groups (parts)
    rootObject.traverse((object) => {
        if (object instanceof THREE.Group && object !== rootObject) {
            parts.push(object);
        }
    });
    
    console.log(`\nFound ${parts.length} parts (THREE.Group objects)`);
    
    // Process each part
    parts.forEach((part) => {
        const partName = part.name || 'unnamed_part';
        console.log(`\nProcessing part: ${partName}`);
        
        // Create directory for this part's meshes
        const partDir = path.join(partMeshesDir, partName);
        if (!fs.existsSync(partDir)) {
            fs.mkdirSync(partDir, { recursive: true });
        }
        
        // Find all direct child meshes in this part
        const meshes = [];
        part.traverse((child) => {
            if (child instanceof THREE.Mesh) {
                meshes.push(child);
            }
        });
        
        console.log(`  Found ${meshes.length} meshes in ${partName}`);
        
        // Export individual meshes
        meshes.forEach((mesh) => {
            const meshName = mesh.name || 'unnamed_mesh';
            const meshPath = path.join(partDir, `${meshName}.obj`);
            
            try {
                // Create a temporary group to properly export the mesh with its transformations
                const tempGroup = new THREE.Group();
                const meshClone = mesh.clone();
                tempGroup.add(meshClone);
                
                exportToOBJ(tempGroup, meshPath);
                console.log(`    Exported: ${meshName}.obj`);
            } catch (error) {
                console.error(`    Error exporting ${meshName}:`, error.message);
            }
        });
        
        // Export the entire part as a combined mesh to links/
        const linkPath = path.join(linksDir, `${partName}.obj`);
        try {
            exportToOBJ(part, linkPath);
            console.log(`  Exported combined part to: links/${partName}.obj`);
        } catch (error) {
            console.error(`  Error exporting combined part ${partName}:`, error.message);
        }
    });
    
    console.log('\n=== Export Summary ===');
    console.log(`Total parts processed: ${parts.length}`);
    console.log(`Output directories:`);
    console.log(`  - Individual meshes: ${partMeshesDir}`);
    console.log(`  - Combined parts: ${linksDir}`);
}

// Process the scene
processScene(scene);

console.log('\nExtraction complete!');
