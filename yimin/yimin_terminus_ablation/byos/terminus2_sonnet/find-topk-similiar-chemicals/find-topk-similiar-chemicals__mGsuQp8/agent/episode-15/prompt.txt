import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k most similar molecules to the target molecule from a pool.
    
    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return
    
    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity (descending),
        then alphabetically for ties
    """
    # Extract molecule names from PDF
    molecule_names = []
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecule_names.append(line)
    
    # Convert target molecule name to SMILES
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    # Generate Morgan fingerprint for target molecule
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Could not create molecule from SMILES: {target_smiles}")
    
    # Use MorganGenerator (new API) - radius=2, chirality=True
    morgan_gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, fpSize=2048, includeChirality=True)
    target_fp = morgan_gen.GetFingerprint(target_mol)
    
    # Calculate similarities for all molecules in the pool
    similarities = []
    for mol_name in molecule_names:
        # Skip if it's the target molecule itself
        if mol_name == target_molecule_name:
            continue
        
        smiles = get_smiles_from_name(mol_name)
        if not smiles:
            continue
        
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
        
        # Generate Morgan fingerprint with radius=2 and chirality
        fp = morgan_gen.GetFingerprint(mol)
        
        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((mol_name, similarity))
    
    # Sort by similarity (descending), then alphabetically by name for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k results
    return similarities[:top_k]


def get_smiles_from_name(chemical_name):
    """
    Get SMILES representation from chemical name using PubChem.
    
    Args:
        chemical_name: Name of the chemical
    
    Returns:
        SMILES string or None if not found
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            # Use the 'smiles' property which includes stereochemistry
            return compounds[0].smiles
    except:
        pass
    return None
