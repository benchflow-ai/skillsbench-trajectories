from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff positions and their multiplicities from a CIF file.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Maps Wyckoff letters to coordinates of first atom
    """
    
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Get space group analyzer
    sga = SpacegroupAnalyzer(structure)
    
    # Get symmetry dataset which contains Wyckoff information
    symmetry_dataset = sga.get_symmetry_dataset()
    
    # Extract Wyckoff positions and standardized positions
    wyckoffs = symmetry_dataset.wyckoffs
    std_positions = symmetry_dataset.std_positions
    
    # Create dictionaries to store results
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    wyckoff_first_occurrence = {}  # Track first occurrence of each Wyckoff letter
    
    # Process each atom and its Wyckoff position
    for i, wyckoff_letter in enumerate(wyckoffs):
        # Count multiplicity for each Wyckoff letter
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff_letter] = wyckoffs.count(wyckoff_letter)
            wyckoff_first_occurrence[wyckoff_letter] = i
    
    # Get coordinates for the first atom of each Wyckoff position
    for wyckoff_letter, atom_index in wyckoff_first_occurrence.items():
        # Get fractional coordinates from standardized positions
        coords = std_positions[atom_index]
        
        # Convert to rational fractions with denominator <= 12
        rational_coords = []
        for coord in coords:
            # Use Fraction to convert to rational number
            frac = Fraction(coord).limit_denominator(12)
            rational_coords.append(str(frac))
        
        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
