import warnings
warnings.filterwarnings('ignore')

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from typing import List

def _extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract all molecule names from the PDF file.
    """
    molecules = []
    try:
        with pdfplumber.open(pdf_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Split by newlines and clean up
                    lines = text.strip().split('\n')
                    for line in lines:
                        line = line.strip()
                        if line and len(line) > 0:
                            molecules.append(line)
    except Exception as e:
        print(f"Error extracting molecules from PDF: {e}")
    return molecules

def _get_smiles_from_pubchem(molecule_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem.
    Returns None if molecule not found.
    """
    try:
        compounds = pcp.get_compounds(molecule_name, 'name')
        if compounds:
            return compounds[0].canonical_smiles
    except Exception as e:
        print(f"Error fetching SMILES for {molecule_name}: {e}")
    return None

def _get_morgan_fingerprint(smiles: str, radius: int = 2):
    """
    Generate Morgan fingerprint from SMILES string.
    Returns None if SMILES is invalid.
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is not None:
            return AllChem.GetMorganFingerprintAsBitVect(mol, radius, nBits=2048, useChirality=True)
    except Exception as e:
        print(f"Error generating fingerprint for SMILES {smiles}: {e}")
    return None

def _calculate_tanimoto_similarity(fp1, fp2) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.
    """
    if fp1 is None or fp2 is None:
        return 0.0
    return DataStructs.TanimotoSimilarity(fp1, fp2)

def topk_tanimoto_similarity_molecules(target_molecule_name: str, molecule_pool_filepath: str, top_k: int) -> List[str]:
    """
    Find the top k similar chemicals to a target molecule from a pool of molecules in a PDF.
    
    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing the molecule pool
        top_k: Number of top similar molecules to return
    
    Returns:
        List of molecule names sorted by similarity (descending) and alphabetically for ties
    """
    # Extract all molecule names from PDF
    pool_molecules = _extract_molecules_from_pdf(molecule_pool_filepath)
    
    # Get SMILES for target molecule
    target_smiles = _get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []
    
    # Get Morgan fingerprint for target molecule
    target_fp = _get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        print(f"Could not generate fingerprint for target molecule: {target_molecule_name}")
        return []
    
    # Calculate similarities for all molecules in pool
    similarities = []
    
    for molecule_name in pool_molecules:
        # Skip the target molecule itself
        if molecule_name.lower() == target_molecule_name.lower():
            continue
        
        # Get SMILES for pool molecule
        smiles = _get_smiles_from_pubchem(molecule_name)
        if smiles is None:
            continue
        
        # Get fingerprint
        fp = _get_morgan_fingerprint(smiles)
        if fp is None:
            continue
        
        # Calculate similarity
        similarity = _calculate_tanimoto_similarity(target_fp, fp)
        similarities.append((molecule_name, similarity))
    
    # Sort by similarity (descending) then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k molecule names
    result = [mol_name for mol_name, _ in similarities[:top_k]]
    return result

if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules("Aspirin", "/root/molecules.pdf", 5)
    print("Top 5 similar molecules to Aspirin:")
    for i, mol in enumerate(result, 1):
        print(f"{i}. {mol}")
