import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from typing import List, Tuple


def topk_tanimoto_similarity_molecules(target_molecule_name: str, molecule_pool_filepath: str, top_k: int) -> list:
    """
    Find the top k similar chemicals from a PDF file to a target molecule.
    
    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return
    
    Returns:
        List of molecule names sorted by similarity (descending), with alphabetical ordering for ties
    """
    
    # Extract molecule names from PDF
    molecule_names = extract_molecules_from_pdf(molecule_pool_filepath)
    
    # Convert target molecule name to SMILES
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    # Convert target SMILES to RDKit mol object
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_smiles}")
    
    # Generate Morgan fingerprint for target (radius=2, useChirality=True)
    target_fp = AllChem.GetMorganFingerprint(target_mol, radius=2, useChirality=True)
    
    # Calculate similarities for all molecules in pool
    similarities = []
    
    for mol_name in molecule_names:
        # Skip if it's the same as target
        if mol_name.strip().lower() == target_molecule_name.strip().lower():
            continue
            
        # Get SMILES for molecule
        mol_smiles = get_smiles_from_name(mol_name)
        if mol_smiles is None:
            continue
        
        # Convert to RDKit mol
        mol = Chem.MolFromSmiles(mol_smiles)
        if mol is None:
            continue
        
        # Generate Morgan fingerprint
        mol_fp = AllChem.GetMorganFingerprint(mol, radius=2, useChirality=True)
        
        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, mol_fp)
        
        similarities.append((mol_name, similarity))
    
    # Sort by similarity (descending), then alphabetically by name for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k molecule names
    return [name for name, sim in similarities[:top_k]]


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract molecule names from a PDF file.
    Assumes molecules are listed one per line or separated by commas/newlines.
    """
    molecule_names = []
    
    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and commas
                lines = text.split('\n')
                for line in lines:
                    # Split by comma if present
                    parts = line.split(',')
                    for part in parts:
                        part = part.strip()
                        if part and len(part) > 0:
                            molecule_names.append(part)
    
    # Remove duplicates while preserving order
    seen = set()
    unique_molecules = []
    for name in molecule_names:
        if name not in seen:
            seen.add(name)
            unique_molecules.append(name)
    
    return unique_molecules


def get_smiles_from_name(molecule_name: str) -> str:
    """
    Convert a molecule name to SMILES using PubChem.
    Returns None if molecule not found.
    """
    try:
        compounds = pcp.get_compounds(molecule_name, 'name')
        if compounds and len(compounds) > 0:
            return compounds[0].canonical_smiles
    except Exception as e:
        print(f"Error fetching SMILES for {molecule_name}: {e}")
    
    return None
