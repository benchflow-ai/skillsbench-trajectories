from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def decimal_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal value to a fraction string with limited denominator.
    """
    # Handle values very close to 0 or 1
    if abs(value) < 1e-6:
        return "0"
    if abs(value - 1.0) < 1e-6:
        return "1"
    
    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)
    
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing wyckoff_multiplicity_dict and wyckoff_coordinates_dict
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    sga = SpacegroupAnalyzer(structure)
    
    # Get symmetrized structure
    symmetrized_structure = sga.get_symmetrized_structure()
    
    # Get symmetry dataset which contains Wyckoff information
    symmetry_dataset = sga.get_symmetry_dataset()
    
    # Extract Wyckoff letters and positions
    wyckoff_letters = symmetry_dataset['wyckoffs']
    equivalent_atoms = symmetry_dataset['equivalent_atoms']
    
    # Group atoms by their Wyckoff position
    wyckoff_groups = {}
    for i, (wyckoff_letter, equiv_idx) in enumerate(zip(wyckoff_letters, equivalent_atoms)):
        if wyckoff_letter not in wyckoff_groups:
            wyckoff_groups[wyckoff_letter] = []
        wyckoff_groups[wyckoff_letter].append(i)
    
    # Calculate multiplicities and get first atom coordinates for each Wyckoff position
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    for wyckoff_letter in sorted(wyckoff_groups.keys()):
        atom_indices = wyckoff_groups[wyckoff_letter]
        
        # Multiplicity is the number of atoms in this Wyckoff position
        wyckoff_multiplicity_dict[wyckoff_letter] = len(atom_indices)
        
        # Get coordinates of the first atom in this Wyckoff position
        first_atom_idx = atom_indices[0]
        coords = structure[first_atom_idx].frac_coords
        
        # Convert to fraction strings
        coord_strings = [
            decimal_to_fraction_string(coord, max_denominator=12)
            for coord in coords
        ]
        
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
