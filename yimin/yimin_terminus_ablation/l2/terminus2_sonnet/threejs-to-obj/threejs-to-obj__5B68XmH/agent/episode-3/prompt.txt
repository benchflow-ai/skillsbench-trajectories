import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

// Create the scene from the imported function
const sceneRoot = createScene();

// Create a wrapper group to apply -90 degree X-rotation for Blender Z-up
const blenderWrapper = new THREE.Group();
blenderWrapper.rotation.x = -Math.PI / 2; // -90 degrees
blenderWrapper.add(sceneRoot);

// Update all world matrices
blenderWrapper.updateMatrixWorld(true);

// Create a new group to hold all expanded geometry
const exportGroup = new THREE.Group();

// Traverse and collect all meshes, expanding InstancedMesh
blenderWrapper.traverse((obj) => {
  if (obj instanceof THREE.InstancedMesh) {
    // Expand InstancedMesh into individual meshes
    const instanceCount = obj.count;
    const geometry = obj.geometry;
    const material = obj.material;
    
    const tempMatrix = new THREE.Matrix4();
    const tempMesh = new THREE.Object3D();
    
    for (let i = 0; i < instanceCount; i++) {
      // Get instance matrix
      obj.getMatrixAt(i, tempMatrix);
      
      // Create a clone of the geometry
      const clonedGeometry = geometry.clone();
      
      // Combine instance matrix with object's world matrix
      const finalMatrix = new THREE.Matrix4();
      finalMatrix.copy(obj.matrixWorld).multiply(tempMatrix);
      
      // Apply the final transform to geometry
      clonedGeometry.applyMatrix4(finalMatrix);
      
      // Create mesh for export
      const mesh = new THREE.Mesh(clonedGeometry, material);
      mesh.name = `${obj.name}_instance_${i}`;
      exportGroup.add(mesh);
    }
  } else if (obj instanceof THREE.Mesh) {
    // Regular mesh - clone geometry and apply world transform
    const clonedGeometry = obj.geometry.clone();
    clonedGeometry.applyMatrix4(obj.matrixWorld);
    
    const mesh = new THREE.Mesh(clonedGeometry, obj.material);
    mesh.name = obj.name || 'mesh';
    exportGroup.add(mesh);
  }
});

console.log(`Collected ${exportGroup.children.length} meshes for export`);

// Export using OBJExporter
const exporter = new OBJExporter();
const objData = exporter.parse(exportGroup);

// Ensure output directory exists
const outputDir = '/root/output';
if (!fs.existsSync(outputDir)) {
  fs.mkdirSync(outputDir, { recursive: true });
}

// Write to file
const outputPath = path.join(outputDir, 'object.obj');
fs.writeFileSync(outputPath, objData);

console.log(`Successfully exported to ${outputPath}`);
