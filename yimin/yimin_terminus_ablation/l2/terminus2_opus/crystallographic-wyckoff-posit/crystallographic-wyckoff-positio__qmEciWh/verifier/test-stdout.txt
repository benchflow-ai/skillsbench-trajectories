from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
import re


def decimal_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal value to a fraction string with denominator <= max_denominator.
    Values are first normalized to [0, 1) range for fractional coordinates.
    """
    # Normalize to [0, 1) range
    value = value % 1.0
    
    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"
    
    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)
    
    # Normalize the fraction to be in [0, 1)
    if frac >= 1:
        frac = frac - int(frac)
    if frac < 0:
        frac = frac + 1
    
    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def parse_wyckoff_symbol(symbol: str) -> tuple[int, str]:
    """
    Parse a Wyckoff symbol like '4a' or '8c' into multiplicity and letter.
    
    Args:
        symbol: Wyckoff symbol string (e.g., '4a', '8c')
        
    Returns:
        Tuple of (multiplicity, letter)
    """
    match = re.match(r'(\d+)([a-zA-Z]+)', symbol)
    if match:
        multiplicity = int(match.group(1))
        letter = match.group(2)
        return multiplicity, letter
    else:
        raise ValueError(f"Could not parse Wyckoff symbol: {symbol}")


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position.
    
    Args:
        filepath: Path to the CIF file
        
    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: mapping of Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: mapping of Wyckoff letter to coordinates as fraction strings
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    analyzer = SpacegroupAnalyzer(structure)
    
    # Get the symmetrized structure which contains Wyckoff information
    symmetrized_struct = analyzer.get_symmetrized_structure()
    
    # Get Wyckoff symbols for each equivalent group
    wyckoff_symbols = symmetrized_struct.wyckoff_symbols
    
    # Get equivalent indices - groups of symmetrically equivalent sites
    equivalent_indices = symmetrized_struct.equivalent_indices
    
    # Build the results
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Iterate over each equivalent group and its Wyckoff symbol
    for i, equiv_group in enumerate(equivalent_indices):
        # Get the Wyckoff symbol for this group
        wyckoff_symbol = wyckoff_symbols[i]
        
        # Parse the symbol to get multiplicity and letter
        multiplicity, letter = parse_wyckoff_symbol(wyckoff_symbol)
        
        # Skip if we've already processed this Wyckoff letter
        # (in case of multiple groups with same letter - shouldn't happen normally)
        if letter in wyckoff_multiplicity_dict:
            continue
        
        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get the first atom index in this equivalent group
        first_idx = equiv_group[0]
        
        # Get fractional coordinates of the first atom
        site = symmetrized_struct[first_idx]
        frac_coords = site.frac_coords
        
        # Convert to fraction strings
        coord_strings = [
            decimal_to_fraction_string(frac_coords[0]),
            decimal_to_fraction_string(frac_coords[1]),
            decimal_to_fraction_string(frac_coords[2])
        ]
        wyckoff_coordinates_dict[letter] = coord_strings
    
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
