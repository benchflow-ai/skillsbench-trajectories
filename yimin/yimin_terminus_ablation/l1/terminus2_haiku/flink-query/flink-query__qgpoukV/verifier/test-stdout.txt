package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {
        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        System.out.println("output  " + outputPath);
        
        // 10 minutes in microseconds = 10 * 60 * 1,000,000 = 600,000,000
        final long sessionGap = 600000000L;

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events - filter for SUBMIT events (eventType == 0)
        var taskEvents = env.addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .filter((FilterFunction<TaskEvent>) event -> event.eventType == 0) // Only SUBMIT events
                .map((MapFunction<TaskEvent, JobTaskEvent>) event -> new JobTaskEvent(true, event.jobId, event.timestamp, 0))
                .keyBy(event -> event.jobId);

        // Read job events to detect job completion
        var jobEvents = env.addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                .filter((FilterFunction<JobEvent>) event -> event.eventType == 4) // FINISH events
                .map((MapFunction<JobEvent, JobTaskEvent>) event -> new JobTaskEvent(false, event.jobId, event.timestamp, 0))
                .keyBy(event -> event.jobId);

        // Union both streams
        var mergedEvents = taskEvents.union(jobEvents).keyBy(event -> event.jobId);

        // Process events to find longest stage per job
        var result = mergedEvents
                .process(new StageTracker(sessionGap))
                .filter((FilterFunction<Tuple2<Long, Integer>>) value -> value.f1 > 0); // Only output non-zero stages

        // Format output
        var formattedResult = result.map((MapFunction<Tuple2<Long, Integer>, String>) value -> "(" + value.f0 + "," + value.f1 + ")");

        // Write output
        formattedResult.addSink((SinkFunction<String>) sinkOrTest(new RichSinkFunction<String>() {
            private java.io.BufferedWriter writer;

            @Override
            public void open(Configuration parameters) throws Exception {
                writer = new java.io.BufferedWriter(new java.io.FileWriter(outputPath));
            }

            @Override
            public void invoke(String value) throws Exception {
                writer.write(value + "\n");
            }

            @Override
            public void close() throws Exception {
                if (writer != null) {
                    writer.close();
                }
            }
        }));

        env.execute("LongestSessionPerJob");
    }

    public static class JobTaskEvent implements Serializable {
        private static final long serialVersionUID = 1L;
        public boolean isTaskEvent;  // true for task, false for job
        public long jobId;
        public long timestamp;
        public int data;  // unused, just for alignment

        public JobTaskEvent() {}

        public JobTaskEvent(boolean isTaskEvent, long jobId, long timestamp, int data) {
            this.isTaskEvent = isTaskEvent;
            this.jobId = jobId;
            this.timestamp = timestamp;
            this.data = data;
        }

    // Source function to read gzipped task events CSV
    public static class TaskEventSource extends RichSourceFunction<TaskEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (GZIPInputStream gzipStream = new GZIPInputStream(Files.newInputStream(Paths.get(filePath)));
                 BufferedReader reader = new BufferedReader(new InputStreamReader(gzipStream))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    String[] fields = line.split(",");
                    if (fields.length >= 13) {
                        try {
                            long timestamp = Long.parseLong(fields[0]);
                            long jobId = Long.parseLong(fields[2]);
                            long taskIndex = Long.parseLong(fields[3]);
                            long machineId = Long.parseLong(fields[4]);
                            int eventType = Integer.parseInt(fields[5]);
                            String userId = fields[6];
                            double cpu = fields.length > 9 ? Double.parseDouble(fields[9]) : 0;
                            double memory = fields.length > 10 ? Double.parseDouble(fields[10]) : 0;
                            double disk = fields.length > 11 ? Double.parseDouble(fields[11]) : 0;
                            int priority = fields.length > 12 ? Integer.parseInt(fields[12]) : 0;

                            TaskEvent event = new TaskEvent(timestamp, jobId, taskIndex, machineId,
                                    eventType, userId, cpu, memory, disk, priority, 0, 0, 0);
                            ctx.collect(event);
                        } catch (NumberFormatException e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    public static class JobEventSource extends RichSourceFunction<JobEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public JobEventSource(String filePath) {
            this.filePath = filePath;

        @Override
        public void run(SourceContext<JobEvent> ctx) throws Exception {

                 BufferedReader reader = new BufferedReader(new InputStreamReader(gzipStream))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    String[] fields = line.split(",");

                        try {

                            long jobId = Long.parseLong(fields[2]);
                            int eventType = Integer.parseInt(fields[3]);
                            String userId = fields.length > 4 ? fields[4] : "";
                            String schedulingClass = fields.length > 5 ? fields[5] : "";
                            String jobName = fields.length > 6 ? fields[6] : "";
                            String logicalJobName = fields.length > 7 ? fields[7] : "";

                            JobEvent event = new JobEvent(timestamp, jobId, eventType, userId,
                                    schedulingClass, jobName, logicalJobName);
                            ctx.collect(event);
                        } catch (NumberFormatException e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        }

    // Process function to track stages and find longest
    public static class StageTracker extends KeyedProcessFunction<Long, JobTaskEvent, Tuple2<Long, Integer>> {
        private long sessionGap; // 10 minutes in microseconds
        private ValueState<StageState> stageState;

        public StageTracker(long sessionGap) {
            this.sessionGap = sessionGap;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            ValueStateDescriptor<StageState> descriptor = new ValueStateDescriptor<>(
                    "stageState",
                    StageState.class);
            stageState = getRuntimeContext().getState(descriptor);
        }

        @Override
        public void processElement(JobTaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            StageState state = stageState.value();
            if (state == null) {
                state = new StageState();
                state.jobId = event.jobId;
            }

            if (event.isTaskEvent) {
                // This is a SUBMIT event
                // Check if this event starts a new stage (gap > sessionGap since last SUBMIT)
                if (state.lastSubmitTime > 0 && event.timestamp - state.lastSubmitTime > sessionGap) {
                    // Stage ended, save it if it's the longest
                    if (state.currentStageSize > state.longestStageSize) {
                        state.longestStageSize = state.currentStageSize;
                    }
                    // Start new stage
                    state.currentStageSize = 1;
                } else if (state.lastSubmitTime == 0) {
                    // First event
                    state.currentStageSize = 1;
                } else {
                    // Continue current stage
                    state.currentStageSize++;
                }
                state.lastSubmitTime = event.timestamp;
            } else {
                // This is a job completion event
                // Finalize the current stage
                if (state.currentStageSize > state.longestStageSize) {
                    state.longestStageSize = state.currentStageSize;
                }
                // Output result
                if (state.longestStageSize > 0) {
                    out.collect(new Tuple2<>(state.jobId, state.longestStageSize));
                }
                stageState.clear();
                return;
            }

            stageState.update(state);
        }
    }

    // State class for tracking stage information
    public static class StageState implements Serializable {
        private static final long serialVersionUID = 1L;
        
        public long jobId;
        public long lastSubmitTime = 0;
        public int currentStageSize = 0;
        public int longestStageSize = 0;
    }
}
