// Numeric tokenizer for numeric types
class NumericTokenizer(
  precision: Int = 6,
  formatOptions: Map[String, Any] = Map.empty
) extends BaseTokenizer[Any] {
  
  def tokenize(value: Any): Token = {
    val strValue = value match {
      case d: Double => 
        val format = s"%.${precision}f"
        format.format(d)
      case f: Float => 
        val format = s"%.${precision}f"
        format.format(f)
      case bd: BigDecimal => 
        bd.setScale(precision, BigDecimal.RoundingMode.HALF_UP).toString
      case other => other.toString
    }
    Token(strValue, TokenType.NUMERIC, Map("original_type" -> value.getClass.getSimpleName))
  }
}

// Temporal tokenizer for date/time types
class TemporalTokenizer(
  formatStr: Option[String] = None
) extends BaseTokenizer[Any] {
  
  private val ISO_FORMAT = "yyyy-MM-dd'T'HH:mm:ss"
  private val DATE_FORMAT = "yyyy-MM-dd"
  
  def tokenize(value: Any): Token = {
    val (formatted, actualFormat) = value match {
      case dt: LocalDateTime =>
        val fmt = formatStr.getOrElse(ISO_FORMAT)
        (dt.format(DateTimeFormatter.ofPattern(fmt)), fmt)
      case d: LocalDate =>
        val fmt = formatStr.getOrElse(DATE_FORMAT)
        (d.format(DateTimeFormatter.ofPattern(fmt)), fmt)
      case _ => throw new IllegalArgumentException(s"Unsupported temporal type: ${value.getClass}")
    }
    Token(formatted, TokenType.TEMPORAL)
  }
