'use client';

import { useState, useEffect } from 'react';

const API_URL = '';

export default function LateBanner() {
  const [text, setText] = useState<string>('');
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch(`${API_URL}/api/late-banner`)
      .then(r => r.json())
      .then(data => {
        setText(data.text);
        setLoading(false);
      })
      .catch(() => setLoading(false));
  }, []);

  // Reserve space during loading to prevent layout shift
  if (loading) {
    return (
      <div
        data-testid="late-banner"
        className="bg-[#ff6b35] text-white py-32 px-4 text-center font-bold text-[28px]"
        style={{ visibility: 'hidden' }}
      >
        &nbsp;
      </div>
    );
  }

  if (!text) return null;

  return (
    <div
      data-testid="late-banner"
      className="bg-[#ff6b35] text-white py-32 px-4 text-center font-bold text-[28px]"
    >
      {text}
    </div>
  );
}
