@layer base {

'use client';

import { useState, useEffect } from 'react';

const API_URL = '';

export default function Banner() {
  const [promo, setPromo] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch(`${API_URL}/api/banner`)
      .then(r => r.json())
      .then(data => {
        setPromo(data.text);
        setLoading(false);
      })
      .catch(() => setLoading(false));
  }, []);
