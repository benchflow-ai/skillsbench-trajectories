New Terminal Output:
(value: T) => {
      // Validate
      validatorsCopy.foreach { validator =>
        if (!validator(value)) {
          throw new IllegalArgumentException(s"Validation failed for $value")
        }
      }
      
      // Convert to string
      var strValue = value.toString
      
      // Normalize
      normalizersCopy.foreach { normalizer =>
        strValue = normalizer(strValue)
      }
      
      Token(strValue, TokenType.STRING, metadataCopy)
    }
  }
}

object TokenizerBuilder {
  def apply[T](): TokenizerBuilder[T] = new TokenizerBuilder[T]()
}

object Tokenizer {
  private val universalTokenizer = UniversalTokenizer()
  
  /**
   * Tokenize any value using the universal tokenizer.
   */
  def tokenize(value: Any): Token = universalTokenizer.tokenize(value)
  
  /**
   * Tokenize a batch of values lazily.
   */
  def tokenizeBatch(values: Iterable[Any]): Iterator[Token] = 
    values.iterator.map(tokenize)
  
  /**
   * Convert a Tokenizable object to a Token.
   */
  def toToken(value: Tokenizable): Token = 
    Token(value.toToken, TokenType.STRUCTURED)
}
