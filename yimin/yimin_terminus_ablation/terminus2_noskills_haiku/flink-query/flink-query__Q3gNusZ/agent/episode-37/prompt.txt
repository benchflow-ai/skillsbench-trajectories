public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        env.setParallelism(1);

        // Create source function for task events
        SourceFunction<TaskEvent> taskSource = new TaskEventSource(taskInput);

        // Read task events, filter for SUBMIT events, and process with custom state logic

                    @Override
                    public boolean filter(TaskEvent event) throws Exception {

    }

    // Custom KeyedProcessFunction to detect stages
    public static class StageDetectionFunction extends KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Long>> {
        private final long sessionGapMs;
        private transient ValueState<StageState> stateState;

        public StageDetectionFunction(long sessionGapMs) {
            this.sessionGapMs = sessionGapMs;
        }

        @Override
        public void open(org.apache.flink.configuration.Configuration parameters) throws Exception {
            ValueStateDescriptor<StageState> descriptor = new ValueStateDescriptor<>(
                    "stageState",
                    StageState.class);
            stateState = getRuntimeContext().getState(descriptor);
        }

        @Override
        public void processElement(TaskEvent event, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            long currentTime = event.timestamp;
            StageState state = stateState.value();

            if (state == null) {
                // Start new stage
                state = new StageState();
                state.currentStageCount = 1;
                state.lastEventTime = currentTime;
                state.maxStageCount = 1;

                // Check if we need to start a new stage
                if (currentTime - state.lastEventTime > sessionGapMs) {

                    }
