// Source function to read task events from gzipped CSV
    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            try (GZIPInputStream gzipInputStream = new GZIPInputStream(
                    Files.newInputStream(Paths.get(filePath)));
                 BufferedReader reader = new BufferedReader(
                    new java.io.InputStreamReader(gzipInputStream))) {

                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    String[] fields = line.split(",");
                    if (fields.length >= 13) {
                        try {
                            long timestamp = Long.parseLong(fields[0]);
                            long jobId = Long.parseLong(fields[2]);
                            long taskIndex = Long.parseLong(fields[3]);
                            long machineId = fields[4].isEmpty() ? -1 : Long.parseLong(fields[4]);
                            int eventType = Integer.parseInt(fields[5]);
                            String userId = fields[6];
                            int priority = Integer.parseInt(fields[7]);
                            int cpu = Integer.parseInt(fields[8]);
                            double memoryGb = Double.parseDouble(fields[9]);
                            double diskGb = Double.parseDouble(fields[10]);
                            double constraintFlags = Double.parseDouble(fields[11]);
                            int missingInfo = Integer.parseInt(fields[12]);

                            TaskEvent event = new TaskEvent(
                                    timestamp, jobId, taskIndex, machineId, eventType,
                                    userId, priority, cpu, memoryGb, diskGb,
                                    constraintFlags, missingInfo);
                            ctx.collect(event);
                        } catch (NumberFormatException e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    // Sink function to write output to file
    public static class FileSink implements SinkFunction<Tuple2<Long, Long>> {
        private String outputPath;
        private transient java.io.FileWriter writer;

        public FileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void invoke(Tuple2<Long, Long> value, Context context) throws Exception {
            if (writer == null) {
                writer = new java.io.FileWriter(outputPath, true);
            }
            writer.write("(" + value.f0 + "," + value.f1 + ")\n");
            writer.flush();
        }

        @Override
        public void finish() throws Exception {
            if (writer != null) {
                writer.close();
            }
        }
    }
