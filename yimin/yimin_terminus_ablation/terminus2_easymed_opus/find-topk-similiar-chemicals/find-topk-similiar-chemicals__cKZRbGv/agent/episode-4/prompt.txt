def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top k most similar molecules to a target molecule from a PDF file.
    
    Uses Morgan fingerprints with Tanimoto similarity (radius=2, include chirality).
    Results are sorted in descending order of similarity, with alphabetical ordering
    for ties.
    
    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return
        
    Returns:
        List of top k molecule names sorted by descending similarity,
        with alphabetical ordering for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    # Create RDKit molecule and fingerprint for target
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_molecule_name}")
    
    target_fp = compute_morgan_fingerprint(target_mol, radius=2, use_chirality=True)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")
    
    # Extract molecules from PDF
    molecule_names = extract_molecules_from_pdf(molecule_pool_filepath)
    
    # Compute similarities for all molecules in pool
    similarities = []
    for mol_name in molecule_names:
        # Get SMILES from PubChem
        smiles = get_smiles_from_name(mol_name)
        if smiles is None:
            continue
        
        # Create RDKit molecule
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            continue
        
        # Compute fingerprint
        fp = compute_morgan_fingerprint(mol, radius=2, use_chirality=True)
        if fp is None:
            continue
        
        # Compute Tanimoto similarity
        sim = tanimoto_similarity(target_fp, fp)
        similarities.append((mol_name, sim))
    
    # Sort by similarity (descending), then by name (alphabetical) for ties
    # Using -sim for descending order, and name for alphabetical tie-breaking
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k molecule names
    top_k_molecules = [name for name, sim in similarities[:top_k]]
    
    return top_k_molecules
