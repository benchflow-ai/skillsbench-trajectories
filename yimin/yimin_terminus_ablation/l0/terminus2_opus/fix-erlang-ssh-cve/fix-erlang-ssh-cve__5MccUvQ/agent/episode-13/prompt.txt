%% Reject connection protocol messages before authentication is complete
handle_event(internal, {conn_msg,_Msg}, StateName, D) when not ?CONNECTED(StateName) ->
    {Shutdown, D1} =
        ?send_disconnect(?SSH_DISCONNECT_PROTOCOL_ERROR,
                         "Connection protocol message received before authentication",
                         StateName, D),
    {stop, Shutdown, D1};
