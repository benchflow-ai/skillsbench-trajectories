import os
# Suppress RDKit deprecation warnings at the C++ level
os.environ['PYTHONWARNINGS'] = 'ignore'

import pdfplumber
import requests
from rdkit import Chem
from rdkit.Chem import AllChem
from typing import List, Tuple
import time
import warnings

# Suppress all Python warnings
warnings.filterwarnings('ignore')

# Cache for chemical name to SMILES conversion
_smiles_cache = {}

def _get_smiles_from_cactus(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES using CACTUS service.
    Uses caching to avoid repeated API calls.
    """
    if chemical_name in _smiles_cache:
        return _smiles_cache[chemical_name]
    
    try:
        # Query CACTUS for the compound
        url = f"https://cactus.nci.nih.gov/chemical/structure/{chemical_name}/smiles"
        response = requests.get(url, timeout=5)
        
        if response.status_code == 200:
            smiles = response.text.strip()
            if smiles and smiles != "":
                _smiles_cache[chemical_name] = smiles
                return smiles
    except Exception as e:
        pass
    
    return None

def _extract_chemical_names_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract all chemical names from the PDF file.
    """
    chemical_names = []
    
    try:
        with pdfplumber.open(pdf_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Split by newlines and filter empty lines
                    names = [name.strip() for name in text.split('\n') if name.strip()]
                    chemical_names.extend(names)
    except Exception as e:
        pass
    
    return chemical_names

def _compute_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule from SMILES string.
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is not None:
            return AllChem.GetMorganFingerprintAsBitVect(
                mol, 
                radius=radius, 
                nBits=2048,
                useChirality=include_chirality
            )
    except Exception as e:
        pass
    
    return None

def _compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.
    """
    if fp1 is None or fp2 is None:
        return 0.0
    
    try:
        from rdkit import DataStructs
        return DataStructs.TanimotoSimilarity(fp1, fp2)
    except Exception as e:
        pass
    
    return 0.0

def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k similar chemicals to a target molecule from a pool of molecules in a PDF.
    
    Uses Morgan fingerprints with Tanimoto similarity (radius=2, include chirality).
    Chemical names are converted to SMILES using external chemistry resources (CACTUS).
    Results are sorted by similarity in descending order, with alphabetical ordering for ties.
    
    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to the PDF file containing the pool of chemicals
        top_k: Number of top similar molecules to return
    
    Returns:
        List of tuples (chemical_name, similarity_score) sorted by similarity (descending)
        and then alphabetically by name when similarities are equal.
    """
    # Get SMILES for target molecule
    target_smiles = _get_smiles_from_cactus(target_molecule_name)
    if target_smiles is None:
        return []
    
    # Compute fingerprint for target
    target_fp = _compute_morgan_fingerprint(target_smiles, radius=2, include_chirality=True)
    if target_fp is None:
        return []
    
    # Extract all chemical names from PDF
    chemical_names = _extract_chemical_names_from_pdf(molecule_pool_filepath)
    
    # Compute similarities for all molecules
    similarities = []
    
    for chem_name in chemical_names:
        # Skip the target molecule itself
        if chem_name.lower() == target_molecule_name.lower():
            continue
        
        # Get SMILES for this molecule
        smiles = _get_smiles_from_cactus(chem_name)
        if smiles is None:
            continue
        
        # Compute fingerprint
        fp = _compute_morgan_fingerprint(smiles, radius=2, include_chirality=True)
        if fp is None:
            continue
        
        # Compute similarity
        similarity = _compute_tanimoto_similarity(target_fp, fp)
        similarities.append((chem_name, similarity))
        
        # Add a small delay to avoid overwhelming the API
        time.sleep(0.1)
    
    # Sort by similarity (descending) and then alphabetically by name
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k
    return similarities[:top_k]

if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        "Ethanol",
        "/root/molecules.pdf",
        5
    )
    
    print("Top 5 similar molecules to Ethanol:")
    for name, similarity in result:
        print(f"{name}: {similarity:.4f}")
