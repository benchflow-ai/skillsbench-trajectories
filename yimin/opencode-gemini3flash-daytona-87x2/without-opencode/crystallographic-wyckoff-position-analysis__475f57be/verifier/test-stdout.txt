import pymatgen.core
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, Any]:
    """
    Analyzes the Wyckoff position multiplicities and calculates approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    """
    try:
        # Load the structure from the CIF file
        structure = pymatgen.core.Structure.from_file(filepath)
        
        # Use SpacegroupAnalyzer to find symmetry and get the standardized dataset
        # Default tolerance of 1e-5 is usually appropriate for CIF files
        sga = SpacegroupAnalyzer(structure, symprec=1e-5)
        dataset = sga.get_symmetry_dataset()
        
        if dataset is None:
            return {}

        wyckoffs = dataset.wyckoffs
        equivalent_atoms = dataset.equivalent_atoms
        std_positions = dataset.std_positions
        
        letters_seen = {} # letter -> (multiplicity, coordinates)
        
        # Identify unique orbits and their corresponding Wyckoff letters
        seen_orbits = set()
        for i, orbit_id in enumerate(equivalent_atoms):
            if orbit_id not in seen_orbits:
                seen_orbits.add(orbit_id)
                letter = wyckoffs[i]
                
                # The multiplicity is the number of equivalent atoms in this orbit
                multiplicity = list(equivalent_atoms).count(orbit_id)
                coords = std_positions[i]
                
                # If multiple orbits have the same letter, we take the first one encountered
                if letter not in letters_seen:
                    letters_seen[letter] = (multiplicity, coords)
        
        # Sort the letters alphabetically for consistent output
        sorted_letters = sorted(letters_seen.keys())
        
        wyckoff_multiplicity_dict = {}
        wyckoff_coordinates_dict = {}
        
        for letter in sorted_letters:
            mult, coords = letters_seen[letter]
            wyckoff_multiplicity_dict[letter] = mult
            
            # Convert coordinates to simplified fraction strings with denominator <= 12
            formatted_coords = []
            for c in coords:
                # Ensure coordinates are in the range [0, 1)
                c_mod = float(c) % 1
                
                # Find the closest rational number with denominator <= 12
                f = Fraction(c_mod).limit_denominator(12)
                
                # Rounding might result in 1.0 (e.g., Fraction(11, 11)), so apply modulo 1 again
                f = f % 1
                
                if f.denominator == 1:
                    formatted_coords.append(str(f.numerator))
                else:
                    formatted_coords.append(f"{f.numerator}/{f.denominator}")
            
            wyckoff_coordinates_dict[letter] = formatted_coords
            
        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict
        }
        
    except Exception:
        return {}
