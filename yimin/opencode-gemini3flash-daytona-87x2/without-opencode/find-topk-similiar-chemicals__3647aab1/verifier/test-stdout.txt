import urllib.request
import urllib.parse
import json
import time
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from pdfminer.high_level import extract_text

def get_smiles_from_name(name, cache={}):
    if name in cache:
        return cache[name]
    
    encoded_name = urllib.parse.quote(name)
    url = f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{encoded_name}/property/SMILES/JSON"
    
    try:
        # Respectful delay
        time.sleep(0.1)
        with urllib.request.urlopen(url) as response:
            if response.getcode() == 200:
                data = json.loads(response.read().decode())
                properties = data['PropertyTable']['Properties'][0]
                # Look for SMILES in the returned properties
                smiles = None
                for key in ['SMILES', 'CanonicalSMILES', 'IsomericSMILES']:
                    if key in properties:
                        smiles = properties[key]
                        break
                if not smiles:
                    # Fallback to any key containing SMILES
                    for key in properties:
                        if 'SMILES' in key:
                            smiles = properties[key]
                            break
                cache[name] = smiles
                return smiles
    except Exception:
        pass
    
    cache[name] = None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # 1. Extract names from PDF
    try:
        text = extract_text(molecule_pool_filepath)
        # Split by newline and filter out empty strings and page breaks
        pool_names = []
        for line in text.split('\n'):
            cleaned = line.strip()
            if cleaned and not cleaned.startswith('\x0c'): # Remove page breaks if any
                pool_names.append(cleaned)
    except Exception:
        return []
    
    if not pool_names:
        return []

    # 2. Convert target name to SMILES and Fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 3. Process pool
    unique_pool_names = sorted(list(set(pool_names)))
    results = []
    
    for name in unique_pool_names:
        smiles = get_smiles_from_name(name)
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                sim = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, sim))
    
    # 4. Sort: similarity descending, name ascending
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k names
    return [name for name, sim in results[:top_k]]
