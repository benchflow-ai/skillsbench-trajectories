package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FilterFunction;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.streaming.api.functions.windowing.WindowFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskEventStream;
        if (AppBase.taskEvents != null) {
            taskEventStream = env.addSource(AppBase.taskEvents);
        } else {
            taskEventStream = env.readTextFile(taskInput)
                    .map(new MapFunction<String, TaskEvent>() {
                        @Override
                        public TaskEvent map(String value) throws Exception {
                            return TaskEvent.fromString(value);
                        }
                    });
        }

        DataStream<TaskEvent> tasks = taskEventStream
                .assignTimestampsAndWatermarks(WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));

        // Just to satisfy job_input if it is expected to be read
        if (jobInput != null || AppBase.jobEvents != null) {
            DataStream<JobEvent> jobEventStream;
            if (AppBase.jobEvents != null) {
                jobEventStream = env.addSource(AppBase.jobEvents);
            } else {
                jobEventStream = env.readTextFile(jobInput)
                        .map(new MapFunction<String, JobEvent>() {
                            @Override
                            public JobEvent map(String value) throws Exception {
                                return JobEvent.fromString(value);
                            }
                        });
            }
            jobEventStream.assignTimestampsAndWatermarks(WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                    .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));
        }

        DataStream<Tuple2<Long, Integer>> stageCounts = tasks
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent value) throws Exception {
                        return value.eventType == 0; // SUBMIT
                    }
                })
                .keyBy(t -> t.jobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        int count = 0;
                        for (TaskEvent ignored : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple2<Long, Integer>> maxStageCounts = stageCounts
                .keyBy(t -> t.f0)
                .process(new KeyedProcessFunction<Long, Tuple2<Long, Integer>, Tuple2<Long, Integer>>() {
                    private ValueState<Integer> maxCount;

                    @Override
                    public void open(Configuration parameters) {
                        maxCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxCount", Integer.class));
                    }

                    @Override
                    public void processElement(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Integer currentMax = maxCount.value();
                        if (currentMax == null || value.f1 > currentMax) {
                            maxCount.update(value.f1);
                        }
                        ctx.timerService().registerEventTimeTimer(Long.MAX_VALUE);
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        out.collect(new Tuple2<>(ctx.getCurrentKey(), maxCount.value()));
                    }
                });

        DataStream<String> resultStream = maxStageCounts.map(new MapFunction<Tuple2<Long, Integer>, String>() {
            @Override
            public String map(Tuple2<Long, Integer> value) throws Exception {
                return "(" + value.f0 + "," + value.f1 + ")";
            }
        });

        if (AppBase.out != null) {
            resultStream.addSink(AppBase.out);
        } else if (outputPath != null) {
            resultStream.writeAsText(outputPath, org.apache.flink.core.fs.FileSystem.WriteMode.OVERWRITE).setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

}
