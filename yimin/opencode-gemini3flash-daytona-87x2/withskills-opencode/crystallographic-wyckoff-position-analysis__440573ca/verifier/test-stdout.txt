import json
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and coordinates for the first atom of each Wyckoff position.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'.
    """
    # Read the structure from the CIF file
    struct = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    # We use the standardized cell to get standard Wyckoff positions and multiplicities
    sga = SpacegroupAnalyzer(struct)
    dataset = sga.get_symmetry_dataset()
    
    # wyckoffs is a list of Wyckoff letters for each atom in the standardized cell
    # std_positions is a list of fractional coordinates for each atom in the standardized cell
    wyckoff_letters = dataset.wyckoffs
    standardized_positions = dataset.std_positions
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    def to_fraction_str(val: float) -> str:
        # Normalize to [0, 1)
        normalized_val = val % 1
            
        # Find the closest rational number with denominator <= 12
        frac = Fraction(normalized_val).limit_denominator(12)
        
        # If the closest fraction is 1/1, it means the value was very close to an integer
        # and we should treat it as 0/1 (since we are in [0, 1)).
        if frac == 1:
            return "0"

        if frac.denominator == 1:
            return str(frac.numerator)
        else:
            return f"{frac.numerator}/{frac.denominator}"

    # Iterate through the atoms in the standardized cell
    for letter, pos in zip(wyckoff_letters, standardized_positions):
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = 0
            # Store the coordinates of the first atom encountered for this Wyckoff position
            wyckoff_coordinates_dict[letter] = [to_fraction_str(p) for p in pos]
        
        # Increment the count (multiplicity) for this Wyckoff letter
        wyckoff_multiplicity_dict[letter] += 1
    
    # Sort the dictionaries by Wyckoff letter for consistency (optional but good practice)
    sorted_letters = sorted(wyckoff_multiplicity_dict.keys())
    
    result = {
        "wyckoff_multiplicity_dict": {letter: wyckoff_multiplicity_dict[letter] for letter in sorted_letters},
        "wyckoff_coordinates_dict": {letter: wyckoff_coordinates_dict[letter] for letter in sorted_letters}
    }
    
    return result

if __name__ == "__main__":
    # Example usage
    import sys
    if len(sys.argv) > 1:
        path = sys.argv[1]
        res = analyze_wyckoff_position_multiplicities_and_coordinates(path)
        print(json.dumps(res, indent=4))
