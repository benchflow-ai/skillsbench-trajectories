import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import warnings

# Suppress warnings from PubChemPy and RDKit for cleaner output
warnings.filterwarnings("ignore", category=DeprecationWarning)
from rdkit import RDLogger
RDLogger.DisableLog('rdApp.*')

def get_smiles_from_name(name):
    """
    Converts a chemical name to its SMILES representation using PubChem.
    """
    try:
        # Search for the compound by name
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            # We use isomeric_smiles to ensure chirality is included as requested.
            # isomeric_smiles attribute contains the SMILES string with stereochemical info.
            return compounds[0].isomeric_smiles or compounds[0].canonical_smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF molecule pool to a target molecule.
    
    Args:
        target_molecule_name (str): The name of the target chemical.
        molecule_pool_filepath (str): The path to the PDF file containing chemical names.
        top_k (int): The number of top similar chemicals to return.
        
    Returns:
        list: A list of the top k chemical names, sorted by similarity (descending) 
              and then by name (alphabetical).
    """
    # 1. Extract chemical names from the PDF
    pool_names = set()
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    for line in text.split('\n'):
                        name = line.strip()
                        if name:
                            pool_names.add(name)
    except Exception:
        return []

    if not pool_names:
        return []

    # 2. Convert target molecule name to SMILES and then to Fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Morgan fingerprints with radius 2 and chirality included
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, radius=2, useChirality=True)

    # 3. Process the pool of molecules
    results = []
    
    for name in pool_names:
        # Get SMILES for each candidate in the pool
        smiles = get_smiles_from_name(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        # Morgan fingerprints for candidate
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, useChirality=True)
        
        # Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # 4. Sort results
    # Primary key: similarity (descending)
    # Secondary key: name (alphabetical)
    results.sort(key=lambda x: (-x[1], x[0]))

    # 5. Return top k names
    return [name for name, sim in results[:top_k]]
