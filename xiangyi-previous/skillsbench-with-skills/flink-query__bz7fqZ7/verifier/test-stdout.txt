package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.TimeCharacteristic;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.sink.RichSinkFunction;
import org.apache.flink.streaming.api.functions.timestamps.BoundedOutOfOrdernessTimestampExtractor;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.util.Collector;
import org.apache.flink.streaming.api.functions.windowing.WindowFunction;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import java.io.FileWriter;
import java.io.IOException;
import java.io.PrintWriter;
import java.util.HashMap;
import java.util.Map;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {
        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input");
        String outputPath = params.get("output");

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setStreamTimeCharacteristic(TimeCharacteristic.EventTime);
        env.getConfig().setGlobalJobParameters(params);
        env.setParallelism(1);

        DataStream<TaskEvent> taskEvents = env.readTextFile(taskInput)
                .map(TaskEvent::fromString)
                .filter(e -> e != null && e.eventType == 0)
                .assignTimestampsAndWatermarks(new BoundedOutOfOrdernessTimestampExtractor<TaskEvent>(Time.seconds(0)) {
                    @Override
                    public long extractTimestamp(TaskEvent element) {
                        return element.timestamp / 1000;
                    }
                });

        taskEvents
                .keyBy(e -> e.jobId)
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent e : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                })
                .addSink(new RichSinkFunction<Tuple2<Long, Integer>>() {
                    private Map<Long, Integer> maxCounts;

                    @Override
                    public void open(Configuration parameters) {
                        maxCounts = new HashMap<>();
                    }

                    @Override
                    public void invoke(Tuple2<Long, Integer> value, Context context) {
                        int currentMax = maxCounts.getOrDefault(value.f0, 0);
                        if (value.f1 > currentMax) {
                            maxCounts.put(value.f0, value.f1);
                        }
                    }

                    @Override
                    public void close() throws IOException {
                        ParameterTool params = (ParameterTool) getRuntimeContext().getExecutionConfig().getGlobalJobParameters();
                        String outPath = params.get("output");
                        try (PrintWriter writer = new PrintWriter(new FileWriter(outPath))) {
                            for (Map.Entry<Long, Integer> entry : maxCounts.entrySet()) {
                                writer.println("(" + entry.getKey() + "," + entry.getValue() + ")");
                            }
                        }
                    }
                }).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }
}
