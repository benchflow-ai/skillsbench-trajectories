   import math
   import time
   from collections import defaultdict
   from concurrent.futures import ProcessPoolExecutor
   import multiprocessing

   from sequential import (
       TFIDFIndex,
       SearchResult,
       IndexingResult,
       tokenize,
       compute_term_frequencies,
       search_sequential,
       Document
   )

   def _index_chunk_stage1(documents: List[Document]) -> Tuple[Dict[int, Dict[str, float]], Dict[str, int]]:
       local_doc_term_freqs = {}
       local_df = defaultdict(int)
       
       for doc in documents:
           text = doc.title + " " + doc.content
           tokens = tokenize(text)
           tf = compute_term_frequencies(tokens)
           local_doc_term_freqs[doc.doc_id] = tf
           for term in set(tf.keys()):
               local_df[term] += 1
               
       return local_doc_term_freqs, dict(local_df)

   def _index_chunk_stage2(chunk_data: Tuple[Dict[int, Dict[str, float]], Dict[str, float]]) -> Tuple[Dict[str, List[Tuple[int, float]]], Dict[int, Dict[str, float]], Dict[int, float]]:
       doc_term_freqs, idf = chunk_data
       
       partial_inverted = defaultdict(list)
       partial_vectors = {}
       partial_norms = {}
       
       for doc_id, tf_dict in doc_term_freqs.items():
           doc_vector = {}
           norm_squared = 0.0
           
           for term, tf in tf_dict.items():
               if term in idf:
                   val = tf * idf[term]
                   doc_vector[term] = val
                   norm_squared += val * val
                   partial_inverted[term].append((doc_id, val))
           
           partial_vectors[doc_id] = doc_vector
           partial_norms[doc_id] = math.sqrt(norm_squared)
           
       return partial_inverted, partial_vectors, partial_norms

   def build_tfidf_index_parallel(documents: List[Document], num_workers=None, chunk_size=500) -> IndexingResult:
       if num_workers is None:
           num_workers = multiprocessing.cpu_count()
           
       start_time = time.perf_counter()
       
       # Split documents
       chunks = [documents[i:i + chunk_size] for i in range(0, len(documents), chunk_size)]
       
       doc_term_freqs = {}
       document_frequencies = defaultdict(int)
       
       # Stage 1: TF and DF
       with ProcessPoolExecutor(max_workers=num_workers) as executor:
           results = executor.map(_index_chunk_stage1, chunks)
           
           for chunk_tf, chunk_df in results:
               doc_term_freqs.update(chunk_tf)
               for term, count in chunk_df.items():
                   document_frequencies[term] += count
                   
       # Compute IDF
       index = TFIDFIndex()
       index.num_documents = len(documents)
       index.document_frequencies = dict(document_frequencies)
       index.vocabulary = set(document_frequencies.keys())
       
       N = index.num_documents
       for term, df in index.document_frequencies.items():
           index.idf[term] = math.log(N / df) + 1
           
       # Stage 2: Inverted Index and Vectors
       # We need to chunk the doc_term_freqs for parallel processing
       # Convert dict items to list to chunk them? No, that's expensive memory-wise.
       # We can just reuse the document chunks logic if we map doc_ids back.
       # Or simpler: Split the doc_term_freqs keys (doc_ids) into chunks.
       
       doc_ids = list(doc_term_freqs.keys())
       doc_id_chunks = [doc_ids[i:i + chunk_size] for i in range(0, len(doc_ids), chunk_size)]
       
       # Prepare data for workers: (subset_of_doc_term_freqs, idf)
       # Passing full IDF (dict) to every worker might be heavy if vocabulary is huge.
       # But let's try.
       
       stage2_inputs = []
       for chunk in doc_id_chunks:
           subset_tf = {doc_id: doc_term_freqs[doc_id] for doc_id in chunk}
           stage2_inputs.append((subset_tf, index.idf))
           
       with ProcessPoolExecutor(max_workers=num_workers) as executor:
           results = executor.map(_index_chunk_stage2, stage2_inputs)
           
           for p_inverted, p_vectors, p_norms in results:
               # Merge inverted index
               for term, postings in p_inverted.items():
                   if term not in index.inverted_index:
                       index.inverted_index[term] = []
                   index.inverted_index[term].extend(postings)
               
               index.doc_vectors.update(p_vectors)
               index.doc_norms.update(p_norms)
               
       # Sort posting lists
       # This can be parallelized too, but maybe fast enough?
       # The sequential code sorts them.
       for term in index.inverted_index:
           index.inverted_index[term].sort(key=lambda x: x[1], reverse=True)
           
       elapsed = time.perf_counter() - start_time
       return IndexingResult(index=index, elapsed_time=elapsed, num_documents=len(documents), vocabulary_size=len(index.vocabulary))

   def _search_chunk(args) -> List[SearchResult]:
       queries, index, top_k, documents = args
       # documents is optional, can be None
       # If documents is passed, we might face pickle issues if it's huge.
       # But let's assume it works.
       return [search_sequential(q, index, top_k, documents) for q in queries]

   def batch_search_parallel(queries: List[str], index: TFIDFIndex, top_k=10, num_workers=None, documents=None) -> Tuple[List[List[SearchResult]], float]:
       if num_workers is None:
           num_workers = multiprocessing.cpu_count()
           
       start_time = time.perf_counter()
       
       chunk_size = max(1, len(queries) // num_workers)
       query_chunks = [queries[i:i + chunk_size] for i in range(0, len(queries), chunk_size)]
       
       # Arguments for each chunk
       # Note: Passing 'index' to every worker.
       process_args = [(chunk, index, top_k, documents) for chunk in query_chunks]
       
       results = []
       with ProcessPoolExecutor(max_workers=num_workers) as executor:
           # map returns results in order
           chunk_results = executor.map(_search_chunk, process_args)
           for res in chunk_results:
               results.extend(res)
               
       elapsed = time.perf_counter() - start_time
       return results, elapsed
