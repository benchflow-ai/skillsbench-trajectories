"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from dataclasses import dataclass
from typing import List
import pubchempy as pcp
import time


@dataclass
class MoleculeResult:
    """Result representing a molecule and its similarity score."""
    name: str
    similarity: float


# Cache for SMILES to avoid repeated PubChem requests
_smiles_cache = {}


def _get_molecule_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to SMILES using PubChem.

    Args:
        chemical_name: The name of the chemical compound

    Returns:
        SMILES string or None if not found
    """
    # Check cache first
    if chemical_name in _smiles_cache:
        return _smiles_cache[chemical_name]

    try:
        # Search for the compound in PubChem
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            # Return the SMILES of the first match
            smiles = compounds[0].canonical_smiles
            _smiles_cache[chemical_name] = smiles
            return smiles
    except Exception:
        pass

    _smiles_cache[chemical_name] = None
    return None


def _extract_molecules_from_pdf(pdf_path: str) -> List[str]:
    """
    Extract chemical names from PDF file.

    Args:
        pdf_path: Path to the PDF file containing chemical names

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and clean up
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def _compute_morgan_fingerprint(smiles: str, radius: int = 2):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string representation
        radius: Radius for Morgan fingerprint (default 2)

    Returns:
        Morgan fingerprint bit vector
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return AllChem.GetMorganFingerprintAsBitVect(mol, radius, useChirality=True)


def _compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First Morgan fingerprint
        fp2: Second Morgan fingerprint

    Returns:
        Tanimoto similarity score (0-1)
    """
    from rdkit import DataStructs
    if fp1 is None or fp2 is None:
        return 0.0
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k similar molecules to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target chemical compound
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of top k similar molecule names, sorted by similarity (descending),
        with alphabetical ordering for ties
    """
    # Get SMILES for target molecule
    target_smiles = _get_molecule_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target molecule
    target_fp = _compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    molecule_names = _extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for mol_name in molecule_names:
        # Skip the target molecule itself
        if mol_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES and fingerprint
        mol_smiles = _get_molecule_from_name(mol_name)
        if mol_smiles is None:
            continue

        mol_fp = _compute_morgan_fingerprint(mol_smiles)
        if mol_fp is None:
            continue

        # Compute similarity
        similarity = _compute_tanimoto_similarity(target_fp, mol_fp)
        similarities.append(MoleculeResult(name=mol_name, similarity=similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x.similarity, x.name))

    # Return top k names
    return [result.name for result in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    for i, mol in enumerate(results, 1):
        print(f"{i}. {mol}")
