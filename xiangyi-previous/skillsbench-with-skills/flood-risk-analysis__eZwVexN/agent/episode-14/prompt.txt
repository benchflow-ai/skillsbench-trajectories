import requests
usgs_id = "04031000"
start_date = '2025-04-01'
end_date = '2025-04-07'
# Try both gage height (00065) and discharge (00060)
for param in ["00065", "00060"]:
    url = f"https://waterservices.usgs.gov/nwis/iv/?format=json&sites={usgs_id}&startDT={start_date}&endDT={end_date}&parameterCd={param}"
    r = requests.get(url)
    print(f"Param {param} status: {r.status_code}")
    if r.status_code == 200:
        data = r.json()
        ts = data.get('value', {}).get('timeSeries', [])
        print(f"Param {param} series count: {len(ts)}")
EOF
