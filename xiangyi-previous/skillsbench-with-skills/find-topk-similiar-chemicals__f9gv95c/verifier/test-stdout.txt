from __future__ import annotations

import argparse
import json
import time
from typing import Dict, Iterable, List, Optional, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


def _extract_molecule_names_from_pdf(pdf_path: str) -> List[str]:
    names: List[str] = []
    seen = set()
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if not text:
                continue
            for line in text.splitlines():
                name = line.strip()
                if not name or name in seen:
                    continue
                seen.add(name)
                names.append(name)
    return names


def _fetch_smiles_from_pubchem(
    name: str,
    cache: Dict[str, Optional[str]],
    delay_seconds: float = 0.2,
    max_retries: int = 3,
) -> Optional[str]:
    if name in cache:
        return cache[name]

    smiles: Optional[str] = None
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(name, "name")
            if compounds:
                compound = compounds[0]
                smiles = compound.smiles or compound.canonical_smiles
            break
        except Exception:
            if attempt < max_retries - 1:
                time.sleep(delay_seconds * (attempt + 1))
            else:
                smiles = None

    cache[name] = smiles
    if delay_seconds:
        time.sleep(delay_seconds)
    return smiles


_MORGAN_GENERATOR = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _morgan_fingerprint_from_smiles(smiles: str):
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _MORGAN_GENERATOR.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> List[Tuple[str, float]]:
    """
    Return the top-k most similar molecules in the pool to a target molecule name.

    Similarity uses Morgan fingerprints (radius=2, chirality enabled) with
    Tanimoto similarity. Results are sorted by descending similarity with
    alphabetical ordering for ties.
    """
    if top_k <= 0:
        return []

    names = _extract_molecule_names_from_pdf(molecule_pool_filepath)
    cache: Dict[str, Optional[str]] = {}

    target_smiles = _fetch_smiles_from_pubchem(target_molecule_name, cache)
    if not target_smiles:
        return []

    target_fp = _morgan_fingerprint_from_smiles(target_smiles)
    if target_fp is None:
        return []

    results: List[Tuple[str, float]] = []
    for name in names:
        smiles = _fetch_smiles_from_pubchem(name, cache)
        if not smiles:
            continue
        fp = _morgan_fingerprint_from_smiles(smiles)
        if fp is None:
            continue
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, float(sim)))

    results.sort(key=lambda item: (-item[1], item[0].casefold()))
    return results[: min(top_k, len(results))]


def _main() -> None:
    parser = argparse.ArgumentParser(
        description="Find top-k similar molecules from a PDF list."
    )
    parser.add_argument("target_name", help="Target molecule name")
    parser.add_argument(
        "molecule_pool_filepath", help="Path to PDF containing molecule names"
    )
    parser.add_argument("top_k", type=int, help="Number of top results to return")
    args = parser.parse_args()

    results = topk_tanimoto_similarity_molecules(
        args.target_name, args.molecule_pool_filepath, args.top_k
    )
    print(json.dumps(results, indent=2))


if __name__ == "__main__":
    _main()
