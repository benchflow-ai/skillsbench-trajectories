def quick_diarize(audio_path):
    y, sr = librosa.load(audio_path, sr=16000, mono=True)
    duration = len(y) / sr
    frame_length = int(0.025 * sr)
    hop_length = int(0.010 * sr)
    S = librosa.feature.melspectrogram(y=y, sr=sr, n_fft=frame_length, hop_length=hop_length)
    energy = np.sqrt(np.mean(S**2, axis=0))
    energy = (energy - np.min(energy)) / (np.max(energy) - np.min(energy) + 1e-8)
    speech_frames = energy > 0.15
    times = librosa.frames_to_time(np.arange(len(speech_frames)), sr=sr, hop_length=hop_length)
    segments = []
    in_speech = False
    start_time = 0
    for i, is_speech in enumerate(speech_frames):
        if is_speech and not in_speech:
            start_time = times[i]
            in_speech = True
        elif not is_speech and in_speech:
            segments.append((start_time, times[i]))
            in_speech = False
    if in_speech:
        segments.append((start_time, duration))
    if not segments:
        segments = [(0, duration)]
    num_speakers = min(2, max(1, len(segments) // 5))
    speakers = []
    for idx, (start, end) in enumerate(segments):
        speaker_id = idx % num_speakers
        speakers.append({'start': start, 'end': end, 'speaker': f'spk{speaker_id:02d}'})
    return speakers, num_speakers, duration
