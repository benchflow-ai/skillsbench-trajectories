d3.csv("data/stock-descriptions.csv").then(data => {
    const width = 600;
    const height = 600;

    // Process data
    data = data.filter(d => d.ticker); // Filter out any empty rows
    data.forEach(d => {
        d.marketCap = d.marketCap ? +d.marketCap : null;
    });

    const sectors = Array.from(new Set(data.map(d => d.sector))).filter(s => s);
    const color = d3.scaleOrdinal().domain(sectors).range(d3.schemeCategory10);

    const sizeScale = d3.scaleSqrt()
        .domain([0, d3.max(data, d => d.marketCap || 0)])
        .range([10, 50]);

    const nodes = data.map(d => ({
        ...d,
        radius: (d.sector === 'ETF' || !d.marketCap) ? 15 : sizeScale(d.marketCap)
    }));

    // Sector centers for clustering
    const sectorCenters = {};
    sectors.forEach((sector, i) => {
        const angle = (i / sectors.length) * 2 * Math.PI;
        const radius = 150;
        sectorCenters[sector] = {
            x: width / 2 + radius * Math.cos(angle),
            y: height / 2 + radius * Math.sin(angle)
        };
    });

    const simulation = d3.forceSimulation(nodes)
        .force("x", d3.forceX(d => sectorCenters[d.sector] ? sectorCenters[d.sector].x : width / 2).strength(0.1))
        .force("y", d3.forceY(d => sectorCenters[d.sector] ? sectorCenters[d.sector].y : height / 2).strength(0.1))
        .force("collide", d3.forceCollide(d => d.radius + 2))
        .force("charge", d3.forceManyBody().strength(-10));

    const svg = d3.select("#bubble-chart")
        .append("svg")
        .attr("width", width)
        .attr("height", height);

    const tooltip = d3.select("body").append("div").attr("class", "tooltip");

    const bubbles = svg.selectAll(".bubble-group")
        .data(nodes)
        .enter()
        .append("g")
        .attr("class", "bubble-group")
        .on("mouseover", (event, d) => {
            if (d.sector !== "ETF") {
                tooltip.style("display", "block")
                    .html("<strong>" + d.ticker + "</strong><br>" + d["full name"] + "<br>" + d.sector);
            }
        })
        .on("mousemove", (event) => {
            tooltip.style("left", (event.pageX + 10) + "px")
                .style("top", (event.pageY + 10) + "px");
        })
        .on("mouseout", () => tooltip.style("display", "none"))
        .on("click", (event, d) => {
            selectStock(d.ticker);
        });

    bubbles.append("circle")
        .attr("class", "bubble")
        .attr("r", d => d.radius)
        .attr("fill", d => color(d.sector))
        .attr("stroke", "#999");

    bubbles.append("text")
        .attr("text-anchor", "middle")
        .attr("dy", ".3em")
        .style("font-size", d => Math.min(d.radius / 2.5, 12) + "px")
        .style("pointer-events", "none")
        .text(d => d.ticker);

    simulation.on("tick", () => {
        bubbles.attr("transform", d => `translate(${d.x},${d.y})`);
    });

    // Legend
    const legend = d3.select("#legend");
    sectors.sort().forEach(sector => {
        const item = legend.append("div").attr("class", "legend-item");
        item.append("div").attr("class", "legend-color").style("background-color", color(sector));
        item.append("span").text(sector);
    });

    // Table
    const tableBody = d3.select("#stock-table tbody");
    const rows = tableBody.selectAll("tr")
        .data(data)
        .enter()
        .append("tr")
        .attr("id", d => "row-" + d.ticker)
        .on("click", (event, d) => {
            selectStock(d.ticker);
        });

    rows.append("td").text(d => d.ticker);
    rows.append("td").text(d => d["full name"]);
    rows.append("td").text(d => d.sector);
    rows.append("td").text(d => formatMarketCap(d.marketCap));

    function formatMarketCap(val) {
        if (val === null || val === undefined || isNaN(val)) return "N/A";
        if (val >= 1e12) return (val / 1e12).toFixed(2) + "T";
        if (val >= 1e9) return (val / 1e9).toFixed(2) + "B";
        if (val >= 1e6) return (val / 1e6).toFixed(2) + "M";
        return val.toLocaleString();
    }

    function selectStock(ticker) {
        // Highlight Table
        d3.selectAll("tr").classed("highlight", false);
        const row = d3.select("#row-" + ticker);
        if (!row.empty()) {
            row.classed("highlight", true);
            row.node().scrollIntoView({ behavior: "smooth", block: "nearest" });
        }

        // Highlight Bubble
        d3.selectAll("circle").style("stroke", "#999").style("stroke-width", "1px");
        d3.selectAll(".bubble-group").filter(d => d.ticker === ticker)
            .select("circle")
            .style("stroke", "black")
            .style("stroke-width", "3px");
    }
});
