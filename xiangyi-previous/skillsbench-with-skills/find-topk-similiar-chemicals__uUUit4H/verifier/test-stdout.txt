import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
from rdkit import RDLogger
import warnings

# Suppress logs and warnings
RDLogger.DisableLog('rdApp.*')
warnings.filterwarnings("ignore")

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Helper to get fingerprint
    def get_fp(name):
        try:
            # Use PubChem to get SMILES
            compounds = pcp.get_compounds(name, 'name')
            if not compounds:
                return None
            # Use smiles (which includes isomerism in newer pcp versions) as isomeric_smiles is deprecated
            smiles = compounds[0].smiles
            if not smiles:
                return None
            
            mol = Chem.MolFromSmiles(smiles)
            if not mol:
                return None
            
            # Morgan fingerprints with Tanimoto similarity (radius = 2, include chirality)
            return AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        except Exception:
            return None

    # Get target fingerprint
    target_fp = get_fp(target_molecule_name)
    if target_fp is None:
        return []

    # Read PDF to get candidate names
    candidates = set()
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    for line in text.split('\n'):
                        line = line.strip()
                        if line:
                            candidates.add(line)
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []

    candidate_scores = []
    for name in candidates:
        fp = get_fp(name)
        if fp:
            sim = DataStructs.TanimotoSimilarity(target_fp, fp)
            candidate_scores.append((name, sim))
    
    # Sort: Descending similarity (-x[1]), then Ascending name (x[0])
    candidate_scores.sort(key=lambda x: (-x[1], x[0]))
    
    return [x[0] for x in candidate_scores[:top_k]]
