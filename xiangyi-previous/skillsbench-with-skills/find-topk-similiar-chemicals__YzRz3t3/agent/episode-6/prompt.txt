import sys
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem
from pdfminer.high_level import extract_text
import time

def get_smiles(name):
    try:
        results = pcp.get_compounds(name, 'name')
        if results:
            return results[0].smiles
    except Exception as e:
        if "503" in str(e):
            time.sleep(1)
            return get_smiles(name)
        print(f"Error fetching SMILES for {name}: {e}", file=sys.stderr)
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Extract names from PDF
    text = extract_text(molecule_pool_filepath)
    pool_names = sorted(list(set([line.strip() for line in text.split('\n') if line.strip()])))

    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []

    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []

    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, nBits=2048, useChirality=True)

    results = []
    for name in pool_names:
        smiles = get_smiles(name)
        if not smiles:
            continue
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, nBits=2048, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # Sort by similarity (descending), then name (ascending)
    results.sort(key=lambda x: (-x[1], x[0]))

    return results[:top_k]

if __name__ == "__main__":
    if len(sys.argv) < 3:
        print("Usage: python solution.py <target_molecule_name> <top_k>")
        sys.exit(1)

    target_name = sys.argv[1]
    top_k = int(sys.argv[2])
    pool_file = "/root/molecules.pdf"

    top_k_results = topk_tanimoto_similarity_molecules(target_name, pool_file, top_k)
    for name, sim in top_k_results:
        print(f"{name}: {sim:.4f}")
