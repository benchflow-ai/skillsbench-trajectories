    structure = Structure.from_file(filepath)

    sga = SpacegroupAnalyzer(structure)
    symmetrized = sga.get_symmetrized_structure()

    # Get Wyckoff symbols and equivalent sites
    wyckoff_symbols = symmetrized.wyckoff_symbols
    equivalent_sites = symmetrized.equivalent_sites

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for wsym, sites in zip(wyckoff_symbols, equivalent_sites):
        # Extract multiplicity (digits) and letter (alphabetic characters)
        multiplicity = int(''.join(c for c in wsym if c.isdigit()))
        letter = ''.join(c for c in wsym if c.isalpha())

        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity

        # Get the first site's fractional coordinates
        first_site = sites[0]
        frac_coords = first_site.frac_coords

        # Convert to fraction strings
        coord_strings = [decimal_to_fraction(c) for c in frac_coords]
        wyckoff_coordinates_dict[letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
