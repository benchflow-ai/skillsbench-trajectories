def fix_ocr_amount(amount_str):
    """Fix common OCR errors in amount strings"""
    fixes = {'u': '0', 'U': '0', 'o': '0', 'O': '0', 'l': '1', 'I': '1', 'S': '5', 'Z': '2'}
    result = amount_str
    for old, new in fixes.items():
        result = result.replace(old, new)
    return result
