import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import time

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find top k most similar molecules to the target molecule from a pool.
    
    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return
        
    Returns:
        List of molecule names sorted by similarity (descending), then alphabetically
    """
    
    # Extract molecule names from PDF
    molecule_names = []
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecule_names.append(line)
    
    # Function to convert chemical name to SMILES using PubChem with retry logic
    def name_to_smiles(name, max_retries=3):
        for attempt in range(max_retries):
            try:
                compounds = pcp.get_compounds(name, 'name')
                if compounds:
                    # Use isomeric_smiles to preserve chirality
                    return compounds[0].isomeric_smiles
                return None
            except Exception as e:
                if attempt < max_retries - 1:
                    # Exponential backoff: wait longer between retries
                    wait_time = (2 ** attempt) * 0.5
                    time.sleep(wait_time)
                else:
                    return None
        return None
    
    # Function to get Morgan fingerprint
    def get_morgan_fingerprint(smiles):
        if smiles is None:
            return None
        try:
            mol = Chem.MolFromSmiles(smiles)
            if mol is None:
                return None
            # Morgan fingerprint with radius=2 and useChirality=True
            fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, nBits=2048, useChirality=True)
            return fp
        except Exception as e:
            return None
    
    # Get target molecule fingerprint
    target_smiles = name_to_smiles(target_molecule_name)
    target_fp = get_morgan_fingerprint(target_smiles)
    
    if target_fp is None:
        return []
    
    # Small delay after getting target to avoid rate limiting
    time.sleep(0.2)
    
    # Calculate similarities
    similarities = []
    for mol_name in molecule_names:
        # Skip if it's the same as target
        if mol_name == target_molecule_name:
            continue
        
        # Add small delay between API calls to avoid rate limiting
        time.sleep(0.15)
        
        mol_smiles = name_to_smiles(mol_name)
        mol_fp = get_morgan_fingerprint(mol_smiles)
        
        if mol_fp is not None:
            # Calculate Tanimoto similarity
            similarity = DataStructs.TanimotoSimilarity(target_fp, mol_fp)
            similarities.append((mol_name, similarity))
    
    # Sort by similarity (descending), then alphabetically by name
    similarities.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k molecule names
    result = [name for name, sim in similarities[:top_k]]
    return result
