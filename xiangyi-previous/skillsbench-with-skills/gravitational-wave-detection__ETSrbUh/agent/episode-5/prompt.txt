import numpy as np
import pandas as pd
from pycbc.frame import read_frame
from pycbc.filter import matched_filter
from pycbc.waveform import get_fd_waveform, get_td_waveform, td_approximants, fd_approximants
from pycbc.psd import interpolate, inverse_spectrum_truncation

# Parameters
filename = "/root/data/PyCBC_T2_2.gwf"
channel = "H1:TEST-STRAIN"
mass_range = range(10, 41)
approximants = ["SEOBNRv4_opt", "IMRPhenomD", "TaylorT4"]
flow = 30.0

# Load data
data = read_frame(filename, channel)

# Condition data
data = data.highpass_fir(15, 512)

# Estimate PSD
psd = data.psd(2)
psd = interpolate(psd, data.delta_f)
psd = inverse_spectrum_truncation(psd, int(2 * data.sample_rate), low_frequency_cutoff=flow)

results = []

# Check which approximants are TD or FD
td_apps = td_approximants()
fd_apps = fd_approximants()

for approx in approximants:
    best_snr = 0
    best_total_mass = 0

    # Handle SEOBNRv4_opt if it is not available but SEOBNRv4 is
    actual_approx = approx
    if approx == "SEOBNRv4_opt" and approx not in td_apps and approx not in fd_apps:
        if "SEOBNRv4" in td_apps or "SEOBNRv4" in fd_apps:
            actual_approx = "SEOBNRv4"

    print(f"Processing {approx} using {actual_approx}...")

    for m1 in mass_range:
        for m2 in range(10, m1 + 1):
            if actual_approx in td_apps:
                hp, hc = get_td_waveform(approximant=actual_approx,
                                         mass1=m1, mass2=m2,
                                         delta_t=data.delta_t, f_lower=flow)
                hp.resize(len(data))
                hp = hp.to_frequencyseries()
            elif actual_approx in fd_apps:
                hp, hc = get_fd_waveform(approximant=actual_approx,
                                         mass1=m1, mass2=m2,
                                         delta_f=data.delta_f, f_lower=flow)
            else:
                continue

            hp.resize(len(psd))
            snr = matched_filter(hp, data, psd=psd, low_frequency_cutoff=flow)
            snr = snr.crop(4, 4)

            max_snr = float(abs(snr).max())
            if max_snr > best_snr:
                best_snr = max_snr
                best_total_mass = m1 + m2

    results.append({"approximant": approx, "snr": best_snr, "total_mass": best_total_mass})
    print(f"Finished {approx}: SNR {best_snr}, Total Mass {best_total_mass}")

# Save results
df = pd.DataFrame(results)
df.to_csv("/root/detection_results.csv", index=False)
