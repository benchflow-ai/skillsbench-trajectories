#!/usr/bin/env python3
"""Fuzz driver for arrow library - date/time parsing and manipulation."""

import sys
import atheris

with atheris.instrument_imports():
    import arrow


def TestOneInput(data):
    """Fuzz target for arrow library."""
    fdp = atheris.FuzzedDataProvider(data)

    try:
        # Test arrow.get() with random string input (main parsing function)
        if fdp.ConsumeBool():
            random_string = fdp.ConsumeUnicodeNoSurrogates(fdp.ConsumeIntInRange(0, 1000))
            try:
                arrow.get(random_string)
            except (arrow.parser.ParserError, ValueError, TypeError, OverflowError):
                pass

        # Test arrow.get() with format string
        elif fdp.ConsumeBool():
            date_string = fdp.ConsumeUnicodeNoSurrogates(fdp.ConsumeIntInRange(0, 100))
            format_string = fdp.ConsumeUnicodeNoSurrogates(fdp.ConsumeIntInRange(0, 100))
            try:
                arrow.get(date_string, format_string)
            except (arrow.parser.ParserError, ValueError, TypeError, OverflowError):
                pass

        # Test arrow.get() with timestamp
        elif fdp.ConsumeBool():
            timestamp = fdp.ConsumeFloat()
            try:
                arrow.get(timestamp)
            except (ValueError, TypeError, OverflowError, OSError):
                pass

        # Test Arrow.format() with random format string
        elif fdp.ConsumeBool():
            try:
                arr = arrow.now()
                format_str = fdp.ConsumeUnicodeNoSurrogates(fdp.ConsumeIntInRange(0, 100))
                arr.format(format_str)
            except (ValueError, TypeError, KeyError):
                pass

        # Test Arrow.shift() with random parameters
        elif fdp.ConsumeBool():
            try:
                arr = arrow.now()
                years = fdp.ConsumeIntInRange(-1000, 1000)
                months = fdp.ConsumeIntInRange(-100, 100)
                days = fdp.ConsumeIntInRange(-1000, 1000)
                arr.shift(years=years, months=months, days=days)
            except (ValueError, TypeError, OverflowError):
                pass

        # Test timezone handling
        else:
            tz_string = fdp.ConsumeUnicodeNoSurrogates(fdp.ConsumeIntInRange(0, 50))
            try:
                arrow.now(tz_string)
            except (ValueError, TypeError, arrow.parser.ParserError):
                pass

    except Exception:
        # Catch any unexpected exceptions that might indicate bugs
        pass


def main():
    atheris.Setup(sys.argv, TestOneInput)
    atheris.Fuzz()


if __name__ == "__main__":
    main()
