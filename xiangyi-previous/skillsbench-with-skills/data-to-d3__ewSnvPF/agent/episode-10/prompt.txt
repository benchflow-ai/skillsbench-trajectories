function createLegend(sectors) {
    const legend = d3.select('#legend');

    sectors.forEach(sector => {
        const item = legend.append('div')
            .attr('class', 'legend-item');

        item.append('div')
            .attr('class', 'legend-color')
            .style('background-color', sectorColors(sector));

        item.append('span')
            .text(sector);
    });
}

function selectStock(ticker) {
    // Toggle selection
    if (selectedTicker === ticker) {
        selectedTicker = null;
    } else {
        selectedTicker = ticker;
    }

    // Update bubble selection
    d3.selectAll('.bubble')
        .classed('selected', d => d.ticker === selectedTicker);

    // Update table selection
    d3.selectAll('#data-table tbody tr')
        .classed('selected', function() {
            return d3.select(this).attr('data-ticker') === selectedTicker;
        });

    // Scroll to selected row in table
    if (selectedTicker) {
        const selectedRow = document.querySelector(`#data-table tr[data-ticker="${selectedTicker}"]`);
        if (selectedRow) {
            selectedRow.scrollIntoView({ behavior: 'smooth', block: 'center' });
        }
    }
}
