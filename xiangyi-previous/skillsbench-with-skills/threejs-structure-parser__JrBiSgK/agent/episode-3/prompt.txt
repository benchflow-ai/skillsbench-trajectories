import * as THREE from 'three';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

function getGeometryData(mesh) {
    const geometry = mesh.geometry;
    mesh.updateMatrix();
    const matrix = mesh.matrix;

    const vertices = [];
    const indices = [];

    const posAttr = geometry.attributes.position;
    const v = new THREE.Vector3();

    for (let i = 0; i < posAttr.count; i++) {
        v.fromBufferAttribute(posAttr, i);
        v.applyMatrix4(matrix);
        vertices.push(v.x, v.y, v.z);
    }

    if (geometry.index) {
        for (let i = 0; i < geometry.index.count; i++) {
            indices.push(geometry.index.getX(i));
        }
    } else {
        for (let i = 0; i < posAttr.count; i++) {
            indices.push(i);
        }
    }

    return { vertices, indices, name: mesh.name };
}

function savePart(group) {
    const partName = group.name;
    const meshes = group.children.filter(c => c.isMesh);
    if (meshes.length === 0) return;

    const partDir = path.join('/root/output/part_meshes', partName);
    if (!fs.existsSync(partDir)) fs.mkdirSync(partDir, { recursive: true });

    let linkVertices = [];
    let linkIndices = [];
    let vertexOffset = 0;

    meshes.forEach(mesh => {
        const data = getGeometryData(mesh);

        // Save individual mesh
        let meshObj = `o ${data.name}\n`;
        for (let i = 0; i < data.vertices.length; i += 3) {
            meshObj += `v ${data.vertices[i]} ${data.vertices[i+1]} ${data.vertices[i+2]}\n`;
        }
        for (let i = 0; i < data.indices.length; i += 3) {
            meshObj += `f ${data.indices[i] + 1} ${data.indices[i+1] + 1} ${data.indices[i+2] + 1}\n`;
        }
        fs.writeFileSync(path.join(partDir, mesh.name + '.obj'), meshObj);

        // Accumulate for link
        linkVertices.push(...data.vertices);
        for (let i = 0; i < data.indices.length; i++) {
            linkIndices.push(data.indices[i] + vertexOffset);
        }
        vertexOffset += data.vertices.length / 3;
    });

    // Save link mesh
    let linkObj = `o ${partName}\n`;
    for (let i = 0; i < linkVertices.length; i += 3) {
        linkObj += `v ${linkVertices[i]} ${linkVertices[i+1]} ${linkVertices[i+2]}\n`;
    }
    for (let i = 0; i < linkIndices.length; i += 3) {
        linkObj += `f ${linkIndices[i] + 1} ${linkIndices[i+1] + 1} ${linkIndices[i+2] + 1}\n`;
    }
    fs.writeFileSync(path.join('/root/output/links', partName + '.obj'), linkObj);
}

try {
    const root = createScene();

    function traverse(node) {
        if (node.isGroup) {
            savePart(node);
        }
        node.children.forEach(child => {
            if (child.isGroup) traverse(child);
        });
    }

    traverse(root);
    console.log('Conversion complete.');
} catch (error) {
    console.error('Error during conversion:', error);
    process.exit(1);
}
