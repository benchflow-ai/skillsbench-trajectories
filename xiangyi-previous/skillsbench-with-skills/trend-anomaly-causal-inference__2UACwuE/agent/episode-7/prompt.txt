> def run_univariate_did(data, feature_col, outcome_col):
>     """Run univariate DiD for a single feature"""
>     data = data.copy()
>     data['Post'] = (data['Period'] == 'Treatment').astype(int)
>     data['Feature'] = data[feature_col]
>     data['Interaction'] = data['Post'] * data['Feature']
>
>     # Simple OLS for DiD
>     try:
>         from sklearn.linear_model import LinearRegression
>         X = data[['Post', 'Feature', 'Interaction']].values
>         y = data[outcome_col].values
>
>         if len(y) < 10 or np.std(y) == 0:
>             return None, None
>
>         model = LinearRegression()
>         model.fit(X, y)
>         did_coef = model.coef_[2]  # Interaction coefficient
>
>         # Calculate p-value using t-test approximation
>         n = len(y)
>         y_pred = model.predict(X)
>         residuals = y - y_pred
>         mse = np.sum(residuals**2) / (n - 4)
>
>         # Standard error approximation
>         X_centered = X - X.mean(axis=0)
>         var_coef = mse * np.linalg.pinv(X_centered.T @ X_centered)
>         se = np.sqrt(np.abs(var_coef[2, 2])) if var_coef.shape[0] > 2 else 1
>
>         if se > 0:
>             t_stat = did_coef / se
>             p_value = 2 * (1 - stats.t.cdf(abs(t_stat), n - 4))
>         else:
>             p_value = 1.0
>
>         return did_coef, p_value
>     except:
>         return None, None
