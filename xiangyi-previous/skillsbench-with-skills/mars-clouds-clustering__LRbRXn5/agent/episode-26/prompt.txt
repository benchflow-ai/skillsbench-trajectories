def run_search(params):
    ms, eps, sw, citsci_grouped, expert_grouped, all_expert_files = params
    f1_scores = []
    deltas = []
    for file_rad in all_expert_files:
        c_pts = citsci_grouped.get(file_rad, np.array([]))
        e_pts = expert_grouped.get(file_rad, np.array([]))
        f1, delta = evaluate_image(c_pts, e_pts, ms, eps, sw)
        f1_scores.append(f1)
        if not np.isnan(delta):
            deltas.append(delta)
    return (np.mean(f1_scores), np.mean(deltas) if deltas else np.nan, ms, eps, sw)
