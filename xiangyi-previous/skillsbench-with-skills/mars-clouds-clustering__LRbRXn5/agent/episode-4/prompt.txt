def greedy_match(cluster_centroids, expert_points, max_dist=100):
    if len(cluster_centroids) == 0 or len(expert_points) == 0:
        return 0, []
    dists = cdist(cluster_centroids, expert_points, metric='euclidean')
    matched_expert_indices = set()
    matched_cluster_indices = set()
    matches = []
    flat_dists = []
    for i in range(len(cluster_centroids)):
        for j in range(len(expert_points)):
            if dists[i, j] <= max_dist:
                flat_dists.append((dists[i, j], i, j))
    flat_dists.sort()
    for d, i, j in flat_dists:
        if i not in matched_cluster_indices and j not in matched_expert_indices:
            matched_cluster_indices.add(i)
            matched_expert_indices.add(j)
            matches.append(d)
    return len(matches), matches

def evaluate_image(citsci_points, expert_points, min_samples, epsilon, shape_weight):
    if len(citsci_points) == 0:
        return 0.0, np.nan
    def custom_dist(a, b):
        dx = a[0] - b[0]
        dy = a[1] - b[1]
        return np.sqrt((shape_weight * dx)**2 + ((2 - shape_weight) * dy)**2)
    # Precompute distance matrix for custom metric
    # Using cdist with a lambda is slow, but for DBSCAN we need a distance matrix or a metric
    # Given we use the same points, we can compute it once.
    # However, sklearn DBSCAN with metric callable is fine if the number of points is small.
    # Let's use precomputed distance matrix to be faster.
    from scipy.spatial.distance import pdist, squareform

    # custom distance between point i and j
    # d(a, b) = sqrt((w * Δx)² + ((2 - w) * Δy)²)
    # We can scale the coordinates and then use standard Euclidean distance
    scaled_points = citsci_points * np.array([shape_weight, 2 - shape_weight])
    # Wait, the formula is (w * dx)^2 + ((2-w) * dy)^2
    # This is equivalent to Euclidean distance on points (w*x, (2-w)*y)
    scaled_citsci = citsci_points.copy()
    scaled_citsci[:, 0] *= shape_weight
    scaled_citsci[:, 1] *= (2 - shape_weight)

    db = DBSCAN(eps=epsilon, min_samples=min_samples, metric='euclidean').fit(scaled_citsci)
    labels = db.labels_
    unique_labels = set(labels) - {-1}
    cluster_centroids = []
    for label in unique_labels:
        cluster_centroids.append(citsci_points[labels == label].mean(axis=0))
    cluster_centroids = np.array(cluster_centroids)
    num_matches, matches = greedy_match(cluster_centroids, expert_points)
    precision = num_matches / len(cluster_centroids) if len(cluster_centroids) > 0 else 0
    recall = num_matches / len(expert_points) if len(expert_points) > 0 else 0
    f1 = 2 * precision * recall / (precision + recall) if (precision + recall) > 0 else 0
    delta = np.mean(matches) if matches else np.nan
    return f1, delta

def run_search(params):
    ms, eps, sw, citsci_grouped, expert_grouped, all_expert_files = params
    f1_scores = []
    deltas = []
    for file_rad in all_expert_files:
        c_pts = citsci_grouped.get(file_rad, np.array([]))
        e_pts = expert_grouped.get(file_rad, np.array([]))
        f1, delta = evaluate_image(c_pts, e_pts, ms, eps, sw)
        f1_scores.append(f1)
        if not np.isnan(delta):
            deltas.append(delta)
    return (np.mean(f1_scores), np.mean(deltas) if deltas else np.nan, ms, eps, sw)
