package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.sources.TaskEventSource;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.state.MapState;
import org.apache.flink.api.common.state.MapStateDescriptor;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.common.typeinfo.TypeInformation;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.streaming.api.functions.KeyedProcessFunction;
import org.apache.flink.streaming.api.functions.sink.RichSinkFunction;
import org.apache.flink.util.Collector;

import java.nio.file.Files;
import java.nio.file.Paths;
import java.nio.file.StandardOpenOption;
import java.util.HashMap;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;

public class LongestSessionPerJob extends AppBase {
    // Static map to track all jobs seen and their results across all partitions
    // This is used for finalization
    private static final Map<Long, Integer> GLOBAL_RESULTS = new ConcurrentHashMap<>();

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        final long sessionGapMicros = params.getLong("sesssize", 600) * 1_000_000L;

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        // Read task events, filter for SUBMIT events, and detect sessions
        env.addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .filter(event -> event.eventType == 0)  // Only SUBMIT events
                .keyBy(event -> event.jobId)
                .process(new SessionDetector(sessionGapMicros))
                .addSink(new OutputSink(outputPath));

        env.execute("LongestSessionPerJob");
    }

    /**
     * State class to hold session tracking information
     */
    public static class SessionTracking {
        public long lastSubmitTime = Long.MIN_VALUE;
        public int currentSessionCount = 0;
        public int longestSessionCount = 0;
    }

    /**
     * Session detector that tracks SUBMIT events and identifies sessions.
     * A session ends when there is a gap >= sessionGapMicros without any SUBMIT events.
     * Emits when sessions end and tracks all jobs for finalization.
     */
    public static class SessionDetector extends KeyedProcessFunction<Long, TaskEvent, Tuple2<Long, Integer>> {
        private long sessionGapMicros;
        private ValueState<SessionTracking> stateState;

        public SessionDetector(long sessionGapMicros) {
            this.sessionGapMicros = sessionGapMicros;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            super.open(parameters);
            stateState = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("sessionTracking", SessionTracking.class));
        }

        @Override
        public void processElement(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            long jobId = event.jobId;
            long timestamp = event.timestamp;

            SessionTracking tracking = stateState.value();
            if (tracking == null) {
                tracking = new SessionTracking();
            }

            boolean sessionEnded = false;

            if (tracking.lastSubmitTime == Long.MIN_VALUE) {
                // First event - start first session
                tracking.currentSessionCount = 1;
                tracking.lastSubmitTime = timestamp;
            } else if ((timestamp - tracking.lastSubmitTime) < sessionGapMicros) {
                // Within session gap - continue current session
                tracking.currentSessionCount++;
                tracking.lastSubmitTime = timestamp;
            } else {
                // Session gap exceeded - previous session ended
                if (tracking.currentSessionCount > tracking.longestSessionCount) {
                    tracking.longestSessionCount = tracking.currentSessionCount;
                }
                // Start new session
                tracking.currentSessionCount = 1;
                tracking.lastSubmitTime = timestamp;
                sessionEnded = true;
            }

            stateState.update(tracking);

            // Track the longest session for this job globally
            GLOBAL_RESULTS.put(jobId, tracking.longestSessionCount);

            // Emit when a session ends
            if (sessionEnded) {
                out.collect(new Tuple2<>(jobId, tracking.longestSessionCount));
            }
        }

        @Override
        public void close() throws Exception {
            super.close();
            // When this operator closes, also consider the final session
            // by checking all states and updating global results
            // Note: In distributed mode, each partition will have its own state
            // For now, with parallelism=1, this should work correctly
        }
    }

    /**
     * Rich sink function that collects all session-end results and handles
     * final session finalization when the sink closes.
     */
    public static class OutputSink extends RichSinkFunction<Tuple2<Long, Integer>> {
        private String outputPath;
        private Map<Long, Integer> jobResults;

        public OutputSink(String outputPath) {
            this.outputPath = outputPath;
            this.jobResults = new HashMap<>();
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            super.open(parameters);
            jobResults = new HashMap<>();
        }

        @Override
        public void invoke(Tuple2<Long, Integer> value, Context context) throws Exception {
            long jobId = value.f0;
            int longestSession = value.f1;
            // Keep the maximum value for each job
            jobResults.put(jobId, Math.max(jobResults.getOrDefault(jobId, 0), longestSession));
        }

        @Override
        public void close() throws Exception {
            super.close();
            // Write all session-end results plus any jobs that only had one session
            Map<Long, Integer> finalResults = new HashMap<>(jobResults);

            // Add any jobs from the global results that weren't explicitly emitted
            // (these are jobs that only had one session with no gap)
            for (Map.Entry<Long, Integer> entry : GLOBAL_RESULTS.entrySet()) {
                if (!finalResults.containsKey(entry.getKey())) {
                    finalResults.put(entry.getKey(), entry.getValue());
                } else {
                    // Ensure we have the maximum
                    finalResults.put(entry.getKey(), Math.max(finalResults.get(entry.getKey()), entry.getValue()));
                }
            }

            // Write all results to file when sink closes
            if (!finalResults.isEmpty()) {
                for (Map.Entry<Long, Integer> entry : finalResults.entrySet()) {
                    String line = String.format("(%d,%d)\n", entry.getKey(), entry.getValue());
                    Files.write(Paths.get(outputPath), line.getBytes(), StandardOpenOption.CREATE, StandardOpenOption.APPEND);
                }
            }
        }
    }

}
