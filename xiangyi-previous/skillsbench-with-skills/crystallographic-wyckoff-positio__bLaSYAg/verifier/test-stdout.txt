from __future__ import annotations

from typing import Any, Dict
import re
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_rational_str(x: float, max_den: int = 12) -> str:
    """Convert float to rational string with denominator <= max_den."""
    # wrap into [0,1)
    x = x % 1.0
    if abs(x) < 1e-8:
        return "0"
    frac = Fraction(x).limit_denominator(max_den)
    # If frac is 1 (e.g., x very close to 1), represent as 0
    if frac.numerator == frac.denominator:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> Dict[str, Dict] | Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position.
    """
    # Read structure from CIF
    structure = Structure.from_file(filepath)
    # Symmetry analysis to get Wyckoff positions
    sga = SpacegroupAnalyzer(structure, symprec=1e-3)
    symm_structure = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list] = {}

    for symbol, sites in zip(symm_structure.wyckoff_symbols, symm_structure.equivalent_sites):
        m = re.match(r"(\d+)([A-Za-z]+)", symbol)
        if not m:
            continue
        multiplicity = int(m.group(1))
        letter = m.group(2)
        # store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity
        # first site's fractional coordinates
        coords = sites[0].frac_coords
        wyckoff_coordinates_dict[letter] = [
            _float_to_rational_str(float(c)) for c in coords
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
