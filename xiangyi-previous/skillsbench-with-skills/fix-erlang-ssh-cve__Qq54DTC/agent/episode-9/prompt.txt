retry_fun(_, undefined, _) ->
    ok;
retry_fun(User, Reason, #data{ssh_params = #ssh{opts = Opts,
                                                peer = {_,Peer}
                                               }}) ->
    {Tag,Info} =
        case Reason of
            {error, Error} ->
                {failfun, Error};
            _ ->
                {infofun, Reason}
        end,
    Fun = ?GET_OPT(Tag, Opts),
    try erlang:fun_info(Fun, arity)
    of
        {arity, 2} -> %% Backwards compatible
            catch Fun(User, Info);
        {arity, 3} ->
            catch Fun(User, Peer, Info);
        _ ->
            ok
    catch
        _:_ ->
            ok
    end.
