"""
Solution for finding top-k similar chemicals using Tanimoto similarity with Morgan fingerprints.
"""

import time
import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file containing molecule names.

    Returns:
        List of molecule names extracted from the PDF.
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Each line contains a molecule name
                for line in text.strip().split('\n'):
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def get_smiles_from_name(molecule_name: str, max_retries: int = 3, base_delay: float = 1.0) -> str | None:
    """
    Convert a chemical name to its SMILES representation using PubChem.

    Args:
        molecule_name: Name of the chemical compound.
        max_retries: Maximum number of retry attempts for API errors.
        base_delay: Base delay in seconds for exponential backoff.

    Returns:
        Canonical SMILES string or None if not found.
    """
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(molecule_name, 'name')
            if compounds:
                return compounds[0].smiles
            return None
        except pcp.PubChemHTTPError:
            if attempt < max_retries - 1:
                delay = base_delay * (2 ** attempt)
                time.sleep(delay)
            else:
                return None
        except Exception:
            return None
    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string of the molecule.
        radius: Radius for Morgan fingerprint (default: 2).
        use_chirality: Whether to include chirality in fingerprint (default: True).

    Returns:
        Morgan fingerprint bit vector or None if molecule is invalid.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    generator = rdFingerprintGenerator.GetMorganGenerator(
        radius=radius,
        includeChirality=use_chirality
    )
    return generator.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a PDF file.

    Uses Morgan fingerprints (radius=2, including chirality) with Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical ordering
    for ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to the PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples containing (molecule_name, similarity_score) for the top-k
        most similar molecules, sorted by descending similarity and alphabetically
        for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities for all molecules in the pool
    similarities = []
    for mol_name in pool_molecules:
        # Get SMILES for pool molecule
        mol_smiles = get_smiles_from_name(mol_name)
        if mol_smiles is None:
            continue

        # Compute fingerprint
        mol_fp = compute_morgan_fingerprint(mol_smiles)
        if mol_fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, mol_fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) >= 4:
        target = sys.argv[1]
        pdf_path = sys.argv[2]
        k = int(sys.argv[3])
    else:
        # Default example
        target = "Aspirin"
        pdf_path = "molecules.pdf"
        k = 5

    print(f"Finding top {k} molecules similar to '{target}'...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for i, (name, sim) in enumerate(results, 1):
        print(f"  {i}. {name}: {sim:.4f}")
