from __future__ import annotations

import re
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _normalize_fractional(value: float, tol: float = 1e-8) -> float:
    if abs(value - 1.0) < tol:
        return 1.0
    if abs(value) < tol:
        return 0.0
    if 0.0 <= value <= 1.0:
        return value
    value = value % 1.0
    if abs(value - 1.0) < tol:
        return 1.0
    if abs(value) < tol:
        return 0.0
    return value


def _format_fraction(value: float, max_denominator: int = 12, tol: float = 1e-8) -> str:
    normalized = _normalize_fractional(value, tol)
    if abs(normalized) < tol:
        return "0"
    if abs(normalized - 1.0) < tol:
        return "1"
    frac = Fraction(normalized).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def _merge_multiplicity(
    multiplicity_dict: dict[str, Any], letter: str, multiplicity: int
) -> None:
    if letter not in multiplicity_dict:
        multiplicity_dict[letter] = multiplicity
        return
    existing = multiplicity_dict[letter]
    if isinstance(existing, list):
        existing.append(multiplicity)
    else:
        multiplicity_dict[letter] = [existing, multiplicity]


def _merge_coord(
    coord_dict: dict[str, Any], letter: str, coordinates: list[str]
) -> None:
    if letter not in coord_dict:
        coord_dict[letter] = coordinates
        return
    existing = coord_dict[letter]
    if isinstance(existing, list) and existing and all(
        isinstance(entry, str) for entry in existing
    ):
        coord_dict[letter] = [existing, coordinates]
    elif isinstance(existing, list):
        existing.append(coordinates)
    else:
        coord_dict[letter] = [existing, coordinates]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> (dict[str, dict] | dict[str, Any]):
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    symm_struct = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict: dict[str, Any] = {}
    wyckoff_coordinates_dict: dict[str, Any] = {}

    for symbol, group in zip(
        symm_struct.wyckoff_symbols, symm_struct.equivalent_sites
    ):
        match = re.match(r"^\s*(\d+)([A-Za-z])", symbol)
        if not match:
            continue
        letter = match.group(2).lower()
        multiplicity = len(group)
        coord = [_format_fraction(float(val)) for val in group[0].frac_coords]

        _merge_multiplicity(wyckoff_multiplicity_dict, letter, multiplicity)
        _merge_coord(wyckoff_coordinates_dict, letter, coord)

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
