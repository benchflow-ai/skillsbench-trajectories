class TokenFunctor[T](val value: T) {
  def map[U](func: T => U): TokenFunctor[U] = new TokenFunctor(func(value))

  def flatMap[U](func: T => TokenFunctor[U]): TokenFunctor[U] = func(value)

  def getOrElse(default: => T): T = if (value != null) value else default

  def get: T = value
}

class TokenMonad[T](value: T) extends TokenFunctor[T](value) {
  def ap[U](funcWrapped: TokenMonad[T => U]): TokenMonad[U] =
    new TokenMonad(funcWrapped.value(value))
}

object TokenMonad {
  def pure[T](value: T): TokenMonad[T] = new TokenMonad(value)
}

class JsonTokenizer(pretty: Boolean = false) {
  private val printer = if (pretty) Printer.spaces2 else Printer.noSpaces

  def tokenize(value: Json): Token = {
    val jsonStr = printer.print(value)
    Token(jsonStr, TokenType.STRUCTURED, Map("json" -> true))
  }

  def tokenizePath(value: Json, path: String): Option[Token] = {
    val parts = path.split("\\.").toList.filter(_.nonEmpty)
    var current: Option[Json] = Some(value)

    parts.foreach { part =>
      current = current.flatMap { json =>
        json.asObject.flatMap(_.apply(part)).orElse {
          json.asArray.flatMap { arr =>
            if (part.forall(_.isDigit)) arr.lift(part.toInt) else None
          }
        }
      }
    }

    current.map(tokenize)
  }
}
