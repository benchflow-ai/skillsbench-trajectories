    val results = ListBuffer.empty[(String, Int, Int)]

    text.split("\\s+").filter(_.nonEmpty).foreach { word =>
      val start = text.indexOf(word, currentPos)

      processToken(word).foreach { processed =>
        results += ((processed, start, end))
