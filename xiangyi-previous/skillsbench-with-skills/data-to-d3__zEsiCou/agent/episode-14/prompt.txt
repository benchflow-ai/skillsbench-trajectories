function highlightBubbleAndRow(ticker) {
    // Clear previous highlights
    d3.selectAll('circle.highlighted').classed('highlighted', false);
    d3.selectAll('tr.highlighted').classed('highlighted', false);

    // Highlight bubble
    d3.selectAll(`[data-ticker="${ticker}"] circle`)
        .classed('highlighted', true);

    // Highlight table row
    const row = d3.select(`tr[data-ticker="${ticker}"]`);
    row.classed('highlighted', true);

    // Scroll table row into view
    const element = document.querySelector(`tr[data-ticker="${ticker}"]`);
    if (element) {
        element.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
    }
}
