final class MutableTokenBatch {
  private val _tokens: mutable.ListBuffer[Token] = mutable.ListBuffer.empty
  private var _processed: Boolean = false

  def tokens: List[Token] = _tokens.toList

  def add(token: Token): Unit = {
    if (_processed) throw new RuntimeException("Batch already processed")
    _tokens += token
  }

  def markProcessed(): Unit = { _processed = true }
  def isProcessed: Boolean = _processed
}

class StringTokenizer(
  lowercase: Boolean = false,
  stripWhitespace: Boolean = true,
  maxLength: Option[Int] = None
) extends BaseTokenizer[String] {

  override def tokenize(value: String): Token = {
    var processed = value
    if (stripWhitespace) processed = processed.trim
    if (lowercase) processed = processed.toLowerCase
    processed = maxLength.filter(_ < processed.length).fold(processed)(processed.take)

    Token(
      value = processed,
      tokenType = TokenType.String,
      metadata = Map(
        "original_length" -> value.length,
        "processed_length" -> processed.length
      )
    )
  }
}

class NumericTokenizer(
  precision: Int = 2,
  useScientificNotation: Boolean = false
) extends BaseTokenizer[BigDecimal] {

  override def tokenize(value: BigDecimal): Token = {
    val formatted = if (useScientificNotation) {
      s"%.${precision}E".format(value.doubleValue)
    } else {
      value.setScale(precision, BigDecimal.RoundingMode.HALF_UP).toString
    }

    Token(
      value = formatted,
      tokenType = TokenType.Numeric,
      metadata = Map(
        "original_value" -> value,
        "precision" -> precision,

    )
  }

  def tokenizeInt(value: Int): Token = tokenize(BigDecimal(value))
  def tokenizeDouble(value: Double): Token = tokenize(BigDecimal(value))
}

class UniversalTokenizer extends BaseTokenizer[Any] {
  private val stringTokenizer = new StringTokenizer()
  private val numericTokenizer = new NumericTokenizer()
  private val temporalTokenizer = new TemporalTokenizer()

    case None => Token(value = "null", tokenType = TokenType.Null)

    case s: String => stringTokenizer.tokenize(s)
    case b: Boolean => Token(b.toString, TokenType.String, Map("boolean_value" -> b))

        tokenType = TokenType.Binary,
        metadata = Map("byte_length" -> bytes.length)

    case seq: Seq[_] =>

      Token(items.mkString("[", ", ", "]"), TokenType.Structured,
        Map("structure_type" -> "sequence", "item_count" -> seq.length))

        Map("structure_type" -> "mapping", "key_count" -> map.size))
    case t: Tokenizable => Token(t.toToken, TokenType.String, Map("source" -> "tokenizable"))
    case other => Token(other.toString, TokenType.String,
        Map("fallback" -> true, "original_type" -> other.getClass.getSimpleName))
  }
}

class WhitespaceTokenizer(
  lowercase: Boolean = false,
  minLength: Int = 0,
  maxLength: Option[Int] = None,
  stripPunctuation: Boolean = false
) {
  private val punctuation: Set[Char] = ".,!?;:'\"()[]{}".toSet

  private def processToken(word: String): Option[String] = {
    var processed = word
    if (stripPunctuation) {
      processed = processed.dropWhile(punctuation).reverse.dropWhile(punctuation).reverse
    }
    if (lowercase) processed = processed.toLowerCase
    if (processed.length < minLength) return None
    processed = maxLength.filter(_ < processed.length).fold(processed)(processed.take)
    if (processed.isEmpty) None else Some(processed)
  }

  def tokenize(text: String): List[Token] = {
    text.split("\\s+").filter(_.nonEmpty).zipWithIndex.flatMap { case (word, i) =>
      processToken(word).map { processed =>
        Token(processed, TokenType.String, Map("position" -> i, "original" -> word))
      }
    }.toList
  }

  def tokenizeToStrings(text: String): List[String] = tokenize(text).map(_.value)

  def tokenizeWithPositions(text: String): List[(String, Int, Int)] = {
    val result = mutable.ListBuffer[(String, Int, Int)]()
    var currentPos = 0
    for (word <- text.split("\\s+").filter(_.nonEmpty)) {
      val start = text.indexOf(word, currentPos)
      val end = start + word.length
      processToken(word).foreach(p => result += ((p, start, end)))
      currentPos = end
    }
    result.toList
  }

  def countTokens(text: String): Int = tokenize(text).length
}

class TokenizerBuilder[T] {
  private val normalizers: mutable.ListBuffer[String => String] = mutable.ListBuffer.empty
  private val validators: mutable.ListBuffer[T => Boolean] = mutable.ListBuffer.empty
  private var _metadata: Map[String, Any] = Map.empty

  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] = {
    normalizers += normalizer
    this
  }

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] = {
    validators += validator
    this
  }

  def withMetadata(entries: (String, Any)*): TokenizerBuilder[T] = {
    _metadata = _metadata ++ entries.toMap
    this
  }

  def build(): T => Token = { value: T =>
    validators.foreach { validator =>
      if (!validator(value)) {
        throw new IllegalArgumentException(s"Validation failed for $value")
      }
    }

    var strValue = value.toString
    normalizers.foreach { normalizer =>
      strValue = normalizer(strValue)
    }

    Token(strValue, TokenType.String, _metadata)
  }
