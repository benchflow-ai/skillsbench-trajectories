  def tokenize(value: Any): Token = {

        Token(nullToken, TokenType.Null, Map("is_null" -> true))

      case Some(v) =>
        tokenize(v)

      case s: String =>
        defaultStringTokenizer.tokenize(s)

      case _: Int | _: Long | _: Float | _: Double | _: BigDecimal =>
        defaultNumericTokenizer.tokenize(value)

      case _: LocalDate | _: LocalDateTime =>
        defaultTemporalTokenizer.tokenize(value)

      case seq: Seq[_] =>
        val tokens = seq.map(tokenize)
        val concatenated = tokens.map(_.value).mkString(",")
        Token(s"[$concatenated]", TokenType.Structured, Map("element_count" -> seq.length))

      case list: List[_] =>
        val tokens = list.map(tokenize)
        val concatenated = tokens.map(_.value).mkString(",")
        Token(s"[$concatenated]", TokenType.Structured, Map("element_count" -> list.length))

      case map: Map[_, _] =>
        val pairs = map.map { case (k, v) =>
          s"${tokenize(k).value}:${tokenize(v).value}"
        }.mkString(",")

      case t: Tokenizable =>
        Token(t.toToken, TokenType.String, Map("custom_tokenizable" -> true))
