"""
Crystal structure analysis module using pymatgen.
Analyzes Wyckoff position multiplicities and calculates coordinates
for the first atom of each Wyckoff position.
"""

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from collections import defaultdict
from typing import Any


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff positions in a crystal structure from a CIF file.

    This function reads a crystallographic information file (CIF) and analyzes
    the crystal structure to determine:
    1. Wyckoff position multiplicities (number of atoms at each position)
    2. Approximate coordinates for the first atom of each Wyckoff position,
       expressed as rational numbers

    Args:
        filepath: Path to the CIF file containing crystal structure data

    Returns:
        A dictionary with two keys:
        - "wyckoff_multiplicity_dict": Maps Wyckoff labels to their multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff labels to coordinate lists,
                                      with fractions expressed as strings

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result["wyckoff_multiplicity_dict"]
        {'a': 4, 'c': 8}
        >>> result["wyckoff_coordinates_dict"]["a"]
        ['0', '1/2', '1/2']
    """

    # Load the crystal structure from CIF file
    structure = Structure.from_file(filepath)

    # Get symmetry information using SpacegroupAnalyzer
    analyzer = SpacegroupAnalyzer(structure)
    symmetry_dataset = analyzer.get_symmetry_dataset()

    # Extract Wyckoff symbols and standard positions
    wyckoffs = symmetry_dataset.wyckoffs
    positions = symmetry_dataset.std_positions

    # Count multiplicities for each Wyckoff position
    wyckoff_multiplicity_dict = defaultdict(int)
    wyckoff_first_coords = {}

    for i, wyckoff_label in enumerate(wyckoffs):
        wyckoff_multiplicity_dict[wyckoff_label] += 1

        # Store the coordinates of the first occurrence of each Wyckoff position
        if wyckoff_label not in wyckoff_first_coords:
            wyckoff_first_coords[wyckoff_label] = positions[i]

    # Convert multiplicities dict to regular dict (from defaultdict)
    wyckoff_multiplicity_dict = dict(wyckoff_multiplicity_dict)

    # Convert coordinates to rational number strings
    wyckoff_coordinates_dict = {}
    for wyckoff_label in sorted(wyckoff_first_coords.keys()):
        coords = wyckoff_first_coords[wyckoff_label]
        frac_coords = [_coord_to_fraction_string(c) for c in coords]
        wyckoff_coordinates_dict[wyckoff_label] = frac_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _coord_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating point coordinate to a fraction string representation.

    Args:
        value: The coordinate value as a float
        max_denominator: Maximum denominator for the fraction representation

    Returns:
        A string representation of the fraction. Integer values are returned
        as strings without denominators (e.g., "0", "1").
        Fractional values are returned as "numerator/denominator" (e.g., "1/2").
    """
    # Convert float to Fraction and limit denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return as string
    if frac.denominator == 1:
        # Integer case
        return str(frac.numerator)
    else:
        # Fractional case
        return f"{frac.numerator}/{frac.denominator}"

    )
    print("Wyckoff Multiplicities:", result["wyckoff_multiplicity_dict"])
    print("Wyckoff Coordinates:", result["wyckoff_coordinates_dict"])
