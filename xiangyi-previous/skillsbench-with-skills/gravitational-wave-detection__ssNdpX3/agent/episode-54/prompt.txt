import numpy as np
from pycbc.frame import read_frame
from pycbc.filter import matched_filter
from pycbc.psd import interpolate
from pycbc.waveform import get_td_waveform

# Read and condition data
strain = read_frame('/root/data/PyCBC_T2_2.gwf','H1:TEST-STRAIN')
strain = strain.crop(4,4)
strain = strain.highpass_fir(15, 8)

# Resample to 2048 Hz
strain = strain.resample(1/2048)

# 2s window around event time
center = 106.8
seg = strain.time_slice(center-1, center+1)

# PSD from segment
psd = seg.psd(2)
psd = interpolate(psd, seg.delta_f)

# Precompute data frequency series
seg_freq = seg.to_frequencyseries()

approximants = ["SEOBNRv4_opt", "IMRPhenomD", "TaylorT4"]
results = []

masses = range(10,41)

for approx in approximants:
    best_snr = 0.0
    best_mass = None
    print('Processing', approx)
    for m1 in masses:
        for m2 in masses:
            if m2 > m1:
                continue
            try:
                hp, hc = get_td_waveform(approximant=approx,
                                         mass1=m1, mass2=m2,
                                         delta_t=seg.delta_t,
                                         f_lower=30)
            except Exception:
                continue
            hp.resize(len(seg))
            snr = matched_filter(hp, seg_freq, psd=psd, low_frequency_cutoff=30)
            snr = snr.crop(0.2,0.2)
            peak = np.abs(snr).max()
            if peak > best_snr:
                best_snr = peak
                best_mass = m1 + m2
    results.append((approx, best_snr, best_mass))

with open('/root/detection_results.csv','w') as f:
    f.write('approximant,snr,total_mass\n')
    for approx, snr, tmass in results:
        f.write(f"{approx},{snr},{tmass}\n")

print('Done, results written.')
