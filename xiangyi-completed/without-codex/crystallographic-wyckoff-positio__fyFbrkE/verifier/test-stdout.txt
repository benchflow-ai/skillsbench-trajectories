_ELEMENT_TO_Z = {symbol: idx + 1 for idx, symbol in enumerate(_ELEMENT_SYMBOLS)}


def _strip_cif_comment(line: str) -> str:
    in_single = False
    in_double = False
    for idx, ch in enumerate(line):
        if ch == "'" and not in_double:
            in_single = not in_single
        elif ch == '"' and not in_single:
            in_double = not in_double
        elif ch == "#" and not in_single and not in_double:
            return line[:idx]
    return line


def _tokenize_cif_line(line: str) -> list[str]:
    lexer = shlex.shlex(line, posix=True)
    lexer.whitespace_split = True
    lexer.commenters = ""
    return list(lexer)


def _parse_cif(filepath: str) -> tuple[dict[str, str], list[tuple[list[str], list[list[str]]]]]:
    text = Path(filepath).read_text().splitlines()
    values: dict[str, str] = {}
    loops: list[tuple[list[str], list[list[str]]]] = []

    i = 0
    while i < len(text):
        raw = _strip_cif_comment(text[i]).strip()
        if not raw:
            i += 1
            continue

        if raw.startswith("loop_"):
            i += 1
            columns: list[str] = []
            while i < len(text):
                line = _strip_cif_comment(text[i]).strip()
                if not line:
                    i += 1
                    continue
                if line.startswith("_"):
                    tokens = _tokenize_cif_line(line)
                    if tokens:
                        columns.append(tokens[0])
                    i += 1
                    continue
                break

            data_tokens: list[str] = []
            while i < len(text):
                line = _strip_cif_comment(text[i]).strip()
                if not line:
                    i += 1
                    continue
                if line.startswith("_") or line.startswith("loop_") or line.startswith("data_"):
                    break
                data_tokens.extend(_tokenize_cif_line(line))
                i += 1

            rows: list[list[str]] = []
            if columns:
                for j in range(0, len(data_tokens), len(columns)):
                    row = data_tokens[j : j + len(columns)]
                    if len(row) == len(columns):
                        rows.append(row)
            loops.append((columns, rows))
            continue

        if raw.startswith("_"):
            tokens = _tokenize_cif_line(raw)
            key = tokens[0]
            if len(tokens) > 1:
                values[key] = " ".join(tokens[1:])
            else:
                # value on next non-empty line
                i += 1
                while i < len(text):
                    line = _strip_cif_comment(text[i]).strip()
                    if line:
                        values[key] = _tokenize_cif_line(line)[0]
                        break
                    i += 1
            i += 1
            continue

        i += 1

    return values, loops


def _parse_number(token: str) -> float | None:
    if token in {"?", ".", ""}:
        return None
    token = token.strip()
    if "/" in token and re.match(r"^[+-]?\d+/\d+$", token):
        return float(Fraction(token))
    # Remove parentheses for standard uncertainty
    cleaned = re.sub(r"\(.*\)$", "", token)
    match = re.match(r"^[+-]?(?:\d+(?:\.\d*)?|\.\d+)(?:[eE][+-]?\d+)?$", cleaned)
    if not match:
        return None
    try:
        return float(cleaned)
    except ValueError:
        return None


def _element_from_token(token: str) -> str | None:
    match = re.match(r"([A-Z][a-z]?)", token)
    if not match:
        return None
    symbol = match.group(1)
    return symbol if symbol in _ELEMENT_TO_Z else None


def _lattice_from_parameters(
    a: float,
    b: float,
    c: float,
    alpha: float,
    beta: float,
    gamma: float,
) -> np.ndarray:
    alpha_r = math.radians(alpha)
    beta_r = math.radians(beta)
    gamma_r = math.radians(gamma)

    v_a = np.array([a, 0.0, 0.0], dtype=float)
    v_b = np.array([b * math.cos(gamma_r), b * math.sin(gamma_r), 0.0], dtype=float)

    cx = c * math.cos(beta_r)
    if abs(math.sin(gamma_r)) < 1e-12:
        cy = 0.0
    else:
        cy = c * (math.cos(alpha_r) - math.cos(beta_r) * math.cos(gamma_r)) / math.sin(gamma_r)
    cz_sq = c * c - cx * cx - cy * cy
    cz = math.sqrt(max(cz_sq, 0.0))

    v_c = np.array([cx, cy, cz], dtype=float)
    return np.vstack([v_a, v_b, v_c])


def _format_fraction(value: float, max_denominator: int = 12) -> str:
    value = value % 1.0
    if abs(value - 1.0) < 1e-10:
        value = 0.0
    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and approximate coordinates for the
    first atom of each Wyckoff position.
    """
    values, loops = _parse_cif(filepath)

    try:
        a = _parse_number(values.get("_cell_length_a", ""))
        b = _parse_number(values.get("_cell_length_b", ""))
        c = _parse_number(values.get("_cell_length_c", ""))
        alpha = _parse_number(values.get("_cell_angle_alpha", ""))
        beta = _parse_number(values.get("_cell_angle_beta", ""))
        gamma = _parse_number(values.get("_cell_angle_gamma", ""))
    except Exception:
        a = b = c = alpha = beta = gamma = None

    if None in {a, b, c, alpha, beta, gamma}:
        return {"error": "Missing or invalid cell parameters in CIF."}

    atom_loop = None
    for columns, rows in loops:
        if {
            "_atom_site_fract_x",
            "_atom_site_fract_y",
            "_atom_site_fract_z",
        }.issubset(columns):
            atom_loop = (columns, rows)
            break

    if atom_loop is None:
        return {"error": "No atom site fractional coordinates found in CIF."}

    columns, rows = atom_loop
    ix = columns.index("_atom_site_fract_x")
    iy = columns.index("_atom_site_fract_y")
    iz = columns.index("_atom_site_fract_z")

    symbol_columns = [
        "_atom_site_type_symbol",
        "_atom_site_symbol",
        "_atom_site_label",
    ]
    sym_idx = None
    for name in symbol_columns:
        if name in columns:
            sym_idx = columns.index(name)
            break

    if sym_idx is None:
        return {"error": "No atom site symbol/label column found in CIF."}

    positions: list[list[float]] = []
    numbers: list[int] = []

    for row in rows:
        if len(row) < max(ix, iy, iz, sym_idx) + 1:
            continue
        symbol = _element_from_token(row[sym_idx])
        if symbol is None:
            continue
        x = _parse_number(row[ix])
        y = _parse_number(row[iy])
        z = _parse_number(row[iz])
        if x is None or y is None or z is None:
            continue
        positions.append([x % 1.0, y % 1.0, z % 1.0])
        numbers.append(_ELEMENT_TO_Z[symbol])

    if not positions:
        return {"error": "No valid atomic positions found in CIF."}

    lattice = _lattice_from_parameters(a, b, c, alpha, beta, gamma)
    cell = (lattice, positions, numbers)

    dataset = None
    for symprec in (1e-3, 1e-2, 5e-2, 1e-1):
        try:
            dataset = spglib.get_symmetry_dataset(cell, symprec=symprec)
        except Exception:
            dataset = None
        if dataset is not None:
            break

    if dataset is None:
        return {"error": "Failed to determine symmetry dataset from CIF."}

    wyckoffs = getattr(dataset, "wyckoffs", None)
    if wyckoffs is None:
        wyckoffs = dataset.get("wyckoffs") if isinstance(dataset, dict) else None

    if wyckoffs is None:
        return {"error": "Wyckoff symbols not available in symmetry dataset."}

    wyckoff_multiplicity: dict[str, int] = {}
    wyckoff_coordinates: dict[str, list[str]] = {}

    for idx, letter in enumerate(wyckoffs):
        if letter not in wyckoff_multiplicity:
            wyckoff_multiplicity[letter] = 0
            coord = positions[idx]
            wyckoff_coordinates[letter] = [
                _format_fraction(coord[0]),
                _format_fraction(coord[1]),
                _format_fraction(coord[2]),
            ]
        wyckoff_multiplicity[letter] += 1

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity,
        "wyckoff_coordinates_dict": wyckoff_coordinates,
    }
