import json
import math
import urllib.error
import urllib.parse
import urllib.request
from functools import lru_cache
from typing import List, Tuple

import pdfplumber
from rdkit import Chem, DataStructs, RDLogger
from rdkit.Chem import rdFingerprintGenerator

RDLogger.DisableLog("rdApp.*")

_MORGAN_GENERATOR = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _extract_molecule_names_from_pdf(molecule_pool_filepath: str) -> List[str]:
    """Extracts molecule names from a PDF (one per line), preserving order and removing blanks."""
    names: List[str] = []
    seen = set()
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.split("\n"):
                name = line.strip()
                if not name:
                    continue
                if name in seen:
                    continue
                seen.add(name)
                names.append(name)
    return names


@lru_cache(maxsize=2048)
def _pubchem_name_to_smiles(name: str) -> str | None:
    """Resolve a chemical name to an isomeric SMILES using PubChem PUG REST."""
    encoded = urllib.parse.quote(name)
    url = (
        "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/"
        f"{encoded}/property/IsomericSMILES,CanonicalSMILES,SMILES/JSON"
    )
    try:
        with urllib.request.urlopen(url, timeout=20) as response:
            payload = json.load(response)
    except (urllib.error.HTTPError, urllib.error.URLError, json.JSONDecodeError):
        return None

    try:
        props = payload["PropertyTable"]["Properties"][0]
    except (KeyError, IndexError, TypeError):
        return None

    for key in ("IsomericSMILES", "CanonicalSMILES", "SMILES", "ConnectivitySMILES"):
        value = props.get(key)
        if value:
            return value
    return None


def _smiles_to_fingerprint(smiles: str):
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _MORGAN_GENERATOR.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Tuple[str, float]]:
    """
    Compute top-k most similar molecules (Tanimoto over Morgan fingerprints) from a pool PDF.

    Returns a list of (molecule_name, similarity) sorted by descending similarity, then name.
    """
    if top_k <= 0:
        return []

    target_smiles = _pubchem_name_to_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")

    target_fp = _smiles_to_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not build fingerprint for target: {target_molecule_name}")

    candidates = _extract_molecule_names_from_pdf(molecule_pool_filepath)

    results: List[Tuple[str, float]] = []
    for name in candidates:
        smiles = _pubchem_name_to_smiles(name)
        if not smiles:
            continue
        fp = _smiles_to_fingerprint(smiles)
        if fp is None:
            continue
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        if math.isnan(similarity):
            continue
        results.append((name, float(similarity)))

    results.sort(key=lambda item: (-item[1], item[0].lower()))
    return results[:top_k]


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(
        description="Top-k Tanimoto similarity against a PDF list of molecule names."
    )
    parser.add_argument("target", help="Target molecule name (e.g., Aspirin)")
    parser.add_argument("pdf", help="Path to molecules.pdf")
    parser.add_argument("k", type=int, help="Number of top results to return")
    args = parser.parse_args()

    topk = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.k)
    for name, sim in topk:
        print(f"{name}\t{sim:.4f}")
