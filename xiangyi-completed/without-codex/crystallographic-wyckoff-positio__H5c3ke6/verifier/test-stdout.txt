from __future__ import annotations

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _rational_str(value: float, max_denominator: int = 12, eps: float = 1e-8) -> str:
    """Return a simplified rational string for a fractional coordinate in [0, 1)."""
    # Normalize into [0, 1)
    val = float(value) % 1.0
    if abs(val) < eps or abs(val - 1.0) < eps:
        return "0"

    frac = Fraction(val).limit_denominator(max_denominator)
    # 1 is equivalent to 0 in fractional coordinates
    if frac.numerator == frac.denominator:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff multiplicities and approximate coordinates for the first atom
    of each Wyckoff position in a CIF file.

    Returns a dict with:
      - "wyckoff_multiplicity_dict": {letter: multiplicity}
      - "wyckoff_coordinates_dict": {letter: [x, y, z]}  # strings
    """
    try:
        structure = Structure.from_file(filepath)
        sga = SpacegroupAnalyzer(structure, symprec=1e-3, angle_tolerance=5)
        dataset = sga.get_symmetry_dataset()
    except Exception as exc:  # pragma: no cover - best-effort error reporting
        return {"error": f"Failed to parse or analyze CIF: {exc}"}

    wyckoffs = dataset.get("wyckoffs", [])
    equivalent_atoms = dataset.get("equivalent_atoms", [])
    if wyckoffs is None or equivalent_atoms is None:
        return {"error": "Symmetry dataset missing Wyckoff or equivalence information."}
    if len(wyckoffs) == 0 or len(equivalent_atoms) == 0 or len(wyckoffs) != len(equivalent_atoms):
        return {"error": "Symmetry dataset missing Wyckoff or equivalence information."}

    # Build equivalence class sizes for multiplicities
    eq_class_indices: dict[int, list[int]] = {}
    for idx, eq in enumerate(equivalent_atoms):
        eq_class_indices.setdefault(int(eq), []).append(idx)

    multiplicity_by_letter: dict[str, int] = {}
    first_coords_by_letter: dict[str, list[str]] = {}

    for idx, letter in enumerate(wyckoffs):
        if letter in multiplicity_by_letter:
            continue
        eq = int(equivalent_atoms[idx])
        multiplicity_by_letter[letter] = len(eq_class_indices[eq])
        frac_coords = structure[idx].frac_coords
        first_coords_by_letter[letter] = [_rational_str(x) for x in frac_coords]

    # Deterministic ordering by Wyckoff letter
    sorted_letters = sorted(multiplicity_by_letter.keys())
    wyckoff_multiplicity_dict = {ltr: multiplicity_by_letter[ltr] for ltr in sorted_letters}
    wyckoff_coordinates_dict = {ltr: first_coords_by_letter[ltr] for ltr in sorted_letters}

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


if __name__ == "__main__":

    if len(sys.argv) != 2:

    result = analyze_wyckoff_position_multiplicities_and_coordinates(sys.argv[1])
