"""Compute top-k Tanimoto similarity (Morgan FP radius=2, chirality on) for chemicals.

Notes:
- Chemical names are converted to SMILES via PubChem PUG REST (external resource).
- Molecule pool is read from a PDF file (one chemical per line).
"""

from __future__ import annotations

from dataclasses import dataclass
from functools import lru_cache
from typing import Iterable, List, Tuple
import json
from urllib.error import HTTPError, URLError
from urllib.parse import quote
from urllib.request import Request, urlopen

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs


@dataclass(frozen=True)
class MoleculeSimilarity:
    name: str
    similarity: float


def _extract_molecule_names_from_pdf(molecule_pool_filepath: str) -> List[str]:
    """Extract molecule names from a PDF, one name per line, preserving order."""
    names: List[str] = []
    seen = set()
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if not name:
                    continue
                if name in seen:
                    continue
                seen.add(name)
                names.append(name)
    return names


@lru_cache(maxsize=2048)
def _fetch_pubchem_smiles(name: str) -> str | None:
    """Fetch IsomericSMILES (preferred) or CanonicalSMILES from PubChem by name."""
    safe_name = quote(name)
    url = (
        "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/"
        f"{safe_name}/property/IsomericSMILES,CanonicalSMILES,SMILES,ConnectivitySMILES/JSON"
    )
    try:
        req = Request(url, headers={"User-Agent": "codex-similarity/1.0"})
        with urlopen(req, timeout=15) as resp:
            data = json.loads(resp.read().decode("utf-8"))
    except HTTPError as exc:
        if exc.code == 404:
            return None
        return None
    except URLError:
        return None
    except Exception:
        return None

    try:
        props = data["PropertyTable"]["Properties"][0]
    except Exception:
        return None

    return (
        props.get("IsomericSMILES")
        or props.get("CanonicalSMILES")
        or props.get("SMILES")
        or props.get("ConnectivitySMILES")
    )


def _mol_from_name(name: str) -> Chem.Mol | None:
    smiles = _fetch_pubchem_smiles(name)
    if not smiles:
        return None
    mol = Chem.MolFromSmiles(smiles)
    return mol


def _morgan_fp(mol: Chem.Mol):
    # 2048 bits is a common default size.
    return AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=2, nBits=2048, useChirality=True
    )


def _iter_pool_fingerprints(
    molecule_pool_filepath: str,
) -> Iterable[Tuple[str, object]]:
    names = _extract_molecule_names_from_pdf(molecule_pool_filepath)
    for name in names:
        mol = _mol_from_name(name)
        if mol is None:
            continue
        yield name, _morgan_fp(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> List[Tuple[str, float]]:
    """Return top-k most similar molecule names with their Tanimoto similarity.

    Args:
        target_molecule_name: Chemical name to compare against.
        molecule_pool_filepath: PDF filepath containing molecule names (one per line).
        top_k: Number of results to return. If <= 0, returns [].

    Returns:
        List of (molecule_name, similarity) tuples sorted by descending similarity,
        with alphabetical ordering as the tie-breaker.
    """
    if top_k <= 0:
        return []

    target_mol = _mol_from_name(target_molecule_name)
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name!r}")
    target_fp = _morgan_fp(target_mol)

    scored: List[MoleculeSimilarity] = []
    for name, fp in _iter_pool_fingerprints(molecule_pool_filepath):
        sim = float(DataStructs.TanimotoSimilarity(target_fp, fp))
        scored.append(MoleculeSimilarity(name=name, similarity=sim))

    scored.sort(key=lambda x: (-x.similarity, x.name.lower(), x.name))

    return [(item.name, item.similarity) for item in scored[:top_k]]

    )
