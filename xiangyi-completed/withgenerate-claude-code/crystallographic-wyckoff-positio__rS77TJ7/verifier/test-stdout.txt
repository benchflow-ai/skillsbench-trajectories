"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.

This module provides functionality to:
1. Read CIF files containing crystal structure data
2. Determine Wyckoff positions and their multiplicities
3. Calculate approximate fractional coordinates for the first atom of each Wyckoff position
"""

from fractions import Fraction
from typing import Any, Dict, List
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def decimal_to_fraction_string(value: float, max_denom: int = 12) -> str:
    """
    Convert a decimal coordinate to a fraction string.

    Args:
        value: Decimal coordinate value
        max_denom: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the coordinate as a fraction or integer
    """
    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle very small values as 0
    if abs(value) < 1e-10:
        return "0"

    # Handle values very close to 1 as 0
    if abs(value - 1.0) < 1e-10:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denom)

    # Return as integer string if denominator is 1, otherwise as fraction
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def coords_to_fraction_strings(coords: List[float], max_denom: int = 12) -> List[str]:
    """
    Convert a list of decimal coordinates to fraction strings.

    Args:
        coords: List of decimal coordinate values [x, y, z]
        max_denom: Maximum denominator for fractions (default: 12)

    Returns:
        List of string representations of coordinates
    """
    return [decimal_to_fraction_string(coord, max_denom) for coord in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    This function reads a CIF file, determines the Wyckoff positions in the crystal
    structure, calculates their multiplicities, and provides the fractional coordinates
    of the first atom in each Wyckoff position.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinate lists

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates("FeS2.cif")
        >>> print(result)
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)

    # Create symmetry analyzer
    analyzer = SpacegroupAnalyzer(structure)

    # Get symmetrized structure with Wyckoff information
    sym_struct = analyzer.get_symmetrized_structure()

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Process each group of equivalent sites
    for i, equiv_sites in enumerate(sym_struct.equivalent_sites):
        # Get the Wyckoff letter for this group
        wyckoff_symbol = sym_struct.wyckoff_symbols[i]

        # Extract just the letter part (remove multiplicity prefix like "4a" -> "a")
        # Wyckoff symbols typically end with the letter
        wyckoff_letter = ''.join(c for c in wyckoff_symbol if c.isalpha())

        # Only process each Wyckoff position once
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            # Store multiplicity (number of equivalent sites)
            multiplicity = len(equiv_sites)
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

            # Get fractional coordinates of the first (representative) atom
            representative_site = equiv_sites[0]
            frac_coords = representative_site.frac_coords.tolist()

            # Convert coordinates to fraction strings
            coord_strings = coords_to_fraction_strings(frac_coords, max_denom=12)
            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    # Return the complete analysis
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
