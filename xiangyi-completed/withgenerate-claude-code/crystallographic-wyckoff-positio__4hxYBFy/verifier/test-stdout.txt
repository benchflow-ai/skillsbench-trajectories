"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.

This module provides functionality to parse CIF files, determine Wyckoff positions,
calculate their multiplicities, and extract coordinates for the first atom of each
Wyckoff position.
"""

from collections import Counter
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(val: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string representation.

    Args:
        val: Float value (typically 0 <= val < 1 for crystallographic coords)
        max_denominator: Maximum allowed denominator

    Returns:
        String representation like "0", "1/2", "3/8", etc.
    """
    # Normalize to [0, 1)
    val = val % 1.0

    # Handle values very close to 0 or 1
    if abs(val) < 1e-10:
        return "0"
    if abs(val - 1.0) < 1e-10:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(val).limit_denominator(max_denominator)

    # Format as string
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def coords_to_fractions(coords: list[float], max_denominator: int = 12) -> list[str]:
    """
    Convert a list of coordinate values to fraction strings.

    Args:
        coords: List of [x, y, z] coordinates
        max_denominator: Maximum allowed denominator

    Returns:
        List of fraction strings ["x", "y", "z"]
    """
    return [float_to_fraction_string(c, max_denominator) for c in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
            - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to their multiplicities
            - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to [x, y, z] fraction strings
    """
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)

    # Create a SpacegroupAnalyzer to determine symmetry
    sga = SpacegroupAnalyzer(structure)

    # Get the symmetry dataset containing Wyckoff information
    symm_data = sga.get_symmetry_dataset()

    # Extract Wyckoff letters and positions
    wyckoffs = symm_data.wyckoffs
    std_positions = symm_data.std_positions

    # Calculate multiplicities for each Wyckoff position
    wyckoff_multiplicity_dict = dict(Counter(wyckoffs))

    # Find the first atom of each Wyckoff position and convert coordinates to fractions
    wyckoff_coordinates_dict = {}
    for i, wyckoff in enumerate(wyckoffs):
        if wyckoff not in wyckoff_coordinates_dict:
            # First occurrence of this Wyckoff position
            coords = std_positions[i]
            wyckoff_coordinates_dict[wyckoff] = coords_to_fractions(coords.tolist())

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
