"""
Materials Science CIF File Analysis
Analyzes Wyckoff position multiplicities and calculates coordinates for crystal structures

This module provides functionality to analyze crystallographic CIF files and extract
Wyckoff position information using the pymatgen library.

Dependencies:
    - pymatgen: For reading CIF files and analyzing crystal symmetry
    - fractions: For converting decimal coordinates to rational fractions

Author: Materials Science Researcher
"""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def decimal_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal coordinate to a fraction string.

    Args:
        value: Decimal value to convert
        max_denominator: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the fraction (e.g., "3/8", "0", "1/2")
    """
    # Round to avoid floating point errors
    value = round(value, 10)

    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return string representation
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def coords_to_fraction_list(coords: list, max_denominator: int = 12) -> list[str]:
    """
    Convert a list of decimal coordinates to fraction strings.

    Args:
        coords: List of decimal coordinates [x, y, z]
        max_denominator: Maximum denominator for fractions

    Returns:
        List of fraction strings
    """
    return [decimal_to_fraction_string(coord, max_denominator) for coord in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    This function reads a CIF file, determines the Wyckoff positions for each
    unique atomic site, calculates their multiplicities, and extracts the
    coordinates of the first atom in each Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Mapping of Wyckoff letters to multiplicities
        - wyckoff_coordinates_dict: Mapping of Wyckoff letters to fractional
          coordinates (as fraction strings)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    try:
        # Read the CIF file and create a Structure object
        structure = Structure.from_file(filepath)

        # Create a SpacegroupAnalyzer to determine symmetry
        analyzer = SpacegroupAnalyzer(structure)

        # Get the symmetrized structure which contains Wyckoff information
        symmetrized_structure = analyzer.get_symmetrized_structure()

        # Initialize dictionaries to store results
        wyckoff_multiplicity_dict = {}
        wyckoff_coordinates_dict = {}

        # Iterate through equivalent sites
        for i, equiv_sites in enumerate(symmetrized_structure.equivalent_sites):
            # Get the Wyckoff symbol for this site (e.g., "4a", "8c")
            wyckoff_symbol = symmetrized_structure.wyckoff_symbols[i]

            # Extract just the letter part (remove multiplicity prefix)
            # Wyckoff symbols are formatted as "{multiplicity}{letter}"
            wyckoff_letter = ''.join(c for c in wyckoff_symbol if c.isalpha())

            # Get the multiplicity (number of equivalent sites)
            multiplicity = len(equiv_sites)

            # Get the fractional coordinates of the first atom in this Wyckoff position
            first_atom_coords = equiv_sites[0].frac_coords

            # Only store the first occurrence of each Wyckoff position
            # (in case the same Wyckoff position appears multiple times for different elements)
            if wyckoff_letter not in wyckoff_multiplicity_dict:
                wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
                wyckoff_coordinates_dict[wyckoff_letter] = coords_to_fraction_list(
                    first_atom_coords
                )

        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
        }

    except Exception as e:
        # Return error information if analysis fails
        return {
            "error": str(e),
            "filepath": filepath,
        }
