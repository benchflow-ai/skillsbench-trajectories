"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.

This module provides functionality to:
1. Parse CIF files using pymatgen
2. Determine space group and Wyckoff positions using spglib
3. Calculate multiplicities and representative coordinates for each Wyckoff position
"""

from typing import Any
from fractions import Fraction

import numpy as np
import spglib
from pymatgen.core import Structure


def _coord_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float coordinate to a fraction string.

    Args:
        value: Float value to convert (should be in range [0, 1))
        max_denominator: Maximum allowed denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "3/8", "1/2", "0")
    """
    # Normalize to [0, 1) range for crystallographic coordinates
    value = value % 1.0

    # Handle very small negative values from floating point errors
    if value < 0:
        value = 0.0

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle edge case where limit_denominator gives 1/1
    if frac >= 1:
        frac = Fraction(0)

    # Format as string
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def _coords_to_fraction_list(coords: list[float], max_denominator: int = 12) -> list[str]:
    """
    Convert a list of float coordinates to fraction strings.

    Args:
        coords: List of [x, y, z] coordinates
        max_denominator: Maximum denominator for fractions

    Returns:
        List of fraction strings ["x_frac", "y_frac", "z_frac"]
    """
    return [_coord_to_fraction_str(c, max_denominator) for c in coords]


def _structure_to_spglib_cell(structure: Structure) -> tuple:
    """
    Convert a pymatgen Structure to spglib cell format.

    Args:
        structure: Pymatgen Structure object

    Returns:
        Tuple of (lattice, positions, numbers) for spglib
    """
    lattice = structure.lattice.matrix
    positions = structure.frac_coords
    numbers = [site.specie.Z for site in structure]
    return (lattice, positions, numbers)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    This function reads a CIF file, determines the space group symmetry,
    identifies Wyckoff positions, and calculates:
    - The multiplicity of each Wyckoff position
    - The coordinates of the first (representative) atom for each Wyckoff position

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with two keys:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letters to their multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff letters to coordinate lists
          (coordinates as fraction strings with denominators <= 12)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/path/to/FeS2_mp-226.cif"
        ... )
        >>> print(result)
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Convert to spglib cell format
    cell = _structure_to_spglib_cell(structure)

    # Get symmetry dataset from spglib
    # Use symprec=0.1 for reasonable tolerance
    dataset = spglib.get_symmetry_dataset(cell, symprec=0.1)

    # Extract Wyckoff letters and equivalent atom mapping
    wyckoffs = dataset.wyckoffs
    equivalent_atoms = dataset.equivalent_atoms
    std_positions = dataset.std_positions

    # Find unique Wyckoff positions
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Get unique representative atom indices
    unique_representatives = np.unique(equivalent_atoms)

    for rep_idx in unique_representatives:
        wyckoff_letter = wyckoffs[rep_idx]

        # Calculate multiplicity (count of atoms equivalent to this representative)
        multiplicity = int(np.sum(equivalent_atoms == rep_idx))

        # Get standardized coordinates for the representative atom
        coords = std_positions[rep_idx].tolist()

        # Store results
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = _coords_to_fraction_list(coords)

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
