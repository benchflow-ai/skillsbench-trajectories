"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to SMILES using PubChem API
3. Compute molecular fingerprints using RDKit
4. Calculate Tanimoto similarity between molecules
5. Return top k most similar molecules
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem
import time


def extract_chemicals_from_pdf(pdf_path):
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_path: Path to the PDF file

    Returns:
        List of chemical names (one per line)
    """
    chemicals = []

    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.split('\n'):
                    line = line.strip()
                    # Skip empty lines
                    if line:
                        chemicals.append(line)

    return chemicals


def get_smiles_from_name(chemical_name, retry_delay=0.5, max_retries=3):
    """
    Convert chemical name to SMILES string using PubChem API.

    Args:
        chemical_name: Name of the chemical
        retry_delay: Delay between retries in seconds
        max_retries: Maximum number of retries

    Returns:
        SMILES string or None if not found
    """
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(chemical_name, "name")
            if compounds:
                # Get SMILES - try different properties to avoid deprecation warnings
                compound = compounds[0]
                # Access the property to check if it exists
                if hasattr(compound, 'smiles') and compound.smiles:
                    return compound.smiles
                # Fallback to accessing via to_dict() to avoid deprecation warnings
                compound_dict = compound.to_dict()
                if 'isomeric_smiles' in compound_dict:
                    return compound_dict['isomeric_smiles']
                elif 'canonical_smiles' in compound_dict:
                    return compound_dict['canonical_smiles']
            return None
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                if attempt < max_retries - 1:
                    time.sleep(retry_delay * (attempt + 1))
                    continue
            print(f"Error looking up {chemical_name}: {e}")
            return None
    return None


def get_morgan_fingerprint(smiles, radius=2, use_chirality=True):
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Include chirality information (default: True)

    Returns:
        Morgan fingerprint or None if invalid SMILES
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    # Use MorganGenerator for newer RDKit versions
    try:
        from rdkit.Chem import rdFingerprintGenerator
        mfpgen = rdFingerprintGenerator.GetMorganGenerator(
            radius=radius,
            fpSize=2048,
            includeChirality=use_chirality
        )
        fp = mfpgen.GetFingerprint(mol)
    except (ImportError, AttributeError):
        # Fallback to legacy method
        fp = AllChem.GetMorganFingerprintAsBitVect(
            mol,
            radius=radius,
            nBits=2048,
            useChirality=use_chirality
        )
    return fp


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k most similar molecules to the target molecule.

    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to PDF file containing molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        List of chemical names sorted by similarity (descending),
        with alphabetical ordering for ties
    """
    # Extract chemicals from PDF
    pool_chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    print(f"Looking up target molecule: {target_molecule_name}")
    target_smiles = get_smiles_from_name(target_molecule_name)

    if target_smiles is None:
        print(f"Could not find SMILES for target: {target_molecule_name}")
        return []

    # Get fingerprint for target
    target_fp = get_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)

    if target_fp is None:
        print(f"Could not generate fingerprint for target: {target_molecule_name}")
        return []

    # Calculate similarities for all molecules in pool
    similarities = []

    print(f"Processing {len(pool_chemicals)} molecules from pool...")

    for i, chem_name in enumerate(pool_chemicals):
        # Skip the target molecule if it's in the pool
        if chem_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES for pool molecule
        smiles = get_smiles_from_name(chem_name)

        if smiles is None:
            print(f"Could not find SMILES for: {chem_name}")
            continue

        # Get fingerprint
        fp = get_morgan_fingerprint(smiles, radius=2, use_chirality=True)

        if fp is None:
            print(f"Could not generate fingerprint for: {chem_name}")
            continue

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((chem_name, similarity))

        # Add delay to avoid rate limiting from PubChem
        time.sleep(0.3)

    # Sort by similarity (descending), then alphabetically for ties
    # Using negative similarity for descending order, then name for alphabetical
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecule names
    top_k_results = [name for name, sim in similarities[:top_k]]

    print(f"\nTop {top_k} similar molecules to {target_molecule_name}:")
    for i, (name, sim) in enumerate(similarities[:top_k], 1):
        print(f"{i}. {name} (similarity: {sim:.4f})")

    return top_k_results


# Example usage
if __name__ == "__main__":

    pdf_path = "/root/molecules.pdf"
    k = 5

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    print(f"\nResults: {results}")
