"""
Crystallographic CIF file analyzer for Wyckoff positions and multiplicities.

This module analyzes CIF files to extract:
- Wyckoff position multiplicities
- Approximate coordinates for the first atom of each Wyckoff position
"""

from fractions import Fraction
from collections import Counter
from typing import Any
import spglib
from pymatgen.core import Structure


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze a CIF file to extract Wyckoff position multiplicities and coordinates.

    Parameters
    ----------
    filepath : str
        Path to the CIF file

    Returns
    -------
    dict[str, dict] | dict[str, Any]
        Dictionary with keys:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letters to multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff letters to first atom coordinates
                                       as fraction strings

    Example
    -------
    >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
    ...     "/root/cif_files/FeS2_mp-226.cif"
    ... )
    >>> result["wyckoff_multiplicity_dict"]
    {'a': 4, 'c': 8}
    >>> result["wyckoff_coordinates_dict"]["a"]
    ['0', '1/2', '1/2']
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Prepare data for spglib
    lattice = structure.lattice.matrix
    positions = structure.frac_coords
    atomic_numbers = [site.specie.number for site in structure.sites]

    # Get Wyckoff positions and symmetry information
    dataset = spglib.get_symmetry_dataset((lattice, positions, atomic_numbers))

    if dataset is None:
        raise ValueError(f"Could not analyze symmetry for file: {filepath}")

    wyckoff_symbols = list(dataset.wyckoffs)

    # Calculate multiplicities for each Wyckoff position
    wyckoff_multiplicity_dict = _get_multiplicities(wyckoff_symbols)

    # Get first atom coordinates for each Wyckoff position
    wyckoff_coordinates_dict = _get_first_atom_coordinates(
        wyckoff_symbols, positions
    )

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _get_multiplicities(wyckoff_symbols: list[str]) -> dict[str, int]:
    """
    Calculate multiplicity (count) for each unique Wyckoff position.

    Parameters
    ----------
    wyckoff_symbols : list[str]
        List of Wyckoff symbols for each atom

    Returns
    -------
    dict[str, int]
        Dictionary mapping Wyckoff letter -> multiplicity count
    """
    counts = Counter(wyckoff_symbols)
    return dict(sorted(counts.items()))


def _get_first_atom_coordinates(
    wyckoff_symbols: list[str], positions: list[list[float]]
) -> dict[str, list[str]]:
    """
    Extract first atom coordinates for each unique Wyckoff position.

    Parameters
    ----------
    wyckoff_symbols : list[str]
        List of Wyckoff symbols for each atom
    positions : list[list[float]]
        Fractional coordinates for each atom

    Returns
    -------
    dict[str, list[str]]
        Dictionary mapping Wyckoff letter -> [x, y, z] as fraction strings
    """
    seen_positions = {}
    result = {}

    for wyckoff, coord in zip(wyckoff_symbols, positions):
        if wyckoff not in seen_positions:
            seen_positions[wyckoff] = True
            # Convert coordinates to fraction strings
            frac_coords = [
                _decimal_to_fraction_string(c, max_denominator=12)
                for c in coord
            ]
            result[wyckoff] = frac_coords

    # Sort by Wyckoff letter (alphabetical order)
    return dict(sorted(result.items()))


def _decimal_to_fraction_string(
    value: float | str, max_denominator: int = 12
) -> str:
    """
    Convert a decimal coordinate to a fraction string.

    Parameters
    ----------
    value : float | str
        Decimal value to convert
    max_denominator : int, optional
        Maximum denominator for the resulting fraction (default: 12)

    Returns
    -------
    str
        Fraction string (e.g., "3/8", "1/2", "0")
    """
    # Parse value if it's a string
    if isinstance(value, str):
        try:
            value = float(value)
        except ValueError:
            return str(value)

    # Convert to fraction with limited denominator
    frac = Fraction(float(value)).limit_denominator(max_denominator)

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"
