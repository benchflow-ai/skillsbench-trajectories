"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.

This module provides functionality to extract and analyze Wyckoff positions
from crystallographic information files (CIF files) generated by SHELX
through X-ray Diffraction.
"""

from fractions import Fraction
from pymatgen.io.cif import CifParser
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from collections import defaultdict
from typing import Dict, Any


def _rationalize_coordinate(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point coordinate to a rationalized fraction string.

    Args:
        value: Floating-point coordinate value
        max_denominator: Maximum denominator for fraction representation (default: 12)

    Returns:
        String representation of the coordinate as a fraction or integer
    """
    # Handle very small values (essentially 0)
    if abs(value) < 1e-10:
        return "0"

    # Round to reasonable precision to avoid floating-point artifacts
    value = round(value, 10)

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle whole numbers
    if frac.denominator == 1:
        return str(frac.numerator)

    return str(frac)


def _get_wyckoff_from_cif_data(cif_file_data: Dict[str, Any]) -> Dict[str, Any]:
    """
    Extract Wyckoff position information directly from CIF file data.

    Args:
        cif_file_data: Dictionary containing raw CIF data for a structure

    Returns:
        Dictionary with wyckoff_multiplicity_dict and wyckoff_coordinates_dict
    """
    labels = cif_file_data.get('_atom_site_label', [])
    multiplicities = cif_file_data.get('_atom_site_symmetry_multiplicity', [])
    wyckoff_symbols = cif_file_data.get('_atom_site_Wyckoff_symbol', [])
    frac_x = cif_file_data.get('_atom_site_fract_x', [])
    frac_y = cif_file_data.get('_atom_site_fract_y', [])
    frac_z = cif_file_data.get('_atom_site_fract_z', [])

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    if multiplicities and wyckoff_symbols:
        # CIF file has explicit Wyckoff information
        for i, (wyck, mult) in enumerate(zip(wyckoff_symbols, multiplicities)):
            if wyck and wyck not in wyckoff_multiplicity_dict:
                mult_val = int(mult)
                wyckoff_multiplicity_dict[wyck] = mult_val

                # Get first atom coordinates for this Wyckoff position
                coords = [
                    _rationalize_coordinate(float(frac_x[i])),
                    _rationalize_coordinate(float(frac_y[i])),
                    _rationalize_coordinate(float(frac_z[i]))
                ]
                wyckoff_coordinates_dict[wyck] = coords

        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict
        }

    return None


def _get_wyckoff_from_symmetry(structure: Structure) -> Dict[str, Any]:
    """
    Determine Wyckoff positions from space group symmetry using SpacegroupAnalyzer.

    Args:
        structure: pymatgen Structure object

    Returns:
        Dictionary with wyckoff_multiplicity_dict and wyckoff_coordinates_dict
    """
    try:
        sga = SpacegroupAnalyzer(structure, symprec=0.1)
        wyckoff_symbols = sga.get_wyckoff_symbols()

        # Count occurrences of each Wyckoff symbol
        wyckoff_multiplicity_dict = {}
        wyckoff_first_site = {}

        for i, wyck in enumerate(wyckoff_symbols):
            if wyck not in wyckoff_multiplicity_dict:
                wyckoff_multiplicity_dict[wyck] = 0
                wyckoff_first_site[wyck] = structure.sites[i]

            wyckoff_multiplicity_dict[wyck] += 1

        # Build coordinates dict
        wyckoff_coordinates_dict = {}
        for wyck, site in wyckoff_first_site.items():
            coords = [
                _rationalize_coordinate(site.frac_coords[0]),
                _rationalize_coordinate(site.frac_coords[1]),
                _rationalize_coordinate(site.frac_coords[2])
            ]
            wyckoff_coordinates_dict[wyck] = coords

        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict
        }
    except Exception as e:
        # If symmetry analysis fails, fall back to grouping by coordinates
        return _get_wyckoff_from_coordinates(structure)


def _get_wyckoff_from_coordinates(structure: Structure) -> Dict[str, Any]:
    """
    Fall back: Group atoms by element type and approximate Wyckoff positions
    based on similar coordinates.

    Args:
        structure: pymatgen Structure object

    Returns:
        Dictionary with wyckoff_multiplicity_dict and wyckoff_coordinates_dict
    """
    # Group atoms by element
    element_groups = defaultdict(list)
    for i, site in enumerate(structure.sites):
        element = site.species
        element_groups[element].append(i)

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # For each element, group into Wyckoff positions
    wyckoff_letter = ord('a')

    for element, indices in sorted(element_groups.items()):
        # Simple approach: use element symbol + letter
        # In a real scenario with multiple Wyckoff positions per element,
        # we'd need more sophisticated clustering
        wyck = chr(wyckoff_letter)
        wyckoff_letter += 1

        wyckoff_multiplicity_dict[wyck] = len(indices)

        # Get first atom's coordinates
        first_idx = indices[0]
        site = structure.sites[first_idx]
        coords = [
            _rationalize_coordinate(site.frac_coords[0]),
            _rationalize_coordinate(site.frac_coords[1]),
            _rationalize_coordinate(site.frac_coords[2])
        ]
        wyckoff_coordinates_dict[wyck] = coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for
    the first atom of each Wyckoff position in a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
            - wyckoff_multiplicity_dict: Dict mapping Wyckoff letter to multiplicity
            - wyckoff_coordinates_dict: Dict mapping Wyckoff letter to [x, y, z] coordinates
                                       as rationalized fractions/integers

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> print(result)
        {
            'wyckoff_multiplicity_dict': {'a': 4, 'c': 8},
            'wyckoff_coordinates_dict': {
                'a': ['0', '1/2', '1/2'],
                'c': ['3/8', '1/9', '8/9']
            }
        }
    """
    # Parse CIF file
    parser = CifParser(filepath)
    cif_file = parser._cif

    # Try to get Wyckoff info from CIF data first
    for block_name, block_data in cif_file.data.items():
        cif_data = block_data.data
        result = _get_wyckoff_from_cif_data(cif_data)
        if result:
            return result

    # If CIF doesn't have explicit Wyckoff data, use symmetry analysis
    structure = Structure.from_file(filepath)
    result = _get_wyckoff_from_symmetry(structure)

    return result
