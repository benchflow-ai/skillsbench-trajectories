"""
Crystal Structure Analysis: Wyckoff Position Multiplicities and Coordinates

This module analyzes CIF (Crystallographic Information File) files to extract
Wyckoff position multiplicities and representative coordinates for crystal structures.
"""

from typing import Dict, Any
from fractions import Fraction
from collections import defaultdict
from pymatgen.core import Structure
import spglib


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> Dict[str, Dict] | Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dictionary mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Dictionary mapping Wyckoff letters to fractional coordinates
          as lists of string fractions (denominators ≤ 12)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates("/path/to/file.cif")
        >>> print(result)
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)

    # Convert to spglib format for symmetry analysis
    cell = (
        structure.lattice.matrix,
        structure.frac_coords,
        structure.atomic_numbers
    )

    # Get symmetry dataset with Wyckoff positions
    dataset = spglib.get_symmetry_dataset(cell, symprec=1e-5)

    # Group atoms by equivalent positions
    wyckoff_groups = defaultdict(list)
    for i, (wyckoff_letter, equiv_idx) in enumerate(
        zip(dataset.wyckoffs, dataset.equivalent_atoms)
    ):
        wyckoff_groups[equiv_idx].append({
            'index': i,
            'wyckoff': wyckoff_letter,
            'coords': structure[i].frac_coords
        })

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Process each Wyckoff position group
    for equiv_idx, atoms in wyckoff_groups.items():
        # Get the Wyckoff letter from the first atom in the group
        wyckoff_letter = atoms[0]['wyckoff']

        # Store the multiplicity (number of atoms in this Wyckoff position)
        wyckoff_multiplicity_dict[wyckoff_letter] = len(atoms)

        # Get the fractional coordinates of the first atom
        first_coords = atoms[0]['coords']

        # Convert coordinates to fraction strings
        wyckoff_coordinates_dict[wyckoff_letter] = coords_to_fraction_strings(
            first_coords, max_denominator=12
        )

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def coords_to_fraction_strings(coords, max_denominator=12):
    """
    Convert coordinate array to list of fraction strings.

    Args:
        coords: Array or list of fractional coordinates
        max_denominator: Maximum allowed denominator for fractions (default: 12)

    Returns:
        List of coordinate strings in fraction format (e.g., ["1/2", "0", "3/8"])
    """
    result = []

    for coord in coords:
        # Normalize coordinate to [0, 1) range
        normalized = coord % 1.0

        # Handle floating point errors near 0
        if abs(normalized) < 1e-10:
            normalized = 0.0

        # Convert to fraction with limited denominator
        frac = Fraction(normalized).limit_denominator(max_denominator)

        # Format as string
        if frac.denominator == 1:
            # Integer case (including 0)
            result.append(str(frac.numerator))
        else:
            # Fraction case
            result.append(f"{frac.numerator}/{frac.denominator}")

    return result
