"""
Solution for finding top k similar molecules using Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from PDF files
2. Convert chemical names to SMILES format using PubChem API
3. Compute Morgan fingerprints with chirality
4. Calculate Tanimoto similarity between molecules
5. Return top k most similar molecules with proper sorting
"""

import time
import requests
import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs


def get_smiles_from_pubchem(chemical_name, max_retries=3):
    """
    Convert a chemical name to SMILES format using PubChem API.

    Args:
        chemical_name: String name of the chemical
        max_retries: Maximum number of retry attempts for rate limiting

    Returns:
        SMILES string or None if not found
    """
    base_url = "https://pubchem.ncbi.nlm.nih.gov/rest/pug"
    url = f"{base_url}/compound/name/{chemical_name}/property/CanonicalSMILES/JSON"

    for attempt in range(max_retries):
        try:
            response = requests.get(url, timeout=10)
            if response.status_code == 200:
                data = response.json()
                # Check if we got a Fault response
                if 'Fault' in data:
                    if 'ServerBusy' in str(data.get('Fault', {})):
                        # Rate limited, wait and retry
                        time.sleep(2 * (attempt + 1))
                        continue
                    return None

                properties = data['PropertyTable']['Properties'][0]
                # PubChem may return different key names
                # Try CanonicalSMILES, ConnectivitySMILES, or IsomericSMILES
                for key in ['CanonicalSMILES', 'ConnectivitySMILES', 'IsomericSMILES']:
                    if key in properties:
                        return properties[key]
                return None
            elif response.status_code == 503:
                # Server busy, wait and retry
                time.sleep(2 * (attempt + 1))
                continue
            else:
                return None
        except Exception as e:
            if attempt == max_retries - 1:
                print(f"Warning: Could not fetch SMILES for '{chemical_name}': {e}")
            else:
                time.sleep(1)
            continue

    return None


def extract_chemical_names_from_pdf(pdf_filepath):
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of chemical names (strings)
    """
    chemicals = []

    try:
        with pdfplumber.open(pdf_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.strip().split('\n')
                    for line in lines:
                        line = line.strip()
                        if line:
                            chemicals.append(line)
    except Exception as e:
        print(f"Error reading PDF file '{pdf_filepath}': {e}")
        return []

    return chemicals


def compute_morgan_fingerprint(smiles, radius=2, use_chirality=True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string representation of molecule
        radius: Fingerprint radius (default: 2)
        use_chirality: Include stereochemical information (default: True)

    Returns:
        Morgan fingerprint or None if invalid SMILES
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    fingerprint = AllChem.GetMorganFingerprint(
        mol,
        radius=radius,
        useChirality=use_chirality
    )
    return fingerprint


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k most similar molecules to a target molecule.

    Uses Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical
    ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule (string)
        molecule_pool_filepath: Path to PDF file containing molecule pool
        top_k: Number of top similar molecules to return (int)

    Returns:
        List of molecule names (strings) representing the top k most similar
        molecules, sorted by similarity (descending) then alphabetically.
        Returns empty list if target molecule cannot be processed.
    """
    # Step 1: Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        print(f"Error: Could not find SMILES for target molecule '{target_molecule_name}'")
        return []

    # Step 2: Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)
    if target_fp is None:
        print(f"Error: Could not compute fingerprint for target molecule '{target_molecule_name}'")
        return []

    # Step 3: Extract chemical names from PDF
    pool_molecules = extract_chemical_names_from_pdf(molecule_pool_filepath)
    if not pool_molecules:
        print(f"Error: No molecules found in '{molecule_pool_filepath}'")
        return []

    # Step 4: Compute similarities
    similarities = []

    for i, molecule_name in enumerate(pool_molecules):
        # Get SMILES for pool molecule
        smiles = get_smiles_from_pubchem(molecule_name)
        if smiles is None:
            continue

        # Compute fingerprint
        fp = compute_morgan_fingerprint(smiles, radius=2, use_chirality=True)
        if fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)

        similarities.append((molecule_name, similarity))

        # Add small delay to avoid rate limiting (max 5 requests/second for PubChem)
        if (i + 1) % 5 == 0:
            time.sleep(1)

    # Step 5: Sort by similarity (descending), then by name (ascending)
    # Using negative similarity for descending order
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Step 6: Return top k molecule names
    top_k_results = [name for name, sim in similarities[:top_k]]

    return top_k_results


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_file = "/root/molecules.pdf"
    k = 10

    print(f"Finding top {k} molecules similar to '{target}'...\n")

    results = topk_tanimoto_similarity_molecules(target, pdf_file, k)

    if results:
        print(f"Top {k} similar molecules:")
        for i, molecule in enumerate(results, 1):
            print(f"{i}. {molecule}")
    else:
        print("No results found.")
