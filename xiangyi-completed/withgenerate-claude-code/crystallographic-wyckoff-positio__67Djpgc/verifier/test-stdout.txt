"""
Crystal Structure Analysis: Wyckoff Position Multiplicities and Coordinates

This module provides functionality to analyze CIF files and extract Wyckoff position
information including multiplicities and coordinates for the first atom of each position.
"""

from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def coord_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point coordinate to a fraction string.

    Args:
        value: The coordinate value (typically in range [0, 1))
        max_denominator: Maximum denominator for the fraction approximation

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    frac = Fraction(float(value)).limit_denominator(max_denominator)

    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def coords_to_fraction_list(coords: list, max_denominator: int = 12) -> list[str]:
    """
    Convert a list of coordinates to fraction strings.

    Args:
        coords: List of coordinate values
        max_denominator: Maximum denominator for fraction approximation

    Returns:
        List of fraction strings
    """
    return [coord_to_fraction_string(c, max_denominator) for c in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinate lists
          where coordinates are represented as fraction strings
    """
    # Load the crystal structure from CIF file
    structure = Structure.from_file(filepath)

    # Create symmetry analyzer to determine space group and Wyckoff positions
    sga = SpacegroupAnalyzer(structure)

    # Get symmetrized structure which contains Wyckoff information
    sym_struct = sga.get_symmetrized_structure()

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Iterate through equivalent sites and their Wyckoff labels
    for sites, wyckoff_label in zip(sym_struct.equivalent_sites, sym_struct.wyckoff_symbols):
        # Extract Wyckoff letter from label (e.g., "4a" -> "a", "8c" -> "c")
        wyckoff_letter = wyckoff_label[-1] if wyckoff_label[-1].isalpha() else wyckoff_label

        # Calculate multiplicity (number of equivalent sites)
        multiplicity = len(sites)

        # Get fractional coordinates of the first site
        first_site_coords = sites[0].frac_coords.tolist()

        # Convert coordinates to fraction strings
        coords_as_fractions = coords_to_fraction_list(first_site_coords)

        # Store in result dictionaries
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coords_as_fractions

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


if __name__ == "__main__":
    # Example usage with FeS2 structure
    result = analyze_wyckoff_position_multiplicities_and_coordinates(
        "/root/cif_files/FeS2_mp-226.cif"
    )
    print("Result:")
    print(result)
