"""
Solution for finding top-k similar molecules using Tanimoto similarity
with Morgan fingerprints.
"""

import time
import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator


def get_smiles_from_name(molecule_name: str, max_retries: int = 5) -> str | None:
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        molecule_name: The chemical name to convert.
        max_retries: Maximum number of retry attempts for API calls.

    Returns:
        SMILES string if found, None otherwise.
    """
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(molecule_name, 'name')
            if compounds:
                return compounds[0].smiles
            return None
        except pcp.PubChemHTTPError:
            if attempt < max_retries - 1:
                time.sleep(2 ** attempt)
            else:
                return None
    return None


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file.

    Returns:
        List of molecule names extracted from the PDF.
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)
    return molecules


def compute_tanimoto_similarity(smiles1: str, smiles2: str, morgan_gen) -> float | None:
    """
    Compute Tanimoto similarity between two molecules using Morgan fingerprints.

    Args:
        smiles1: SMILES string of the first molecule.
        smiles2: SMILES string of the second molecule.
        morgan_gen: MorganGenerator object for fingerprint generation.

    Returns:
        Tanimoto similarity score, or None if molecules cannot be parsed.
    """
    mol1 = Chem.MolFromSmiles(smiles1)
    mol2 = Chem.MolFromSmiles(smiles2)

    if mol1 is None or mol2 is None:
        return None

    fp1 = morgan_gen.GetFingerprint(mol1)
    fp2 = morgan_gen.GetFingerprint(mol2)

    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a PDF pool.

    Uses Morgan fingerprints with radius=2 and chirality included to compute
    Tanimoto similarity. Results are sorted by descending similarity, with
    alphabetical ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score) for the top-k
        most similar molecules, sorted by descending similarity with
        alphabetical ordering for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Extract molecule names from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Create Morgan fingerprint generator with radius=2, include chirality
    morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=2,
        includeChirality=True
    )

    # Compute similarities for all molecules in the pool
    similarities = []
    for mol_name in pool_molecules:
        mol_smiles = get_smiles_from_name(mol_name)
        if mol_smiles is None:
            continue

        similarity = compute_tanimoto_similarity(target_smiles, mol_smiles, morgan_gen)
        if similarity is not None:
            similarities.append((mol_name, similarity))

    # Sort by descending similarity, then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        target_molecule_name="Aspirin",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5
    )

    print("Top 5 molecules similar to Aspirin:")
    for mol_name, similarity in result:
        print(f"  {mol_name}: {similarity:.4f}")
