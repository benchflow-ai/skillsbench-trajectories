package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ListState;
import org.apache.flink.api.common.state.ListStateDescriptor;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;
import org.apache.flink.util.Collector;

import java.io.BufferedReader;
import java.io.FileInputStream;
import java.io.InputStreamReader;
import java.time.Duration;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    // 10 minutes in microseconds
    private static final long SESSION_GAP_MICROS = 10L * 60L * 1000L * 1000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events from gzipped CSV
        DataStream<TaskEvent> taskEvents = env
                .addSource(taskSourceOrTest(new TaskEventSource(taskInput)))
                .name("TaskEventSource");

        // Read job events from gzipped CSV
        DataStream<JobEvent> jobEvents = env
                .addSource(jobSourceOrTest(new JobEventSource(jobInput)))
                .name("JobEventSource");

        // Filter task events to only SUBMIT events and assign watermarks
        // Timestamp is in microseconds, convert to milliseconds for Flink
        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(event -> event != null && event.getEventType() == TaskEvent.SUBMIT)
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000)
                );

        // Filter for finished jobs and assign watermarks
        DataStream<JobEvent> finishedJobs = jobEvents
                .filter(event -> event != null && event.isFinished())
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000)
                );

        // Connect submit events with finished jobs keyed by job ID
        DataStream<Tuple2<Long, Long>> result = submitEvents
                .keyBy(TaskEvent::getJobId)
                .connect(finishedJobs.keyBy(JobEvent::getJobId))
                .process(new KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Long>>() {

                    // Store timestamps of SUBMIT events for this job
                    private ListState<Long> submitTimestampsState;
                    // The timestamp when the job finished (0 if not finished)
                    private ValueState<Long> jobFinishedTimestampState;
                    // Whether we have already outputted for this job
                    private ValueState<Boolean> outputtedState;
                    // Whether timer has been registered
                    private ValueState<Boolean> timerRegisteredState;

                    @Override
                    public void open(Configuration parameters) {
                        submitTimestampsState = getRuntimeContext().getListState(
                                new ListStateDescriptor<>("submitTimestamps", Long.class));
                        jobFinishedTimestampState = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("jobFinishedTimestamp", Long.class));
                        outputtedState = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("outputted", Boolean.class));
                        timerRegisteredState = getRuntimeContext().getState(
                                new ValueStateDescriptor<>("timerRegistered", Boolean.class));
                    }

                    @Override
                    public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
                        Boolean outputted = outputtedState.value();
                        if (outputted != null && outputted) {
                            return;
                        }

                        // Store the timestamp of this SUBMIT event
                        submitTimestampsState.add(event.getTimestamp());

                        // Check if we should register a timer
                        registerTimerIfNeeded(ctx);
                    }

                    @Override
                    public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
                        Boolean outputted = outputtedState.value();
                        if (outputted != null && outputted) {
                            return;
                        }

                        // Store the job finish timestamp
                        jobFinishedTimestampState.update(event.getTimestamp());

                        // Register a timer to output after a delay
                        registerTimerIfNeeded(ctx);
                    }

                    private void registerTimerIfNeeded(Context ctx) throws Exception {
                        Long jobFinishedTs = jobFinishedTimestampState.value();
                        Boolean timerRegistered = timerRegisteredState.value();

                        if (jobFinishedTs != null && jobFinishedTs > 0 && (timerRegistered == null || !timerRegistered)) {
                            // Register an event-time timer for 11 minutes after job finish
                            // This ensures all session windows have closed (10 min gap)
                            long timerTime = (jobFinishedTs / 1000) + (11 * 60 * 1000);
                            ctx.timerService().registerEventTimeTimer(timerTime);
                            timerRegisteredState.update(true);
                        }
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
                        Boolean outputted = outputtedState.value();
                        if (outputted != null && outputted) {
                            return;
                        }

                        Long jobFinishedTs = jobFinishedTimestampState.value();
                        if (jobFinishedTs == null || jobFinishedTs == 0) {
                            return;
                        }

                        // Collect all timestamps and compute sessions
                        List<Long> timestamps = new ArrayList<>();
                        for (Long ts : submitTimestampsState.get()) {
                            timestamps.add(ts);
                        }

                        if (timestamps.isEmpty()) {
                            return;
                        }

                        // Compute the longest session
                        long longestSession = computeLongestSession(timestamps);

                        out.collect(new Tuple2<>(ctx.getCurrentKey(), longestSession));
                        outputtedState.update(true);
                    }

                    private long computeLongestSession(List<Long> timestamps) {
                        // Sort timestamps
                        Collections.sort(timestamps);

                        long longestSession = 0;
                        long currentSessionCount = 1;

                        for (int i = 1; i < timestamps.size(); i++) {
                            long gap = timestamps.get(i) - timestamps.get(i - 1);
                            if (gap >= SESSION_GAP_MICROS) {
                                // New session (gap is 10 minutes or more)
                                longestSession = Math.max(longestSession, currentSessionCount);
                                currentSessionCount = 1;
                            } else {
                                // Same session
                                currentSessionCount++;
                            }
                        }
                        // Don't forget the last session
                        longestSession = Math.max(longestSession, currentSessionCount);

                        return longestSession;
                    }
                });

        // Format output as (jobId,count)
        DataStream<String> formattedResult = result
                .map(new MapFunction<Tuple2<Long, Long>, String>() {
                    @Override
                    public String map(Tuple2<Long, Long> value) {
                        return "(" + value.f0 + "," + value.f1 + ")";
                    }
                });

        // Write to output file
        if (outputPath != null) {
            formattedResult.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                    .setParallelism(1);
        } else {
            formattedResult.print();
        }

        env.execute("LongestSessionPerJob");
    }

    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String filePath;
        private volatile boolean isRunning = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            if (filePath == null) {
                return;
            }
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(new GZIPInputStream(new FileInputStream(filePath))))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

        }
    }
