"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.
"""
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def rationalize_coordinate(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating point coordinate to a rational string representation.

    Args:
        value: The coordinate value (0 <= value < 1)
        max_denominator: Maximum denominator for the fraction

    Returns:
        String representation of the coordinate as a fraction or integer
    """
    # Normalize to [0, 1)
    value = value % 1.0

    # Handle special case of very small values (treat as 0)
    if abs(value) < 1e-6:
        return "0"

    # Handle special case of values very close to 1 (treat as 0)
    if abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return appropriate string representation
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: mapping Wyckoff letters to coordinates [x, y, z]
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get the symmetrized structure with space group information
    sga = SpacegroupAnalyzer(structure)
    symmetrized_structure = sga.get_symmetrized_structure()

    # Dictionary to store results
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Iterate through equivalent sites and their Wyckoff symbols
    for equiv_sites, wyckoff_symbol in zip(symmetrized_structure.equivalent_sites,
                                            symmetrized_structure.wyckoff_symbols):
        # Parse the Wyckoff symbol (e.g., "4a" -> multiplicity=4, letter="a")
        # Extract letter from symbol (last character)
        wyckoff_letter = wyckoff_symbol[-1]

        # Extract multiplicity from symbol (all characters except last)
        multiplicity = int(wyckoff_symbol[:-1])

        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get coordinates of the first atom in this Wyckoff position
        first_site = equiv_sites[0]
        coords = first_site.frac_coords

        # Convert coordinates to rational string format
        coord_strings = [
            rationalize_coordinate(coords[0]),
            rationalize_coordinate(coords[1]),
            rationalize_coordinate(coords[2])
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
