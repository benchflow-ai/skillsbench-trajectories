"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from typing import List, Tuple
import time


def get_smiles_from_name(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string representation of the molecule

    Raises:
        ValueError: If the chemical name cannot be found in PubChem
    """
    max_retries = 3
    retry_delay = 0.5

    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(chemical_name, 'name')
            if not compounds:
                raise ValueError(f"Chemical '{chemical_name}' not found in PubChem")
            # Use the non-deprecated smiles property
            smiles = compounds[0].isomeric_smiles
            return smiles
        except Exception as e:
            if attempt < max_retries - 1 and ('503' in str(e) or 'ServerBusy' in str(e) or 'Too many' in str(e)):
                time.sleep(retry_delay)
                retry_delay *= 2  # Exponential backoff
                continue
            raise ValueError(f"Error fetching SMILES for '{chemical_name}': {str(e)}")


def get_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES string representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality information (default: True)

    Returns:
        Morgan fingerprint

    Raises:
        ValueError: If SMILES string is invalid
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError(f"Invalid SMILES string: {smiles}")

    # Use newer MorganGenerator API if available, otherwise fall back to deprecated function
    try:
        from rdkit.Chem import rdFingerprintGenerator
        generator = rdFingerprintGenerator.GetMorganGenerator(radius=radius, includeChirality=use_chirality)
        return generator.GetFingerprint(mol)
    except (ImportError, AttributeError):
        # Fall back to older API
        return AllChem.GetMorganFingerprint(mol, radius, useChirality=use_chirality)


def calculate_tanimoto_similarity(fp1, fp2) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0-1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract list of molecule names from PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of molecule names
    """
    molecules = []

    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and filter out empty lines
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)

    return molecules


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k most similar molecules to a target molecule from a pool.

    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of top k most similar molecule names, sorted by similarity (descending),
        with alphabetical ordering for ties

    Raises:
        ValueError: If target molecule or pool molecules cannot be processed
    """
    # Get target molecule SMILES and fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    target_fp = get_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities: List[Tuple[str, float]] = []

    for molecule_name in pool_molecules:
        # Skip if it's the same as target
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        try:
            mol_smiles = get_smiles_from_name(molecule_name)
            mol_fp = get_morgan_fingerprint(mol_smiles, radius=2, use_chirality=True)
            similarity = calculate_tanimoto_similarity(target_fp, mol_fp)
            similarities.append((molecule_name, similarity))
            # Small delay to avoid rate limiting
            time.sleep(0.2)
        except Exception as e:
            # Skip molecules that cannot be processed
            print(f"Warning: Skipping '{molecule_name}': {str(e)}")
            continue

    # Sort by similarity (descending), then by name (alphabetical) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecule names
    top_k_molecules = [mol_name for mol_name, _ in similarities[:top_k]]

    return top_k_molecules
