"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit.DataStructs import TanimotoSimilarity
import pubchempy as pcp


def _extract_molecules_from_pdf(filepath):
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Each line in the PDF is a chemical name
                lines = text.strip().split('\n')
                molecules.extend([line.strip() for line in lines if line.strip()])
    return molecules


def _get_smiles_from_name(chemical_name):
    """
    Convert a chemical name to SMILES using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string or None if not found
    """
    try:
        # Search for the compound by name
        results = pcp.get_compounds(chemical_name, 'name')
        if results:
            # Get the first match
            compound = results[0]
            return compound.smiles
    except Exception:
        pass
    return None


def _get_morgan_fingerprint(smiles, radius=2):
    """
    Calculate Morgan fingerprint for a SMILES string.

    Args:
        smiles: SMILES representation of the molecule
        radius: Radius for Morgan fingerprint (default 2)

    Returns:
        Morgan fingerprint or None if invalid SMILES
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        # Generate Morgan fingerprint with specified radius and include chirality
        return AllChem.GetMorganFingerprintAsBitVect(mol, radius, nBits=2048, useChirality=True)
    except Exception:
        return None


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k similar chemicals to a target molecule using Morgan fingerprints
    and Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by:
        1. Similarity score (descending)
        2. Alphabetically by name (ascending) for ties
    """
    # Get target molecule SMILES and fingerprint
    target_smiles = _get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        return []

    target_fp = _get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        return []

    # Extract all molecules from the PDF
    molecule_names = _extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarity for each molecule
    similarities = []
    for mol_name in molecule_names:
        # Skip the target molecule itself
        if mol_name.lower() == target_molecule_name.lower():
            continue

        mol_smiles = _get_smiles_from_name(mol_name)
        if mol_smiles is None:
            continue

        mol_fp = _get_morgan_fingerprint(mol_smiles)
        if mol_fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = TanimotoSimilarity(target_fp, mol_fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending) and then alphabetically (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        target_molecule_name="Aspirin",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5
    )

    print(f"Top 5 molecules similar to Aspirin:")
    for mol_name, similarity in result:
        print(f"  {mol_name}: {similarity:.4f}")
