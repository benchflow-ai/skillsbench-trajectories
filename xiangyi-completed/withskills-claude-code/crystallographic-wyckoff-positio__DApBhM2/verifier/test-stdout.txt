"""
Wyckoff Position Analysis Module

This module provides functionality to analyze crystal structures from CIF files
and extract Wyckoff position multiplicities and representative coordinates.
"""

from typing import Any
from fractions import Fraction
from collections import defaultdict

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point coordinate to a fraction string.

    The value is normalized to [0, 1) range and then converted to the nearest
    fraction with denominator <= max_denominator.

    Args:
        value: The floating-point coordinate value
        max_denominator: Maximum allowed denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize to [0, 1) range
    normalized = value % 1.0
    if normalized < 0:
        normalized += 1.0

    # Handle values very close to 0 or 1
    if abs(normalized) < 1e-9 or abs(normalized - 1.0) < 1e-9:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(normalized).limit_denominator(max_denominator)

    if frac.denominator == 1:
        return str(frac.numerator % 1)  # Return "0" for whole numbers in [0,1)

    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze a CIF file to extract Wyckoff position multiplicities and
    representative coordinates.

    This function reads a crystal structure from a CIF file, performs symmetry
    analysis to identify the space group and Wyckoff positions, and returns
    the multiplicity and first atom coordinates for each unique Wyckoff position.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Dict mapping Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": Dict mapping Wyckoff letter to coordinates
          as a list of fraction strings ["x", "y", "z"]

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result["wyckoff_multiplicity_dict"]
        {'a': 4, 'c': 8}
        >>> result["wyckoff_coordinates_dict"]
        {'a': ['0', '1/2', '1/2'], 'c': ['3/8', '1/9', '8/9']}
    """
    # Load the crystal structure from CIF file
    structure = Structure.from_file(filepath)

    # Perform symmetry analysis
    spacegroup_analyzer = SpacegroupAnalyzer(structure)
    symmetry_dataset = spacegroup_analyzer.get_symmetry_dataset()

    # Extract Wyckoff letters for each site
    wyckoff_letters = symmetry_dataset.wyckoffs

    # Group sites by Wyckoff letter, keeping track of multiplicities
    # and storing the first occurrence's coordinates
    wyckoff_data = defaultdict(lambda: {"count": 0, "first_coords": None})

    for wyckoff_letter, site in zip(wyckoff_letters, structure.sites):
        data = wyckoff_data[wyckoff_letter]
        data["count"] += 1

        # Store coordinates of the first site encountered for this Wyckoff position
        if data["first_coords"] is None:
            data["first_coords"] = site.frac_coords.tolist()

    # Build the result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Sort by Wyckoff letter for consistent ordering
    for wyckoff_letter in sorted(wyckoff_data.keys()):
        data = wyckoff_data[wyckoff_letter]

        # Store multiplicity
        wyckoff_multiplicity_dict[wyckoff_letter] = data["count"]

        # Convert coordinates to fraction strings
        coords = data["first_coords"]
        fraction_coords = [float_to_fraction_string(c) for c in coords]
        wyckoff_coordinates_dict[wyckoff_letter] = fraction_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
