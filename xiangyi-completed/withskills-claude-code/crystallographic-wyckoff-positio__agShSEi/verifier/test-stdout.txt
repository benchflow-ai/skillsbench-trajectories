"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.

Uses pymatgen for crystal structure analysis and symmetry detection.
"""

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_rational_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a rational string with denominator <= max_denominator.

    Args:
        value: The floating point value to convert (expected in range [0, 1))
        max_denominator: Maximum denominator allowed for the fraction

    Returns:
        String representation of the fraction (e.g., "1/3", "0", "1/2")
    """
    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle edge cases where fraction becomes 0 or 1
    if frac.numerator == 0:
        return "0"
    if frac.numerator == frac.denominator:
        return "0"

    # Return as fraction string or integer if denominator is 1
    if frac.denominator == 1:
        return str(frac.numerator)

    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        Dictionary containing:
        - "wyckoff_multiplicity_dict": dict mapping Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": dict mapping Wyckoff letter to coordinates
          of the first atom, as list of rational strings

    Example output:
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Analyze symmetry
    sga = SpacegroupAnalyzer(structure)

    # Get symmetrized structure which contains Wyckoff information
    symmetrized_structure = sga.get_symmetrized_structure()

    # Get Wyckoff symbols - these are per-group, formatted as "multiplicity+letter"
    # e.g., ['4a', '8c'] for FeS2
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Get equivalent sites - groups of symmetry-equivalent atom indices
    equivalent_indices = symmetrized_structure.equivalent_indices

    # Build dictionaries for multiplicities and coordinates
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Track which Wyckoff positions we've already processed
    processed_wyckoffs = set()

    # Iterate through equivalent groups with their Wyckoff symbols
    for group_idx, group_indices in enumerate(equivalent_indices):
        # Get the full Wyckoff symbol for this group (e.g., "4a", "8c")
        full_wyckoff = wyckoff_symbols[group_idx]

        # Extract the letter part (last character) and multiplicity
        wyckoff_letter = full_wyckoff[-1]
        multiplicity = int(full_wyckoff[:-1])

        # Skip if we've already processed this Wyckoff position
        if wyckoff_letter in processed_wyckoffs:
            continue
        processed_wyckoffs.add(wyckoff_letter)

        # Store multiplicity
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get coordinates of the first atom in this Wyckoff position
        first_site_idx = group_indices[0]
        first_site = symmetrized_structure[first_site_idx]
        frac_coords = first_site.frac_coords

        # Convert to rational strings
        coord_strings = [
            float_to_rational_string(coord) for coord in frac_coords
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
