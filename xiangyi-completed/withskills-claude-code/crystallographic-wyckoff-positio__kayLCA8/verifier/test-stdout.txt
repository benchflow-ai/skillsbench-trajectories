#!/usr/bin/env python3
"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.
Uses pymatgen for crystallographic analysis to extract symmetry information.
"""

from fractions import Fraction
from typing import Any
from collections import defaultdict
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        A dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Maps Wyckoff letters to coordinates of first atom
                                   (as strings in fraction format)
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get spacegroup analyzer for symmetry analysis
    analyzer = SpacegroupAnalyzer(structure, symprec=0.1)

    # Get symmetrized structure to identify Wyckoff positions
    sym_struct = analyzer.get_symmetrized_structure()

    # Dictionary to store info by Wyckoff letter
    wyckoff_groups = defaultdict(list)

    # Get the Wyckoff symbols (one per unique atom/position)
    wyckoff_symbols = sym_struct.wyckoff_symbols

    # Track which atoms belong to which Wyckoff position
    # Use the symmetry dataset to identify equivalent atoms
    dataset = analyzer.get_symmetry_dataset()
    equivalent_atoms = dataset.equivalent_atoms

    # Process each site and group by Wyckoff position letter
    for i, site in enumerate(sym_struct):
        # Find the unique atom index this site corresponds to
        unique_idx = equivalent_atoms[i] if i < len(equivalent_atoms) else i

        # Get the Wyckoff symbol for this atom's position
        if unique_idx < len(wyckoff_symbols):
            wyckoff_symbol = wyckoff_symbols[unique_idx]
        else:
            # Fallback: find the representative Wyckoff symbol
            wyckoff_symbol = wyckoff_symbols[0] if wyckoff_symbols else "a"

        # Extract letter from symbol (e.g., "2a" -> "a", "8c" -> "c")
        wyckoff_letter = wyckoff_symbol.lstrip('0123456789')

        wyckoff_groups[wyckoff_letter].append({
            'coords': site.frac_coords,
            'species': str(site.species),
            'full_symbol': wyckoff_symbol
        })

    # Dictionary to store multiplicity for each Wyckoff letter
    wyckoff_multiplicity_dict = {}

    # Dictionary to store coordinates for first atom of each Wyckoff position
    wyckoff_coordinates_dict = {}

    # Process each Wyckoff position group
    for wyckoff_letter in sorted(wyckoff_groups.keys()):
        sites = wyckoff_groups[wyckoff_letter]

        # Multiplicity is the number of equivalent sites
        multiplicity = len(sites)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get coordinates of the first site in this Wyckoff position
        first_coords = sites[0]['coords']

        # Convert coordinates to fractions with denominator <= 12
        fractional_coords = [
            _to_rational_string(coord, max_denominator=12) for coord in first_coords
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = fractional_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _to_rational_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float value to a rational string representation with limited denominator.

    Args:
        value: Float value to convert
        max_denominator: Maximum denominator to use for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", or "0")
    """
    # Use Fraction to approximate the float as a rational number
    fraction = Fraction(value).limit_denominator(max_denominator)

    # Handle special cases
    if fraction.denominator == 1:
        return str(fraction.numerator)
    else:
        return f"{fraction.numerator}/{fraction.denominator}"


if __name__ == "__main__":
    # Example usage with the provided CIF file
    result = analyze_wyckoff_position_multiplicities_and_coordinates(
        "/root/cif_files/FeS2_mp-226.cif"
    )

    print("Wyckoff Multiplicity Dictionary:")
    print(result["wyckoff_multiplicity_dict"])
    print("\nWyckoff Coordinates Dictionary:")
    print(result["wyckoff_coordinates_dict"])
