"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.
"""
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dictionary mapping Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Dictionary mapping Wyckoff letter to coordinates [x, y, z]
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get the symmetrized structure to determine Wyckoff positions
    sga = SpacegroupAnalyzer(structure)
    symmetrized_structure = sga.get_symmetrized_structure()

    # Dictionary to store Wyckoff multiplicities
    wyckoff_multiplicity_dict = {}

    # Dictionary to store coordinates of the first atom in each Wyckoff position
    wyckoff_coordinates_dict = {}

    # Iterate through equivalent sites
    for equiv_sites in symmetrized_structure.equivalent_sites:
        # Get Wyckoff letter for this set of equivalent sites
        wyckoff_letter = equiv_sites[0].properties.get('wyckoff', None)

        if wyckoff_letter is None:
            # If wyckoff is not in properties, use wyckoff_symbol from SpacegroupAnalyzer
            continue

        # Get multiplicity (number of equivalent sites)
        multiplicity = len(equiv_sites)

        # Get coordinates of the first site in this Wyckoff position
        first_site = equiv_sites[0]
        coords = first_site.frac_coords

        # Convert coordinates to simplified fraction strings
        coord_strings = []
        for coord in coords:
            coord_strings.append(_decimal_to_fraction_string(coord))

        # Store in dictionaries
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    # If the above method didn't work (wyckoff not in properties),
    # use an alternative approach
    if not wyckoff_multiplicity_dict:
        wyckoff_multiplicity_dict, wyckoff_coordinates_dict = _analyze_wyckoff_alternative(
            structure, sga
        )

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def _analyze_wyckoff_alternative(structure, sga):
    """
    Alternative method to analyze Wyckoff positions using SpacegroupAnalyzer.
    """
    symmetrized_structure = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Get Wyckoff symbols for each site
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Group sites by element and Wyckoff position
    for i, equiv_sites in enumerate(symmetrized_structure.equivalent_sites):
        wyckoff_symbol = wyckoff_symbols[i]

        # Extract just the letter part (e.g., "4a" -> "a", "8c" -> "c")
        wyckoff_letter = _extract_wyckoff_letter(wyckoff_symbol)

        # Get multiplicity
        multiplicity = len(equiv_sites)

        # Get coordinates of the first site
        first_site = equiv_sites[0]
        coords = first_site.frac_coords

        # Convert coordinates to fraction strings
        coord_strings = []
        for coord in coords:
            coord_strings.append(_decimal_to_fraction_string(coord))

        # Store in dictionaries (keyed by Wyckoff letter only)
        # If there are multiple elements with the same Wyckoff letter,
        # we keep the first one encountered
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return wyckoff_multiplicity_dict, wyckoff_coordinates_dict


def _extract_wyckoff_letter(wyckoff_symbol: str) -> str:
    """
    Extract the letter part from a Wyckoff symbol.

    Args:
        wyckoff_symbol: Wyckoff symbol (e.g., "4a", "8c", "a", "c")

    Returns:
        The letter part only (e.g., "a", "c")
    """
    # Remove all digits from the symbol to get just the letter
    return ''.join(char for char in wyckoff_symbol if char.isalpha())


def _decimal_to_fraction_string(decimal_value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal value to a simplified fraction string.

    Args:
        decimal_value: The decimal number to convert
        max_denominator: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Handle values very close to 0 or 1
    if abs(decimal_value) < 1e-6:
        return "0"
    if abs(decimal_value - 1.0) < 1e-6:
        return "1"

    # Use Fraction to find the best rational approximation
    frac = Fraction(decimal_value).limit_denominator(max_denominator)

    # Format the output
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"
