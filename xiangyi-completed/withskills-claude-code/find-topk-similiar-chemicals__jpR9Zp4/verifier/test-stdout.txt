"""
Solution for finding top-k similar chemicals using Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to SMILES using PubChem
3. Compute Morgan fingerprints and Tanimoto similarity using RDKit
4. Return top-k most similar molecules sorted by similarity (descending),
   with alphabetical ordering for ties.
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem.rdFingerprintGenerator import GetMorganGenerator
from rdkit import DataStructs


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file containing chemical names.

    Returns:
        List of chemical names extracted from the PDF.
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Each line contains one chemical name
                for line in text.strip().split('\n'):
                    name = line.strip()
                    if name:
                        molecules.append(name)
    return molecules


def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to SMILES using PubChem.

    Args:
        chemical_name: The name of the chemical compound.

    Returns:
        Canonical SMILES string or None if not found.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


# Create Morgan fingerprint generator with radius=2 and chirality enabled
_morgan_generator = GetMorganGenerator(radius=2, includeChirality=True)


def get_morgan_fingerprint(smiles: str):
    """
    Generate Morgan fingerprint for a molecule from its SMILES.

    Uses Morgan fingerprints with radius=2 and chirality enabled.

    Args:
        smiles: SMILES string of the molecule.

    Returns:
        Morgan fingerprint bit vector or None if molecule cannot be parsed.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_generator.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to the target molecule.

    This function:
    1. Extracts molecule names from the given PDF file
    2. Converts each name to SMILES using PubChem
    3. Computes Morgan fingerprints (radius=2, with chirality)
    4. Calculates Tanimoto similarity between target and pool molecules
    5. Returns top-k results sorted by similarity (descending),
       with alphabetical ordering for ties

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to PDF file containing pool of molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score) for top-k similar molecules.
        Sorted by similarity in descending order, with alphabetical ordering for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Get fingerprint for target molecule
    target_fp = get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Extract molecule names from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities for all molecules in the pool
    similarities = []
    for mol_name in pool_molecules:
        # Get SMILES from PubChem
        smiles = get_smiles_from_name(mol_name)
        if smiles is None:
            continue

        # Get fingerprint
        fp = get_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (alphabetically) for ties
    # Using negative similarity for descending order, and name for alphabetical
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        print("Example: python solution.py Aspirin molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} molecules similar to '{target}':")
    print("-" * 50)
    for i, (name, sim) in enumerate(results, 1):
        print(f"{i}. {name}: {sim:.4f}")
