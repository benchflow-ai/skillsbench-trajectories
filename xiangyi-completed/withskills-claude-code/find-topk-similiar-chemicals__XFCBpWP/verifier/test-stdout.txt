"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to SMILES using PubChem
3. Compute Morgan fingerprints (radius=2, with chirality)
4. Calculate Tanimoto similarity between molecules
5. Return top-k most similar molecules sorted by similarity (descending),
   with alphabetical ordering for ties
"""

import time
from typing import Optional

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one per line."""
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)
    return molecules


def get_smiles_from_name(chemical_name: str, max_retries: int = 3, retry_delay: float = 2.0) -> Optional[str]:
    """
    Convert a chemical name to SMILES using PubChem.

    Includes retry logic to handle rate limiting (503 errors).
    Returns SMILES representation of the molecule.
    """
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(chemical_name, 'name')
            if compounds:
                return compounds[0].smiles
            return None
        except pcp.ServerBusyError:
            if attempt < max_retries - 1:
                time.sleep(retry_delay * (attempt + 1))
            else:
                raise
        except Exception:
            return None
    return None


def compute_morgan_fingerprint(mol):
    """
    Compute Morgan fingerprint for a molecule.

    Uses radius=2 and includes chirality as specified.
    """
    gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, includeChirality=True)
    return gen.GetFingerprint(mol)


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """Compute Tanimoto similarity between two fingerprints."""
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a pool of molecules in a PDF.

    Args:
        target_molecule_name: The name of the target molecule to compare against
        molecule_pool_filepath: Path to a PDF file containing molecule names (one per line)
        top_k: Number of top similar molecules to return

    Returns:
        A list of the top-k most similar molecule names, sorted by:
        1. Tanimoto similarity (descending)
        2. Alphabetical order (ascending) for ties

    The function uses:
    - PubChem to convert chemical names to SMILES
    - RDKit Morgan fingerprints (radius=2, with chirality)
    - Tanimoto similarity metric
    """
    # Get target molecule SMILES and create RDKit mol object
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_molecule_name}")

    target_fp = compute_morgan_fingerprint(target_mol)

    # Extract molecule names from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities for each molecule in the pool
    similarities = []
    for mol_name in pool_molecules:
        # Skip if it's the same molecule as target
        if mol_name.lower() == target_molecule_name.lower():
            continue

        try:
            smiles = get_smiles_from_name(mol_name)
            if smiles is None:
                continue

            mol = Chem.MolFromSmiles(smiles)
            if mol is None:
                continue

            fp = compute_morgan_fingerprint(mol)
            similarity = compute_tanimoto_similarity(target_fp, fp)
            similarities.append((mol_name, similarity))
        except Exception:
            # Skip molecules that can't be processed
            continue

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [mol_name for mol_name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_filepath> <top_k>")
        print("Example: python solution.py Aspirin molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    filepath = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, filepath, k)
    print(f"Top {k} molecules similar to {target}:")
    for i, mol in enumerate(results, 1):
        print(f"  {i}. {mol}")
