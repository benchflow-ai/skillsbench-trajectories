"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.
"""

import re
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_simple_fraction(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a simple fraction string with denominator <= max_denominator.

    Args:
        value: The float value to convert (expected to be in range [0, 1) for coordinates)
        max_denominator: Maximum allowed denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize to [0, 1) range
    value = value % 1.0
    if value < 0:
        value += 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # If the fraction is 0 or 1, return "0"
    if frac.numerator == 0 or frac.numerator == frac.denominator:
        return "0"

    # Return as string fraction
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with two keys:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letter to multiplicity count
        - "wyckoff_coordinates_dict": Maps Wyckoff letter to list of 3 coordinate strings
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Analyze symmetry to find the space group and Wyckoff positions
    analyzer = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure which contains Wyckoff information
    symmetrized_structure = analyzer.get_symmetrized_structure()

    # Get Wyckoff symbols - indexed by equivalent group, not by site
    # Format is like '4a', '8c' where the number is the multiplicity
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Get equivalent sites - groups of sites that are symmetrically equivalent
    equivalent_sites = symmetrized_structure.equivalent_sites

    # Build dictionaries for multiplicities and coordinates
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Iterate through each group of equivalent sites with corresponding Wyckoff symbol
    for group_idx, (wyckoff_full, sites) in enumerate(
        zip(wyckoff_symbols, equivalent_sites)
    ):
        # Parse the Wyckoff symbol (e.g., '4a' -> multiplicity=4, letter='a')
        # The multiplicity is the leading digits, the letter is at the end
        match = re.match(r"(\d+)([a-zA-Z]+)", wyckoff_full)
        if match:
            multiplicity = int(match.group(1))
            wyckoff_letter = match.group(2)
        else:
            # Fallback if format is unexpected
            wyckoff_letter = wyckoff_full
            multiplicity = len(sites)

        # Get fractional coordinates of the first site in this group
        first_site = sites[0]
        frac_coords = first_site.frac_coords

        # Convert coordinates to simple fractions
        coord_strings = [
            float_to_simple_fraction(coord) for coord in frac_coords
        ]

        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
