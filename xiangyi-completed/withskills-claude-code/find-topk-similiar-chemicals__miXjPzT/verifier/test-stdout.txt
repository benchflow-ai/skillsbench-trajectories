"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, rdMolDescriptors
from rdkit import DataStructs
from typing import List, Tuple, Optional
import time
import warnings


def get_smiles_from_name(chemical_name: str) -> Optional[str]:
    """
    Convert a chemical name to SMILES representation using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string if found, None otherwise
    """
    max_retries = 3
    retry_delay = 0.5

    for attempt in range(max_retries):
        try:
            # Add delay to avoid rate limiting
            time.sleep(0.2)
            compounds = pcp.get_compounds(chemical_name, 'name')
            if compounds:
                # Use smiles to get molecular representation
                return compounds[0].smiles
            return None
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                if attempt < max_retries - 1:
                    time.sleep(retry_delay * (attempt + 1))
                    continue
            return None
    return None


def smiles_to_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Convert SMILES to Morgan fingerprint.

    Args:
        smiles: SMILES representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality information (default: True)

    Returns:
        Morgan fingerprint or None if conversion fails
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        # Suppress deprecation warnings
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius, useChirality=use_chirality)
        return fp
    except Exception as e:
        return None


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract molecule names from PDF file.

    Args:
        pdf_filepath: Path to PDF file containing molecule names

    Returns:
        List of molecule names
    """
    molecules = []
    try:
        with pdfplumber.open(pdf_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.strip().split('\n')
                    for line in lines:
                        line = line.strip()
                        if line:
                            molecules.append(line)
    except Exception as e:
        print(f"Error reading PDF: {e}")

    return molecules


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find top k most similar molecules to target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of top k most similar molecule names, sorted by similarity (descending),
        with alphabetical ordering for ties
    """
    # Get target molecule SMILES and fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []

    target_fp = smiles_to_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)
    if target_fp is None:
        print(f"Could not generate fingerprint for target molecule: {target_molecule_name}")
        return []

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for molecule_name in pool_molecules:
        # Skip if it's the target molecule itself
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES and fingerprint
        smiles = get_smiles_from_name(molecule_name)
        if smiles is None:
            continue

        fp = smiles_to_morgan_fingerprint(smiles, radius=2, use_chirality=True)
        if fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((molecule_name, similarity))

    # Sort by similarity (descending), then by name (alphabetical) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecule names
    top_k_molecules = [name for name, _ in similarities[:top_k]]

    return top_k_molecules


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5

    print(f"Finding top {k} molecules similar to {target}...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for i, molecule in enumerate(results, 1):
        print(f"{i}. {molecule}")
