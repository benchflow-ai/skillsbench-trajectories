"""
Materials Science Crystal Structure Analysis
Analyzes Wyckoff position multiplicities and coordinates from CIF files

This module provides functionality to:
1. Read CIF (Crystallographic Information File) files
2. Determine the symmetry and space group of crystal structures
3. Identify Wyckoff positions and their multiplicities
4. Calculate approximate fractional coordinates as rational numbers

Dependencies:
- pymatgen: For reading CIF files and symmetry analysis
- fractions: For converting floating-point coordinates to rational form

Author: Materials Science Researcher
"""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_rational_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a rational string representation.

    Args:
        value: Float value to convert
        max_denominator: Maximum denominator for fraction simplification

    Returns:
        String representation of the rational number
    """
    # Handle values very close to integers
    rounded_int = round(value)
    if abs(value - rounded_int) < 1e-6:
        return str(rounded_int)

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates from CIF file.

    This function reads a CIF file, determines the crystal symmetry using pymatgen's
    SpacegroupAnalyzer, and extracts Wyckoff position information including:
    - The multiplicity of each Wyckoff position
    - The fractional coordinates of the first atom at each position (as rational strings)

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Mapping of Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Mapping of Wyckoff letters to coordinates of first atom
                                   (coordinates are represented as rational strings)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> print(result)
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }

    Notes:
        - Coordinates are converted to rational format with denominators ≤ 12
        - The function uses a symmetry precision (symprec) of 0.1 for analysis
        - In case of errors, returns a dict with 'error' and 'filepath' keys
    """
    try:
        # Read the structure from CIF file
        structure = Structure.from_file(filepath)

        # Get symmetrized structure using SpacegroupAnalyzer
        # Use a reasonable symmetry precision
        sga = SpacegroupAnalyzer(structure, symprec=0.1)

        # Get symmetrized structure
        symmetrized_structure = sga.get_symmetrized_structure()

        # Get Wyckoff symbols and equivalent sites
        wyckoff_symbols = symmetrized_structure.wyckoff_symbols
        equivalent_sites = symmetrized_structure.equivalent_sites

        # Dictionary to store Wyckoff multiplicities
        wyckoff_multiplicity_dict = {}

        # Dictionary to store first atom coordinates for each Wyckoff position
        wyckoff_coordinates_dict = {}

        # Process each group of equivalent sites
        for wyckoff_symbol, site_group in zip(wyckoff_symbols, equivalent_sites):
            # Parse Wyckoff symbol (e.g., '4a' -> multiplicity=4, letter='a')
            # Extract letter (last character) and multiplicity (remaining digits)
            wyckoff_letter = wyckoff_symbol[-1]
            multiplicity = int(wyckoff_symbol[:-1])

            # Get the first site coordinates
            first_site = site_group[0]
            frac_coords = first_site.frac_coords

            # Convert coordinates to rational string format
            coord_strings = [
                _float_to_rational_string(coord, max_denominator=12)
                for coord in frac_coords
            ]

            # Store in dictionaries
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

        # Return the result
        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict
        }

    except Exception as e:
        # Return error information
        return {
            "error": str(e),
            "filepath": filepath
        }
