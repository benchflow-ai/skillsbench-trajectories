"""
Wyckoff position analysis for crystal structures from CIF files.

This module analyzes Wyckoff position multiplicities and calculates approximate
coordinates for the first atom of each Wyckoff position in a crystal structure.
"""

from fractions import Fraction
from typing import Any, Dict

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates.

    This function reads a CIF file, analyzes the crystal structure, and extracts
    Wyckoff position information including multiplicities and coordinates of the
    first atom in each Wyckoff position.

    Args:
        filepath: Path to the CIF file to analyze.

    Returns:
        A dictionary with two keys:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letters to their multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff letters to coordinate lists
          for the first atom in that position (as fractional strings).

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result["wyckoff_multiplicity_dict"]
        {'a': 4, 'c': 8}
        >>> result["wyckoff_coordinates_dict"]
        {'a': ['0', '1/2', '1/2'], 'c': ['3/8', '1/9', '8/9']}
    """
    # Read the crystal structure from CIF file
    structure = Structure.from_file(filepath)

    # Get space group analyzer
    analyzer = SpacegroupAnalyzer(structure)

    # Get symmetry dataset
    dataset = analyzer.get_symmetry_dataset()

    # Extract Wyckoff positions and letters
    wyckoff_letters = dataset.wyckoffs

    # Build mapping of Wyckoff letters to atom indices
    wyckoff_to_atoms: Dict[str, list] = {}
    for atom_idx, wyckoff_letter in enumerate(wyckoff_letters):
        if wyckoff_letter not in wyckoff_to_atoms:
            wyckoff_to_atoms[wyckoff_letter] = []
        wyckoff_to_atoms[wyckoff_letter].append(atom_idx)

    # Calculate multiplicities and coordinates
    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list] = {}

    for wyckoff_letter, atom_indices in wyckoff_to_atoms.items():
        # Multiplicity is the number of equivalent atoms at this Wyckoff position
        multiplicity = len(atom_indices)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get the first atom's fractional coordinates
        first_atom_idx = atom_indices[0]
        frac_coords = structure.frac_coords[first_atom_idx]

        # Convert fractional coordinates to rational fractions with denominator <= 12
        rational_coords = [
            _float_to_fraction(coord) for coord in frac_coords
        ]

        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _float_to_fraction(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating-point number to a simplified fraction string.

    Args:
        value: The floating-point number to convert.
        max_denominator: Maximum denominator allowed (default 12).

    Returns:
        A string representation of the fraction (e.g., "1/2", "0", "3/8").
    """
    # Handle floating point precision issues
    value = round(value, 10)

    # Use Fraction to convert to rational number
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


if __name__ == "__main__":
    # Example usage
    test_file = "/root/cif_files/FeS2_mp-226.cif"
    try:
        result = analyze_wyckoff_position_multiplicities_and_coordinates(
            test_file
        )
        print("Wyckoff Multiplicity Dictionary:")
        print(result["wyckoff_multiplicity_dict"])
        print("\nWyckoff Coordinates Dictionary:")
        print(result["wyckoff_coordinates_dict"])
    except FileNotFoundError:
        print(f"File not found: {test_file}")
        print("Please provide a valid CIF file path.")
