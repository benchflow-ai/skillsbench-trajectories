"""
Wyckoff Position Analysis Module

This module provides functions to analyze crystal structures from CIF files,
extracting Wyckoff position multiplicities and coordinates for each unique
Wyckoff site.
"""

from typing import Any
from fractions import Fraction
from collections import defaultdict

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string with denominator constraint.

    Args:
        value: The float value to convert (assumed to be in range [0, 1])
        max_denominator: Maximum allowed denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize value to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return string representation
    if frac.denominator == 1:
        return str(frac.numerator % 1)  # Normalize to 0 if it's 1
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        Dictionary containing:
            - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to their multiplicities
            - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinates
              of the first atom in that Wyckoff position (as list of fraction strings)

    Example output:
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Create SpacegroupAnalyzer with Materials Project tolerance
    # Using symprec=0.1 as recommended for DFT-relaxed structures
    analyzer = SpacegroupAnalyzer(structure, symprec=0.1)

    # Get the symmetry dataset containing Wyckoff letters
    symmetry_dataset = analyzer.get_symmetry_dataset()
    wyckoff_letters = symmetry_dataset.wyckoffs

    # Track multiplicities and first coordinates for each Wyckoff position
    wyckoff_multiplicity = defaultdict(int)
    wyckoff_first_coords = {}

    # Iterate through all sites and their Wyckoff letters
    for i, wyckoff_letter in enumerate(wyckoff_letters):
        # Count multiplicity
        wyckoff_multiplicity[wyckoff_letter] += 1

        # Store coordinates of the first atom encountered for each Wyckoff position
        if wyckoff_letter not in wyckoff_first_coords:
            frac_coords = structure[i].frac_coords
            wyckoff_first_coords[wyckoff_letter] = frac_coords

    # Convert multiplicities to regular dict and sort by Wyckoff letter
    wyckoff_multiplicity_dict = dict(sorted(wyckoff_multiplicity.items()))

    # Convert coordinates to fraction strings
    wyckoff_coordinates_dict = {}
    for wyckoff_letter in sorted(wyckoff_first_coords.keys()):
        coords = wyckoff_first_coords[wyckoff_letter]
        coord_strings = [
            _float_to_fraction_string(coord) for coord in coords
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
