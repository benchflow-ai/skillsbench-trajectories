import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
import pubchempy as pcp


def extract_molecules_from_pdf(pdf_filepath):
    """
    Extract all chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file containing chemical names

    Returns:
        A list of chemical names (strings)
    """
    molecules = []

    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            # Split by newlines and filter empty lines
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            molecules.extend(lines)

    return molecules


def name_to_smiles(chemical_name):
    """
    Convert a chemical name to SMILES representation using PubChem.

    Args:
        chemical_name: The chemical name as a string

    Returns:
        SMILES string if found, None otherwise
    """
    try:
        # Query PubChem for the compound
        results = pcp.get_compounds(chemical_name, 'name')
        if results:
            # Return the canonical SMILES of the first result
            # Use canonical_smiles or fallback to isomeric_smiles
            smiles = results[0].canonical_smiles
            if smiles is None:
                smiles = results[0].isomeric_smiles
            return smiles
        return None
    except Exception:
        return None


def calculate_morgan_fingerprint(smiles, radius=2):
    """
    Calculate Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)

    Returns:
        Morgan fingerprint as ExplicitBitVect, or None if invalid SMILES
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        # radius=2, nBits=2048, useChirality=True
        fingerprint = AllChem.GetMorganFingerprintAsBitVect(mol, radius, nBits=2048, useChirality=True)
        return fingerprint
    except Exception:
        return None


def tanimoto_similarity(fp1, fp2):
    """
    Calculate Tanimoto similarity between two Morgan fingerprints.

    Args:
        fp1: First Morgan fingerprint
        fp2: Second Morgan fingerprint

    Returns:
        Tanimoto similarity score (0-1)
    """
    if fp1 is None or fp2 is None:
        return 0.0

    from rdkit import DataStructs
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k molecules in a PDF that are most similar to a target molecule
    using Morgan fingerprints and Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to PDF file containing the molecule pool
        top_k: Number of top similar molecules to return

    Returns:
        A list of (chemical_name, similarity_score) tuples, sorted by:
        1. Similarity score (descending)
        2. Chemical name (alphabetical ascending) for ties
    """
    # Extract all molecules from the PDF
    all_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Convert target molecule name to SMILES
    target_smiles = name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES representation for: {target_molecule_name}")

    # Calculate fingerprint for target molecule
    target_fp = calculate_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not calculate fingerprint for: {target_molecule_name}")

    # Calculate similarity scores for all molecules in the pool
    similarities = []
    for molecule_name in all_molecules:
        # Skip if it's the target molecule itself
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        # Convert to SMILES
        smiles = name_to_smiles(molecule_name)
        if smiles is None:
            continue

        # Calculate fingerprint
        fp = calculate_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Calculate similarity
        similarity = tanimoto_similarity(target_fp, fp)
        similarities.append((molecule_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    # Using negative similarity for descending order, name for ascending order
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]
