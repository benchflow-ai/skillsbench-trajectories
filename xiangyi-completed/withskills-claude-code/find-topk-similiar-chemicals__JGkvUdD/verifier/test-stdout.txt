"""
Solution for finding top k similar chemicals using Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from pubchempy import get_compounds
from typing import List, Tuple


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract all chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and filter empty lines
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)
    return molecules


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Get SMILES representation of a chemical using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string, or None if not found
    """
    try:
        compounds = get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].canonical_smiles
    except Exception:
        pass
    return None


def calculate_tanimoto_similarity(smiles1: str, smiles2: str, radius: int = 2) -> float:
    """
    Calculate Tanimoto similarity between two molecules using Morgan fingerprints.

    Args:
        smiles1: SMILES string of first molecule
        smiles2: SMILES string of second molecule
        radius: Radius for Morgan fingerprint (default: 2)

    Returns:
        Tanimoto similarity score (0-1)
    """
    try:
        mol1 = Chem.MolFromSmiles(smiles1)
        mol2 = Chem.MolFromSmiles(smiles2)

        if mol1 is None or mol2 is None:
            return 0.0

        # Generate Morgan fingerprints with chirality
        fp1 = AllChem.GetMorganFingerprintAsBitVect(mol1, radius, useChirality=True)
        fp2 = AllChem.GetMorganFingerprintAsBitVect(mol2, radius, useChirality=True)

        # Calculate Tanimoto similarity
        similarity = Chem.DataStructs.TanimotoSimilarity(fp1, fp2)
        return similarity
    except Exception:
        return 0.0


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k similar chemicals in a PDF to a target chemical.

    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to the PDF file containing chemicals
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (chemical_name, similarity_score) sorted by similarity (descending)
        and alphabetically when ties occur. Does not include the target molecule itself.
    """
    # Extract molecules from PDF
    molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        return []

    # Calculate similarities
    similarities = []
    for molecule_name in molecules:
        # Skip the target molecule itself
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        molecule_smiles = get_smiles_from_pubchem(molecule_name)
        if molecule_smiles is None:
            continue

        similarity = calculate_tanimoto_similarity(target_smiles, molecule_smiles)
        similarities.append((molecule_name, similarity))

    # Sort by similarity (descending), then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    pdf_path = "/root/molecules.pdf"

    # Test with a molecule
    result = topk_tanimoto_similarity_molecules("Caffeine", pdf_path, 5)
    print(f"Top 5 similar molecules to Caffeine:")
    for name, similarity in result:
        print(f"  {name}: {similarity:.4f}")
