package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ListState;
import org.apache.flink.api.common.state.ListStateDescriptor;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.CoProcessFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;
import org.apache.flink.util.Collector;

import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.time.Duration;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    // 10 minutes in microseconds
    private static final long SESSION_GAP_MICROS = 10L * 60L * 1000L * 1000L;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events and filter to SUBMIT events only
        DataStream<TaskEvent> taskEvents = env
                .addSource(taskSourceOrTest(new GzippedCsvSource<>(taskInput, TaskEvent::fromString)))
                .filter(event -> event != null && event.getEventType() == TaskEvent.SUBMIT)
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000) // Convert microseconds to milliseconds
                );

        // Read job events and filter to finished jobs
        DataStream<JobEvent> jobEvents = env
                .addSource(jobSourceOrTest(new GzippedCsvSource<>(jobInput, JobEvent::fromString)))
                .filter(event -> event != null && event.isJobFinished())
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000) // Convert microseconds to milliseconds
                );

        // Key by job ID and connect the two streams
        SingleOutputStreamOperator<Tuple2<Long, Integer>> result = taskEvents
                .keyBy(TaskEvent::getJobId)
                .connect(jobEvents.keyBy(JobEvent::getJobId))
                .process(new SessionProcessor());

        // Write output
        result.map(new MapFunction<Tuple2<Long, Integer>, String>() {
            @Override
            public String map(Tuple2<Long, Integer> value) throws Exception {
                return "(" + value.f0 + "," + value.f1 + ")";
            }
        }).writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    /**
     * Source function for reading gzipped CSV files.
     */
    public static class GzippedCsvSource<T> implements SourceFunction<T> {
        private final String filePath;
        private final java.util.function.Function<String, T> parser;
        private volatile boolean isRunning = true;

        public GzippedCsvSource(String filePath, java.util.function.Function<String, T> parser) {
            this.filePath = filePath;
            this.parser = parser;
        }

        @Override
        public void run(SourceContext<T> ctx) throws Exception {
            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(
                            new GZIPInputStream(
                                    new java.io.FileInputStream(filePath))))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    T event = parser.apply(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    /**
     * Process function that collects SUBMIT events and calculates longest stage when job finishes.
     * A stage is a sequence of SUBMIT events with gaps less than 10 minutes.
     */
    public static class SessionProcessor extends CoProcessFunction<TaskEvent, JobEvent, Tuple2<Long, Integer>> {

        // Store timestamps of all SUBMIT events for this job (in microseconds)
        private transient ListState<Long> submitTimestamps;
        // Flag to track if job has finished
        private transient ValueState<Boolean> jobFinished;

        @Override
        public void open(Configuration parameters) throws Exception {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Long.class));
            jobFinished = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("jobFinished", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Check if job already finished - if so, ignore late task events
            Boolean finished = jobFinished.value();
            if (finished != null && finished) {
                return;
            }
            // Add the SUBMIT event timestamp
            submitTimestamps.add(event.getTimestamp());
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Job finished event received
            Boolean finished = jobFinished.value();
            if (finished != null && finished) {
                return; // Already processed
            }
            jobFinished.update(true);

            // Calculate the longest stage
            int longestStage = calculateLongestStage();

            // Output result only if there were any SUBMIT events
            if (longestStage > 0) {
                out.collect(Tuple2.of(event.getJobId(), longestStage));
            }

            // Clear state
            submitTimestamps.clear();
        }

        private int calculateLongestStage() throws Exception {
            // Collect all timestamps
            List<Long> timestamps = new ArrayList<>();
            for (Long ts : submitTimestamps.get()) {
                timestamps.add(ts);
            }

            if (timestamps.isEmpty()) {
                return 0;
            }

            // Sort timestamps
            Collections.sort(timestamps);

            int longestStage = 1;
            int currentStage = 1;

            for (int i = 1; i < timestamps.size(); i++) {
                long gap = timestamps.get(i) - timestamps.get(i - 1);
                if (gap < SESSION_GAP_MICROS) {
                    // Same stage (gap less than 10 minutes)
                    currentStage++;
                } else {
                    // New stage (gap is 10 minutes or more)
                    longestStage = Math.max(longestStage, currentStage);
                    currentStage = 1;
                }
            }
            longestStage = Math.max(longestStage, currentStage);

            return longestStage;
        }
    }
}
