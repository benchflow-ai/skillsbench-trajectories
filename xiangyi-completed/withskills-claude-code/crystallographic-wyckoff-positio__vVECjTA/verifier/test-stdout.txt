"""
Module to analyze Wyckoff position multiplicities and coordinates from CIF files.
Uses pymatgen for crystal structure reading and symmetry analysis.
"""

from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string with denominator <= max_denominator.

    Args:
        value: The float value to convert (assumed to be in range [0, 1) for fractional coords)
        max_denominator: Maximum denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle edge case where fraction is 1/1 (should be 0 in fractional coords)
    if frac.numerator == frac.denominator:
        return "0"

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary with:
        - wyckoff_multiplicity_dict: dict mapping Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: dict mapping Wyckoff letter to [x, y, z] as fraction strings
    """
    # Read structure from CIF file
    structure = Structure.from_file(filepath)

    # Analyze symmetry
    sga = SpacegroupAnalyzer(structure)
    symmetry_dataset = sga.get_symmetry_dataset()

    # Get Wyckoff letters for each site
    wyckoff_letters = symmetry_dataset.wyckoffs

    # Get standard positions (these are the positions in the standard setting)
    std_positions = symmetry_dataset.std_positions

    # Dictionary to collect Wyckoff multiplicities
    wyckoff_multiplicity_dict: dict[str, int] = {}

    # Dictionary to collect first atom coordinates for each Wyckoff position
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    # Group sites by their Wyckoff letter and equivalent atom index
    # We need to find unique Wyckoff orbits
    wyckoff_orbit_first_atom: dict[str, int] = {}

    for i, wyckoff in enumerate(wyckoff_letters):
        # Count multiplicity for each unique Wyckoff letter
        if wyckoff not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff] = 0
        wyckoff_multiplicity_dict[wyckoff] += 1

        # Store the first atom index for each Wyckoff position
        if wyckoff not in wyckoff_orbit_first_atom:
            wyckoff_orbit_first_atom[wyckoff] = i

    # Get coordinates for the first atom of each Wyckoff position
    for wyckoff, first_idx in wyckoff_orbit_first_atom.items():
        coords = std_positions[first_idx]
        coord_strs = [float_to_fraction_str(c) for c in coords]
        wyckoff_coordinates_dict[wyckoff] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
