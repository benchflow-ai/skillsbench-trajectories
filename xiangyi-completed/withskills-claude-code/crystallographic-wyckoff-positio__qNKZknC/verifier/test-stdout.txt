"""
Script to analyze Wyckoff positions and coordinates from CIF files.
"""

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Dict, Any


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the first atom
    of each Wyckoff position in a crystal structure.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
            - "wyckoff_multiplicity_dict": Mapping of Wyckoff letters to multiplicities
            - "wyckoff_coordinates_dict": Mapping of Wyckoff letters to fractional coordinates
              of the first atom in each position (as strings in fraction format)
    """

    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get space group analyzer
    analyzer = SpacegroupAnalyzer(structure)

    # Get symmetry dataset which contains Wyckoff information
    dataset = analyzer.get_symmetry_dataset()

    # Get the standard positions and Wyckoff letters from the dataset
    std_positions = dataset.std_positions
    wyckoff_letters = dataset.wyckoffs

    # Dictionary to store multiplicity for each Wyckoff position
    wyckoff_multiplicity_dict = {}

    # Dictionary to store the first atom coordinates for each Wyckoff position
    wyckoff_coordinates_dict = {}

    # Track which Wyckoff positions we've already processed
    processed_wyckoffs = set()

    # Process each site and its Wyckoff position
    for site_index, (coords, wyckoff_letter) in enumerate(zip(std_positions, wyckoff_letters)):
        # If we haven't seen this Wyckoff position yet
        if wyckoff_letter not in processed_wyckoffs:
            processed_wyckoffs.add(wyckoff_letter)

            # Calculate the multiplicity by counting occurrences of this Wyckoff letter
            wyckoff_multiplicity = sum(1 for w in wyckoff_letters if w == wyckoff_letter)
            wyckoff_multiplicity_dict[wyckoff_letter] = wyckoff_multiplicity

            # Store the fractional coordinates of this first atom of this Wyckoff position
            rational_coords = [_to_rational_string(coord) for coord in coords]
            wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def _to_rational_string(decimal_value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal coordinate to a rational fraction string with denominator <= max_denominator.

    Args:
        decimal_value: The decimal coordinate value
        max_denominator: Maximum denominator for the fraction (default 12)

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0", etc.)
    """

    # Handle very small values (numerical errors)
    if abs(decimal_value) < 1e-10:
        return "0"

    # Handle values very close to 1
    if abs(decimal_value - 1.0) < 1e-10:
        return "1"

    # Find the best rational approximation
    frac = Fraction(decimal_value).limit_denominator(max_denominator)

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"
