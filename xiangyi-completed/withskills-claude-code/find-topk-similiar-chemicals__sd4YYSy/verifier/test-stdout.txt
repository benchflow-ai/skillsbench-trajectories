import pdfplumber
import pubchempy as pcp
import time
from rdkit import Chem
from rdkit.Chem import rdMolDescriptors
from rdkit.Chem import DataStructs


def get_smiles_from_name(chemical_name):
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        chemical_name: The name of the chemical

    Returns:
        SMILES string or None if not found
    """
    max_retries = 3
    retry_delay = 1

    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(chemical_name, 'name')
            if compounds:
                return compounds[0].isomeric_smiles
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                if attempt < max_retries - 1:
                    time.sleep(retry_delay)
                    retry_delay *= 2
                    continue
            print(f"Error fetching SMILES for {chemical_name}: {e}")
            break
    return None


def get_morgan_fingerprint(smiles, radius=2, include_chirality=True):
    """
    Generate Morgan fingerprint from SMILES string using MorganGenerator.

    Args:
        smiles: SMILES representation of the molecule
        radius: Morgan fingerprint radius (default=2)
        include_chirality: Whether to include chirality information (default=True)

    Returns:
        Morgan fingerprint or None if conversion fails
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        fp = rdMolDescriptors.GetMorganFingerprint(mol, radius, useChirality=include_chirality)
        return fp
    except Exception as e:
        print(f"Error generating fingerprint for {smiles}: {e}")
    return None


def calculate_tanimoto_similarity(fp1, fp2):
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First Morgan fingerprint
        fp2: Second Morgan fingerprint

    Returns:
        Tanimoto similarity score (0 to 1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def extract_molecules_from_pdf(pdf_filepath):
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k most similar molecules to a target molecule from a pool of molecules in a PDF file.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing the pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity (descending),
        then alphabetically for ties
    """
    # Get target molecule SMILES and fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    target_fp = get_morgan_fingerprint(target_smiles, radius=2, include_chirality=True)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for molecule_name in pool_molecules:
        # Get SMILES for pool molecule
        smiles = get_smiles_from_name(molecule_name)
        if smiles is None:
            print(f"Warning: Could not find SMILES for {molecule_name}, skipping")
            continue

        # Generate fingerprint
        fp = get_morgan_fingerprint(smiles, radius=2, include_chirality=True)
        if fp is None:
            print(f"Warning: Could not generate fingerprint for {molecule_name}, skipping")
            continue

        # Calculate similarity
        similarity = calculate_tanimoto_similarity(target_fp, fp)
        similarities.append((molecule_name, similarity))

    # Sort by similarity (descending), then alphabetically for ties
    # Using negative similarity for descending order, then molecule name for alphabetical order
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k results
    return similarities[:top_k]
