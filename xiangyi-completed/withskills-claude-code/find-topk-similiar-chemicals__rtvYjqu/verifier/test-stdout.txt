"""
Solution for finding top-k similar chemicals using Morgan fingerprints with Tanimoto similarity.
"""

import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs
import pdfplumber
from typing import List, Optional
from functools import lru_cache

# Create Morgan fingerprint generator with radius=2 and chirality enabled
_morgan_generator = rdFingerprintGenerator.GetMorganGenerator(
    radius=2,
    fpSize=2048,
    includeChirality=True
)


@lru_cache(maxsize=1000)
def get_smiles_from_name(chemical_name: str) -> Optional[str]:
    """
    Convert a chemical name to its canonical SMILES representation using PubChem.

    Args:
        chemical_name: The name of the chemical compound.

    Returns:
        The canonical SMILES string, or None if not found.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def extract_molecules_from_pdf(filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file containing chemical names.

    Returns:
        A list of chemical names extracted from the PDF.
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Each line contains a chemical name
                for line in text.strip().split('\n'):
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def compute_morgan_fingerprint(mol):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        mol: RDKit molecule object.

    Returns:
        Morgan fingerprint as a bit vector.
    """
    return _morgan_generator.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top-k most similar molecules to a target molecule from a pool of molecules
    stored in a PDF file, using Morgan fingerprints with Tanimoto similarity.

    Args:
        target_molecule_name: The name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing a list of molecule names.
        top_k: The number of top similar molecules to return.

    Returns:
        A list of molecule names sorted by descending similarity to the target.
        When similarities are equal, molecules are sorted alphabetically.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Create RDKit molecule and fingerprint for target
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_molecule_name}")

    target_fp = compute_morgan_fingerprint(target_mol)

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities for each molecule in the pool
    similarities = []
    for mol_name in pool_molecules:
        # Get SMILES for pool molecule
        pool_smiles = get_smiles_from_name(mol_name)
        if pool_smiles is None:
            continue

        # Create RDKit molecule and fingerprint
        pool_mol = Chem.MolFromSmiles(pool_smiles)
        if pool_mol is None:
            continue

        pool_fp = compute_morgan_fingerprint(pool_mol)

        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, pool_fp)

        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [mol_name for mol_name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        print("Example: python solution.py Aspirin /root/molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"Top {k} similar molecules to {target}:")
    for i, mol in enumerate(results, 1):
        print(f"  {i}. {mol}")
