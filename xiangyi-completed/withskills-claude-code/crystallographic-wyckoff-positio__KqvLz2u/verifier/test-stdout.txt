"""
Crystal structure analysis tool for Wyckoff position multiplicities and coordinates.
"""

from pymatgen.core.structure import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the first
    atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Maps Wyckoff letters to fractional coordinates
                                    of the first atom, as rational fractions (str format)
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get spacegroup analyzer
    analyzer = SpacegroupAnalyzer(structure)

    # Get symmetry data
    symmetry_data = analyzer.get_symmetry_dataset()

    # Get Wyckoff positions from symmetry data
    wyckoff_letters = symmetry_data.wyckoffs

    # Dictionary to store multiplicity for each Wyckoff position
    wyckoff_multiplicity_dict = {}
    # Dictionary to store first atom coordinates for each Wyckoff position
    wyckoff_coordinates_dict = {}
    # Track first atom for each Wyckoff position
    wyckoff_first_atom = {}

    # Process each atom and its Wyckoff position
    for i, (atom, wyckoff) in enumerate(zip(structure, wyckoff_letters)):
        if wyckoff not in wyckoff_multiplicity_dict:
            # First time seeing this Wyckoff position
            # Count how many atoms have this Wyckoff letter
            count = sum(1 for w in wyckoff_letters if w == wyckoff)
            wyckoff_multiplicity_dict[wyckoff] = count
            wyckoff_first_atom[wyckoff] = atom

    # Convert coordinates to rational fractions
    for wyckoff, atom in wyckoff_first_atom.items():
        coords = []
        for coord in atom.frac_coords:
            # Convert to fraction with denominator <= 12
            frac = _to_rational_fraction(coord, max_denominator=12)
            coords.append(frac)
        wyckoff_coordinates_dict[wyckoff] = coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _to_rational_fraction(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a rational fraction string representation.

    Args:
        value: Float value to convert
        max_denominator: Maximum denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Handle values very close to integers
    if abs(value - round(value)) < 1e-10:
        return str(int(round(value)))

    # Use Fraction to get rational representation
    frac = Fraction(value).limit_denominator(max_denominator)

    # Format the fraction
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"
