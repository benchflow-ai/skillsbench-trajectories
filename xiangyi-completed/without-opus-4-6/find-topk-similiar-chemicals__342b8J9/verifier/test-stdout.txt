import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator, DataStructs

_morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _extract_chemical_names(filepath: str) -> list[str]:
    """Extract chemical names from a PDF file, one name per line."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split("\n"):
                    name = line.strip()
                    if name:
                        names.append(name)
    return names


def _name_to_smiles(name: str) -> str | None:
    """Convert a chemical name to SMILES using PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            return compounds[0].isomeric_smiles
    except Exception:
        pass
    return None


def _smiles_to_morgan_fp(smiles: str):
    """Convert SMILES to a Morgan fingerprint."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_gen.GetFingerprint(mol)


def _tanimoto_similarity(fp1, fp2) -> float:
    """Compute Tanimoto similarity between two fingerprints."""
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """
    Find the top-k most similar chemicals from a PDF file to a target molecule.

    Uses Morgan fingerprints (radius=2, chirality-aware) with Tanimoto similarity.
    Results are sorted by descending similarity, with alphabetical ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing chemical names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of the top-k most similar molecule names.
    """
    # Get the SMILES and fingerprint for the target molecule
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not find SMILES for target molecule: {target_molecule_name}"
        )
    target_fp = _smiles_to_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Could not generate fingerprint for target molecule: {target_molecule_name}"
        )

    # Extract chemical names from the PDF
    pool_names = _extract_chemical_names(molecule_pool_filepath)

    # Compute similarity for each molecule in the pool
    similarities = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _smiles_to_morgan_fp(smiles)
        if fp is None:
            continue
        sim = _tanimoto_similarity(target_fp, fp)
        similarities.append((name, sim))

    # Sort by descending similarity, then alphabetically by name for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in similarities[:top_k]]
