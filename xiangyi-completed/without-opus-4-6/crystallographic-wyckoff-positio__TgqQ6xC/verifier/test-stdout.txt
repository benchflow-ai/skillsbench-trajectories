from typing import Any
from fractions import Fraction
from collections import Counter

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _coord_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """Convert a fractional coordinate to a rational string representation.

    Constrains denominators to at most ``max_denominator`` so that results
    stay human-readable (e.g. "3/8" instead of "1927/5000").
    """
    # Wrap into [0, 1) so negative or >1 coords are normalised
    value = value % 1.0
    # Snap values very close to 0 or 1 after modding
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    frac = Fraction(value).limit_denominator(max_denominator)

    # After rounding, the fraction could be 0/1 or 1/1; both map to "0"
    if frac.denominator == 1:
        return str(frac.numerator % 1)  # 0 % 1 = 0, 1 % 1 = 0
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyse Wyckoff position multiplicities and first-atom coordinates.

    Parameters
    ----------
    filepath : str
        Path to a CIF file (may be in P1 symmetry).

    Returns
    -------
    dict
        ``wyckoff_multiplicity_dict`` – mapping from Wyckoff key to its
        multiplicity (number of equivalent sites generated by symmetry).

        ``wyckoff_coordinates_dict`` – mapping from Wyckoff key to the
        fractional coordinates of the first atom, expressed as rational
        strings with denominators ≤ 12.
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    sym_struct = sga.get_symmetrized_structure()

    wyckoff_symbols = sym_struct.wyckoff_symbols  # e.g. ['4a', '8c']
    equiv_sites = sym_struct.equivalent_sites     # list of lists of PeriodicSite

    # Extract the letter part from each Wyckoff symbol (e.g. '4a' -> 'a')
    wyckoff_letters = []
    for ws in wyckoff_symbols:
        letter = "".join(c for c in ws if c.isalpha())
        wyckoff_letters.append(letter)

    # Count how many times each letter appears to decide on numbering
    letter_counts = Counter(wyckoff_letters)

    # Track how many times we've seen each letter so far (for indexing)
    letter_seen: dict[str, int] = {}

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for idx, (letter, sites) in enumerate(zip(wyckoff_letters, equiv_sites)):
        multiplicity = len(sites)

        # Build a key: plain letter if unique, else letter + 1-based index
        if letter_counts[letter] == 1:
            key = letter
        else:
            seen = letter_seen.get(letter, 0) + 1
            letter_seen[letter] = seen
            key = f"{letter}{seen}"

        wyckoff_multiplicity_dict[key] = multiplicity

        # First site coordinates
        coords = sites[0].frac_coords
        coord_strs = [_coord_to_fraction_str(c) for c in coords]
        wyckoff_coordinates_dict[key] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
