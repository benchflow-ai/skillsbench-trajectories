from typing import Any

import spglib
import pymatgen.core as mg
from fractions import Fraction


def _float_to_frac_str(val: float, max_denom: int = 12) -> str:
    """Convert a float to a simplified fraction string with denominator <= max_denom.

    Values are normalized to [0, 1) first, then converted to the nearest
    rational approximation with denominator at most ``max_denom``.
    """
    # Normalize to [0, 1)
    val = val % 1.0
    # Clamp near-integer values
    if abs(val) < 1e-6 or abs(val - 1.0) < 1e-6:
        return "0"
    frac = Fraction(val).limit_denominator(max_denom)
    if frac.denominator == 1:
        return str(frac.numerator % 1)  # should be 0
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and first-atom coordinates from a CIF file.

    Parameters
    ----------
    filepath : str
        Path to a CIF file (typically generated by pymatgen in P1 setting).

    Returns
    -------
    dict
        A dictionary with two keys:
        - ``wyckoff_multiplicity_dict``: mapping from Wyckoff letter to its
          multiplicity (number of symmetry-equivalent positions).
        - ``wyckoff_coordinates_dict``: mapping from Wyckoff letter to the
          fractional coordinates of the first atom in that Wyckoff position,
          expressed as simplified fraction strings.
    """
    # Read the structure
    struct = mg.Structure.from_file(filepath)

    # Build spglib cell tuple
    lattice = struct.lattice.matrix.tolist()
    positions = [site.frac_coords.tolist() for site in struct]
    numbers = [site.specie.Z for site in struct]
    cell = (lattice, positions, numbers)

    # Determine symmetry
    dataset = spglib.get_symmetry_dataset(cell)

    wyckoff_letters = dataset.wyckoffs
    orbits = dataset.crystallographic_orbits

    # Collect unique Wyckoff letters in order of first appearance,
    # and compute multiplicity as atoms-per-orbit.
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    seen_letters: set[str] = set()

    for i, letter in enumerate(wyckoff_letters):
        if letter not in seen_letters:
            seen_letters.add(letter)

            # Multiplicity = number of atoms sharing the same orbit as this atom
            this_orbit = orbits[i]
            multiplicity = int(sum(1 for o in orbits if o == this_orbit))
            wyckoff_multiplicity_dict[letter] = multiplicity

            # Coordinates of the first atom in this Wyckoff position
            coords = positions[i]
            wyckoff_coordinates_dict[letter] = [
                _float_to_frac_str(c) for c in coords
            ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
