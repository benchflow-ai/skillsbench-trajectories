from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_frac_str(value: float, max_denominator: int = 12) -> str:
    """Convert a float to a fraction string with denominator ≤ max_denominator.

    Values are wrapped into [0, 1) before conversion.  A result of 1
    (which is equivalent to 0 in fractional crystallographic coordinates)
    is mapped back to "0".
    """
    value = value % 1.0
    frac = Fraction(value).limit_denominator(max_denominator)
    if frac >= 1:
        frac = Fraction(0)
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and calculate approximate
    coordinates for the first atom of each Wyckoff position from a CIF file.

    Parameters
    ----------
    filepath : str
        Path to a CIF file.

    Returns
    -------
    dict with keys:
        "wyckoff_multiplicity_dict" : dict mapping Wyckoff letter -> multiplicity (int)
        "wyckoff_coordinates_dict"  : dict mapping Wyckoff letter -> [x, y, z] as
                                      fraction strings with denominators ≤ 12.
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    dataset = sga.get_symmetry_dataset()

    wyckoff_letters = dataset.wyckoffs
    equivalent_atoms = dataset.equivalent_atoms
    std_positions = dataset.std_positions

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    seen: set[str] = set()
    for idx in range(len(wyckoff_letters)):
        letter = wyckoff_letters[idx]
        if letter in seen:
            continue
        seen.add(letter)

        # Count how many atoms share this Wyckoff letter
        multiplicity = sum(1 for w in wyckoff_letters if w == letter)
        wyckoff_multiplicity_dict[letter] = multiplicity

        # Use the first representative atom (the one pointed to by equivalent_atoms)
        rep_idx = equivalent_atoms[idx]
        coords = std_positions[rep_idx]
        wyckoff_coordinates_dict[letter] = [
            _float_to_frac_str(c) for c in coords
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
