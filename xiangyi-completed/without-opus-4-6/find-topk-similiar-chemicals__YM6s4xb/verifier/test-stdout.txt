import PyPDF2
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator

_morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _extract_molecule_names(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one name per line."""
    reader = PyPDF2.PdfReader(filepath)
    names = []
    for page in reader.pages:
        text = page.extract_text()
        if text:
            for line in text.strip().splitlines():
                name = line.strip()
                if name:
                    names.append(name)
    return names


def _name_to_smiles(name: str) -> str | None:
    """Convert a chemical name to a canonical SMILES string using PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def _smiles_to_morgan_fp(smiles: str):
    """Convert a SMILES string to a Morgan fingerprint bit vector."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """Find the top-k most similar molecules from a PDF pool to a target molecule.

    Uses Morgan fingerprints (radius=2, chirality-aware) with Tanimoto similarity.
    Results are sorted by descending similarity, with alphabetical ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names (strings) most similar to the target, length <= top_k.
    """
    # Get SMILES and fingerprint for the target molecule
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not resolve target molecule '{target_molecule_name}' to a SMILES."
        )
    target_fp = _smiles_to_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Could not generate fingerprint for target molecule '{target_molecule_name}'."
        )

    # Extract pool molecule names from the PDF
    pool_names = _extract_molecule_names(molecule_pool_filepath)

    # Compute similarities
    results: list[tuple[float, str]] = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _smiles_to_morgan_fp(smiles)
        if fp is None:
            continue
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((similarity, name))

    # Sort by descending similarity, then alphabetically by name for ties
    results.sort(key=lambda x: (-x[0], x[1]))

    # Return the top-k molecule names
    return [name for _, name in results[:top_k]]


if __name__ == "__main__":
    # Quick demo
    import sys

    target = sys.argv[1] if len(sys.argv) > 1 else "Aspirin"
    pdf_path = sys.argv[2] if len(sys.argv) > 2 else "/root/molecules.pdf"
    k = int(sys.argv[3]) if len(sys.argv) > 3 else 5

    print(f"Target: {target}")
    print(f"Pool: {pdf_path}")
    print(f"Top-{k} similar molecules:")
    for i, name in enumerate(
        topk_tanimoto_similarity_molecules(target, pdf_path, k), 1
    ):
        print(f"  {i}. {name}")
