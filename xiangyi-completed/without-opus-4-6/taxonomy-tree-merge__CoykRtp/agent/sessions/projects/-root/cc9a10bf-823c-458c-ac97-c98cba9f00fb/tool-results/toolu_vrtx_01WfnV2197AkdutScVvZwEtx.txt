    25→def load_categories(filepath, source):
    26→    """Load categories from a CSV file and return list of (source, path_parts, original_path)."""
    27→    results = []
    28→    with open(filepath, 'r', encoding='utf-8-sig') as f:
    29→        reader = csv.DictReader(f)
    30→        for row in reader:
    31→            raw = row['category_path'].strip().strip('"')
    32→            parts = [p.strip() for p in raw.split('>')]
    33→            parts = [p for p in parts if p]
    34→            if parts:
    35→                results.append((source, parts, raw))
    36→    return results
    37→
    38→# ---------------------------------------------------------------------------
    39→# 2. TEXT NORMALIZATION
    40→# ---------------------------------------------------------------------------
    41→
    42→STOP_WORDS = {'&', 'and', 'or', 'the', 'a', 'an', 'of', 'for', 'in', 'on', 'to', 'with', 'by'}
    43→
    44→def normalize_text(text):
    45→    """Normalize category text to Title Case, strip extra whitespace."""
    46→    text = text.strip()
    47→    text = re.sub(r'\s+', ' ', text)
    48→    # Title-case each word
    49→    words = text.split()
    50→    result = []
    51→    for w in words:
    52→        if w.lower() in STOP_WORDS and len(words) > 1:
    53→            result.append(w.lower())
    54→        else:
    55→            result.append(w.capitalize() if w.islower() or w.isupper() else w)
    56→    return ' '.join(result)
    57→
    58→def standardize_name(name):
    59→    """Further standardize a category name."""
    60→    name = normalize_text(name)
    61→    # Common replacements for consistency
    62→    replacements = {
    63→        'Tvs': 'TVs',

    79→        'Suv': 'SUV',
    80→    }
    81→    for old, new in replacements.items():
    82→        name = name.replace(old, new)
    83→    return name

   170→def get_word_set(text):
   171→    """Get set of meaningful words from text."""
   172→    words = re.findall(r'[a-zA-Z]+', text.lower())
   173→    return set(w for w in words if w not in STOP_WORDS and len(w) > 1)
   174→
   175→def word_overlap_ratio(a, b):
   176→    """Calculate word overlap ratio between two strings."""
   177→    wa = get_word_set(a)
   178→    wb = get_word_set(b)
   179→    if not wa or not wb:
   180→        return 0.0
   181→    intersection = wa & wb
   182→    return len(intersection) / min(len(wa), len(wb))
   183→
   184→
   185→def pick_representative_name(names, parent_name=""):
   186→    """Pick the most representative name from a cluster of names.
   187→
   188→    The representative should cover 70%+ of the concepts in the group.
   189→    """
   190→    if not names:
   191→        return "Other"
   192→
   193→    if len(names) == 1:
   194→        return standardize_name(names[0])
   195→
   196→    # Count word frequencies across all names
   197→    word_freq = Counter()
   198→    for n in names:
   199→        words = get_word_set(n)
   200→        word_freq.update(words)
   201→
   202→    # Score each name by how many of the overall concepts it covers
   203→    best_name = names[0]
   204→    best_score = -1
   205→    for n in names:
   206→        words = get_word_set(n)
   207→        score = sum(word_freq[w] for w in words)
   208→        # Penalize very long names
   209→        if len(n.split()) > 5:
   210→            score *= 0.7
   211→        if score > best_score:
   212→            best_score = score
   213→            best_name = n
   214→
   215→    result = standardize_name(best_name)
   216→
   217→    # Ensure no overlap with parent
   218→    if parent_name:
   219→        parent_words = get_word_set(parent_name)
   220→        result_words = get_word_set(result)
   221→        if result_words and result_words.issubset(parent_words):
   222→            # All words overlap with parent - need to differentiate
   223→            # Find a distinguishing word from the children
   224→            all_child_words = Counter()
   225→            for n in names:
   226→                all_child_words.update(get_word_set(n) - parent_words)
   227→            if all_child_words:
   228→                top_word = all_child_words.most_common(1)[0][0]
   229→                result = result + " | " + top_word.capitalize()
   230→
   231→    return result
   232→
   233→def truncate_name(name, max_words=5):
   234→    """Truncate name to at most max_words, using ' | ' as separator if needed."""
   235→    # Split by ' | '
   236→    pipe_parts = [p.strip() for p in name.split('|')]
   237→    words = []
   238→    for p in pipe_parts:
   239→        words.extend(p.split())
   240→
   241→    if len(words) <= max_words:
   242→        return name
   243→
   244→    # Take up to max_words meaningful words
   245→    kept = words[:max_words]
   246→    return ' '.join(kept)

   494→    def match_to_l2_cluster(l1, source_l2_text):
   495→        """Match a source level-2 category text to a unified L2 cluster."""
   496→        text_lower = source_l2_text.lower()
   497→        clusters = L2_CLUSTER_RULES.get(l1, {})
   498→
   499→        best_match = None
   500→        best_score = 0
   501→
   502→        for cluster_name, keywords in clusters.items():
   503→            score = 0
   504→            for kw in keywords:
   505→                if kw in text_lower:
   506→                    score += len(kw)  # longer keyword matches score higher
   507→            if score > best_score:
   508→                best_score = score
   509→                best_match = cluster_name
   510→
   511→        if best_match and best_score > 0:
   512→            return best_match
   513→
   514→        # Fallback: try word overlap
   515→        for cluster_name, keywords in clusters.items():
   516→            cluster_words = set()
   517→            for kw in keywords:
   518→                cluster_words.update(kw.split())
   519→            text_words = set(text_lower.split())
   520→            overlap = len(cluster_words & text_words)
   521→            if overlap > best_score:
   522→                best_score = overlap
   523→                best_match = cluster_name
   524→
   525→        if best_match and best_score >= 1:
   526→            return best_match
   527→
   528→        return "Other " + l1.split(' | ')[0]
   529→
   530→    # Now process every entry
   531→    # For each entry, we map: source path -> (L1, L2, L3, L4, L5)
   532→
   533→    # Strategy for L3-L5:
   534→    # - Use the source category hierarchy levels, normalized and cleaned
   535→    # - L3 comes from source level 3 (or level 2 sub-detail)
   536→    # - L4 comes from source level 4
   537→    # - L5 comes from source level 5+
   538→
   539→    raw_mappings = []  # (source, raw_path, depth, l1, source_parts)
   540→
   541→    for source, parts, raw in all_entries:
   542→        l1 = UNIFIED_L1_MAP.get(parts[0], 'General | Miscellaneous')
   543→        depth = len(parts)
   544→
   545→        # Determine L2 from parts[1] if available, else from parts[0]
   546→        if len(parts) >= 2:
   547→            # Combine parts[0] and parts[1] context for better matching
   548→            l2_text = parts[1]
   549→            # Also include parts[0] for context if it adds info
   550→            full_context = parts[0] + ' ' + parts[1]
   551→            l2 = match_to_l2_cluster(l1, full_context)
   552→        else:
   553→            l2 = match_to_l2_cluster(l1, parts[0])
   554→
   555→        # For L3-L5, use source hierarchy parts, standardized
   556→        # Map source levels to unified levels:
   557→        # Source L1 -> Unified L1 (already mapped)
   558→        # Source L2 -> Unified L2 (cluster mapped)
   559→        # Source L3+ -> Unified L3-L5
   560→
   561→        # We need to condense potentially deep hierarchies into 5 levels
   562→        # Source depth can be 1-9, but unified is always exactly 5
   563→
   564→        # Collect remaining levels after L1 and L2 mapping
   565→        remaining = parts[1:]  # everything after source L1
   566→
   567→        # We've already used parts[1] for L2 clustering, so L3 starts from parts[2]
   568→        # But sometimes parts[1] contains important sub-info not captured by L2 name
   569→
   570→        l3 = ""
   571→        l4 = ""
   572→        l5 = ""
   573→
   574→        if len(parts) >= 3:
   575→            l3 = standardize_name(parts[2])
   576→        elif len(parts) >= 2:
   577→            # Use parts[1] as L3 if no deeper level
   578→            l3 = standardize_name(parts[1])
   579→
   580→        if len(parts) >= 4:
   581→            l4 = standardize_name(parts[3])
   582→        elif len(parts) >= 3:
   583→            l4 = ""
   584→
   585→        if len(parts) >= 5:
   586→            # Combine remaining levels into L5
   587→            l5_parts = [standardize_name(p) for p in parts[4:]]
   588→            l5 = standardize_name(' | '.join(l5_parts[:2]))  # Take at most 2 extra levels
   589→
   590→        raw_mappings.append((source, raw, depth, l1, l2, l3, l4, l5))
   591→

   593→    # POST-PROCESS: Clean up L3-L5 names, ensure no parent-child overlap,
   594→    # ensure sibling distinctness, and truncate to 5 words max
   595→    # ---------------------------------------------------------------------------
   596→
   597→    # Group by (L1, L2) to clean up L3 names
   598→    l1_l2_groups = defaultdict(list)
   599→    for i, (source, raw, depth, l1, l2, l3, l4, l5) in enumerate(raw_mappings):
   600→        l1_l2_groups[(l1, l2)].append(i)

   602→    # Clean up L3: group similar L3 names within each (L1, L2) and consolidate
   603→    def clean_level_names(mappings, level_idx, parent_level_idx):
   604→        """Consolidate similar names at a given level within each parent group."""
   605→        # Group by parent
   606→        parent_groups = defaultdict(lambda: defaultdict(list))
   607→        for i, m in enumerate(mappings):
   608→            parent_key = tuple(m[3:3+parent_level_idx])  # l1..parent
   609→            level_name = m[3 + level_idx]  # current level name
   610→            if level_name:
   611→                parent_groups[parent_key][level_name].append(i)
   612→
   613→        # For each parent, check if there are too many children or overlapping names
   614→        for parent_key, children in parent_groups.items():
   615→            if len(children) <= 20:
   616→                continue
   617→
   618→            # Too many children - need to merge similar ones
   619→            names = list(children.keys())
   620→            # Simple clustering: group by first word overlap
   621→            clusters = defaultdict(list)
   622→            used = set()
   623→
   624→            for n in sorted(names, key=lambda x: -len(children[x])):
   625→                if n in used:
   626→                    continue
   627→                cluster_key = n
   628→                clusters[cluster_key].append(n)
   629→                used.add(n)
   630→
   631→                n_words = get_word_set(n)
   632→                for m in names:
   633→                    if m in used:
   634→                        continue
   635→                    m_words = get_word_set(m)
   636→                    if n_words and m_words:
   637→                        overlap = len(n_words & m_words) / min(len(n_words), len(m_words))
   638→                        if overlap >= 0.5:
   639→                            clusters[cluster_key].append(m)
   640→                            used.add(m)
   641→
   642→            # Rename merged clusters
   643→            for cluster_key, cluster_names in clusters.items():
   644→                if len(cluster_names) > 1:
   645→                    rep_name = pick_representative_name(cluster_names, '|'.join(parent_key))
   646→                    for cn in cluster_names:
   647→                        for idx in children[cn]:
   648→                            m = list(mappings[idx])
   649→                            m[3 + level_idx] = rep_name
   650→                            mappings[idx] = tuple(m)
   651→
   652→    # Clean L3, L4, L5
   653→    clean_level_names(raw_mappings, 2, 2)  # L3 within (L1,L2)
   654→    clean_level_names(raw_mappings, 3, 3)  # L4 within (L1,L2,L3)
   655→    clean_level_names(raw_mappings, 4, 4)  # L5 within (L1,L2,L3,L4)
   656→
   657→    # Now fix parent-child name overlaps and truncate
   658→    final_mappings = []
   659→    for source, raw, depth, l1, l2, l3, l4, l5 in raw_mappings:
   660→        # Truncate each level to max 5 words
   661→        l1 = truncate_name(l1)
   662→        l2 = truncate_name(l2)
   663→        l3 = truncate_name(l3)
   664→        l4 = truncate_name(l4)
   665→        l5 = truncate_name(l5)
   666→
   667→        # Fix parent-child overlaps
   668→        # L3 should not duplicate L2
   669→        if l3 and word_overlap_ratio(l2, l3) > 0.7:
   670→            # Try to differentiate l3
   671→            l3_words = get_word_set(l3)
   672→            l2_words = get_word_set(l2)
   673→            diff_words = l3_words - l2_words
   674→            if diff_words:
   675→                l3 = ' | '.join(w.capitalize() for w in sorted(diff_words)[:3])
   676→            else:
   677→                l3 = l3 + " Items"
   678→
   679→        # L4 should not duplicate L3
   680→        if l4 and l3 and word_overlap_ratio(l3, l4) > 0.7:
   681→            l4_words = get_word_set(l4)
   682→            l3_words = get_word_set(l3)
   683→            diff_words = l4_words - l3_words
   684→            if diff_words:
   685→                l4 = ' | '.join(w.capitalize() for w in sorted(diff_words)[:3])
   686→
   687→        # L5 should not duplicate L4
   688→        if l5 and l4 and word_overlap_ratio(l4, l5) > 0.7:
   689→            l5_words = get_word_set(l5)
   690→            l4_words = get_word_set(l4)
   691→            diff_words = l5_words - l4_words
   692→            if diff_words:
   693→                l5 = ' | '.join(w.capitalize() for w in sorted(diff_words)[:3])
   694→
   695→        # Truncate again after modifications
   696→        l3 = truncate_name(l3)
   697→        l4 = truncate_name(l4)
   698→        l5 = truncate_name(l5)
   699→
   700→        final_mappings.append((source, raw, depth, l1, l2, l3, l4, l5))
