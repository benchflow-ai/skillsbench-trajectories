import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator


def _extract_molecule_names(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one name per line."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split("\n"):
                    name = line.strip()
                    if name:
                        names.append(name)
    return names


def _name_to_smiles(name: str) -> str | None:
    """Convert a chemical name to SMILES using PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def _smiles_to_fingerprint(smiles: str):
    """Convert SMILES to a Morgan fingerprint (radius=2, chirality=True)."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=2, fpSize=2048, includeChirality=True
    )
    return gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """Find the top-k most similar molecules from a PDF pool to the target molecule.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names sorted by descending Tanimoto similarity,
        with alphabetical ordering to break ties.
    """
    # Get target SMILES and fingerprint
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not resolve target molecule: {target_molecule_name}"
        )
    target_fp = _smiles_to_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Could not generate fingerprint for: {target_molecule_name}"
        )

    # Extract pool molecule names from PDF
    pool_names = _extract_molecule_names(molecule_pool_filepath)

    # Compute similarities
    similarities: list[tuple[float, str]] = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _smiles_to_fingerprint(smiles)
        if fp is None:
            continue
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((sim, name))

    # Sort by descending similarity, then alphabetical name for ties
    similarities.sort(key=lambda x: (-x[0], x[1]))

    # Return top-k names
    return [name for _, name in similarities[:top_k]]
