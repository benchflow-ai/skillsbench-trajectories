from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_frac_str(value: float, max_denominator: int = 12) -> str:
    """Convert a float to a fractional string representation.

    Constrains the denominator to be at most max_denominator for simpler results.
    Values are normalized to [0, 1) before conversion.
    """
    # Normalize to [0, 1)
    value = value % 1.0
    # Handle near-zero after mod
    if abs(value) < 1e-8 or abs(value - 1.0) < 1e-8:
        return "0"

    frac = Fraction(value).limit_denominator(max_denominator)

    # Normalize fraction to [0, 1)
    frac = frac % 1
    if frac == 0:
        return "0"

    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and calculate approximate
    coordinates for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to a CIF file.

    Returns:
        A dict with keys:
            - "wyckoff_multiplicity_dict": mapping Wyckoff letter -> multiplicity (int)
            - "wyckoff_coordinates_dict": mapping Wyckoff letter -> [x, y, z] as fraction strings
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    sym_dataset = sga.get_symmetry_dataset()

    wyckoff_letters = sym_dataset.wyckoffs
    equivalent_atoms = sym_dataset.equivalent_atoms

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for i, letter in enumerate(wyckoff_letters):
        if letter in wyckoff_multiplicity_dict:
            continue

        # Multiplicity = number of atoms sharing the same equivalent atom representative
        representative = equivalent_atoms[i]
        multiplicity = int((equivalent_atoms == representative).sum())

        wyckoff_multiplicity_dict[letter] = multiplicity

        # Coordinates of the first atom with this Wyckoff letter
        coords = struct[i].frac_coords
        frac_coords = [_float_to_frac_str(c) for c in coords]
        wyckoff_coordinates_dict[letter] = frac_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
