    16→def load_amazon(filepath):
    17→    rows = []
    18→    with open(filepath, 'r', encoding='utf-8') as f:
    19→        reader = csv.DictReader(f)
    20→        for row in reader:
    21→            path = row['category_path'].strip().strip('"')
    22→            parts = [p.strip() for p in path.split('>')]
    23→            rows.append({'source': 'amazon', 'category_path': path, 'parts': parts})
    24→    return rows
    25→
    26→def load_facebook(filepath):

    46→# ============================================================
    47→# STEP 2: Text standardization
    48→# ============================================================
    49→
    50→def standardize_text(text):
    51→    """Standardize category text: title case, consistent separators."""
    52→    text = text.strip()
    53→    # Remove possessives for consistency
    54→    text = re.sub(r"'s\b", '', text)
    55→    # Replace & with &
    56→    text = text.replace('&amp;', '&')
    57→    # Title case
    58→    words = text.split()
    59→    result = []
    60→    small_words = {'a', 'an', 'the', 'and', 'or', 'for', 'of', 'in', 'on', 'to', 'with', 'by', 'at', 'from'}
    61→    for i, w in enumerate(words):
    62→        if w == '&':
    63→            result.append('&')
    64→        elif i == 0 or w.lower() not in small_words:
    65→            result.append(w.capitalize())
    66→        else:
    67→            result.append(w.lower())
    68→    return ' '.join(result)
