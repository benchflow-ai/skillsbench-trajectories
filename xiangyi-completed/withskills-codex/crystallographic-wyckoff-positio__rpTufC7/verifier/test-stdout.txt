from __future__ import annotations

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_rational_str(value: float, max_denominator: int = 12, tol: float = 1e-6) -> str:
    """Convert a float to a rational string with bounded denominator."""
    # Wrap into [0, 1)
    value = value % 1.0
    if abs(value) < tol or abs(value - 1.0) < tol:
        return "0"

    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.numerator == frac.denominator:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and approximate coordinates
    for the first atom in each Wyckoff position.
    """
    structure = Structure.from_file(filepath)

    # Symmetry analysis
    sga = SpacegroupAnalyzer(structure, symprec=1e-3, angle_tolerance=5)
    symm_structure = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for wyckoff_symbol, sites in zip(
        symm_structure.wyckoff_symbols, symm_structure.equivalent_sites
    ):
        # Pymatgen typically reports symbols like "4a"; keep only the letter.
        letter = wyckoff_symbol[-1]

        if letter in wyckoff_multiplicity_dict:
            # If a letter appears multiple times, keep the first occurrence.
            continue

        wyckoff_multiplicity_dict[letter] = len(sites)
        coords = sites[0].frac_coords
        wyckoff_coordinates_dict[letter] = [
            _float_to_rational_str(float(v)) for v in coords
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
