"""Chemical similarity search using PubChem + RDKit.

Requirements implemented:
- Convert names to molecular representations via PubChem (no manual mapping).
- Compute Morgan fingerprints (radius=2, include chirality) and Tanimoto similarity.
- Sort results by similarity desc, alphabetical for ties.
"""

from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Dict, Iterable, List, Optional, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem


@dataclass
class SimilarityResult:
    name: str
    similarity: float


def _extract_names_from_pdf(pdf_path: str) -> List[str]:
    """Extract molecule names from a PDF, one per line."""
    names: List[str] = []
    seen_lower = set()
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if not name:
                    continue
                key = name.lower()
                if key in seen_lower:
                    continue
                seen_lower.add(key)
                names.append(name)
    return names


def _pubchem_name_to_smiles(
    name: str,
    cache: Dict[str, Optional[str]],
    delay_seconds: float = 0.2,
) -> Optional[str]:
    """Resolve a chemical name to an isomeric SMILES string via PubChem.

    Uses a small delay to respect PubChem rate limits.
    """
    cache_key = name.lower()
    if cache_key in cache:
        return cache[cache_key]

    smiles: Optional[str] = None
    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            compound = compounds[0]
            smiles = compound.isomeric_smiles or compound.canonical_smiles
    except Exception:
        smiles = None

    cache[cache_key] = smiles
    time.sleep(delay_seconds)
    return smiles


def _smiles_to_mol(smiles: str) -> Optional[Chem.Mol]:
    mol = Chem.MolFromSmiles(smiles)
    return mol


def _morgan_fp(mol: Chem.Mol):
    return AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=2, nBits=2048, useChirality=True
    )


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> List[Tuple[str, float]]:
    """Return the top-k most similar molecules from a PDF list.

    Args:
        target_molecule_name: Chemical name to compare against.
        molecule_pool_filepath: Path to a PDF containing one molecule name per line.
        top_k: Number of top results to return.

    Returns:
        List of (molecule_name, similarity) tuples sorted by similarity desc,
        then alphabetical for ties.
    """
    if top_k <= 0:
        return []

    names = _extract_names_from_pdf(molecule_pool_filepath)
    if not names:
        return []

    cache: Dict[str, Optional[str]] = {}
    target_smiles = _pubchem_name_to_smiles(target_molecule_name, cache)
    if not target_smiles:
        return []

    target_mol = _smiles_to_mol(target_smiles)
    if target_mol is None:
        return []

    target_fp = _morgan_fp(target_mol)

    results: List[SimilarityResult] = []
    target_name_lower = target_molecule_name.lower()

    for name in names:
        if name.lower() == target_name_lower:
            continue
        smiles = _pubchem_name_to_smiles(name, cache)
        if not smiles:
            continue
        mol = _smiles_to_mol(smiles)
        if mol is None:
            continue
        fp = _morgan_fp(mol)
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append(SimilarityResult(name=name, similarity=sim))

    results.sort(key=lambda r: (-r.similarity, r.name.lower()))

    top = results[: top_k]
    return [(r.name, r.similarity) for r in top]


def _format_results(results: Iterable[Tuple[str, float]]) -> str:
    lines = []
    for name, sim in results:
        lines.append(f"{name}\t{sim:.4f}")
    return "\n".join(lines)


if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description="Top-k chemical similarity search")
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument(
        "--pdf",
        default="/root/molecules.pdf",
        help="Path to PDF containing molecule names",
    )
    parser.add_argument("-k", type=int, default=5, help="Number of top results")
    args = parser.parse_args()

    results = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.k)
    print(_format_results(results))
