import argparse
from functools import lru_cache
from pathlib import Path
from typing import List, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


def _extract_molecule_names_from_pdf(pdf_path: Path) -> List[str]:
    names: List[str] = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if name:
                    names.append(name)
    return names


@lru_cache(maxsize=2048)
def _name_to_smiles(name: str) -> str | None:
    # Prefer property lookup for speed; fall back to full compound fetch.
    try:
        props = pcp.get_properties(["IsomericSMILES", "CanonicalSMILES"], name, "name")
        if props:
            smiles = props[0].get("IsomericSMILES") or props[0].get("CanonicalSMILES")
            if smiles:
                return smiles
    except Exception:
        pass

    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            compound = compounds[0]
            smiles = getattr(compound, "smiles", None)
            if smiles:
                return smiles
            return compound.canonical_smiles
    except Exception:
        pass

    return None


_MORGAN_GENERATOR = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _smiles_to_fingerprint(smiles: str):
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _MORGAN_GENERATOR.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> List[Tuple[str, float]]:
    """
    Compute the top-k most similar molecules to the target molecule name.

    Similarity uses Morgan fingerprints (radius=2, include chirality)
    with Tanimoto similarity. Results are sorted by descending similarity,
    then alphabetical name for ties.
    """
    if top_k <= 0:
        return []

    pool_path = Path(molecule_pool_filepath)
    if not pool_path.exists():
        raise FileNotFoundError(f"Molecule pool file not found: {pool_path}")

    if pool_path.suffix.lower() != ".pdf":
        raise ValueError("molecule_pool_filepath must be a PDF")

    pool_names = _extract_molecule_names_from_pdf(pool_path)

    target_smiles = _name_to_smiles(target_molecule_name)
    if not target_smiles:
        return []

    target_fp = _smiles_to_fingerprint(target_smiles)
    if target_fp is None:
        return []

    results: List[Tuple[str, float]] = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if not smiles:
            continue
        fp = _smiles_to_fingerprint(smiles)
        if fp is None:
            continue
        sim = float(DataStructs.TanimotoSimilarity(target_fp, fp))
        results.append((name, sim))

    results.sort(key=lambda x: (-x[1], x[0].lower()))
    return results[:top_k]


def _main() -> None:
    parser = argparse.ArgumentParser(
        description="Top-k Tanimoto similarity against molecules listed in a PDF."
    )
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument("pdf", help="Path to molecules.pdf")
    parser.add_argument("top_k", type=int, help="Number of results to return")
    args = parser.parse_args()

    results = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.top_k)
    for name, score in results:
        print(f"{name}\t{score:.4f}")


if __name__ == "__main__":
    _main()
