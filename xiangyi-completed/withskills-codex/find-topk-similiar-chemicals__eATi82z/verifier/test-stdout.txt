"""Molecule similarity search using PubChem + RDKit.

Uses PubChem to resolve chemical names to SMILES, then computes Morgan
fingerprints (radius=2, use chirality) and Tanimoto similarity.
"""

from __future__ import annotations

from functools import lru_cache
from pathlib import Path
import random
import time
from typing import Iterable, List, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


def _load_molecule_names(molecule_pool_filepath: str) -> List[str]:
    path = Path(molecule_pool_filepath)
    if not path.exists():
        raise FileNotFoundError(f"Molecule pool file not found: {molecule_pool_filepath}")

    names: List[str] = []
    if path.suffix.lower() == ".pdf":
        with pdfplumber.open(str(path)) as pdf:
            for page in pdf.pages:
                text = page.extract_text() or ""
                for line in text.splitlines():
                    name = line.strip()
                    if name:
                        names.append(name)
    else:
        content = path.read_text(encoding="utf-8", errors="ignore")
        for line in content.splitlines():
            name = line.strip()
            if name:
                names.append(name)

    # Deduplicate while preserving order (case-insensitive)
    seen = set()
    deduped: List[str] = []
    for name in names:
        key = name.lower()
        if key in seen:
            continue
        seen.add(key)
        deduped.append(name)
    return deduped


_MIN_PUBCHEM_INTERVAL_SEC = 0.2
_last_pubchem_call = 0.0


def _rate_limited_pubchem_call(func, *args, **kwargs):
    global _last_pubchem_call
    elapsed = time.time() - _last_pubchem_call
    if elapsed < _MIN_PUBCHEM_INTERVAL_SEC:
        time.sleep(_MIN_PUBCHEM_INTERVAL_SEC - elapsed)
    result = func(*args, **kwargs)
    _last_pubchem_call = time.time()
    return result


@lru_cache(maxsize=2048)
def _smiles_from_pubchem(name: str) -> str | None:
    retries = 4
    backoff = 0.6
    for attempt in range(retries):
        try:
            compounds = _rate_limited_pubchem_call(pcp.get_compounds, name, "name")
            if not compounds:
                return None
            compound = compounds[0]
            return compound.smiles or compound.canonical_smiles
        except Exception:
            if attempt == retries - 1:
                return None
            sleep_for = backoff * (2**attempt) + random.uniform(0, 0.2)
            time.sleep(sleep_for)
    return None


_MORGAN_GENERATOR = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def _fingerprint_from_smiles(smiles: str):
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _MORGAN_GENERATOR.GetFingerprint(mol)


def _iter_pool_fps(pool_names: Iterable[str]):
    for name in pool_names:
        smiles = _smiles_from_pubchem(name)
        if not smiles:
            continue
        fp = _fingerprint_from_smiles(smiles)
        if fp is None:
            continue
        yield name, fp


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> list:
    """Return top-k most similar molecules from pool to the target name.

    Returns a list of (molecule_name, similarity) tuples sorted by descending
    similarity, then alphabetical order for ties.
    """
    if top_k <= 0:
        return []

    pool_names = _load_molecule_names(molecule_pool_filepath)
    target_smiles = _smiles_from_pubchem(target_molecule_name)
    if not target_smiles:
        return []

    target_fp = _fingerprint_from_smiles(target_smiles)
    if target_fp is None:
        return []

    results: List[Tuple[str, float]] = []
    for name, fp in _iter_pool_fps(pool_names):
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, float(similarity)))

    results.sort(key=lambda item: (-item[1], item[0].lower()))
    return results[:top_k]


if __name__ == "__main__":
    import argparse
    import json

    parser = argparse.ArgumentParser(
        description="Top-k Tanimoto similarity search against a molecule pool."
    )
    parser.add_argument("target_name", help="Target molecule name")
    parser.add_argument("pool_file", help="Path to molecule pool (PDF or text)")
    parser.add_argument("top_k", type=int, help="Number of results to return")
    args = parser.parse_args()

    results = topk_tanimoto_similarity_molecules(
        args.target_name, args.pool_file, args.top_k
    )
    print(json.dumps(results, indent=2))
