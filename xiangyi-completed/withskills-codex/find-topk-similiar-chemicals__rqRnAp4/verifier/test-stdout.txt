"""Compute top-k Tanimoto similarity against a molecule pool in a PDF."""
from __future__ import annotations

from functools import lru_cache
from typing import List, Tuple

import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem


def _extract_molecule_names_from_pdf(pdf_path: str) -> List[str]:
    """Extract molecule names from a PDF, one name per line.

    Preserves original ordering and removes duplicates.
    """
    names: List[str] = []
    seen = set()
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text() or ""
            for line in text.splitlines():
                name = line.strip()
                if not name:
                    continue
                if name in seen:
                    continue
                seen.add(name)
                names.append(name)
    return names


@lru_cache(maxsize=4096)
def _name_to_smiles(name: str) -> str | None:
    """Resolve a chemical name to an isomeric/canonical SMILES via PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
    except Exception:
        return None
    if not compounds:
        return None
    for compound in compounds:
        smiles = getattr(compound, "smiles", None) or getattr(
            compound, "canonical_smiles", None
        )
        if smiles:
            return smiles
    return None


def _name_to_mol(name: str) -> Chem.Mol | None:
    smiles = _name_to_smiles(name)
    if not smiles:
        return None
    return Chem.MolFromSmiles(smiles)


def _mol_to_fingerprint(mol: Chem.Mol):
    return AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=2, nBits=2048, useChirality=True
    )


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> List[Tuple[str, float]]:
    """Return the top-k most similar molecules by Tanimoto similarity.

    Uses Morgan fingerprints (radius=2, chirality enabled).

    Args:
        target_molecule_name: Chemical name to compare against.
        molecule_pool_filepath: Path to PDF containing molecule names.
        top_k: Number of top results to return.

    Returns:
        List of (molecule_name, similarity) sorted by similarity desc, then name.
    """
    if top_k <= 0:
        return []

    names = _extract_molecule_names_from_pdf(molecule_pool_filepath)
    if not names:
        return []

    target_mol = _name_to_mol(target_molecule_name)
    if target_mol is None:
        raise ValueError(f"Could not resolve target molecule name: {target_molecule_name}")

    target_fp = _mol_to_fingerprint(target_mol)

    results: List[Tuple[str, float]] = []
    for name in names:
        mol = _name_to_mol(name)
        if mol is None:
            continue
        fp = _mol_to_fingerprint(mol)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, float(similarity)))

    results.sort(key=lambda item: (-item[1], item[0].casefold(), item[0]))
    return results[:top_k]


if __name__ == "__main__":
    import argparse
    import json

    parser = argparse.ArgumentParser(
        description="Top-k Tanimoto similarity against molecules in a PDF"
    )
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument("pdf", help="Path to PDF with molecule names")
    parser.add_argument("top_k", type=int, help="Number of results to return")
    args = parser.parse_args()

    output = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.top_k)
    print(json.dumps(output, indent=2))
