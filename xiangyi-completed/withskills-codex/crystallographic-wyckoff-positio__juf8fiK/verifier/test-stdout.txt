from __future__ import annotations

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _rational_str(value: float, max_denominator: int = 12) -> str:
    """Convert a float to a simple rational string with bounded denominator."""
    # Normalize to [0, 1) for fractional coordinates
    normalized = value - int(value // 1)
    frac = Fraction(normalized).limit_denominator(max_denominator)
    if frac.numerator == 0:
        return "0"
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff multiplicities and approximate coordinates for the first
    atom of each Wyckoff position.
    """
    try:
        structure = Structure.from_file(filepath)
        sga = SpacegroupAnalyzer(structure)
        sym_struct = sga.get_symmetrized_structure()

        wyckoff_multiplicity_dict: dict[str, int] = {}
        wyckoff_coordinates_dict: dict[str, list[str]] = {}

        for eq_sites, letter in zip(
            sym_struct.equivalent_sites, sym_struct.wyckoff_symbols
        ):
            multiplicity = len(eq_sites)
            wyckoff_multiplicity_dict[letter] = (
                wyckoff_multiplicity_dict.get(letter, 0) + multiplicity
            )

            if letter not in wyckoff_coordinates_dict:
                coords = eq_sites[0].frac_coords
                wyckoff_coordinates_dict[letter] = [
                    _rational_str(float(coords[0])),
                    _rational_str(float(coords[1])),
                    _rational_str(float(coords[2])),
                ]

        return {
            "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
            "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
        }
    except Exception as exc:  # pragma: no cover - defensive for malformed inputs
        return {"error": str(exc)}
