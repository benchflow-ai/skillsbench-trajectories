from __future__ import annotations

from fractions import Fraction
import re
from typing import Any

import numpy as np
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _rationalize_fraction(value: float, max_denominator: int = 12) -> str:
    """Return a rational string with a limited denominator."""
    if np.isnan(value):
        return "0"

    # Map to [0, 1) and clean up numerical noise at 0/1 boundaries.
    value = value % 1.0
    if abs(value - 1.0) < 1e-8 or abs(value) < 1e-8:
        value = 0.0

    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    try:
        structure = Structure.from_file(filepath)
    except Exception as exc:  # pragma: no cover - error path for invalid input
        return {"error": f"failed to read structure: {exc}"}

    try:
        sga = SpacegroupAnalyzer(structure, symprec=1e-2, angle_tolerance=5.0)
        symm_struct = sga.get_symmetrized_structure()
    except Exception as exc:  # pragma: no cover - error path for bad symmetry
        return {"error": f"failed to analyze symmetry: {exc}"}

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    wyckoff_symbols = symm_struct.wyckoff_symbols
    for indices, sites, wyckoff_symbol in zip(
        symm_struct.equivalent_indices,
        symm_struct.equivalent_sites,
        wyckoff_symbols,
        strict=False,
    ):
        if not indices or not sites:
            continue
        match = re.search(r"([A-Za-z]+)$", wyckoff_symbol)
        wyckoff_letter = match.group(1).lower() if match else wyckoff_symbol
        wyckoff_multiplicity_dict.setdefault(wyckoff_letter, len(indices))

        # Keep the first orbit for each Wyckoff letter for deterministic output.
        if wyckoff_letter in wyckoff_coordinates_dict:
            continue
        site = sites[0]
        coords = [_rationalize_fraction(val) for val in site.frac_coords]
        wyckoff_coordinates_dict[wyckoff_letter] = coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
