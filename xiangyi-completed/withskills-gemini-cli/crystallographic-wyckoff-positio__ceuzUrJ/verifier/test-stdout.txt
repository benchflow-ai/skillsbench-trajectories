from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates 
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath (str): Path to the CIF file.
        
    Returns:
        dict: A dictionary containing 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'.
    """
    # Load the structure from the CIF file
    struct = Structure.from_file(filepath)
    
    # Analyze the symmetry of the structure
    sga = SpacegroupAnalyzer(struct)
    symm_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    def to_rational_str(val: float, max_den: int = 12) -> str:
        """Converts a float to a rational string with a maximum denominator."""
        # Fraction.limit_denominator(max_den) finds the closest fraction with denominator <= max_den
        f = Fraction(val).limit_denominator(max_den)
        # Normalize to [0, 1) range
        f = f % 1
        if f.denominator == 1:
            return str(f.numerator)
        else:
            return f"{f.numerator}/{f.denominator}"

    # symm_struct.equivalent_indices groups sites into Wyckoff sets.
    # symm_struct.wyckoff_symbols contains the symbol (e.g., '4a') for each set.
    for i, indices in enumerate(symm_struct.equivalent_indices):
        wyckoff_symbol = symm_struct.wyckoff_symbols[i]
        
        # Extract multiplicity and label (e.g., '4a' -> 4, 'a')
        multiplicity = int(''.join(filter(str.isdigit, wyckoff_symbol)))
        label = ''.join(filter(str.isalpha, wyckoff_symbol))
        
        # Store multiplicity (assuming letters are unique keys as per example)
        wyckoff_multiplicity_dict[label] = multiplicity
        
        # Get coordinates of the first atom in this Wyckoff position
        first_site = symm_struct[indices[0]]
        frac_coords = first_site.frac_coords
        
        # Convert coordinates to rational strings
        rational_coords = [to_rational_str(x) for x in frac_coords]
        wyckoff_coordinates_dict[label] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
