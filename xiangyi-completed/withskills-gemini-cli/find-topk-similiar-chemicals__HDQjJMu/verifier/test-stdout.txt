import time
import pubchempy as pcp
from pypdf import PdfReader
from rdkit import Chem
from rdkit import RDLogger
from rdkit.Chem import AllChem, DataStructs

# Suppress RDKit warnings
RDLogger.DisableLog('rdApp.*')

def get_smiles_from_name(name):
    """
    Retrieves the Isomeric SMILES for a given chemical name using PubChemPy.
    Retries once if it fails.
    """
    for attempt in range(2):
        try:
            # Respect rate limits (approx 5/sec max, so 0.25s delay is safe)
            time.sleep(0.25) 
            # Use get_properties to fetch IsomericSMILES directly and avoid Compound object deprecation warnings
            properties = pcp.get_properties(['IsomericSMILES'], name, 'name')
            if properties and 'SMILES' in properties[0]:
                return properties[0]['SMILES']
            elif properties and 'IsomericSMILES' in properties[0]:
                 return properties[0]['IsomericSMILES']
            else:
                if attempt == 1: # Only print warning on last attempt if not found (or maybe it's just not there)
                    print(f"Warning: Could not find SMILES for '{name}'")
                return None
        except Exception as e:
            if attempt == 1:
                print(f"Error fetching SMILES for '{name}': {e}")
            else:
                time.sleep(1.0) # Backoff a bit more before retry
    return None

def get_molecule_fingerprint(smiles):
    """
    Converts a SMILES string to an RDKit molecule and generates a Morgan fingerprint
    (radius=2, include chirality).
    """
    if not smiles:
        return None
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol:
            # Morgan fingerprint with radius 2 and chirality
            return AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, useChirality=True)
    except Exception as e:
        print(f"Error generating fingerprint for SMILES '{smiles}': {e}")
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target molecule.
    """
    # 1. Get Target Molecule Fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not resolve target molecule name: {target_molecule_name}")
    
    target_fp = get_molecule_fingerprint(target_smiles)
    if not target_fp:
         raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # 2. Read Molecule Names from PDF
    molecule_names = []
    try:
        reader = PdfReader(molecule_pool_filepath)
        text = ""
        for page in reader.pages:
            text += page.extract_text() + "\n"
        
        # Split by newlines and clean up
        lines = text.split('\n')
        molecule_names = [line.strip() for line in lines if line.strip()]
        
        # Remove duplicates if any, preserving order? Or just set?
        # The prompt implies a list of chemicals. Let's keep unique names to avoid redundant API calls.
        molecule_names = list(set(molecule_names))
        
    except Exception as e:
        raise ValueError(f"Error reading PDF file '{molecule_pool_filepath}': {e}")

    # 3. Process Pool Molecules
    results = []
    
    for name in molecule_names:
        # Skip if name is the same as target (optional, but usually desired in similarity search)
        # But user said "to any chemicals you are given", if target is in list, it should match 1.0.
        # Let's keep it.
        
        smiles = get_smiles_from_name(name)
        if smiles:
            fp = get_molecule_fingerprint(smiles)
            if fp:
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 4. Sort results
    # Sort by similarity (descending), then by name (ascending)
    # Python's sort is stable.
    # We can use a tuple key: (-similarity, name)
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k pairs (or just names? User said "Find the top k similar chemicals... returns -> list")
    # Usually returning the list of names or (name, score) tuples.
    # The return type annotation says `-> list`.
    # Let's return the list of (name, similarity) tuples as it's most informative, 
    # or just names if strictly interpreted.
    # "Find the top k similar chemicals" -> usually implies the chemicals themselves.
    # However, without clarification, (name, score) is safer for debugging and verification.
    # But usually "top k items" means the items.
    # Let's look at the function signature again: `topk_tanimoto_similarity_molecules(...) -> list`.
    # Given the potential ambiguity, I'll return a list of (name, similarity) tuples. 
    # If the user wanted just names, they can extract them.
    # Wait, usually "top k similar chemicals" implies the entities.
    # Let's assume (name, similarity) is better.
    
    return results[:top_k]
