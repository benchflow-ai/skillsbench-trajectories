from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
import re
from collections import defaultdict

def float_to_rational_str(value, max_denom=12):
    """
    Converts a float to a rational string representation with a maximum denominator.
    Constraints: denominator <= 12.
    """
    # Wrap to [0, 1) to handle periodic boundary conditions for coordinates
    # Using modulo 1.0, but handling edge cases for numerical stability
    val = value % 1.0
    
    # If very close to 0 or 1, treat as 0
    if abs(val) < 1e-4 or abs(val - 1.0) < 1e-4:
        return "0"
        
    f = Fraction(val).limit_denominator(max_denom)
    
    if f.numerator == 0:
        return "0"
        
    # Check if rounding pushed it to 1 (which is 0 in fractional coords)
    if f == 1:
        return "0"
        
    return str(f)

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing multiplicity and coordinates for each Wyckoff position.
    """
    try:
        struct = Structure.from_file(filepath)
    except Exception as e:
        raise ValueError(f"Failed to load structure from {filepath}: {e}")
        
    # Use SpacegroupAnalyzer to get symmetry data
    # symprec=0.1 is usually good for experimental data, but default is 0.01.
    # Let's stick to default unless needed.
    sga = SpacegroupAnalyzer(struct)
    
    try:
        sym_struct = sga.get_symmetrized_structure()
    except Exception as e:
        raise ValueError(f"Symmetry analysis failed: {e}")
        
    wyckoff_multiplicity_dict = defaultdict(int)
    wyckoff_coordinates_dict = {}
    
    # sym_struct.equivalent_indices is a list of lists.
    # Each inner list contains indices of sites that are equivalent.
    # sym_struct.wyckoff_symbols corresponds to the groups (one symbol per group).
    for i, group in enumerate(sym_struct.equivalent_indices):
        if not group:
            continue
            
        # Get the index of the first atom in this equivalent group
        first_idx = group[0]
        
        # Get the Wyckoff symbol for this group
        symbol = sym_struct.wyckoff_symbols[i]
        
        # Extract the letter (e.g., "4a" -> "a")
        # Usually the last character, but using regex to be safe
        match = re.search(r'([a-zA-Z]+)$', symbol)
        if match:
            letter = match.group(1)
        else:
            letter = symbol # Fallback
            
        # Add to multiplicity count
        wyckoff_multiplicity_dict[letter] += len(group)
        
        # Store coordinates if not already recorded for this letter
        if letter not in wyckoff_coordinates_dict:
            # Get coordinates of the first atom in the group
            site = sym_struct[first_idx]
            coords = site.frac_coords
            formatted_coords = [float_to_rational_str(x) for x in coords]
            wyckoff_coordinates_dict[letter] = formatted_coords
            
    return {
        "wyckoff_multiplicity_dict": dict(wyckoff_multiplicity_dict),
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
