import time
import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs

def get_smiles(name):
    """
    Retrieves the canonical SMILES for a given chemical name using PubChemPy.
    Includes a small delay to respect API rate limits.
    """
    try:
        # Respect PubChem API rate limits (limit is ~5 req/s)
        time.sleep(0.25) 
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            return compounds[0].canonical_smiles
    except Exception as e:
        # In a real scenario, we might want to log this
        # print(f"Warning: Could not find SMILES for '{name}': {e}")
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target molecule.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing the pool of molecules.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: A list of tuples (molecule_name, similarity_score) sorted by 
              similarity (descending) and then name (ascending).
    """
    
    # 1. Extract molecule names from the PDF
    molecule_names = []
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Assuming names are on separate lines based on inspection
                    lines = [line.strip() for line in text.split('\n') if line.strip()]
                    molecule_names.extend(lines)
    except Exception as e:
        raise ValueError(f"Failed to read PDF file {molecule_pool_filepath}: {e}")

    # Remove duplicates if any, but keep original case for display? 
    # Usually sets are good, but we want to return the name as found or normalized.
    # Let's use a set to avoid redundant API calls.
    unique_names = sorted(list(set(molecule_names)))

    # 2. Get Target SMILES and Fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_molecule_name}")

    # Morgan fingerprint, radius=2, include chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, radius=2, useChirality=True)

    # 3. Process the pool of molecules
    results = []
    
    for name in unique_names:
        # Skip if it's the target itself? The prompt implies "similar to", usually excluding self,
        # but if the target is IN the list, it is maximally similar (1.0).
        # Often "top k similar" includes the item itself if present. 
        # I will include it.
        
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, useChirality=True)
        
        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # 4. Sort results
    # Descending order of similarity, alphabetical ordering when ties happen
    results.sort(key=lambda x: (-x[1], x[0]))

    # 5. Return top k
    return results[:top_k]
