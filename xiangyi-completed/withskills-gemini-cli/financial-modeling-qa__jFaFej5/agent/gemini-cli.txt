    rolls_arr = np.array(rolls)
    s = set(rolls)
    
    # 1. High and Often
    max_val = np.max(rolls_arr)
    count_max = np.sum(rolls_arr == max_val)
    score1 = max_val * count_max
    
    # 2. Summation
    score2 = np.sum(rolls_arr)
    
    # 3. Highs and Lows
    min_val = np.min(rolls_arr)
    score3 = max_val * min_val * (max_val - min_val)
    
    # 4. Only two numbers

    # 5. All the numbers
    score5 = 40 if s == {1, 2, 3, 4, 5, 6} else 0
