def extract_pdf_data(pdf_path):
    employees = {}
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            tables = page.extract_tables()
            for table in tables:
                if not table:
                    continue
                # Assuming first row is header
                header = table[0]
                for row in table[1:]:
                    if not any(row):
                        continue
                    employee = dict(zip(header, row))
                    emp_id = employee.get('ID')
                    if emp_id:
                        # Convert numeric strings to numbers if possible
                        for key, value in employee.items():
                            if key in ['Salary', 'Years', 'Score']:
                                try:
                                    # Handle cases where value might be like "$50,000" or similar
                                    clean_val = str(value).replace('$', '').replace(',', '')
                                    if '.' in clean_val:
                                        employee[key] = float(clean_val)
                                    else:
                                        employee[key] = int(clean_val)
                                except (ValueError, TypeError):
                                    pass
                        employees[emp_id] = employee
    return employees

def extract_xlsx_data(xlsx_path):
    df = pd.read_excel(xlsx_path)
    # Ensure ID is treated as string and other fields as appropriate
    df['ID'] = df['ID'].astype(str)
    employees = {}
    for _, row in df.iterrows():
        emp_id = row['ID']
        employees[emp_id] = row.to_dict()
    return employees
