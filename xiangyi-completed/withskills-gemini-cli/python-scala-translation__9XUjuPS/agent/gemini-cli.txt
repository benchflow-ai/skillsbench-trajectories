class StringTokenizer(
    encoding: String = "utf-8",
    normalizer: String => String = identity
) extends BaseTokenizer[String] {

  def tokenize(value: String): Token = {
    // In Scala strings are already decoded sequences of chars, 
    // but if we were handling byte arrays we'd decode.
    // Assuming input is String here as per signature.
    val normalized = normalizer(value)
    Token(normalized, TokenType.STRING)
  }
  
  // Overload for bytes if needed, but BaseTokenizer[String] constrains it.
  def tokenize(value: Array[Byte]): Token = {
     val strValue = new String(value, encoding)
     tokenize(strValue)
  }
}

class NumericTokenizer(
    precision: Int = 6,
    formatOptions: Map[String, Any] = Map.empty
) extends BaseTokenizer[Any] { // Using Any to cover Int, Double, BigDecimal like Python's Union

  override def tokenize(value: Any): Token = {
    val strValue = value match {
      case d: BigDecimal => d.setScale(precision, scala.math.BigDecimal.RoundingMode.HALF_UP).toString()
      case f: Float      => s"%.${precision}f".format(f)
      case d: Double     => s"%.${precision}f".format(d)
      case n: Number     => n.toString
      case other         => other.toString
    }

    Token(
      strValue,
      TokenType.NUMERIC,
      Map("original_type" -> value.getClass.getSimpleName)
    )
  }
}

class TemporalTokenizer(
    formatStr: Option[String] = None
) extends BaseTokenizer[Temporal] {

  private val ISO_FORMAT = DateTimeFormatter.ISO_LOCAL_DATE_TIME
  private val DATE_FORMAT = DateTimeFormatter.ISO_LOCAL_DATE

  override def tokenize(value: Temporal): Token = {
    val formatter = formatStr match {
      case Some(fmt) => DateTimeFormatter.ofPattern(fmt)
      case None => value match {
        case _: LocalDateTime => ISO_FORMAT
        case _: LocalDate => DATE_FORMAT
        case _ => ISO_FORMAT // Fallback
      }
    }
    
    // TemporalAccessor handling for formatting
    val strValue = try {
      formatter.format(value)
    } catch {
       // Fallback for simple toString if format fails or types mismatch
       case _: Exception => value.toString 
    }

    Token(strValue, TokenType.TEMPORAL)
  }
}

class TokenRegistry[T] {
  private val _registry = mutable.Map.empty[String, TokenContainer[T]]
  private val _handlers = mutable.ListBuffer.empty[T => Option[Token]]

  def register(key: String, container: TokenContainer[T]): Unit = {
    _registry(key) = container
  }

  def addHandler(handler: T => Option[Token]): Unit = {
    _handlers += handler
  }

  def process(key: String): List[Option[Token]] = {
    _registry.get(key) match {
      case None => List.empty
      case Some(container) =>
        container.getAll.map { item =>
          // Find first handler that returns Some
          _handlers.iterator.map(h => h(item)).find(_.isDefined).flatten
        }.toList
    }
  }
}

class JsonTokenizer(pretty: Boolean = false) {
  
  def tokenize(value: Json): Token = {
    val jsonStr = if (pretty) value.spaces2 else value.noSpaces
    Token(jsonStr, TokenType.STRUCTURED, Map("json" -> true))
  }

  def tokenizePath(value: Json, path: String): Option[Token] = {
    val parts = path.split("\\.")
    
    // Recursive navigation
    def navigate(current: Json, remainingParts: Seq[String]): Option[Json] = {
      if (remainingParts.isEmpty) Some(current)
      else {
        val head = remainingParts.head
        val tail = remainingParts.tail
        
        // Check for array index or object key
        if (current.isArray && head.toIntOption.isDefined) {
          val idx = head.toInt
          current.asArray.flatMap(arr => if (idx >= 0 && idx < arr.length) navigate(arr(idx), tail) else None)
        } else if (current.isObject) {
          current.asObject.flatMap(_(head)).flatMap(navigate(_, tail))
        } else {
          None
        }
      }
    }

    navigate(value, parts.toSeq).map(tokenize)
  }
}

class WhitespaceTokenizer(
    lowercase: Boolean = false,
    minLength: Int = 0,
    maxLength: Option[Int] = None,
    stripPunctuation: Boolean = false
) {
  private val _punctuation = ".,!?;:'\"()[]{}"

  private def processToken(word: String): Option[String] = {
    var processed = word
    
    if (stripPunctuation) {
      processed = processed.dropWhile(_punctuation.contains(_)).reverse.dropWhile(_punctuation.contains(_)).reverse
    }

    if (lowercase) {
      processed = processed.toLowerCase
    }

    if (processed.length < minLength) {
      None
    } else {
      maxLength match {
        case Some(max) if processed.length > max => Some(processed.take(max))
        case _ => if (processed.nonEmpty) Some(processed) else None
      }
    }
  }

  def tokenize(text: String): List[Token] = {
    val words = text.split("\\s+")
    words.zipWithIndex.flatMap { case (word, i) =>
      processToken(word).map { processed =>
        Token(processed, TokenType.STRING, Map("position" -> i, "original" -> word))
      }
    }.toList
  }

  def tokenizeToStrings(text: String): List[String] = {
    tokenize(text).map(_.value)
  }
  
  def countTokens(text: String): Int = {
    tokenize(text).size
  }
}

class TokenizerBuilder[T] {
  private val _normalizers = mutable.ListBuffer.empty[String => String]
  private val _validators = mutable.ListBuffer.empty[T => Boolean]
  private val _metadata = mutable.Map.empty[String, Any]

  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] = {
    _normalizers += normalizer
    this
  }

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] = {
    _validators += validator
    this
  }

  def withMetadata(meta: (String, Any)*): TokenizerBuilder[T] = {
    _metadata ++= meta
    this
  }

  def build(): T => Token = {
    val normalizers = _normalizers.toList
    val validators = _validators.toList
    val metadata = _metadata.toMap

    (value: T) => {
      // Validate
      validators.foreach { validator =>
        if (!validator(value)) {
          throw new IllegalArgumentException(s"Validation failed for $value")
        }
      }

      // Convert to string
      var strValue = value.toString

      // Normalize
      normalizers.foreach { normalizer =>
        strValue = normalizer(strValue)
      }

      Token(strValue, TokenType.STRING, metadata)
    }
  }
}

object TokenizerBuilder {
  def apply[T](): TokenizerBuilder[T] = new TokenizerBuilder[T]()
}
