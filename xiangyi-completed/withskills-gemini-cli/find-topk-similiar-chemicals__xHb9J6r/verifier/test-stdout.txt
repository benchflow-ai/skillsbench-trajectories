import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import os

import time

def get_smiles_from_name(name):
    """Convert chemical name to SMILES using PubChem."""
    try:
        props = pcp.get_properties(['CanonicalSMILES', 'IsomericSMILES'], name, 'name')
        if props:
            # Try to get any SMILES available
            return props[0].get('CanonicalSMILES') or props[0].get('IsomericSMILES') or props[0].get('ConnectivitySMILES')
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find top k similar chemicals in a PDF to a target molecule name.
    """
    # 1. Extract names from PDF
    pool_names = []
    if os.path.exists(molecule_pool_filepath):
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Assume each line is a chemical name
                    names = [line.strip() for line in text.split('\n') if line.strip()]
                    pool_names.extend(names)
    
    unique_pool_names = []
    seen = set()
    for name in pool_names:
        if name not in seen:
            unique_pool_names.append(name)
            seen.add(name)
    
    # 2. Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, radius=2, useChirality=True)
    
    # 3. Retrieve SMILES for the pool
    name_to_smiles = {}
    for name in unique_pool_names:
        smiles = get_smiles_from_name(name)
        if smiles:
            name_to_smiles[name] = smiles
    
    # 4. Compute similarities
    results = []
    for name, smiles in name_to_smiles.items():
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
        
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, radius=2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        
        results.append({
            'name': name,
            'similarity': similarity
        })
    
    # 5. Sort results
    # Sort by similarity descending, then name ascending
    results.sort(key=lambda x: (-x['similarity'], x['name']))
    
    return results[:top_k]
