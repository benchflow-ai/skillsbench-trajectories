# Fill Rows
# Columns mapping (0-based in Logic, 1-based in Openpyxl):
# B: Year
# C: Real GDP (Link WEO)
# D: Allocation (Input)
# E: Project Inv (Calc)
# F: Imports (Calc)
# G: GDP Impact (Calc)
# J: Baseline Growth (Link WEO)
# K: Projected Growth (Calc)

# WEO mapping: Row 2 is NGDP_R, Row 3 is NGDP_D, Row 4 is NGDP_RPCH...
# Years are in columns B(2020), C(2021)... in WEO sheet?
# No, `dataframe_to_rows` writes columns as years.
# WEO Sheet:
# Row 1: Index, 2020, 2021...
# Row 2: NGDP_R, val, val...
# Row 4: NGDP_RPCH, val, val...

# Helper to find WEO column for a year
# WEO columns: A=Index, B=2020, C=2021... 
# Year X column index = (X - 2020) + 2. (2020->2(B))

start_year = 2020
end_year = 2043

for i, year in enumerate(range(start_year, end_year + 1)):
    row_idx = header_row + 1 + i
    
    # Year
    ws_na.cell(row=row_idx, column=2).value = year
    
    # WEO Column Letter
    from openpyxl.utils import get_column_letter
    weo_col = get_column_letter( (year - 2020) + 2 )
    
    # Link Real GDP (WEO Row 2)
    ws_na.cell(row=row_idx, column=3).value = f"=WEO_Data!{weo_col}2"
    
    # Link Baseline Growth (WEO Row 4)
    ws_na.cell(row=row_idx, column=10).value = f"=WEO_Data!{weo_col}4"
    
    # Allocation
    alloc = allocations.get(year, 0)
    ws_na.cell(row=row_idx, column=4).value = alloc
    ws_na.cell(row=row_idx, column=4).number_format = '0.0%'
    
    # Project Inv (Nominal GEL) = Total(USD) * Alloc * ExRate
    # Cell D30 is Total Inv, D33 is ExRate.
    ws_na.cell(row=row_idx, column=5).value = f"=$D$30*D{row_idx}*$D$33"
    
    # Imports (Nominal GEL) = Inv * ImportShare
    ws_na.cell(row=row_idx, column=6).value = f"=E{row_idx}*$D$31"
    
    # GDP Impact (Real) = (Inv - Imp) * Multiplier / (Deflator/100)
    # Deflator is in WEO Row 3.
    ws_na.cell(row=row_idx, column=7).value = f"=(E{row_idx}-F{row_idx})*$D$32/(WEO_Data!{weo_col}3/100)"
    
    # Projected Growth
    # ((BaseReal + Impact) / (PrevBaseReal + PrevImpact)) - 1
    if year > start_year:
        ws_na.cell(row=row_idx, column=11).value = f"=((C{row_idx}+G{row_idx})/(C{row_idx-1}+G{row_idx-1}))-1"
        ws_na.cell(row=row_idx, column=11).number_format = '0.0%'
