    public static void main(String[] args) throws Exception {

        String taskInput = params.get("task_input");
        String jobInput = params.get("job_input");

        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Task Stream
        DataStream<TaskEvent> taskEvents;
        if (AppBase.taskEvents != null) {
            taskEvents = env.addSource(AppBase.taskEvents);
        } else {
            taskEvents = env.readTextFile(taskInput)
                    .map(new MapFunction<String, TaskEvent>() {
                        @Override
                        public TaskEvent map(String s) throws Exception {
                            if (s == null || s.isEmpty()) return null;
                            String[] tokens = s.split(",");
                            if (tokens.length < 6) return null;
                            try {
                                long ts = Long.parseLong(tokens[0]);
                                long jid = Long.parseLong(tokens[2]);
                                int type = Integer.parseInt(tokens[5]);
                                // Convert micros to millis
                                return new TaskEvent(ts / 1000, jid, type);
                            } catch (NumberFormatException e) {
                                return null;
                            }
                        }
                    })
                    .filter(t -> t != null);
        }

        DataStream<TaskEvent> validTasks = taskEvents
                .assignTimestampsAndWatermarks(WatermarkStrategy
                        .<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp))
                .filter(t -> t.eventType == 0); // SUBMIT

        // Job Stream
        DataStream<JobEvent> jobEvents;
        if (AppBase.jobEvents != null) {
            jobEvents = env.addSource(AppBase.jobEvents);
        } else {
            jobEvents = env.readTextFile(jobInput)
                    .map(new MapFunction<String, JobEvent>() {
                        @Override
                        public JobEvent map(String s) throws Exception {
                            if (s == null || s.isEmpty()) return null;
                            String[] tokens = s.split(",");
                            if (tokens.length < 4) return null;
                            try {
                                long ts = Long.parseLong(tokens[0]);
                                long jid = Long.parseLong(tokens[2]);
                                int type = Integer.parseInt(tokens[3]);
                                return new JobEvent(ts / 1000, jid, type);
                            } catch (NumberFormatException e) {
                                return null;
                            }
                        }

                        .<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp))

    public static class SessionResult {
        public long jobId;
        public int count;
        public long windowEnd;
        public SessionResult() {}
        public SessionResult(long j, int c, long w) { jobId=j; count=c; windowEnd=w; }
    }

    public static class CountSessionTasks extends ProcessWindowFunction<TaskEvent, SessionResult, Long, TimeWindow> {
        @Override
        public void process(Long key, Context context, Iterable<TaskEvent> elements, Collector<SessionResult> out) {
            int count = 0;
            for (TaskEvent e : elements) count++;
            out.collect(new SessionResult(key, count, context.window().getEnd()));
        }
    }

    public static class MergeJobAndSessions extends KeyedCoProcessFunction<Long, SessionResult, JobEvent, String> {
        private final long waitTime;
        private ValueState<Integer> maxCountState;
        private ValueState<Boolean> jobFinishedState;

        public MergeJobAndSessions(long waitTime) {
            this.waitTime = waitTime;
        }

        @Override
        public void open(Configuration parameters) {
            maxCountState = getRuntimeContext().getState(new ValueStateDescriptor<>("maxCount", Integer.class));
            jobFinishedState = getRuntimeContext().getState(new ValueStateDescriptor<>("jobFinished", Boolean.class));
        }

        @Override
        public void processElement1(SessionResult value, Context ctx, Collector<String> out) throws Exception {
            Integer currentMax = maxCountState.value();
            if (currentMax == null || value.count > currentMax) {
                maxCountState.update(value.count);
            }
        }

        @Override
        public void processElement2(JobEvent value, Context ctx, Collector<String> out) throws Exception {
            jobFinishedState.update(true);
            // Set timer to wait for any lagging sessions or for the last session to close.
            long timerTime = value.timestamp + waitTime + 1000; // +1 sec buffer
            ctx.timerService().registerEventTimeTimer(timerTime);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            Boolean finished = jobFinishedState.value();
            if (finished != null && finished) {
                Integer max = maxCountState.value();
                if (max != null) {
                    out.collect("(" + ctx.getCurrentKey() + "," + max + ")");
                }
                // Clear state
                maxCountState.clear();
                jobFinishedState.clear();
            }
        }
    }
