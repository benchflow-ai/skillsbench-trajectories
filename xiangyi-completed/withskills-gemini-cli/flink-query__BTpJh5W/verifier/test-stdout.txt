package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.AggregateFunction;
import org.apache.flink.api.common.serialization.SimpleStringEncoder;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.Path;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.datastream.SingleOutputStreamOperator;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.sink.SinkFunction;
import org.apache.flink.streaming.api.functions.sink.filesystem.StreamingFileSink;
import org.apache.flink.streaming.api.functions.windowing.ProcessWindowFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;

import java.util.Objects;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input");
        String jobInput = params.get("job_input");
        String outputPath = params.get("output");

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        // Read Task Events
        DataStream<TaskEvent> tasks = env.readTextFile(taskInput)
                .map(TaskEvent::fromString)
                .filter(Objects::nonNull)
                .assignTimestampsAndWatermarks(WatermarkStrategy
                        .<TaskEvent>forMonotonousTimestamps()
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));

        // Session Windows on Task Submits
        SingleOutputStreamOperator<Tuple2<Long, Integer>> sessionCounts = tasks
                .filter(t -> t.eventType == 0) // SUBMIT
                .keyBy(t -> t.jobId)
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .aggregate(new CountAgg(), new WindowResult());

        // Read Job Events
        DataStream<JobEvent> jobs = env.readTextFile(jobInput)
                .map(JobEvent::fromString)
                .filter(Objects::nonNull)
                .assignTimestampsAndWatermarks(WatermarkStrategy
                        .<JobEvent>forMonotonousTimestamps()
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));

        // Filter Terminal Job Events
        DataStream<JobEvent> jobTerminals = jobs
                .filter(j -> j.eventType == 3 || j.eventType == 4 || j.eventType == 5 || j.eventType == 6);

        // Connect and find Max
        DataStream<String> result = sessionCounts
                .connect(jobTerminals)
                .keyBy(t -> t.f0, j -> j.jobId)
                .process(new MaxSessionProcessFunction());

        // Output
        if (AppBase.out != null) {
            result.addSink(AppBase.out);
        } else if (outputPath != null) {
            final StreamingFileSink<String> sink = StreamingFileSink
                    .forRowFormat(new Path(outputPath), new SimpleStringEncoder<String>("UTF-8"))
                    .build();
            // AppBase.sinkOrTest handles SinkFunction, StreamingFileSink implements SinkFunction
            result.addSink(sink);
        } else {
             result.print();
        }

        env.execute("LongestSessionPerJob");
    }

    public static class CountAgg implements AggregateFunction<TaskEvent, Long, Long> {
        @Override
        public Long createAccumulator() { return 0L; }
        @Override
        public Long add(TaskEvent value, Long accumulator) { return accumulator + 1; }
        @Override
        public Long getResult(Long accumulator) { return accumulator; }
        @Override
        public Long merge(Long a, Long b) { return a + b; }
    }

    public static class WindowResult extends ProcessWindowFunction<Long, Tuple2<Long, Integer>, Long, TimeWindow> {
        @Override
        public void process(Long key, Context context, Iterable<Long> elements, Collector<Tuple2<Long, Integer>> out) {
            long count = elements.iterator().next();
            out.collect(Tuple2.of(key, (int)count));
        }
    }

    public static class MaxSessionProcessFunction extends KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, JobEvent, String> {
        private ValueState<Integer> maxTasks;
        private ValueState<Long> timerTs;

        @Override
        public void open(Configuration parameters) {
            maxTasks = getRuntimeContext().getState(new ValueStateDescriptor<>("maxTasks", Integer.class));
            timerTs = getRuntimeContext().getState(new ValueStateDescriptor<>("timerTs", Long.class));
        }

        @Override
        public void processElement1(Tuple2<Long, Integer> value, Context ctx, Collector<String> out) throws Exception {
            Integer current = maxTasks.value();
            if (current == null || value.f1 > current) {
                maxTasks.update(value.f1);
            }
        }

        @Override
        public void processElement2(JobEvent value, Context ctx, Collector<String> out) throws Exception {
            // Job Finished. Register timer.
            // Timestamp is in millis (from assigner).
            // Add 10 mins (600000ms) to ensure session windows have closed.
            long targetTime = ctx.timestamp() + 600000 + 1;
            
            Long existing = timerTs.value();
            if (existing != null) {
                ctx.timerService().deleteEventTimeTimer(existing);
            }
            
            ctx.timerService().registerEventTimeTimer(targetTime);
            timerTs.update(targetTime);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
            Integer count = maxTasks.value();
            if (count == null) count = 0;
            out.collect("(" + ctx.getCurrentKey() + "," + count + ")");
            maxTasks.clear();
            timerTs.clear();
        }
    }
}
