import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import time

def name_to_smiles(name):
    try:
        results = pcp.get_compounds(name, 'name')
        if results:
            # Use smiles to get the best representation available in modern PubChemPy
            return results[0].smiles or results[0].isomeric_smiles or results[0].canonical_smiles
    except Exception as e:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # 1. Extract names from PDF
    pool_names = []
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                names = [line.strip() for line in text.split('\n') if line.strip()]
                pool_names.extend(names)
    
    # Remove duplicates but maintain alphabetical order
    pool_names = sorted(list(set(pool_names)))
    
    # 2. Convert target molecule to SMILES
    target_smiles = name_to_smiles(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Use MorganGenerator for modern RDKit
    morgan_gen = AllChem.GetMorganGenerator(radius=2, includeChirality=True)
    target_fp = morgan_gen.GetFingerprint(target_mol)
    
    # 3. Convert pool names to SMILES and compute similarity
    results = []
    for name in pool_names:
        smiles = name_to_smiles(name)
        if not smiles:
            continue
        
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
        
        fp = morgan_gen.GetFingerprint(mol)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))
        # Be nice to PubChem API
        time.sleep(0.2)
    
    # 4. Sort results
    # Primary sort: similarity descending
    # Secondary sort: name ascending
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k
    return results[:top_k]
