from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from typing import Any

def to_rational_string(val: float) -> str:
    """
    Converts a float coordinate to a rational string representation
    with denominator <= 12, as required. Coordinates are normalized to [0, 1).
    """
    # Normalize to [0, 1)
    val_norm = float(val) % 1.0
    # Handle floating point precision issues near 1.0
    if abs(val_norm - 1.0) < 1e-6:
        val_norm = 0.0
    
    # Find the closest fraction with denominator <= 12
    f = Fraction(val_norm).limit_denominator(12)
    
    # If the fraction rounds to 1/1, it corresponds to 0 in fractional coordinates
    if f.numerator == 0 or f.numerator == f.denominator:
        return "0"
    
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    """
    # Load the structure from the CIF file
    struct = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to find the symmetry and group equivalent sites (orbits)
    # symprec=0.01 is a standard tolerance for symmetry detection
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Iterate through each orbit (set of equivalent sites) found in the structure
    for i, orbit in enumerate(sym_struct.equivalent_sites):
        # wyckoff_symbols contains strings like "4a", "8c", etc.
        symbol = sym_struct.wyckoff_symbols[i]
        multiplicity = int(symbol[:-1])
        letter = symbol[-1]
        
        # If multiple orbits have the same Wyckoff letter, we follow the requirement 
        # to provide coordinates for the "first atom of each Wyckoff position".
        # Based on the example, we map unique Wyckoff letters to their properties.
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = multiplicity
            # Get the fractional coordinates of the first atom in this orbit
            first_atom_coords = orbit[0].frac_coords
            wyckoff_coordinates_dict[letter] = [to_rational_string(c) for c in first_atom_coords]
            
    # Sort the resulting dictionaries by Wyckoff letter for consistent output
    sorted_multiplicities = dict(sorted(wyckoff_multiplicity_dict.items()))
    sorted_coordinates = dict(sorted(wyckoff_coordinates_dict.items()))
    
    return {
        "wyckoff_multiplicity_dict": sorted_multiplicities,
        "wyckoff_coordinates_dict": sorted_coordinates
    }
