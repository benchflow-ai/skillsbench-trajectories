def get_similarity(a, b):
    return difflib.SequenceMatcher(None, a.lower(), b.lower()).ratio()

def score_candidate(segment, candidate, record):
    score = 0.0
    
    # 1. Station compatibility
    if candidate['station_scope'] and record['station'] not in candidate['station_scope']:
        return -1.0 # Reject
    
    seg_lower = segment.lower()
    label_lower = candidate['standard_label'].lower()
    
    # 2. Keyword/Term overlap
    keyword_matches = 0
    for kw in candidate['keywords']:
        if kw in seg_lower:
            # Check if it's a whole word or significant part
            keyword_matches += 1
            score += 0.4
            
    # 3. Fuzzy similarity with standard label or category
    fuzzy_sim = get_similarity(segment, candidate['standard_label'])
    score += fuzzy_sim * 0.4
    
    # 4. Context alignment
    # Fail code category match
    # E101, E207, etc. often align with categories
    fail_code = record['fail_code']
    if fail_code:
        # Example: E1xx often means OPEN/SHORT in some codebooks? 
        # Let's see if fail_code appears in keywords or labels
        if fail_code.lower() in seg_lower:
            score += 0.1
            
    # 5. Test Item match
    test_item = record['test_item'].lower()
    if test_item in label_lower or test_item in seg_lower:
        score += 0.2

    return score
