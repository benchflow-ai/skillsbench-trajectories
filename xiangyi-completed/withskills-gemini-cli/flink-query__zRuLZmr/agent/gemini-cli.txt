package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.windowing.ProcessWindowFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        
        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1); // Usually good for small file processing, but can be higher.

        // Task Stream
        DataStream<TaskEvent> taskStream;
        if (taskEvents != null) {
            taskStream = env.addSource(taskEvents);
        } else {
            taskStream = env.readTextFile(taskInput)
                    .map(new MapFunction<String, TaskEvent>() {
                        @Override
                        public TaskEvent map(String value) throws Exception {
                            return TaskEvent.fromString(value);
                        }
                    })
                    .filter(t -> t != null);
        }

        DataStream<TaskEvent> tasks = taskStream
                .assignTimestampsAndWatermarks(
                    WatermarkStrategy
                        .<TaskEvent>forMonotonousTimestamps()
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                )
                .filter(t -> t.eventType == 0); // SUBMIT

        // Job Stream
        DataStream<JobEvent> jobStream;
        if (jobEvents != null) {
            jobStream = env.addSource(jobEvents);
        } else {
            jobStream = env.readTextFile(jobInput)
                    .map(new MapFunction<String, JobEvent>() {
                        @Override
                        public JobEvent map(String value) throws Exception {
                            return JobEvent.fromString(value);
                        }
                    })
                    .filter(j -> j != null);
        }

        DataStream<JobEvent> jobs = jobStream
                .assignTimestampsAndWatermarks(
                    WatermarkStrategy
                        .<JobEvent>forMonotonousTimestamps()
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                )
                .filter(j -> j.eventType == 3 || j.eventType == 4 || j.eventType == 5); // FAIL, FINISH, KILL

        // Calculate session counts
        DataStream<Tuple2<Long, Integer>> sessionCounts = tasks
                .keyBy(t -> t.jobId)
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .process(new ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void process(Long jobId, Context context, Iterable<TaskEvent> elements, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        int count = 0;
                        for (TaskEvent ignored : elements) {
                            count++;
                        }
                        out.collect(Tuple2.of(jobId, count));
                    }
                });

        // Join with Job Finish events to find max
        DataStream<String> result = sessionCounts
                .connect(jobs)
                .keyBy(t -> t.f0, j -> j.jobId)
                .process(new KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, JobEvent, Tuple2<Long, Integer>>() {
                    private ValueState<Integer> maxTasksState;
                    private ValueState<Boolean> jobFinishedState;

                    @Override
                    public void open(Configuration parameters) throws Exception {
                        maxTasksState = getRuntimeContext().getState(new ValueStateDescriptor<>("maxTasks", Integer.class));
                        jobFinishedState = getRuntimeContext().getState(new ValueStateDescriptor<>("jobFinished", Boolean.class));
                    }

                    @Override
                    public void processElement1(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Integer currentMax = maxTasksState.value();
                        if (currentMax == null || value.f1 > currentMax) {
                            maxTasksState.update(value.f1);
                        }
                    }

                    @Override
                    public void processElement2(JobEvent value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        jobFinishedState.update(true);
                        // Register timer to emit result. 
                        // We wait 10 minutes (session gap) to be sure all sessions are closed? 
                        // Actually, if job finishes at T, any session active at T could extend to T+10m. 
                        // So we wait until watermark passes T+10m.
                        // Or just use the current watermark + 10m?
                        // A safe bet is: timestamp of finish event + 10 minutes.
                        long finishTime = value.timestamp / 1000;
                        ctx.timerService().registerEventTimeTimer(finishTime + Time.minutes(10).toMilliseconds());
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Boolean finished = jobFinishedState.value();
                        if (finished != null && finished) {
                            Integer max = maxTasksState.value();
                            if (max != null) {
                                out.collect(Tuple2.of(ctx.getCurrentKey(), max));
                            }
                            // Cleanup
                            maxTasksState.clear();
                            jobFinishedState.clear();
                        }
                    }
                })
                .map(t -> "(" + t.f0 + "," + t.f1 + ")");

        if (out != null) {
            result.addSink(out);
        } else {
            result.writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

}
