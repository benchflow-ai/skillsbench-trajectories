from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from collections import defaultdict
from typing import Any, Dict

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates 
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'.
    """
    try:
        structure = Structure.from_file(filepath)
    except Exception as e:
        # Depending on requirements, we might want to log this or re-raise.
        # For this task, we assume valid input or fail noisily.
        raise ValueError(f"Failed to load structure from {filepath}: {e}")

    sga = SpacegroupAnalyzer(structure)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_symbols = sym_struct.wyckoff_symbols
    equivalent_sites = sym_struct.equivalent_sites
    
    multiplicity_dict = {}
    coordinates_dict = {}
    
    # Counter to handle duplicate Wyckoff letters (e.g. multiple '4a' sites)
    wyckoff_counts = defaultdict(int)
    
    for symbol, sites in zip(wyckoff_symbols, equivalent_sites):
        # Extract Wyckoff letter (last character of the symbol, e.g., '4a' -> 'a')
        letter = symbol[-1]
        
        wyckoff_counts[letter] += 1
        
        if wyckoff_counts[letter] == 1:
            key = letter
        else:
            key = f"{letter}_{wyckoff_counts[letter]}"
            
        # Multiplicity is the number of atoms in this set
        multiplicity_dict[key] = len(sites)
        
        # Coordinates of the first atom in the set
        first_site_coords = sites[0].frac_coords
        
        formatted_coords = []
        for coord in first_site_coords:
            # Approximate to nearest fraction with denominator <= 12
            frac = Fraction(coord).limit_denominator(12)
            formatted_coords.append(str(frac))
            
        coordinates_dict[key] = formatted_coords
        
    return {
        "wyckoff_multiplicity_dict": multiplicity_dict,
        "wyckoff_coordinates_dict": coordinates_dict
    }
