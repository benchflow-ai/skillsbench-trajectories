import pdfplumber
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import AllChem
import os
import warnings

# Suppress PubChemPy deprecation warnings and RDKit warnings for cleaner output
warnings.filterwarnings("ignore", category=DeprecationWarning)

def get_smiles(name, cache):
    if name in cache:
        return cache[name]
    try:
        results = pcp.get_compounds(name, 'name')
        if results:
            # Using canonical_smiles as requested for Morgan fingerprints
            # (Note: pubchempy might warn about canonical_smiles, but it's still widely used)
            smiles = results[0].canonical_smiles
            cache[name] = smiles
            return smiles
    except Exception:
        pass
    cache[name] = None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF molecule pool to a target molecule.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: A list of chemical names (str) sorted by similarity desc and name asc.
    """
    # 1. Extract names from PDF
    if not os.path.exists(molecule_pool_filepath):
        return []
        
    all_names = []
    with pdfplumber.open(molecule_pool_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.split('\n')
                for line in lines:
                    name = line.strip()
                    if name:
                        all_names.append(name)
    
    if not all_names:
        return []

    # 2. Convert target molecule to fingerprint
    cache = {}
    target_smiles = get_smiles(target_molecule_name, cache)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 3. Process pool molecules
    pool_data = [] # List of (name, similarity)
    name_to_fp = {} 

    for name in all_names:
        if name not in name_to_fp:
            smiles = get_smiles(name, cache)
            if smiles:
                mol = Chem.MolFromSmiles(smiles)
                if mol:
                    fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
                    name_to_fp[name] = fp
                else:
                    name_to_fp[name] = None
            else:
                name_to_fp[name] = None
        
        fp = name_to_fp[name]
        if fp:
            similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
            pool_data.append((name, similarity))
    
    # 4. Sort results
    # descending order of similarity, with alphabetical ordering when ties happen
    pool_data.sort(key=lambda x: (-x[1], x[0]))
    
    # Return only the names
    top_k_names = [item[0] for item in pool_data[:top_k]]
    
    return top_k_names

if __name__ == "__main__":
    # The script should be usable as a module.
    pass
