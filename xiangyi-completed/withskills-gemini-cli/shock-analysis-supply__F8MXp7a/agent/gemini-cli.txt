# Implies K_proj = Y_proj * avg_ky.
for i in range(len(df)):
    if df.loc[i, 'Year'] > 2023: # PWT ends ~2019/2021, but let's say we have data or need to project
        # If rnna is NaN, calculate it
        if pd.isna(df.loc[i, 'rnna']):
             df.loc[i, 'rnna'] = df.loc[i, 'GDP_Real_WEO'] * avg_ky

# Extend LnZ Trend (Linear Trend)
# Use the HP Trend to project forward.
# Excel 'TREND' uses linear regression on the known y's (hp_trend) and known x's (years).
# Fit line to calculated hp_trend
z_y = hp_data['lnZ_Trend'].values
z_x = hp_data['Year'].values
# Fit linear model
if len(z_x) > 0:
    poly = np.polyfit(z_x, z_y, 1) # Linear
    p = np.poly1d(poly)
    
    # Project LnZ Trend for 2024 onwards (actually to 2041)
    proj_mask = df['Year'] > 2023
    df.loc[proj_mask, 'lnZ_Trend'] = p(df.loc[proj_mask, 'Year'])

# --- 7. Prepare Output Dictionary ---
output = {
    'pwt': df[['Year', 'rnna', 'xr']].dropna(subset=['rnna']).to_dict(orient='records'),
    'weo': df[['Year', 'GDP_Real_WEO', 'GDP_Growth_WEO']].dropna(subset=['GDP_Real_WEO']).to_dict(orient='records'),
    'ecb': df[['Year', 'CFC_LCU', 'rnna', 'Implied_Deprec']].dropna(subset=['CFC_LCU']).to_dict(orient='records'),
    'production': {
        'avg_deprec': avg_deprec,
        'hp_data': hp_data[['Year', 'lnK', 'lnY', 'lnZ_Trend']].to_dict(orient='records'),
        'projection': df[['Year', 'GDP_Real_WEO', 'rnna', 'lnZ_Trend']].to_dict(orient='records')
    }
}

with open('model_data.json', 'w') as f:
