from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict

def float_to_fraction_str(x: float, max_denom: int = 12) -> str:
    f = Fraction(x).limit_denominator(max_denom)
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes a CIF file to determine Wyckoff position multiplicities and coordinates.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'.
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    sym_struct = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # sym_struct.equivalent_sites is a list of lists of PeriodicSite objects.
    # sym_struct.wyckoff_symbols is a corresponding list of Wyckoff symbols (e.g., "4a", "8c").
    
    for i, sites in enumerate(sym_struct.equivalent_sites):
        symbol = sym_struct.wyckoff_symbols[i]
        
        # Extract the Wyckoff letter. 
        # Wyckoff symbols usually end with the letter.
        # Examples: "4a", "8c", "24d", "96i"
        # We strip digits from the left.
        letter = symbol.lstrip("0123456789")
        
        # Multiplicity is the number of equivalent sites in the group
        multiplicity = len(sites)
        
        # Get the first site's fractional coordinates
        first_site = sites[0]
        frac_coords = first_site.frac_coords
        
        # Convert coordinates to simplified fractions
        coords_strs = [float_to_fraction_str(c) for c in frac_coords]
        
        wyckoff_multiplicity_dict[letter] = multiplicity
        wyckoff_coordinates_dict[letter] = coords_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
