package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.core.fs.FileSystem;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.util.Collector;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);
        final long gap = sesssize * 1_000_000L;

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskStream = env.readTextFile(taskInput)
                .map(TaskEvent::fromString)
                .filter(e -> e != null && e.eventType == 0) // SUBMIT
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp)
                );

        DataStream<JobEvent> jobStream = env.readTextFile(jobInput)
                .map(JobEvent::fromString)
                .filter(e -> e != null && (e.eventType == 4 || e.eventType == 3 || e.eventType == 5)) // FINISH, FAIL, KILL
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp)
                );

        taskStream.connect(jobStream)
                .keyBy(e -> e.jobId, e -> e.jobId)
                .process(new StageProcessor(gap))
                .map(t -> "(" + t.f0 + "," + t.f1 + ")")
                .writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE)
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    public static class StageProcessor extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {
        private final long gap;
        private ValueState<Integer> maxTasks;
        private ValueState<Integer> currentTasks;
        private ValueState<Long> lastSubmitTime;
        private ValueState<Boolean> isFinished;

        public StageProcessor(long gap) {
            this.gap = gap;
        }

        @Override
        public void open(Configuration parameters) {
            maxTasks = getRuntimeContext().getState(new ValueStateDescriptor<>("maxTasks", Integer.class, 0));
            currentTasks = getRuntimeContext().getState(new ValueStateDescriptor<>("currentTasks", Integer.class, 0));
            lastSubmitTime = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTime", Long.class, 0L));
            isFinished = getRuntimeContext().getState(new ValueStateDescriptor<>("isFinished", Boolean.class, false));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            long lastTime = lastSubmitTime.value();
            if (lastTime > 0 && event.timestamp - lastTime > gap) {
                if (currentTasks.value() > maxTasks.value()) {
                    maxTasks.update(currentTasks.value());
                }
                currentTasks.update(1);
            } else {
                currentTasks.update(currentTasks.value() + 1);
            }
            lastSubmitTime.update(event.timestamp);
            ctx.timerService().registerEventTimeTimer(event.timestamp + gap);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            isFinished.update(true);
            if (lastSubmitTime.value() == 0 || ctx.timerService().currentWatermark() >= lastSubmitTime.value() + gap) {
                emitResult(ctx, out);
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            if (timestamp == lastSubmitTime.value() + gap) {
                if (currentTasks.value() > maxTasks.value()) {
                    maxTasks.update(currentTasks.value());
                }
                currentTasks.update(0);
                if (isFinished.value()) {
                    emitResult(ctx, out);
                }
            }
        }

        private void emitResult(Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            int finalMax = Math.max(maxTasks.value(), currentTasks.value());
            if (finalMax > 0) {
                out.collect(new Tuple2<>(ctx.getCurrentKey(), finalMax));
            }
            maxTasks.clear();
            currentTasks.clear();
            lastSubmitTime.clear();
            isFinished.clear();
        }
    }

}
