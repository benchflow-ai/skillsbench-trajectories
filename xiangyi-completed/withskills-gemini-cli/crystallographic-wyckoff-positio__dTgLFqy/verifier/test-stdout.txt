import warnings
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict, List

# Suppress warnings from pymatgen to keep output clean
warnings.filterwarnings("ignore")

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes a CIF file to determine Wyckoff position multiplicities and 
    calculates approximate rational coordinates for the first atom of each Wyckoff position.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Mapping of Wyckoff letter to count of atoms.
        - "wyckoff_coordinates_dict": Mapping of Wyckoff letter to list of coordinate strings.
    """
    try:
        struct = Structure.from_file(filepath)
    except Exception as e:
        raise ValueError(f"Failed to load structure from {filepath}: {e}")

    # Initialize SpacegroupAnalyzer
    # Default symprec=0.01 is usually sufficient for standard CIFs
    sga = SpacegroupAnalyzer(struct)
    
    # Get symmetry dataset
    dataset = sga.get_symmetry_dataset()
    
    if dataset is None:
        raise ValueError("Could not determine symmetry dataset for the structure.")
        
    # 'wyckoffs' is a list of Wyckoff letters (e.g., ['a', 'a', 'c', ...]) 
    # corresponding to the atoms in the input structure order.
    wyckoffs = dataset['wyckoffs']
    
    multiplicity_dict = {}
    coordinates_dict = {}
    
    # Iterate through atoms to count multiplicities and find first coordinates
    for i, letter in enumerate(wyckoffs):
        # Update multiplicity count
        multiplicity_dict[letter] = multiplicity_dict.get(letter, 0) + 1
        
        # If this is the first time we see this Wyckoff letter, record its coordinates
        if letter not in coordinates_dict:
            coords = struct[i].frac_coords
            rational_coords = []
            
            for c in coords:
                # Normalize coordinate to [0, 1)
                val = c % 1.0
                
                # Convert to fraction with denominator <= 12
                f = Fraction(val).limit_denominator(12)
                
                # If rounding resulted in 1, wrap back to 0
                if f == 1:
                    f = Fraction(0, 1)
                    
                rational_coords.append(str(f))
            
            coordinates_dict[letter] = rational_coords
            
    return {
        "wyckoff_multiplicity_dict": multiplicity_dict,
        "wyckoff_coordinates_dict": coordinates_dict
    }
