import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from pypdf import PdfReader
import os
import time
import random
import warnings

# Suppress warnings
warnings.filterwarnings("ignore", category=DeprecationWarning) 
warnings.filterwarnings("ignore", category=UserWarning)

def get_smiles(name):
    max_retries = 5
    base_delay = 1
    
    for attempt in range(max_retries):
        try:
            # Respect rate limits (approx 5 requests/sec max, so 0.2s delay minimum)
            # Add a bit of jitter
            time.sleep(0.25 + random.random() * 0.1)
            
            # Use pubchempy to get SMILES
            # There might be multiple results, we usually take the first one
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                return compounds[0].canonical_smiles
            return None # Not found
            
        except Exception as e:
            error_msg = str(e)
            if "503" in error_msg or "ServerBusy" in error_msg or "Timeout" in error_msg:
                if attempt < max_retries - 1:
                    sleep_time = base_delay * (2 ** attempt) + random.random()
                    print(f"Server busy/error for {name}, retrying in {sleep_time:.2f}s...")
                    time.sleep(sleep_time)
                    continue
            
            print(f"Error getting SMILES for {name}: {e}")
            return None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # 1. Get target SMILES
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        # Fallback or strict error? Prompt implies we are given chemical names.
        # If target is not found, we probably can't proceed.
        print(f"Warning: Could not find SMILES for target molecule: {target_molecule_name}")
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        print(f"Warning: Could not parse SMILES for target molecule: {target_molecule_name}")
        return []
        
    # Morgan fingerprint for target
    # radius = 2, include chirality (useChirality=True)
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    # 2. Read molecules from PDF
    pool_molecules = []
    try:
        reader = PdfReader(molecule_pool_filepath)
        text = ""
        for page in reader.pages:
            text += page.extract_text() + "\n"
        
        # Split by newlines and filter empty lines
        lines = [line.strip() for line in text.split('\n') if line.strip()]
        # Remove duplicates if any, but keep order? 
        # The prompt doesn't say unique, but usually we don't want duplicates.
        # However, list(set()) loses order.
        # Let's keep them as is from the PDF, assuming the PDF is the source of truth.
        pool_molecules = lines
    except Exception as e:
        print(f"Error reading PDF file: {e}")
        return []

    results = []
    
    # 3. Process pool molecules
    for name in pool_molecules:
        # Skip if name is exactly the target? Usually similarity search includes the target if it's in the pool.
        # But if the target is "Guaiacol" and "Guaiacol" is in the list, similarity is 1.0.
        # The prompt says "Find the top k similar chemicals... to any chemicals you are given."
        # It doesn't explicitly exclude self. I will include it.
        
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # 4. Sort results
    # Sort by similarity (descending), then by name (ascending)
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # Return top k names
    return [name for name, score in results[:top_k]]
