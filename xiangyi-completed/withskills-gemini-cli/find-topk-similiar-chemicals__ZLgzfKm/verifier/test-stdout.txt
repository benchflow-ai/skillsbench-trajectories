import time
from pypdf import PdfReader
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs

def get_smiles(name):
    try:
        # Rate limit handling: Max 5 requests/sec -> 0.2s delay minimum
        time.sleep(0.25) 
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            return compounds[0].canonical_smiles
    except Exception as e:
        # Silently fail or log? For a solution, printing might be noisy but helpful for debug.
        # print(f"Warning: Error fetching {name}: {e}")
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target molecule. 
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing the pool of molecules.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: List of chemical names sorted by similarity (desc) and name (asc).
    """
    
    # 1. Extract names from PDF
    reader = PdfReader(molecule_pool_filepath)
    names = set() # Use set to avoid duplicates immediately
    for page in reader.pages:
        text = page.extract_text()
        if text:
            lines = text.split('\n')
            for line in lines:
                clean_line = line.strip()
                if clean_line:
                    names.add(clean_line)
    
    sorted_names = sorted(list(names)) # Sort for deterministic processing order
    
    # 2. Get Target SMILES and Fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        # If target cannot be found, we can't compute similarity. 
        # Returning empty list or raising error. Raising error is safer.
        raise ValueError(f"Could not find SMILES for target: {target_molecule_name}")
        
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
         raise ValueError(f"Invalid SMILES/Structure for target: {target_molecule_name}")

    # Morgan fingerprint, radius 2, include chirality (useChirality=True)
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 3. Process Pool
    results = []
    
    for name in sorted_names:
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, sim))
        
    # 4. Sort
    # Primary: Similarity (Descending) -> -x[1]
    # Secondary: Name (Ascending) -> x[0]
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k names
    return [x[0] for x in results[:top_k]]
