from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates 
    for the first atom of each Wyckoff position from a CIF file.
    """
    # Load the structure from the CIF file
    structure = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    sga = SpacegroupAnalyzer(structure)
    sym_data = sga.get_symmetry_dataset()
    
    # sym_data.wyckoffs contains the Wyckoff letter for each atom in the standardized cell
    # sym_data.std_positions contains the coordinates of these atoms in the standardized cell
    wyckoff_letters = sym_data.wyckoffs
    std_positions = sym_data.std_positions
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track which letters we've already found a coordinate for
    seen_letters = set()
    
    for letter, coord in zip(wyckoff_letters, std_positions):
        # Count multiplicity for each Wyckoff letter
        wyckoff_multiplicity_dict[letter] = wyckoff_multiplicity_dict.get(letter, 0) + 1
        
        # If this is the first time we see this letter, store its coordinates
        if letter not in seen_letters:
            formatted_coords = []
            for c in coord:
                # Wrap coordinate to [0, 1) range
                c_wrapped = c % 1
                # Handle potential floating point precision issues near 1.0
                if abs(c_wrapped - 1.0) < 1e-7:
                    c_wrapped = 0.0
                
                # Convert to fraction with denominator <= 12
                frac = Fraction(c_wrapped).limit_denominator(12)
                
                # Format as string: "0" instead of "0/1"
                if frac.numerator == 0:
                    formatted_coords.append("0")
                else:
                    formatted_coords.append(str(frac))
                    
            wyckoff_coordinates_dict[letter] = formatted_coords
            seen_letters.add(letter)
            
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
