package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.SerializableTimestampAssigner;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.FilterFunction;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.functions.KeySelector;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.TimeCharacteristic;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.windowing.ProcessWindowFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.util.Collector;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input");
        String jobInput = params.get("job_input");
        String outputPath = params.get("output");

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setStreamTimeCharacteristic(TimeCharacteristic.EventTime);

        // Read and parse Task Events
        DataStream<TaskEvent> taskEvents = env.readTextFile(taskInput)
                .map(new MapFunction<String, TaskEvent>() {
                    @Override
                    public TaskEvent map(String value) throws Exception {
                        return TaskEvent.fromString(value);
                    }
                })
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent value) throws Exception {
                        return value != null && value.eventType == 0; // Filter for SUBMIT events
                    }
                })
                .assignTimestampsAndWatermarks(WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                        .withTimestampAssigner(new SerializableTimestampAssigner<TaskEvent>() {
                            @Override
                            public long extractTimestamp(TaskEvent element, long recordTimestamp) {
                                return element.timestamp / 1000; // Convert micros to millis
                            }
                        }));

        // Calculate session counts
        DataStream<Tuple2<Long, Integer>> sessionCounts = taskEvents
                .keyBy(new KeySelector<TaskEvent, Long>() {
                    @Override
                    public Long getKey(TaskEvent value) throws Exception {
                        return value.jobId;
                    }
                })
                .window(EventTimeSessionWindows.withGap(Time.minutes(10)))
                .process(new ProcessWindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void process(Long key, Context context, Iterable<TaskEvent> elements, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        int count = 0;
                        for (TaskEvent ignored : elements) {
                            count++;
                        }
                        out.collect(Tuple2.of(key, count));
                    }
                });

        // Read and parse Job Events
        DataStream<JobEvent> jobEvents = env.readTextFile(jobInput)
                .map(new MapFunction<String, JobEvent>() {
                    @Override
                    public JobEvent map(String value) throws Exception {
                        return JobEvent.fromString(value);
                    }
                })
                .filter(new FilterFunction<JobEvent>() {
                    @Override
                    public boolean filter(JobEvent value) throws Exception {
                        // Filter for FINISH (4). 
                        // Assuming 4 is FINISH based on standard cluster data format.
                        return value != null && value.eventType == 4;
                    }
                })
                .assignTimestampsAndWatermarks(WatermarkStrategy.<JobEvent>forMonotonousTimestamps()
                        .withTimestampAssigner(new SerializableTimestampAssigner<JobEvent>() {
                            @Override
                            public long extractTimestamp(JobEvent element, long recordTimestamp) {
                                return element.timestamp / 1000; // Convert micros to millis
                            }
                        }));

        // Connect streams to find max session count per finished job
        DataStream<Tuple2<Long, Integer>> result = sessionCounts
                .connect(jobEvents)
                .keyBy(new KeySelector<Tuple2<Long, Integer>, Long>() {
                           @Override
                           public Long getKey(Tuple2<Long, Integer> value) throws Exception {
                               return value.f0;
                           }
                       },
                        new KeySelector<JobEvent, Long>() {
                            @Override
                            public Long getKey(JobEvent value) throws Exception {
                                return value.jobId;
                            }
                        })
                .process(new KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, JobEvent, Tuple2<Long, Integer>>() {

                    private ValueState<Integer> maxCountState;
                    private ValueState<Boolean> jobFinishedState;

                    @Override
                    public void open(Configuration parameters) throws Exception {
                        maxCountState = getRuntimeContext().getState(new ValueStateDescriptor<>("maxCount", Integer.class, 0));
                        jobFinishedState = getRuntimeContext().getState(new ValueStateDescriptor<>("jobFinished", Boolean.class, false));
                    }

                    @Override
                    public void processElement1(Tuple2<Long, Integer> value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        // Process session count
                        Integer currentMax = maxCountState.value();
                        if (currentMax == null) currentMax = 0;
                        if (value.f1 > currentMax) {
                            maxCountState.update(value.f1);
                        }
                    }

                    @Override
                    public void processElement2(JobEvent value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        // Process job finish
                        jobFinishedState.update(true);
                        // Register timer to emit result.
                        long timerTime = ctx.timestamp() + Time.minutes(10).toMilliseconds() + 1;
                        ctx.timerService().registerEventTimeTimer(timerTime);
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
                        Boolean finished = jobFinishedState.value();
                        if (finished != null && finished) {
                            Integer count = maxCountState.value();
                            if (count != null && count > 0) {
                                out.collect(Tuple2.of(ctx.getCurrentKey(), count));
                            }
                            // Clean up
                            maxCountState.clear();
                            jobFinishedState.clear();
                        }
                    }
                });

        if (outputPath != null) {
            result.map(new MapFunction<Tuple2<Long, Integer>, String>() {
                @Override
                public String map(Tuple2<Long, Integer> value) throws Exception {
                    return "(" + value.f0 + "," + value.f1 + ")";
                }
            }).writeAsText(outputPath).setParallelism(1);
        } else {
             // For testing or standard out if needed, but prompt requires file output
             result.print();
        }

        env.execute("LongestSessionPerJob");
    }

}
