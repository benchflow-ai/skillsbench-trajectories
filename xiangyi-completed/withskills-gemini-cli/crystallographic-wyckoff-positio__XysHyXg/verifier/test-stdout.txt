from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath (str): Path to the CIF file.
        
    Returns:
        dict: A dictionary containing 'wyckoff_multiplicity_dict' and 'wyckoff_coordinates_dict'.
    """
    # Load the structure from the CIF file
    # pymatgen handles symmetry operations in CIF files to reconstruct the full unit cell if needed
    struct = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to find the symmetry and Wyckoff positions
    # We use a default symprec of 0.01 which is usually good for XRD data
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    dataset = sga.get_symmetry_dataset()
    
    # dataset.wyckoffs contains the Wyckoff label for each site in struct
    wyckoff_labels = dataset.wyckoffs
    
    wyckoff_counts = {}
    wyckoff_coords = {}
    
    for i, label in enumerate(wyckoff_labels):
        if label not in wyckoff_counts:
            wyckoff_counts[label] = 0
            # Store the first fractional coordinate found for this label
            # Ensure coordinates are in the [0, 1) range
            raw_coord = struct[i].frac_coords
            wrapped_coord = [c % 1 for c in raw_coord]
            # Clean up near-zero or near-one values
            cleaned_coord = []
            for c in wrapped_coord:
                if abs(c) < 1e-7 or abs(c - 1) < 1e-7:
                    cleaned_coord.append(0.0)
                else:
                    cleaned_coord.append(c)
            wyckoff_coords[label] = cleaned_coord
            
        wyckoff_counts[label] += 1
        
    # Format coordinates as simplified fractions with denominator <= 12
    formatted_wyckoff_coords = {}
    for label, coord in wyckoff_coords.items():
        frac_strings = []
        for c in coord:
            # Round to the closest rational number with denominator <= 12
            f = Fraction(c).limit_denominator(12)
            
            if f.numerator == 0 or f == 1:
                frac_strings.append("0")
            elif f.denominator == 1:
                frac_strings.append(str(f.numerator))
            else:
                frac_strings.append(f"{f.numerator}/{f.denominator}")
        formatted_wyckoff_coords[label] = frac_strings
        
    # Sort by Wyckoff label alphabetically for consistent output
    sorted_labels = sorted(wyckoff_counts.keys())
    
    return {
        "wyckoff_multiplicity_dict": {label: wyckoff_counts[label] for label in sorted_labels},
        "wyckoff_coordinates_dict": {label: formatted_wyckoff_coords[label] for label in sorted_labels}
    }
