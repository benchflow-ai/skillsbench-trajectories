class StringTokenizer(val encoding: String = "utf-8", val normalizer: String => String = identity) extends BaseTokenizer[Any] {
  // In Python, input is StrOrBytes. Here we accept Any and check, or could use Either.
  // Using Any to be compatible with UniversalTokenizer's dispatch.

  override def tokenize(value: Any): Token = {
    val strValue = value match {
      case bytes: Array[Byte] => new String(bytes, encoding)
      case s: String => s
      case other => other.toString
    }

    val normalized = normalizer(strValue)
    Token(normalized, TokenType.STRING)
  }
}

class NumericTokenizer(

class TokenRegistry[T] {
  private val registry = mutable.Map.empty[String, TokenContainer[T]]
  private val handlers = mutable.ListBuffer.empty[T => Option[Token]]

  def register(key: String, container: TokenContainer[T]): Unit = {
    registry(key) = container
  }

  def addHandler(handler: T => Option[Token]): Unit = {
    handlers += handler
  }

  def process(key: String): List[Option[Token]] = {
    registry.get(key) match {
      case None => List.empty
      case Some(container) =>
        container.getAll.map { item =>
          // Find first handler that returns Some, or None if all return None
          handlers.view.map(h => h(item)).find(_.isDefined).flatten
        }.toList
    }
  }
}

class JsonTokenizer(val pretty: Boolean = false) {
  
  def tokenize(value: Json): Token = {
    val jsonStr = if (pretty) value.spaces2 else value.noSpaces
    Token(jsonStr, TokenType.STRUCTURED, Map("json" -> true))
  }

  def tokenizePath(value: Json, path: String): Option[Token] = {
    val parts = path.split("\\.")
    
    // Recursive navigation using Circe cursors would be better, but implementing logic akin to Python's simple loop
    var current: io.circe.ACursor = value.hcursor
    
    for (part <- parts) {
      // Check if part is an index (for arrays) or key (for objects)
      // Circe's downField works for objects, downN for arrays.
      // We need to try both or guess. Python code checks isinstance dict vs list.
      // Circe HCursor captures the state.
      
      // Try array index first if it looks like a number
      val indexOpt = Try(part.toInt).toOption
      
      current = indexOpt match {
        case Some(idx) => 
           // If current focus is Array, downN. If Object, might still be a key "123"?
           // Python logic: `if isinstance(current, list) and part.isdigit()`
           // In Circe, we can check focus.
           current.focus match {
             case Some(j) if j.isArray => current.downN(idx)
             case _ => current.downField(part)
           }
        case None => current.downField(part)
      }
    }

    current.focus.map(tokenize)
  }
}

class WhitespaceTokenizer(
  val lowercase: Boolean = false,
  val minLength: Int = 0,
  val maxLength: Option[Int] = None,
  val stripPunctuation: Boolean = false
) {
  private val punctuation = ".,!?;:'\"()[]{}"

  private def processToken(word: String): Option[String] = {
    var processed = word
    if (stripPunctuation) {
      // Scala doesn't have a direct "strip" for set of chars, use dropWhile/reverse
      processed = processed.dropWhile(c => punctuation.contains(c))
                           .reverse.dropWhile(c => punctuation.contains(c)).reverse
    }

    if (lowercase) {
      processed = processed.toLowerCase
    }

    if (processed.length < minLength) {
      return None
    }

    if (maxLength.exists(max => processed.length > max)) {
      processed = processed.take(maxLength.get)
    }

    if (processed.isEmpty) None else Some(processed)
  }

  def tokenize(text: String): List[Token] = {
    val words = text.split("\\s+") // Split by whitespace
    words.zipWithIndex.flatMap { case (word, i) =>
      processToken(word).map { processed =>
        Token(processed, TokenType.STRING, Map("position" -> i, "original" -> word))
      }
    }.toList
  }
  
  def tokenizeToStrings(text: String): List[String] = {
    tokenize(text).map(_.value)
  }

  def countTokens(text: String): Int = {
    tokenize(text).length
  }
}

class TokenizerBuilder[T] {
  private val normalizers = mutable.ListBuffer.empty[String => String]
  private val validators = mutable.ListBuffer.empty[T => Boolean]
  private var metadata = Map.empty[String, Any]

  def withNormalizer(normalizer: String => String): TokenizerBuilder[T] = {
    normalizers += normalizer
    this
  }

  def withValidator(validator: T => Boolean): TokenizerBuilder[T] = {
    validators += validator
    this
  }

  def withMetadata(entries: (String, Any)*): TokenizerBuilder[T] = {
    metadata ++= entries
    this
  }

  def build(): T => Token = {
    // Capture state
    val finalNormalizers = normalizers.toList
    val finalValidators = validators.toList
    val finalMetadata = metadata

    (value: T) => {
      // Validate
      for (validator <- finalValidators) {
        if (!validator(value)) {
          throw new IllegalArgumentException(s"Validation failed for $value")
        }
      }

      // Convert to string
      var strValue = value.toString

      // Normalize
      for (normalizer <- finalNormalizers) {
        strValue = normalizer(strValue)
      }

      Token(strValue, TokenType.STRING, finalMetadata)
    }
  }
}

object TokenizerBuilder {
  def apply[T](): TokenizerBuilder[T] = new TokenizerBuilder[T]()
}
