package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.KeyedCoProcessFunction;
import org.apache.flink.streaming.api.functions.sink.RichSinkFunction;
import org.apache.flink.streaming.api.functions.sink.SinkFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;
import org.apache.flink.streaming.api.windowing.assigners.EventTimeSessionWindows;
import org.apache.flink.streaming.api.windowing.time.Time;
import org.apache.flink.streaming.api.windowing.windows.TimeWindow;
import org.apache.flink.streaming.api.functions.windowing.WindowFunction;
import org.apache.flink.util.Collector;

import java.io.*;
import java.time.Duration;
import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input");
        String jobInput = params.get("job_input");
        String outputPath = params.get("output");
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskEvents = env.addSource(AppBase.taskSourceOrTest(new TaskEventSource(taskInput)))
                .assignTimestampsAndWatermarks(WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));

        DataStream<JobEvent> jobEvents = env.addSource(AppBase.jobSourceOrTest(new JobEventSource(jobInput)))
                .assignTimestampsAndWatermarks(WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofSeconds(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000));

        DataStream<Tuple2<Long, Integer>> stageCounts = taskEvents
                .filter(t -> t.eventType == 0) // SUBMIT
                .keyBy(t -> t.jobId)
                .window(EventTimeSessionWindows.withGap(Time.seconds(sesssize)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent e : input) count++;
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<JobEvent> jobFinishes = jobEvents
                .filter(j -> j.eventType == 4); // FINISH

        stageCounts
                .keyBy(t -> t.f0)
                .connect(jobFinishes.keyBy(j -> j.jobId))
                .process(new KeyedCoProcessFunction<Long, Tuple2<Long, Integer>, JobEvent, String>() {
                    private ValueState<Integer> maxCount;
                    private ValueState<Boolean> finished;

                    @Override
                    public void open(Configuration parameters) {
                        maxCount = getRuntimeContext().getState(new ValueStateDescriptor<>("max", Integer.class));
                        finished = getRuntimeContext().getState(new ValueStateDescriptor<>("finished", Boolean.class));
                    }

                    @Override
                    public void processElement1(Tuple2<Long, Integer> value, Context ctx, Collector<String> out) throws Exception {
                        Integer currentMax = maxCount.value();
                        if (currentMax == null || value.f1 > currentMax) {
                            maxCount.update(value.f1);
                        }
                    }

                    @Override
                    public void processElement2(JobEvent value, Context ctx, Collector<String> out) throws Exception {
                        finished.update(true);
                        // Register a timer to output the final result.
                        // Wait for session gap to ensure all stages have been counted.
                        ctx.timerService().registerEventTimeTimer(ctx.timestamp() + sesssize * 1000L + 1000L);
                    }

                    @Override
                    public void onTimer(long timestamp, OnTimerContext ctx, Collector<String> out) throws Exception {
                        Integer currentMax = maxCount.value();
                        if (currentMax != null) {
                            out.collect("(" + ctx.getCurrentKey() + "," + currentMax + ")");
                        }
                        maxCount.clear();
                        finished.clear();
                    }
                })
                .addSink((SinkFunction<String>) AppBase.sinkOrTest(new TextSink(outputPath)))
                .setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String path;
        private volatile boolean running = true;

        public TaskEventSource(String path) {
            this.path = path;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            if (path == null) return;
            try (InputStream is = new GZIPInputStream(new FileInputStream(path));
                 BufferedReader br = new BufferedReader(new InputStreamReader(is))) {
                String line;
                while (running && (line = br.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    public static class JobEventSource implements SourceFunction<JobEvent> {
        private final String path;
        private volatile boolean running = true;

        public JobEventSource(String path) {
            this.path = path;
        }

        @Override
        public void run(SourceContext<JobEvent> ctx) throws Exception {
            if (path == null) return;
            try (InputStream is = new GZIPInputStream(new FileInputStream(path));
                 BufferedReader br = new BufferedReader(new InputStreamReader(is))) {
                String line;
                while (running && (line = br.readLine()) != null) {
                    JobEvent event = JobEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    public static class TextSink extends RichSinkFunction<String> {
        private final String path;
        private transient PrintWriter writer;

        public TextSink(String path) {
            this.path = path;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            writer = new PrintWriter(new BufferedWriter(new FileWriter(path)));
        }

        @Override
        public void invoke(String value, Context context) throws Exception {
            writer.println(value);
            writer.flush();
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.close();
            }
        }
    }
}
