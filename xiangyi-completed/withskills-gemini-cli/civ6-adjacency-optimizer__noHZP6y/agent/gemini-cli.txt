def parse_map(db_path):
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()

    cursor.execute("SELECT Width, Height FROM Map LIMIT 1")
    row = cursor.fetchone()
    width, height = row['Width'], row['Height']

    tiles = {}
    
    # Base plot data
    cursor.execute("SELECT ID, TerrainType FROM Plots")
    for row in cursor.fetchall():
        pid = row['ID']
        x = pid % width
        y = pid // width
        terrain = row['TerrainType'].replace('TERRAIN_', '')
        is_hills = 'HILLS' in terrain
        base_terrain = terrain.replace('_HILLS', '')
        tiles[(x, y)] = Tile(x=x, y=y, terrain=base_terrain, is_hills=is_hills)

    # Features
    cursor.execute("SELECT PlotID, FeatureType FROM PlotFeatures")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].feature = row['FeatureType']
        if row['FeatureType'] == 'FEATURE_FLOODPLAINS':
            tiles[(x, y)].is_floodplains = True

    # Resources
    cursor.execute("SELECT pr.PlotID, pr.ResourceType, r.ResourceClassType FROM PlotResources pr JOIN Resources r ON pr.ResourceType = r.ResourceType")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].resource = row['ResourceType']
        tiles[(x, y)].resource_type = row['ResourceClassType'].replace('RESOURCECLASS_', '')

    # Rivers
    cursor.execute("SELECT PlotID, Edge FROM PlotRivers")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].river_edges.append(row['Edge'])

    conn.close()
    return tiles, width, height

def parse_map(db_path):
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()

    cursor.execute("SELECT Width, Height FROM Map LIMIT 1")
    row = cursor.fetchone()
    width, height = row['Width'], row['Height']

    tiles = {}
    
    # Base plot data
    cursor.execute("SELECT ID, TerrainType FROM Plots")
    for row in cursor.fetchall():
        pid = row['ID']
        x = pid % width
        y = pid // width
        terrain = row['TerrainType'].replace('TERRAIN_', '')
        is_hills = 'HILLS' in terrain
        base_terrain = terrain.replace('_HILLS', '')
        tiles[(x, y)] = Tile(x=x, y=y, terrain=base_terrain, is_hills=is_hills)

    # Features
    cursor.execute("SELECT PlotID, FeatureType FROM PlotFeatures")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].feature = row['FeatureType']
        if row['FeatureType'] == 'FEATURE_FLOODPLAINS':
            tiles[(x, y)].is_floodplains = True

    # Resources
    cursor.execute("SELECT pr.PlotID, pr.ResourceType, r.ResourceClassType FROM PlotResources pr JOIN Resources r ON pr.ResourceType = r.ResourceType")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].resource = row['ResourceType']
        tiles[(x, y)].resource_type = row['ResourceClassType'].replace('RESOURCECLASS_', '')

    # Rivers
    cursor.execute("SELECT PlotID, Edge FROM PlotRivers")
    for row in cursor.fetchall():
        pid = row['PlotID']
        x, y = pid % width, pid // width
        tiles[(x, y)].river_edges.append(row['Edge'])

    conn.close()
    return tiles, width, height
