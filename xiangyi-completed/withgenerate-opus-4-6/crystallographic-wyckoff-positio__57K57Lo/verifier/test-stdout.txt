"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.

Uses pymatgen for CIF parsing and spglib for symmetry analysis to determine
Wyckoff positions, their multiplicities, and representative atomic coordinates.
"""

from __future__ import annotations

import warnings
from collections import OrderedDict
from fractions import Fraction
from typing import Any

import numpy as np
import spglib
from pymatgen.core import Structure


def _float_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """Convert a floating-point coordinate to a fraction string.

    Args:
        value: Fractional coordinate value.
        max_denominator: Maximum denominator for the fraction.

    Returns:
        String representation of the closest fraction (e.g., "0", "1/2", "3/8").
    """
    frac = Fraction(value).limit_denominator(max_denominator)
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    Reads a CIF file, determines the space group symmetry using spglib,
    identifies distinct Wyckoff positions, calculates their multiplicities,
    and finds representative coordinates for the first atom of each position.

    Args:
        filepath: Path to the CIF file.

    Returns:
        Dictionary with two keys:
          - "wyckoff_multiplicity_dict": maps Wyckoff letter -> multiplicity (int)
          - "wyckoff_coordinates_dict": maps Wyckoff letter -> [x, y, z] as fraction strings
    """
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        struct = Structure.from_file(filepath)

    # Prepare cell tuple for spglib: (lattice, positions, atomic_numbers)
    lattice = struct.lattice.matrix
    positions = struct.frac_coords
    numbers = [s.Z for s in struct.species]
    cell = (lattice, positions, numbers)

    # Get symmetry dataset from spglib
    dataset = spglib.get_symmetry_dataset(cell)

    wyckoff_letters = dataset.wyckoffs
    equivalent_atoms = dataset.equivalent_atoms
    std_positions = dataset.std_positions

    # Identify unique Wyckoff letters in order of first appearance
    # and determine multiplicity from the first orbit of each letter
    wyckoff_multiplicity_dict = OrderedDict()
    wyckoff_coordinates_dict = OrderedDict()

    seen_letters = set()
    unique_orbits = np.unique(equivalent_atoms)

    for orbit_rep in unique_orbits:
        letter = wyckoff_letters[orbit_rep]
        if letter in seen_letters:
            continue
        seen_letters.add(letter)

        # Multiplicity = number of atoms in this orbit
        multiplicity = int(np.sum(equivalent_atoms == orbit_rep))
        wyckoff_multiplicity_dict[letter] = multiplicity

        # Representative coordinates from the first atom of this orbit
        coords = std_positions[orbit_rep]
        coord_strs = [_float_to_fraction_str(c) for c in coords]
        wyckoff_coordinates_dict[letter] = coord_strs

    return {
        "wyckoff_multiplicity_dict": dict(wyckoff_multiplicity_dict),
        "wyckoff_coordinates_dict": dict(wyckoff_coordinates_dict),
    }
