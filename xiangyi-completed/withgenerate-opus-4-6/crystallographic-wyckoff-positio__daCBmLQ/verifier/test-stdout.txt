from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and calculate approximate
    coordinates for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file.

    Returns:
        A dict with two keys:
        - "wyckoff_multiplicity_dict": mapping Wyckoff letter to its multiplicity
        - "wyckoff_coordinates_dict": mapping Wyckoff letter to a list of three
          fraction strings for the first atom's fractional coordinates
    """
    structure = Structure.from_file(filepath)
    analyzer = SpacegroupAnalyzer(structure)
    symm_struct = analyzer.get_symmetrized_structure()

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for indices, wyckoff_symbol in zip(
        symm_struct.equivalent_indices, symm_struct.wyckoff_symbols
    ):
        letter = wyckoff_symbol[-1]
        multiplicity = int(wyckoff_symbol[:-1])

        if letter in wyckoff_multiplicity_dict:
            continue

        wyckoff_multiplicity_dict[letter] = multiplicity

        first_idx = indices[0]
        frac_coords = structure[first_idx].frac_coords

        coord_strs = [
            str(Fraction(float(c)).limit_denominator(12)) for c in frac_coords
        ]
        wyckoff_coordinates_dict[letter] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
