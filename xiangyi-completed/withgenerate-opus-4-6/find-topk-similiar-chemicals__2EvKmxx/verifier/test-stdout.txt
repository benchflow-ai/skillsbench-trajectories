import time

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator

_morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def extract_molecules_from_pdf(filepath: str) -> list:
    """Extract molecule names from a PDF file, one name per line."""
    text = ""
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + "\n"
    return [line.strip() for line in text.split("\n") if line.strip()]


def name_to_smiles(name: str, max_retries: int = 5) -> str:
    """Convert a chemical name to its isomeric SMILES using PubChem with retry logic."""
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(name, "name")
            if compounds:
                return compounds[0].smiles
            return None
        except pcp.PubChemHTTPError:
            if attempt < max_retries - 1:
                time.sleep(1.0 * (attempt + 1))
            else:
                raise
    return None


def get_morgan_fp(smiles: str):
    """Generate a Morgan fingerprint from a SMILES string."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a PDF pool.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names sorted by descending Tanimoto similarity,
        with alphabetical ordering for ties.
    """
    # 1. Extract molecule names from the PDF
    pool_names = extract_molecules_from_pdf(molecule_pool_filepath)

    # 2. Get SMILES for the target molecule
    target_smiles = name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not find SMILES for target molecule: {target_molecule_name}"
        )

    # 3. Create fingerprint for target
    target_fp = get_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Invalid SMILES for target molecule: {target_smiles}"
        )

    # 4. Compute similarity for each molecule in the pool
    results = []
    for name in pool_names:
        smiles = name_to_smiles(name)
        if smiles is None:
            continue
        fp = get_morgan_fp(smiles)
        if fp is None:
            continue
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, sim))

    # 5. Sort by similarity descending, then alphabetically ascending for ties
    results.sort(key=lambda x: (-x[1], x[0]))

    # 6. Return top-k molecule names
    return [name for name, _ in results[:top_k]]


if __name__ == "__main__":
    target = "Aspirin"
    filepath = "/root/molecules.pdf"
    k = 5
    top_molecules = topk_tanimoto_similarity_molecules(target, filepath, k)
    print(f"Top {k} molecules similar to {target}:")
    for i, name in enumerate(top_molecules, 1):
        print(f"  {i}. {name}")
