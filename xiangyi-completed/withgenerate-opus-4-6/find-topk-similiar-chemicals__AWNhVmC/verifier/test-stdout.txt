import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator


def extract_molecule_names_from_pdf(filepath: str) -> list:
    """Extract molecule names from a PDF file, one name per line."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split("\n"):
                    line = line.strip()
                    if line:
                        names.append(line)
    return names


def get_smiles_from_name(chemical_name: str) -> str:
    """Convert a chemical name to its canonical SMILES using PubChem."""
    compounds = pcp.get_compounds(chemical_name, "name")
    if compounds:
        return compounds[0].smiles
    raise ValueError(f"Could not find SMILES for: {chemical_name}")


_morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, fpSize=2048, includeChirality=True
)


def get_morgan_fingerprint(smiles: str):
    """Generate a Morgan fingerprint bit vector from a SMILES string."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError(f"Invalid SMILES: {smiles}")
    return _morgan_gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> list:
    """
    Find the top-k most similar molecules to the target molecule from a PDF pool.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to the PDF containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names sorted by descending Tanimoto similarity,
        with alphabetical ordering for ties.
    """
    # Extract molecule names from the PDF
    pool_names = extract_molecule_names_from_pdf(molecule_pool_filepath)

    # Get SMILES for the target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    target_fp = get_morgan_fingerprint(target_smiles)

    # Compute similarity for each molecule in the pool
    similarities = []
    for name in pool_names:
        try:
            smiles = get_smiles_from_name(name)
            fp = get_morgan_fingerprint(smiles)
            sim = DataStructs.TanimotoSimilarity(target_fp, fp)
            similarities.append((name, sim))
        except Exception:
            continue

    # Sort by descending similarity, then alphabetical for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in similarities[:top_k]]


if __name__ == "__main__":
    results = topk_tanimoto_similarity_molecules(
        "Aspirin", "/root/molecules.pdf", 5
    )
    print("Top 5 similar molecules to Aspirin:")
    for i, name in enumerate(results, 1):
        print(f"  {i}. {name}")
