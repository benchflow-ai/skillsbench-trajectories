from typing import Any
from fractions import Fraction

from pymatgen.core.structure import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_frac_str(val: float, max_denom: int = 12) -> str:
    """Convert a floating-point coordinate to the closest rational fraction string.

    Crystallographic coordinates are periodic modulo 1, so values are wrapped
    to [0, 1) before conversion. The result is expressed as a fraction with
    denominator at most *max_denom*.
    """
    val = val % 1.0
    if abs(val) < 1e-6 or abs(val - 1.0) < 1e-6:
        return "0"
    frac = Fraction(val).limit_denominator(max_denom)
    # After rounding, the fraction may equal 0 or 1 (e.g., 0.9999 -> 1)
    if frac == 0 or frac == 1:
        return "0"
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    Parameters
    ----------
    filepath : str
        Path to a CIF file (typically generated by SHELX / Materials Project).

    Returns
    -------
    dict
        A dictionary with two keys:
        - ``wyckoff_multiplicity_dict``: maps each Wyckoff letter to its
          multiplicity (int).
        - ``wyckoff_coordinates_dict``: maps each Wyckoff letter to a list of
          three strings representing the fractional coordinates of the first
          atom in that Wyckoff orbit, expressed as simplified fractions with
          denominators <= 12.
    """
    # 1. Load the structure from the CIF file
    struct = Structure.from_file(filepath)

    # 2. Determine symmetry using spglib via pymatgen
    sga = SpacegroupAnalyzer(struct)
    dataset = sga.get_symmetry_dataset()

    # 3. Extract per-atom Wyckoff letters and fractional coordinates
    wyckoff_letters = dataset.wyckoffs
    positions = struct.frac_coords

    # 4. Collect multiplicity and first-atom coordinates for each Wyckoff letter
    wyckoff_data: dict[str, dict] = {}
    for i, letter in enumerate(wyckoff_letters):
        if letter not in wyckoff_data:
            wyckoff_data[letter] = {
                "multiplicity": 0,
                "first_coords": positions[i],
            }
        wyckoff_data[letter]["multiplicity"] += 1

    # 5. Build the output dictionaries (sorted by Wyckoff letter)
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for letter in sorted(wyckoff_data.keys()):
        data = wyckoff_data[letter]
        wyckoff_multiplicity_dict[letter] = data["multiplicity"]
        wyckoff_coordinates_dict[letter] = [
            _float_to_frac_str(c) for c in data["first_coords"]
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
