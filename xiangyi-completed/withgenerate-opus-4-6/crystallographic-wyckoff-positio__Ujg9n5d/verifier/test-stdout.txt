import re
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_frac_str(val: float, max_denom: int = 12) -> str:
    """Convert a float to the closest fraction string with denominator <= max_denom."""
    return str(Fraction(val).limit_denominator(max_denom))


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to a CIF file.

    Returns:
        Dictionary with two keys:
        - "wyckoff_multiplicity_dict": maps Wyckoff letter to its multiplicity (int)
        - "wyckoff_coordinates_dict": maps Wyckoff letter to a list of three
          fraction strings for the first atom's fractional coordinates
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    sym_struct = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for sites, wyckoff_symbol in zip(
        sym_struct.equivalent_sites, sym_struct.wyckoff_symbols
    ):
        match = re.match(r"(\d+)([a-zA-Z])", wyckoff_symbol)
        multiplicity = int(match.group(1))
        letter = match.group(2)

        if letter in wyckoff_multiplicity_dict:
            continue

        wyckoff_multiplicity_dict[letter] = multiplicity

        first_coords = sites[0].frac_coords
        wyckoff_coordinates_dict[letter] = [
            _float_to_frac_str(c) for c in first_coords
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
