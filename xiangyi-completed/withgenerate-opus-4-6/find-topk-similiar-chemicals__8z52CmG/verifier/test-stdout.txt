import fitz
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs


def _extract_molecule_names_from_pdf(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one name per non-empty line."""
    doc = fitz.open(filepath)
    names = []
    for page in doc:
        text = page.get_text()
        for line in text.strip().split("\n"):
            line = line.strip()
            if line:
                names.append(line)
    doc.close()
    return names


def _name_to_smiles(name: str) -> str | None:
    """Convert a chemical name to its canonical SMILES using PubChem."""
    try:
        compounds = pcp.get_compounds(name, "name")
        if compounds:
            return compounds[0].canonical_smiles
    except Exception:
        pass
    return None


def _compute_morgan_fp(smiles: str, radius: int = 2, use_chirality: bool = True):
    """Compute a Morgan fingerprint bit vector from a SMILES string."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return AllChem.GetMorganFingerprintAsBitVect(
        mol, radius=radius, useChirality=use_chirality
    )


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """
    Find the top-k most similar molecules from a PDF pool to a target molecule.

    Uses Morgan fingerprints (radius=2, chirality=True) with Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical
    ordering used to break ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names (strings) sorted by descending Tanimoto
        similarity to the target, with ties broken alphabetically.
    """
    # Get target SMILES and fingerprint
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not resolve target molecule: {target_molecule_name}"
        )

    target_fp = _compute_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Could not compute fingerprint for: {target_molecule_name}"
        )

    # Extract pool molecule names from PDF
    pool_names = _extract_molecule_names_from_pdf(molecule_pool_filepath)

    # Compute similarities
    results = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _compute_morgan_fp(smiles)
        if fp is None:
            continue
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # Sort: descending by similarity, then ascending alphabetically by name
    results.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in results[:top_k]]
