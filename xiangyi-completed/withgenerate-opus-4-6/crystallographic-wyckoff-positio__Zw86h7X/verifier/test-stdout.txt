from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _float_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """Convert a floating-point fractional coordinate to a rational string.

    The value is first wrapped into [0, 1) and then approximated by the
    closest fraction whose denominator does not exceed *max_denominator*.
    """
    value = float(value) % 1.0
    frac = Fraction(value).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyse Wyckoff position multiplicities and first-atom coordinates.

    Parameters
    ----------
    filepath : str
        Path to a CIF file (may be stored in P1 with all atoms expanded).

    Returns
    -------
    dict
        ``wyckoff_multiplicity_dict`` maps each Wyckoff letter to its
        multiplicity.  ``wyckoff_coordinates_dict`` maps each Wyckoff
        letter to the fractional coordinates of the first representative
        atom, expressed as simplified fraction strings.
    """
    structure = Structure.from_file(filepath)
    analyzer = SpacegroupAnalyzer(structure)
    sym_structure = analyzer.get_symmetrized_structure()

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for sites, wsym in zip(sym_structure.equivalent_sites, sym_structure.wyckoff_symbols):
        # wsym is e.g. '4a', '8c', '16h'
        letter = wsym.lstrip("0123456789")
        multiplicity = int(wsym[: len(wsym) - len(letter)])

        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = multiplicity
            coords = sites[0].frac_coords
            wyckoff_coordinates_dict[letter] = [
                _float_to_fraction_str(v) for v in coords
            ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
