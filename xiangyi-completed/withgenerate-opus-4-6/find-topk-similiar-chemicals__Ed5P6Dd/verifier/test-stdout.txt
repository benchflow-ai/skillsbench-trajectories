import PyPDF2
import pubchempy as pcp
from rdkit import Chem, DataStructs
from rdkit.Chem import rdFingerprintGenerator


_morgan_gen = rdFingerprintGenerator.GetMorganGenerator(
    radius=2, includeChirality=True
)


def _extract_chemical_names_from_pdf(filepath):
    """Extract chemical names from a PDF file, one name per line."""
    names = []
    with open(filepath, 'rb') as f:
        reader = PyPDF2.PdfReader(f)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split('\n'):
                    line = line.strip()
                    if line:
                        names.append(line)
    return names


_smiles_cache = {}


def _get_smiles(name):
    """Convert a chemical name to its SMILES string using PubChem."""
    if name in _smiles_cache:
        return _smiles_cache[name]
    try:
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            smiles = compounds[0].isomeric_smiles
            _smiles_cache[name] = smiles
            return smiles
    except Exception:
        pass
    _smiles_cache[name] = None
    return None


def _compute_morgan_fp(smiles):
    """Compute a Morgan fingerprint (radius=2, chirality=True) from a SMILES string."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top-k most similar chemicals from a PDF pool to the target molecule.

    Args:
        target_molecule_name: Name of the target chemical compound.
        molecule_pool_filepath: Path to a PDF file containing chemical names (one per line).
        top_k: Number of top similar molecules to return.

    Returns:
        A list of chemical names (strings) from the pool, sorted by descending
        Tanimoto similarity (Morgan fingerprints, radius=2, chirality=True).
        Ties are broken alphabetically (ascending).
    """
    # Get SMILES for the target molecule
    target_smiles = _get_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not resolve target molecule: {target_molecule_name}")

    target_fp = _compute_morgan_fp(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_molecule_name}")

    # Extract chemical names from the PDF
    pool_names = _extract_chemical_names_from_pdf(molecule_pool_filepath)

    # Compute similarity for each molecule in the pool
    results = []
    for name in pool_names:
        smiles = _get_smiles(name)
        if smiles is None:
            continue
        fp = _compute_morgan_fp(smiles)
        if fp is None:
            continue
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # Sort by descending similarity, then alphabetically for ties
    results.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k names
    return [name for name, _ in results[:top_k]]


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5
    top_molecules = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    print(f"Top {k} molecules similar to {target}:")
    for i, name in enumerate(top_molecules, 1):
        print(f"  {i}. {name}")
