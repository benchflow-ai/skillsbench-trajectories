import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def _extract_molecule_names(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file (one name per line)."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split('\n'):
                    name = line.strip()
                    if name:
                        names.append(name)
    return names


def _name_to_smiles(name: str) -> str | None:
    """Convert a chemical name to a SMILES string using PubChem."""
    try:
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def _get_morgan_fingerprint(smiles: str):
    """Generate a Morgan fingerprint from a SMILES string."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    gen = rdFingerprintGenerator.GetMorganGenerator(
        radius=2, fpSize=2048, includeChirality=True
    )
    return gen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """Find the top-k most similar molecules to the target from a PDF pool.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of molecule names sorted by descending Tanimoto similarity,
        with alphabetical ordering for ties.
    """
    # Extract pool molecule names from PDF
    pool_names = _extract_molecule_names(molecule_pool_filepath)

    # Convert target molecule name to SMILES and fingerprint
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        return []
    target_fp = _get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        return []

    # Compute similarities for each pool molecule
    results = []
    for name in pool_names:
        # Skip if the pool molecule is the same as the target
        if name.lower() == target_molecule_name.lower():
            continue

        smiles = _name_to_smiles(name)
        if smiles is None:
            continue

        fp = _get_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))

    # Sort by similarity descending, then alphabetically ascending for ties
    results.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k names
    return [name for name, _ in results[:top_k]]


if __name__ == '__main__':
    import sys

    target = sys.argv[1] if len(sys.argv) > 1 else 'Aspirin'
    pdf_path = sys.argv[2] if len(sys.argv) > 2 else '/root/molecules.pdf'
    k = int(sys.argv[3]) if len(sys.argv) > 3 else 5

    top_molecules = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    print(f"Top {k} molecules similar to '{target}':")
    for i, name in enumerate(top_molecules, 1):
        print(f"  {i}. {name}")
