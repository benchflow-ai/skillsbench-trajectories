import argparse
import json
import re
from functools import lru_cache
from typing import Iterable, List, Tuple
from urllib.parse import quote
from urllib.request import Request, urlopen
from urllib.error import HTTPError, URLError


class MoleculeLookupError(RuntimeError):
    pass


def _extract_lines_from_pdf(pdf_path: str) -> List[str]:
    lines: List[str] = []
    try:
        import pdfplumber  # type: ignore

        with pdfplumber.open(pdf_path) as pdf:
            for page in pdf.pages:
                text = page.extract_text() or ""
                if text:
                    lines.extend(text.splitlines())
    except Exception:
        try:
            from pypdf import PdfReader  # type: ignore

            reader = PdfReader(pdf_path)
            for page in reader.pages:
                text = page.extract_text() or ""
                if text:
                    lines.extend(text.splitlines())
        except Exception as exc:
            raise MoleculeLookupError(
                "Failed to read PDF. Install pdfplumber or pypdf."
            ) from exc

    return [line.strip() for line in lines if line and line.strip()]


def _split_columns(line: str) -> List[str]:
    # Split columns by tabs or 2+ spaces to keep single spaces in names
    parts = re.split(r"\t+|\s{2,}", line)
    return [p.strip() for p in parts if p and p.strip()]


def _clean_name(name: str) -> str:
    # Remove list numbering and simple bullets
    name = re.sub(r"^\s*(?:\d+[\.)]|[-*])\s*", "", name)
    return name.strip()


def _dedupe_preserve_order(items: Iterable[str]) -> List[str]:
    seen = set()
    out: List[str] = []
    for item in items:
        key = item.casefold()
        if key in seen:
            continue
        seen.add(key)
        out.append(item)
    return out


def _molecule_names_from_pdf(pdf_path: str) -> List[str]:
    lines = _extract_lines_from_pdf(pdf_path)
    names: List[str] = []
    for line in lines:
        for part in _split_columns(line):
            cleaned = _clean_name(part)
            if not cleaned:
                continue
            # Skip lines without letters (likely page numbers)
            if not re.search(r"[A-Za-z]", cleaned):
                continue
            names.append(cleaned)
    return _dedupe_preserve_order(names)


@lru_cache(maxsize=4096)
def _pubchem_name_to_smiles(name: str, timeout: int = 20) -> str | None:
    url = (
        "https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/"
        f"{quote(name)}/property/IsomericSMILES,CanonicalSMILES,SMILES/JSON"
    )
    headers = {"User-Agent": "molecule-similarity-script/1.0"}
    for _ in range(3):
        try:
            req = Request(url, headers=headers)
            with urlopen(req, timeout=timeout) as resp:
                status = getattr(resp, "status", 200)
                if status != 200:
                    continue
                data = json.loads(resp.read().decode("utf-8"))
                props = data.get("PropertyTable", {}).get("Properties", [])
                if props:
                    return (
                        props[0].get("IsomericSMILES")
                        or props[0].get("CanonicalSMILES")
                        or props[0].get("SMILES")
                    )
                return None
        except HTTPError as exc:
            if exc.code == 404:
                return None
        except URLError:
            continue
    return None


def _require_rdkit():
    try:
        from rdkit import Chem, DataStructs  # type: ignore
        from rdkit.Chem import AllChem  # type: ignore
    except Exception as exc:  # pragma: no cover - environment-specific
        raise MoleculeLookupError(
            "RDKit is required. Install rdkit-pypi or conda-forge rdkit."
        ) from exc
    return Chem, DataStructs, AllChem


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> List[Tuple[str, float]]:
    """
    Return the top-k most similar molecules from the pool to the target.

    Similarity: Morgan fingerprints (radius=2, useChirality=True), Tanimoto.
    Ordering: descending similarity, then alphabetical by name for ties.
    """

    if top_k <= 0:
        return []

    Chem, DataStructs, AllChem = _require_rdkit()

    target_smiles = _pubchem_name_to_smiles(target_molecule_name)
    if not target_smiles:
        raise MoleculeLookupError(
            f"Could not resolve target molecule name: {target_molecule_name}"
        )
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise MoleculeLookupError(
            f"Failed to parse SMILES for target: {target_molecule_name}"
        )
    target_fp = AllChem.GetMorganFingerprintAsBitVect(
        target_mol, radius=2, nBits=2048, useChirality=True
    )

    pool_names = _molecule_names_from_pdf(molecule_pool_filepath)

    results: List[Tuple[str, float]] = []
    for name in pool_names:
        smiles = _pubchem_name_to_smiles(name)
        if not smiles:
            continue
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            continue
        fp = AllChem.GetMorganFingerprintAsBitVect(
            mol, radius=2, nBits=2048, useChirality=True
        )
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, float(sim)))

    results.sort(key=lambda x: (-x[1], x[0].casefold()))
    return results[: min(top_k, len(results))]


def _main() -> int:
    parser = argparse.ArgumentParser(
        description="Top-k similar molecules in a PDF using Morgan/Tanimoto."
    )
    parser.add_argument("target", help="Target molecule name")
    parser.add_argument(
        "--pdf",
        default="/root/molecules.pdf",
        help="Path to molecules PDF (default: /root/molecules.pdf)",
    )
    parser.add_argument("--top-k", type=int, default=5, help="Number of results")
    args = parser.parse_args()

    results = topk_tanimoto_similarity_molecules(args.target, args.pdf, args.top_k)
    print(json.dumps(results, indent=2))
    return 0


if __name__ == "__main__":
    raise SystemExit(_main())
