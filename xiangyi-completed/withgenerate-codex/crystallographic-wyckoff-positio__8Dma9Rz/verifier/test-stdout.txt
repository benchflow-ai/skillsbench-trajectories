from __future__ import annotations

from typing import Any, Dict, Tuple
from fractions import Fraction
import math


def _normalize_fractional(x: float, tol: float = 1e-8) -> float:
    """
    Map a fractional coordinate into [0, 1) with a small tolerance.
    """
    if not math.isfinite(x):
        return x
    x = x - math.floor(x)
    # Snap values extremely close to 1 to 0 for periodicity
    if abs(x - 1.0) < tol:
        return 0.0
    if abs(x) < tol:
        return 0.0
    return x


def _to_rational_str(x: float, max_den: int = 12) -> str:
    """
    Convert a float to a simplified rational string with bounded denominator.
    """
    x = _normalize_fractional(x)
    fr = Fraction(x).limit_denominator(max_den)
    if fr.denominator == 1:
        return str(fr.numerator)
    return f"{fr.numerator}/{fr.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> Dict[str, Dict] | Dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and first-atom coordinates for each Wyckoff letter.

    Returns:
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8, ...},
            "wyckoff_coordinates_dict": {"a": ["0", "1/2", "1/2"], ...}
        }
    """
    try:
        from pymatgen.core import Structure
        from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
    except Exception as exc:  # pragma: no cover - environment-dependent
        raise RuntimeError(
            "pymatgen is required. Install with `pip install pymatgen`."
        ) from exc

    structure = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(structure, symprec=1e-3)
    dataset = sga.get_symmetry_dataset()

    wyckoffs = list(dataset["wyckoffs"])

    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list[str]] = {}

    for wyckoff_letter, site in zip(wyckoffs, structure.sites):
        wyckoff_multiplicity_dict[wyckoff_letter] = (
            wyckoff_multiplicity_dict.get(wyckoff_letter, 0) + 1
        )

        if wyckoff_letter not in wyckoff_coordinates_dict:
            frac = site.frac_coords
            wyckoff_coordinates_dict[wyckoff_letter] = [
                _to_rational_str(float(frac[0])),
                _to_rational_str(float(frac[1])),
                _to_rational_str(float(frac[2])),
            ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


__all__ = ["analyze_wyckoff_position_multiplicities_and_coordinates"]
