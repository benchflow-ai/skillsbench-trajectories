from __future__ import annotations

from fractions import Fraction
import re
from typing import Any, Dict

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _wyckoff_letter(symbol: str) -> str:
    match = re.search(r"[A-Za-z]+$", symbol)
    if not match:
        return symbol
    return match.group(0)


def _rational_str(x: float, max_denominator: int = 12) -> str:
    # Wrap into [0, 1) for fractional coordinates.
    x = float(x) % 1.0
    if abs(x) < 1e-8 or abs(x - 1.0) < 1e-8:
        x = 0.0
    frac = Fraction(x).limit_denominator(max_denominator)
    if frac.denominator == 1:
        return str(frac.numerator)
    return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> Dict[str, Dict[str, Any]]:
    structure = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(structure, symprec=1e-3)
    symm = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, list[str]] = {}

    for group_idx, wyckoff_symbol in enumerate(symm.wyckoff_symbols):
        letter = _wyckoff_letter(wyckoff_symbol)
        indices = symm.equivalent_indices[group_idx]
        multiplicity = len(indices)

        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = multiplicity
        elif wyckoff_multiplicity_dict[letter] != multiplicity:
            # Keep the largest multiplicity seen for the letter.
            wyckoff_multiplicity_dict[letter] = max(
                wyckoff_multiplicity_dict[letter], multiplicity
            )

        if letter not in wyckoff_coordinates_dict:
            first_index = indices[0]
            frac_coords = structure[first_index].frac_coords
            wyckoff_coordinates_dict[letter] = [
                _rational_str(coord) for coord in frac_coords
            ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
