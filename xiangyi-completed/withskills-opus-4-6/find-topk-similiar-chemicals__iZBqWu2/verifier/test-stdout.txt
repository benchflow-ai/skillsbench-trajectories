import time

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator, DataStructs


def _extract_molecule_names(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one name per line."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split("\n"):
                    name = line.strip()
                    if name:
                        names.append(name)
    return names


def _name_to_smiles(name: str, max_retries: int = 5) -> str | None:
    """Convert a chemical name to a SMILES string using PubChem with retry logic."""
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(name, "name")
            if compounds:
                return compounds[0].isomeric_smiles
            return None
        except pcp.PubChemHTTPError:
            if attempt < max_retries - 1:
                time.sleep(1.0 * (attempt + 1))
            else:
                raise


def _smiles_to_fingerprint(smiles: str, generator):
    """Convert a SMILES string to a Morgan fingerprint."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return generator.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int,
) -> list:
    """Find the top-k most similar molecules to a target molecule.

    Args:
        target_molecule_name: Name of the target molecule.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of molecule names sorted by descending Tanimoto similarity,
        with alphabetical ordering for ties.
    """
    fp_generator = rdFingerprintGenerator.GetMorganGenerator(
        radius=2, includeChirality=True
    )

    # Convert target molecule name to fingerprint
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not find SMILES for target molecule: {target_molecule_name}"
        )
    target_fp = _smiles_to_fingerprint(target_smiles, fp_generator)
    if target_fp is None:
        raise ValueError(
            f"Could not parse SMILES for target molecule: {target_molecule_name}"
        )

    # Extract pool molecule names from the PDF
    pool_names = _extract_molecule_names(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _smiles_to_fingerprint(smiles, fp_generator)
        if fp is None:
            continue
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((name, sim))

    # Sort by descending similarity, then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k names
    return [name for name, _ in similarities[:top_k]]
