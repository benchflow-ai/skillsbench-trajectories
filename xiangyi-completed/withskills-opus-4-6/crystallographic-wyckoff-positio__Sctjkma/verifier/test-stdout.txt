from __future__ import annotations

from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _coord_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """Convert a float coordinate to a rational string representation.

    Approximates the value as a fraction with denominator <= max_denominator,
    then returns a string like "0", "1/2", "3/8", etc.
    """
    # Wrap into [0, 1) range for fractional coordinates
    value = value % 1.0
    # Handle near-zero and near-one after wrapping
    frac = Fraction(value).limit_denominator(max_denominator)
    # Ensure result is in [0, 1) by taking modulo 1
    if frac >= 1:
        frac = frac - 1
    if frac < 0:
        frac = frac + 1
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and coordinates from a CIF file.

    Reads the CIF file, determines the space group symmetry using spglib,
    identifies distinct Wyckoff positions (crystallographic orbits), and
    returns their multiplicities and the fractional coordinates of the first
    atom in each orbit approximated as fractions with denominators <= 12.

    Parameters
    ----------
    filepath : str
        Path to the CIF file.

    Returns
    -------
    dict
        Dictionary with keys:
        - "wyckoff_multiplicity_dict": mapping from Wyckoff label to multiplicity
        - "wyckoff_coordinates_dict": mapping from Wyckoff label to [x, y, z]
          as fraction strings
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct)
    sym_dataset = sga.get_symmetry_dataset()

    wyckoff_letters = sym_dataset.wyckoffs
    crystallographic_orbits = sym_dataset.crystallographic_orbits
    std_positions = sym_dataset.std_positions

    # Group atoms by crystallographic orbit (each orbit is a distinct
    # Wyckoff position occupation in the structure)
    orbit_info: dict[int, dict] = {}
    for idx, (letter, orbit_id) in enumerate(
        zip(wyckoff_letters, crystallographic_orbits)
    ):
        if orbit_id not in orbit_info:
            orbit_info[orbit_id] = {
                "letter": letter,
                "count": 0,
                "first_idx": idx,
            }
        orbit_info[orbit_id]["count"] += 1

    # Count how many distinct orbits share each Wyckoff letter
    letter_orbit_count: dict[str, int] = {}
    for info in orbit_info.values():
        letter_orbit_count[info["letter"]] = (
            letter_orbit_count.get(info["letter"], 0) + 1
        )

    # Build labels: if a letter is unique, use it directly;
    # if multiple orbits share the same letter, append a numeric suffix.
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    # Track how many times we've seen each letter so far for numbering
    letter_counter: dict[str, int] = {}

    for orbit_id in sorted(orbit_info.keys()):
        info = orbit_info[orbit_id]
        letter = info["letter"]

        if letter_orbit_count[letter] == 1:
            label = letter
        else:
            letter_counter[letter] = letter_counter.get(letter, 0) + 1
            label = f"{letter}{letter_counter[letter]}"

        wyckoff_multiplicity_dict[label] = info["count"]

        coords = std_positions[info["first_idx"]]
        wyckoff_coordinates_dict[label] = [
            _coord_to_fraction_str(c) for c in coords
        ]

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
