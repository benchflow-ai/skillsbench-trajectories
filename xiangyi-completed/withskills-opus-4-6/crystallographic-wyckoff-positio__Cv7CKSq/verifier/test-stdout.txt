from collections import Counter
from fractions import Fraction
from typing import Any

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _coord_to_fraction_str(value: float) -> str:
    """Convert a float coordinate to a fraction string with denominator <= 12."""
    frac = Fraction(value).limit_denominator(12)
    return str(frac)


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    sym_struct = sga.get_symmetrized_structure()

    wyckoff_symbols = sym_struct.wyckoff_symbols
    equivalent_sites = sym_struct.equivalent_sites

    # Extract letters from wyckoff symbols (e.g. "4a" -> "a", "8c" -> "c")
    letters = [ws[-1] for ws in wyckoff_symbols]

    # Count occurrences of each letter to decide if numbering is needed
    letter_counts = Counter(letters)
    # Track how many times we've seen each letter so far for numbering
    letter_seen = {}

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for ws, sites in zip(wyckoff_symbols, equivalent_sites):
        letter = ws[-1]
        multiplicity = int(ws[:-1])

        # Build the key: use just the letter if unique, else append a number
        if letter_counts[letter] == 1:
            key = letter
        else:
            letter_seen[letter] = letter_seen.get(letter, 0) + 1
            key = f"{letter}{letter_seen[letter]}"

        representative = sites[0]
        coords = representative.frac_coords

        coord_strs = [_coord_to_fraction_str(c) for c in coords]

        wyckoff_multiplicity_dict[key] = multiplicity
        wyckoff_coordinates_dict[key] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


if __name__ == "__main__":
