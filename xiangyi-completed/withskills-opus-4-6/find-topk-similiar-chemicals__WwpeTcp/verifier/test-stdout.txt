import time

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs, rdFingerprintGenerator


def _extract_molecule_names(filepath: str) -> list[str]:
    """Extract molecule names from a PDF file, one name per line."""
    names = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().splitlines():
                    name = line.strip()
                    if name:
                        names.append(name)
    return names


def _name_to_smiles(name: str, max_retries: int = 5) -> str | None:
    """Convert a chemical name to a SMILES string using PubChem with retry logic."""
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(name, "name")
            if compounds:
                return compounds[0].isomeric_smiles
            return None
        except pcp.PubChemHTTPError:
            if attempt < max_retries - 1:
                wait = 2 ** attempt
                time.sleep(wait)
            else:
                raise


def _smiles_to_fingerprint(smiles: str):
    """Convert a SMILES string to a Morgan fingerprint (radius=2, chirality=True)."""
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    generator = rdFingerprintGenerator.GetMorganGenerator(
        radius=2, fpSize=2048, includeChirality=True
    )
    return generator.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str, molecule_pool_filepath: str, top_k: int
) -> list:
    """Find the top-k most similar molecules from a PDF pool to a target molecule.

    Uses Morgan fingerprints (radius=2, chirality=True) with Tanimoto similarity.
    Results are sorted by descending similarity, with alphabetical ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        A list of the top-k most similar molecule names (strings).
    """
    # Get SMILES and fingerprint for the target molecule
    target_smiles = _name_to_smiles(target_molecule_name)
    if target_smiles is None:
        raise ValueError(
            f"Could not find SMILES for target molecule: {target_molecule_name}"
        )
    target_fp = _smiles_to_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(
            f"Could not generate fingerprint for target molecule: {target_molecule_name}"
        )

    # Extract molecule names from the PDF pool
    pool_names = _extract_molecule_names(molecule_pool_filepath)

    # Compute similarity for each molecule in the pool
    similarities = []
    for name in pool_names:
        smiles = _name_to_smiles(name)
        if smiles is None:
            continue
        fp = _smiles_to_fingerprint(smiles)
        if fp is None:
            continue
        sim = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((name, sim))

    # Sort by descending similarity, then alphabetically by name for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in similarities[:top_k]]

    for i, name in enumerate(results, 1):
        print(f"  {i}. {name}")
