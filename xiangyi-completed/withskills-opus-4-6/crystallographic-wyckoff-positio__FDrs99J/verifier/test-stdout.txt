from typing import Any

import numpy as np
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """Analyze Wyckoff position multiplicities and calculate approximate
    coordinates for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to a CIF file.

    Returns:
        A dict with two keys:
          - "wyckoff_multiplicity_dict": maps each Wyckoff letter to its
            multiplicity (number of equivalent positions generated by symmetry).
          - "wyckoff_coordinates_dict": maps each Wyckoff letter to a list of
            three strings representing the fractional coordinates of the first
            atom, expressed as fractions with denominators <= 12.
    """
    struct = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    dataset = sga.get_symmetry_dataset()

    wyckoff_letters = dataset.wyckoffs
    equivalent_atoms = dataset.equivalent_atoms
    std_positions = dataset.std_positions

    # Collect unique orbit representatives in order of first appearance
    unique_representatives = []
    seen_reps = set()
    for eq in equivalent_atoms:
        if eq not in seen_reps:
            seen_reps.add(eq)
            unique_representatives.append(eq)

    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for rep_idx in unique_representatives:
        wl = wyckoff_letters[rep_idx]

        # Only record the first orbit encountered for each Wyckoff letter
        if wl in wyckoff_multiplicity_dict:
            continue

        # Multiplicity = number of atoms in this symmetry orbit
        orbit_size = int(np.sum(equivalent_atoms == rep_idx))
        wyckoff_multiplicity_dict[wl] = orbit_size

        # Convert standardised fractional coordinates to rational strings
        pos = std_positions[rep_idx]
        coords: list[str] = []
        for v in pos:
            v_mod = v % 1.0
            # Snap values very close to 0 or 1 to exactly 0
            if abs(v_mod) < 1e-6 or abs(v_mod - 1.0) < 1e-6:
                v_mod = 0.0
            frac = Fraction(v_mod).limit_denominator(12)
            # Ensure result is in [0, 1)
            if frac >= 1:
                frac = Fraction(0)
            coords.append(str(frac))
        wyckoff_coordinates_dict[wl] = coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
