"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        chemical_name: The name of the chemical compound

    Returns:
        The SMILES string if found, None otherwise
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


# Create Morgan fingerprint generator (radius=2, include chirality)
_morgan_generator = rdFingerprintGenerator.GetMorganGenerator(
    radius=2,
    fpSize=2048,
    includeChirality=True
)


def get_morgan_fingerprint(smiles: str):
    """
    Generate Morgan fingerprint for a molecule given its SMILES.

    Args:
        smiles: SMILES representation of the molecule

    Returns:
        Morgan fingerprint object or None if molecule is invalid
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    return _morgan_generator.GetFingerprint(mol)


def calculate_tanimoto_similarity(fp1, fp2) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0.0 to 1.0)
    """
    if fp1 is None or fp2 is None:
        return 0.0
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of molecule names
    """
    molecules = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                for line in text.strip().split('\n'):
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a PDF file.

    Uses Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Results are sorted by descending similarity, with alphabetical ordering for ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) for top-k similar molecules
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Get fingerprint for target molecule
    target_fp = get_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities for all molecules in pool
    similarities = []
    for mol_name in pool_molecules:
        # Get SMILES for pool molecule
        mol_smiles = get_smiles_from_name(mol_name)
        if mol_smiles is None:
            continue

        # Get fingerprint for pool molecule
        mol_fp = get_morgan_fingerprint(mol_smiles)
        if mol_fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = calculate_tanimoto_similarity(target_fp, mol_fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) >= 4:
        target = sys.argv[1]
        pdf_path = sys.argv[2]
        k = int(sys.argv[3])
    else:
        target = "Aspirin"
        pdf_path = "/root/molecules.pdf"
        k = 5

    print(f"Finding top {k} molecules similar to '{target}'...")
    print()

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"Top {len(results)} similar molecules:")
    for i, (name, score) in enumerate(results, 1):
        print(f"  {i}. {name}: {score:.4f}")
