"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import PyPDF2
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit.DataStructs import TanimotoSimilarity
import pubchempy as pcp
from typing import List, Tuple
import logging

# Configure logging to suppress verbose output
logging.basicConfig(level=logging.WARNING)


def extract_chemicals_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file containing chemical names

    Returns:
        List of chemical names
    """
    chemicals = []

    with open(pdf_filepath, 'rb') as f:
        pdf_reader = PyPDF2.PdfReader(f)
        for page in pdf_reader.pages:
            text = page.extract_text()
            # Split by newlines and clean up
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            chemicals.extend(lines)

    return chemicals


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem API.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string for the chemical, or None if not found
    """
    try:
        compound = pcp.get_compounds(chemical_name, 'name', max_results=1)
        if compound:
            # Try to use smiles first (newer API), fall back to isomeric_smiles
            return getattr(compound[0], 'smiles', None) or compound[0].isomeric_smiles
    except Exception as e:
        logging.warning(f"Error fetching SMILES for {chemical_name}: {e}")

    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string of the molecule
        radius: Radius for Morgan fingerprint
        include_chirality: Whether to include chirality information

    Returns:
        Morgan fingerprint object
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    fp = AllChem.GetMorganFingerprintAsBitVect(
        mol,
        radius=radius,
        useChirality=include_chirality
    )
    return fp


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k molecules most similar to a target molecule using Tanimoto similarity
    based on Morgan fingerprints.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of top k most similar molecule names, sorted by similarity (descending)
        and alphabetically when ties occur
    """

    # Extract all chemicals from the PDF
    chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Compute similarities
    similarities = []

    for chemical in chemicals:
        # Skip the target molecule itself if it's in the pool
        if chemical.lower() == target_molecule_name.lower():
            continue

        # Get SMILES for this chemical
        smiles = get_smiles_from_pubchem(chemical)
        if smiles is None:
            continue

        # Compute fingerprint
        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Compute Tanimoto similarity
        similarity = TanimotoSimilarity(target_fp, fp)
        similarities.append((chemical, similarity))

    # Sort by similarity (descending), then alphabetically by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecule names
    return [name for name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    try:
        result = topk_tanimoto_similarity_molecules(
            target_molecule_name="Aspirin",
            molecule_pool_filepath="/root/molecules.pdf",
            top_k=5
        )
        print(f"Top 5 molecules similar to Aspirin:")
        for i, mol in enumerate(result, 1):
            print(f"{i}. {mol}")
    except Exception as e:
        print(f"Error: {e}")
