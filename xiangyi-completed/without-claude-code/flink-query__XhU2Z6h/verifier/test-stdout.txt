        KeyedStream<TaskEvent, Long> keyedSubmits = submitEvents.keyBy(event -> event.jobId);

        // Apply session windows and count tasks in each session
        DataStream<Tuple2<Long, Long>> sessionCounts = keyedSubmits
                .window(EventTimeSessionWindows.withGap(Time.milliseconds(600000)))
                .aggregate(new CountAggregator(), new SessionWindowProcessor());

        DataStream<Tuple2<Long, Long>> maxPerJob = sessionCounts
