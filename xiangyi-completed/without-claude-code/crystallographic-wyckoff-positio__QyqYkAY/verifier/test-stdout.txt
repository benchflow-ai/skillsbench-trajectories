from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Wyckoff letter -> multiplicity mapping
        - wyckoff_coordinates_dict: Wyckoff letter -> first atom coordinates
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Use SpacegroupAnalyzer to determine symmetry
    sga = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure
    symmetrized_structure = sga.get_symmetrized_structure()

    # Extract Wyckoff information
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Iterate over equivalent site groups and their Wyckoff symbols
    for i, equiv_sites in enumerate(symmetrized_structure.equivalent_sites):
        # Get Wyckoff symbol for this group
        wyckoff_symbol = symmetrized_structure.wyckoff_symbols[i]

        # Extract just the letter from Wyckoff symbol (e.g., "4a" -> "a")
        wyckoff_letter = wyckoff_symbol[-1]

        # Multiplicity is the number of equivalent sites
        multiplicity = len(equiv_sites)

        # Get the first site in this equivalent group
        first_site = equiv_sites[0]

        # Get fractional coordinates of the first atom
        frac_coords = first_site.frac_coords

        # Convert to simplified fraction strings
        coord_strings = [
            _float_to_fraction_string(coord, max_denominator=12)
            for coord in frac_coords
        ]

        # Store in dictionaries (use letter as key)
        if wyckoff_letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a simplified fraction string.

    Args:
        value: Float value to convert
        max_denominator: Maximum allowed denominator

    Returns:
        String representation of the fraction (e.g., "1/2", "0", "3/8")
    """
    # Use Fraction to convert float to rational with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle whole numbers
    if frac.denominator == 1:
        return str(frac.numerator)

    # Return as fraction string
    return f"{frac.numerator}/{frac.denominator}"
