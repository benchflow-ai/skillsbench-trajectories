"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import time
import warnings
import PyPDF2
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
from typing import List, Tuple

# Suppress RDKit and PubChemPy deprecation warnings
warnings.filterwarnings('ignore', category=DeprecationWarning)
try:
    from pubchempy import PubChemPyDeprecationWarning
    warnings.filterwarnings('ignore', category=PubChemPyDeprecationWarning)
except ImportError:
    pass


def get_smiles_from_name(chemical_name: str, max_retries: int = 3) -> str:
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        chemical_name: Name of the chemical
        max_retries: Maximum number of retry attempts (default: 3)

    Returns:
        SMILES string representation of the molecule

    Raises:
        ValueError: If the chemical name cannot be found in PubChem
    """
    for attempt in range(max_retries):
        try:
            compounds = pcp.get_compounds(chemical_name, 'name')
            if not compounds:
                raise ValueError(f"Chemical '{chemical_name}' not found in PubChem")
            # Use isomeric_smiles for better chirality support, fallback to canonical
            try:
                return compounds[0].isomeric_smiles or compounds[0].canonical_smiles
            except AttributeError:
                return compounds[0].canonical_smiles
        except ValueError:
            # Re-raise ValueError (not found) without retry
            raise
        except Exception as e:
            if attempt < max_retries - 1:
                # Exponential backoff for rate limiting
                wait_time = (2 ** attempt) * 1.0
                time.sleep(wait_time)
                continue
            raise ValueError(f"Error retrieving SMILES for '{chemical_name}': {str(e)}")


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file containing chemical names

    Returns:
        List of chemical names (stripped of whitespace)
    """
    molecules = []
    with open(pdf_filepath, 'rb') as file:
        pdf_reader = PyPDF2.PdfReader(file)
        for page in pdf_reader.pages:
            text = page.extract_text()
            lines = text.strip().split('\n')
            for line in lines:
                line = line.strip()
                if line:
                    molecules.append(line)
    return molecules


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES representation of the molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality information (default: True)

    Returns:
        Morgan fingerprint object

    Raises:
        ValueError: If SMILES cannot be parsed
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError(f"Invalid SMILES: {smiles}")
    return AllChem.GetMorganFingerprint(mol, radius, useChirality=use_chirality)


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0.0 to 1.0)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top k most similar molecules to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity (descending)
        and alphabetically for ties
    """
    # Extract molecules from PDF
    molecule_pool = extract_molecules_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    target_fp = compute_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)

    # Compute similarities for all molecules in the pool
    similarities = []

    for molecule_name in molecule_pool:
        # Skip if it's the same as target molecule
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        try:
            mol_smiles = get_smiles_from_name(molecule_name)
            mol_fp = compute_morgan_fingerprint(mol_smiles, radius=2, use_chirality=True)
            similarity = compute_tanimoto_similarity(target_fp, mol_fp)
            similarities.append((molecule_name, similarity))
            # Small delay to avoid overwhelming PubChem API
            time.sleep(0.2)
        except Exception as e:
            # Skip molecules that cannot be processed
            print(f"Warning: Skipping '{molecule_name}': {str(e)}")
            time.sleep(0.5)  # Longer delay on errors (rate limiting)
            continue

    # Sort by similarity (descending), then alphabetically by name for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5

    print(f"Finding top {k} molecules similar to '{target}'...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for i, (name, similarity) in enumerate(results, 1):
        print(f"{i}. {name}: {similarity:.4f}")
