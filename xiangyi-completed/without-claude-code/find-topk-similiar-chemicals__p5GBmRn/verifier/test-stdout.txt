"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import subprocess
import sys


def extract_molecules_from_pdf(pdf_path: str) -> list[str]:
    """
    Extract molecule names from a PDF file using pdftotext.

    Args:
        pdf_path: Path to the PDF file containing molecule names

    Returns:
        List of molecule names extracted from the PDF
    """
    result = subprocess.run(
        ["pdftotext", pdf_path, "-"],
        capture_output=True,
        text=True,
        check=True
    )

    molecules = []
    for line in result.stdout.strip().split("\n"):
        line = line.strip()
        if line:
            molecules.append(line)

    return molecules


def get_smiles_from_pubchem(molecule_name: str) -> str | None:
    """
    Get SMILES representation of a molecule from PubChem.

    Args:
        molecule_name: Common name of the molecule

    Returns:
        SMILES string if found, None otherwise
    """
    import pubchempy as pcp

    try:
        compounds = pcp.get_compounds(molecule_name, "name")
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass

    return None


def compute_tanimoto_similarity(smiles1: str, smiles2: str, radius: int = 2, use_chirality: bool = True) -> float:
    """
    Compute Tanimoto similarity between two molecules using Morgan fingerprints.

    Args:
        smiles1: SMILES string of first molecule
        smiles2: SMILES string of second molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality in fingerprint (default: True)

    Returns:
        Tanimoto similarity score between 0 and 1
    """
    from rdkit import Chem
    from rdkit.Chem import rdFingerprintGenerator
    from rdkit import DataStructs

    mol1 = Chem.MolFromSmiles(smiles1)
    mol2 = Chem.MolFromSmiles(smiles2)

    if mol1 is None or mol2 is None:
        return 0.0

    generator = rdFingerprintGenerator.GetMorganGenerator(radius=radius, includeChirality=use_chirality)
    fp1 = generator.GetFingerprint(mol1)
    fp2 = generator.GetFingerprint(mol2)

    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to the target molecule from a PDF file.

    Uses Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical
    ordering when ties happen.

    Args:
        target_molecule_name: Name of the target molecule to compare against
        molecule_pool_filepath: Path to PDF file containing pool of molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of top-k most similar molecule names, sorted by descending similarity
        (alphabetically for ties)
    """
    # Get SMILES for target molecule from PubChem
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Extract molecule names from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for molecule_name in pool_molecules:
        smiles = get_smiles_from_pubchem(molecule_name)
        if smiles is not None:
            similarity = compute_tanimoto_similarity(target_smiles, smiles)
            similarities.append((molecule_name, similarity))

    # Sort by descending similarity, then alphabetically for ties
    # Using negative similarity for descending order, molecule name for alphabetical
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        print("Example: python solution.py Aspirin molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)
    print(f"Top {k} similar molecules to {target}:")
    for i, mol in enumerate(results, 1):
        print(f"  {i}. {mol}")
