"""
Crystal structure analyzer for CIF files.

This module provides functionality to analyze Wyckoff positions and coordinates
from CIF (Crystallographic Information File) files generated by X-ray diffraction.
"""

from typing import Any
from fractions import Fraction
from pymatgen.core.structure import Structure


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position in a crystal structure.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        A dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff labels to their multiplicities
        - wyckoff_coordinates_dict: Maps Wyckoff labels to fractional coordinates
                                    of the first atom in that position, as strings
                                    with rational representations (fractions with
                                    denominators ≤ 12)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result["wyckoff_multiplicity_dict"]
        {"a": 4, "c": 8}
        >>> result["wyckoff_coordinates_dict"]
        {"a": ["0", "1/2", "1/2"], "c": ["3/8", "1/9", "8/9"]}
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Get Wyckoff information using pymatgen's SpacegroupAnalyzer
    from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

    sga = SpacegroupAnalyzer(structure)

    # Get the refined structure with symmetry information
    symmetry_dataset = sga.get_symmetry_dataset()

    # Group atoms by Wyckoff position
    wyckoff_labels = symmetry_dataset.wyckoffs
    positions = structure.frac_coords
    atoms = structure.species

    # Dictionary to store results
    wyckoff_multiplicity_dict = {}
    wyckoff_first_coords = {}

    # Count multiplicities and track first coordinates for each Wyckoff position
    wyckoff_count = {}
    for i, wyckoff_label in enumerate(wyckoff_labels):
        if wyckoff_label not in wyckoff_count:
            wyckoff_count[wyckoff_label] = 0
            wyckoff_first_coords[wyckoff_label] = positions[i]

        wyckoff_count[wyckoff_label] += 1

    # Store multiplicities
    wyckoff_multiplicity_dict = wyckoff_count

    # Convert coordinates to rational representation
    wyckoff_coordinates_dict = {}
    for wyckoff_label, coords in wyckoff_first_coords.items():
        rational_coords = _coords_to_rational_strings(coords)
        wyckoff_coordinates_dict[wyckoff_label] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _coords_to_rational_strings(coords: list[float], max_denominator: int = 12) -> list[str]:
    """
    Convert fractional coordinates to rational string representations.

    Args:
        coords: List of fractional coordinates (floats between 0 and 1)
        max_denominator: Maximum denominator for rational representation

    Returns:
        List of coordinate strings in rational format (e.g., "1/2", "3/8", "0")
    """
    rational_strings = []

    for coord in coords:
        # Convert to Fraction with limited denominator
        frac = Fraction(coord).limit_denominator(max_denominator)

        # Format as string
        if frac.denominator == 1:
            coord_str = str(frac.numerator)
        else:
            coord_str = f"{frac.numerator}/{frac.denominator}"

        rational_strings.append(coord_str)

    return rational_strings
