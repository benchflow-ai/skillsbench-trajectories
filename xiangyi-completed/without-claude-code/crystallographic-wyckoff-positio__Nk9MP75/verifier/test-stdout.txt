from fractions import Fraction
from typing import Any
from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position in a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        A dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff position labels to their multiplicities
        - wyckoff_coordinates_dict: Maps Wyckoff position labels to fractional coordinates
                                   of the first atom, expressed as rational numbers
    """

    # Parse the CIF file
    parser = CifParser(filepath)
    structure = parser.parse_structures()[0]

    # Get space group analyzer for Wyckoff position information
    sga = SpacegroupAnalyzer(structure)

    # Get symmetrized structure with Wyckoff position information
    sym_struct = sga.get_symmetrized_structure()
    wyckoff_letters = sym_struct.wyckoff_letters

    # Dictionary to store results
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Group atoms by their Wyckoff letters
    wyckoff_groups = {}

    # Get the Wyckoff letters for each atom
    for site, wyckoff_letter in zip(sym_struct, wyckoff_letters):
        if wyckoff_letter not in wyckoff_groups:
            wyckoff_groups[wyckoff_letter] = {
                'count': 0,
                'first_coords': None
            }

        wyckoff_groups[wyckoff_letter]['count'] += 1

        # Store the first atom's coordinates for this Wyckoff position
        if wyckoff_groups[wyckoff_letter]['first_coords'] is None:
            wyckoff_groups[wyckoff_letter]['first_coords'] = site.frac_coords

    # Process each Wyckoff position
    for wyckoff_letter in sorted(wyckoff_groups.keys()):
        group = wyckoff_groups[wyckoff_letter]
        multiplicity = group['count']
        coords = group['first_coords']

        # Store multiplicity
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Convert coordinates to rational format
        rational_coords = []
        for coord in coords:
            # Convert to fraction with denominator ≤ 12
            frac = Fraction(coord).limit_denominator(12)
            rational_coords.append(str(frac))

        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
