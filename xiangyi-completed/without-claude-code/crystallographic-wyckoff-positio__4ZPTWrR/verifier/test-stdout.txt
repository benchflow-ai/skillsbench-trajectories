from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letter to coordinates [x, y, z]
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get the space group analyzer with symmetry detection
    # symprec controls the tolerance for symmetry detection
    sga = SpacegroupAnalyzer(structure, symprec=0.1)

    # Get the symmetrized structure which identifies equivalent sites
    symmetrized_structure = sga.get_symmetrized_structure()

    # Get Wyckoff symbols and their information
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols

    # Dictionary to store multiplicity for each unique Wyckoff position
    wyckoff_multiplicity_dict = {}
    # Dictionary to store coordinates for the first atom of each Wyckoff position
    wyckoff_coordinates_dict = {}

    # Track which Wyckoff positions we've already seen
    seen_wyckoff = {}

    # Iterate through equivalent sites
    for idx, equiv_sites in enumerate(symmetrized_structure.equivalent_sites):
        wyckoff_symbol = wyckoff_symbols[idx]

        # Extract just the letter part (remove multiplicity number prefix)
        # Wyckoff symbols can be like '4a', '8c', etc. We want just 'a', 'c'
        wyckoff_letter = ''.join(c for c in wyckoff_symbol if c.isalpha())

        # Only process each Wyckoff position once
        if wyckoff_letter not in seen_wyckoff:
            # Multiplicity is the number of equivalent sites
            multiplicity = len(equiv_sites)
            wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

            # Get the first site's coordinates
            first_site = equiv_sites[0]
            coords = first_site.frac_coords

            # Convert coordinates to simple fractions
            coord_strings = []
            for coord in coords:
                coord_str = decimal_to_fraction_string(coord)
                coord_strings.append(coord_str)

            wyckoff_coordinates_dict[wyckoff_letter] = coord_strings
            seen_wyckoff[wyckoff_letter] = True

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


def decimal_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a decimal value to a simplified fraction string.

    Args:
        value: The decimal value to convert
        max_denominator: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the fraction (e.g., "1/2", "0", "3/8")
    """
    # Normalize the value to [0, 1) range for periodic boundary conditions
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle the case where the fraction rounds to 1 (should be 0 due to periodicity)
    if frac.numerator == frac.denominator:
        return "0"

    # Return simplified string representation
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"
