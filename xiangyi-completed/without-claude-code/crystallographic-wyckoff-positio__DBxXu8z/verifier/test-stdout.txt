"""
Crystal structure analyzer for CIF files.
Analyzes Wyckoff position multiplicities and calculates coordinates for the first atom of each position.
"""

from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff positions from a CIF file.

    This function reads a CIF file, identifies unique Wyckoff positions,
    calculates their multiplicities, and provides approximate coordinates
    for the first atom of each Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Maps Wyckoff letters to their multiplicities
        - "wyckoff_coordinates_dict": Maps Wyckoff letters to fractional coordinates
                                       as strings with rational numbers (denominators ≤ 12)
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get symmetry information
    spacegroup_analyzer = SpacegroupAnalyzer(structure)

    # Get symmetry dataset which contains Wyckoff information
    symmetry_dataset = spacegroup_analyzer.get_symmetry_dataset()

    # Extract Wyckoff positions and their multiplicities
    wyckoff_letters = symmetry_dataset.wyckoffs

    # Create a mapping from Wyckoff letter to indices of atoms at that position
    wyckoff_to_indices: dict[str, list[int]] = {}
    for idx, wyckoff_letter in enumerate(wyckoff_letters):
        if wyckoff_letter not in wyckoff_to_indices:
            wyckoff_to_indices[wyckoff_letter] = []
        wyckoff_to_indices[wyckoff_letter].append(idx)

    # Build the output dictionaries
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    # Process each unique Wyckoff position
    for wyckoff_letter in sorted(wyckoff_to_indices.keys()):
        indices = wyckoff_to_indices[wyckoff_letter]

        # Multiplicity is the number of atoms at this Wyckoff position
        multiplicity = len(indices)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get the coordinates of the first atom at this Wyckoff position
        first_atom_idx = indices[0]
        coords = structure[first_atom_idx].frac_coords

        # Convert to rational numbers with denominators ≤ 12
        rational_coords = []
        for coord in coords:
            # Convert to fraction with max denominator of 12
            frac = Fraction(coord).limit_denominator(12)
            # Format as string
            if frac.denominator == 1:
                coord_str = str(frac.numerator)
            else:
                coord_str = f"{frac.numerator}/{frac.denominator}"
            rational_coords.append(coord_str)

        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
