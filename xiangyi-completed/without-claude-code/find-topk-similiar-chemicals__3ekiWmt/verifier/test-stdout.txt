"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from typing import List, Tuple
import time
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
import pubchempy as pcp


def _get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem API.

    Args:
        chemical_name: Name of the chemical compound

    Returns:
        SMILES string representation of the molecule

    Raises:
        ValueError: If chemical name not found or SMILES cannot be obtained
    """
    try:
        # Query PubChem for the compound by name
        compounds = pcp.get_compounds(chemical_name, "name")

        if not compounds:
            raise ValueError(f"Compound not found: {chemical_name}")

        compound = compounds[0]

        # Get SMILES - use smiles attribute (newer API)
        smiles = getattr(compound, 'smiles', None) or getattr(compound, 'canonical_smiles', None)

        if not smiles:
            raise ValueError(f"Could not get SMILES for {chemical_name}")

        return smiles

    except (IndexError, AttributeError) as e:
        raise ValueError(f"Error processing SMILES for {chemical_name}: {e}")
    except Exception as e:
        raise ValueError(f"Failed to fetch SMILES for {chemical_name}: {e}")


def _get_molecules_from_pdf(filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []

    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and filter empty lines
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(lines)

    return molecules


def _calculate_morgan_fingerprint(smiles: str, radius: int = 2) -> object:
    """
    Calculate Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string representation of the molecule
        radius: Radius for Morgan fingerprint (default: 2)

    Returns:
        Morgan fingerprint object

    Raises:
        ValueError: If SMILES is invalid
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError(f"Invalid SMILES: {smiles}")

    # useChirality=True to include chirality information
    # Using MorganGenerator for newer RDKit API compatibility
    try:
        fingerprint = AllChem.GetMorganFingerprintAsBitVect(mol, radius, useChirality=True, nBits=2048)
    except TypeError:
        # Fallback for older RDKit versions
        fingerprint = AllChem.GetMorganFingerprintAsBitVect(mol, radius, useChirality=True)
    return fingerprint


def _calculate_tanimoto_similarity(fp1: object, fp2: object) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0-1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(target_molecule_name: str,
                                       molecule_pool_filepath: str,
                                       top_k: int) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing the pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity descending,
        then alphabetically by molecule name for ties
    """
    # Get target molecule SMILES
    print(f"Fetching SMILES for target molecule: {target_molecule_name}")
    target_smiles = _get_smiles_from_pubchem(target_molecule_name)
    target_fp = _calculate_morgan_fingerprint(target_smiles)

    # Get all molecules from PDF
    print(f"Extracting molecules from {molecule_pool_filepath}")
    molecules = _get_molecules_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    retry_count = 0
    max_retries = 3

    for molecule_name in molecules:
        try:
            # Skip the target molecule itself
            if molecule_name.lower() == target_molecule_name.lower():
                continue

            # Get SMILES for molecule in pool with retry logic
            molecule_smiles = None
            for attempt in range(max_retries):
                try:
                    molecule_smiles = _get_smiles_from_pubchem(molecule_name)
                    break
                except ValueError as e:
                    if "503" in str(e) or "ServerBusy" in str(e):
                        # Server is busy, wait and retry
                        if attempt < max_retries - 1:
                            time.sleep(2)
                            continue
                    raise

            if molecule_smiles is None:
                raise ValueError(f"Could not get SMILES for {molecule_name}")

            molecule_fp = _calculate_morgan_fingerprint(molecule_smiles)

            # Calculate similarity
            similarity = _calculate_tanimoto_similarity(target_fp, molecule_fp)
            similarities.append((molecule_name, similarity))

            # Add a delay to avoid overwhelming the API
            time.sleep(0.15)
            retry_count = 0

        except (ValueError, Exception) as e:
            print(f"Warning: Could not process {molecule_name}: {e}")
            continue

    # Sort by similarity (descending), then alphabetically by name for ties
    sorted_similarities = sorted(similarities, key=lambda x: (-x[1], x[0]))

    # Return top k
    return sorted_similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        target_molecule_name="Aspirin",
        molecule_pool_filepath="/root/molecules.pdf",
        top_k=5
    )

    print("\nTop 5 similar molecules to Aspirin:")
    for i, (name, similarity) in enumerate(result, 1):
        print(f"{i}. {name}: {similarity:.4f}")
