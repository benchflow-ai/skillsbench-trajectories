import pdfplumber
import requests
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs, rdFingerprintGenerator
from typing import List, Tuple
import time


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert chemical name to SMILES format using PubChem API.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string representation of the molecule, or None if not found
    """
    base_url = "https://pubchem.ncbi.nlm.nih.gov/rest/pug"
    url = f"{base_url}/compound/name/{chemical_name}/property/CanonicalSMILES,IsomericSMILES/JSON"

    try:
        response = requests.get(url, timeout=10)
        if response.status_code == 200:
            data = response.json()
            props = data['PropertyTable']['Properties'][0]
            for smiles_key in ['CanonicalSMILES', 'IsomericSMILES', 'SMILES', 'ConnectivitySMILES']:
                if smiles_key in props:
                    return props[smiles_key]
            return None
        else:
            return None
    except Exception as e:
        return None


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = text.strip().split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecules.append(line)
    return molecules


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a SMILES string.

    Args:
        smiles: SMILES string representation of the molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Include chirality in fingerprint (default: True)

    Returns:
        Morgan fingerprint object, or None if conversion fails
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    mg = rdFingerprintGenerator.GetMorganGenerator(radius=radius, includeChirality=use_chirality)
    return mg.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(target_molecule_name: str, molecule_pool_filepath: str, top_k: int) -> list:
    """
    Find the top k most similar molecules to a target molecule from a pool of molecules in a PDF file.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing the pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples containing (molecule_name, similarity_score) sorted by similarity (descending)
        and alphabetically for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute target fingerprint
    target_fp = compute_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    molecule_names = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []

    for mol_name in molecule_names:
        # Skip if it's the same as target
        if mol_name.lower() == target_molecule_name.lower():
            continue

        # Get SMILES from PubChem
        smiles = get_smiles_from_pubchem(mol_name)
        if smiles is None:
            continue

        # Skip if it's the same molecule as target (same SMILES)
        if smiles == target_smiles:
            continue

        # Compute fingerprint
        fp = compute_morgan_fingerprint(smiles, radius=2, use_chirality=True)
        if fp is None:
            continue

        # Compute Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((mol_name, similarity))

        # Add small delay to avoid overwhelming PubChem API
        time.sleep(0.2)

    # Sort by similarity (descending), then alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]
