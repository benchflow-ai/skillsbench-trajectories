from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file to analyze.

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Dict mapping Wyckoff letters to multiplicities
        - "wyckoff_coordinates_dict": Dict mapping Wyckoff letters to coordinate lists
          (coordinates as fractional strings with denominators <= 12)
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Analyze symmetry to get the proper space group
    sga = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure which contains Wyckoff information
    sym_struct = sga.get_symmetrized_structure()

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Process each unique Wyckoff position
    for idx, wyckoff_symbol in enumerate(sym_struct.wyckoff_symbols):
        # Parse the Wyckoff symbol (e.g., "4a" -> multiplicity=4, letter="a")
        multiplicity = int("".join(filter(str.isdigit, wyckoff_symbol)))
        letter = "".join(filter(str.isalpha, wyckoff_symbol))

        # Get the first atom index for this Wyckoff position
        first_atom_idx = sym_struct.equivalent_indices[idx][0]

        # Get fractional coordinates of the first atom
        coords = sym_struct.sites[first_atom_idx].frac_coords

        # Convert coordinates to rational number format with denominator <= 12
        coord_strs = []
        for coord in coords:
            frac = Fraction(coord).limit_denominator(12)
            if frac.denominator == 1:
                coord_strs.append(str(frac.numerator))
            else:
                coord_strs.append(f"{frac.numerator}/{frac.denominator}")

        wyckoff_multiplicity_dict[letter] = multiplicity
        wyckoff_coordinates_dict[letter] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


if __name__ == "__main__":
    # Example usage
    result = analyze_wyckoff_position_multiplicities_and_coordinates(
        "/root/cif_files/FeS2_mp-226.cif"
    )
    print(result)
