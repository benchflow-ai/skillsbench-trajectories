"""
CIF File Wyckoff Position Analyzer

This module provides functionality to analyze crystal structure CIF files
and extract Wyckoff position multiplicities and coordinates.
"""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_rational_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating point coordinate to a rational string representation.

    Args:
        value: The floating point value to convert (should be in range [0, 1))
        max_denominator: Maximum denominator for the fraction (default 12)

    Returns:
        A string representation of the rational number (e.g., "1/2", "3/8", "0")
    """
    # Normalize value to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle the case where fraction equals 1 (should be 0)
    if frac.numerator == frac.denominator:
        return "0"

    # Return as string
    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze a CIF file to extract Wyckoff position multiplicities and
    coordinates for the first atom of each Wyckoff position.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        A dictionary containing:
        - wyckoff_multiplicity_dict: Maps Wyckoff letter to multiplicity
        - wyckoff_coordinates_dict: Maps Wyckoff letter to list of 3 coordinate strings
          (as rational numbers with denominators ≤ 12)

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result["wyckoff_multiplicity_dict"]
        {"a": 4, "c": 8}
        >>> result["wyckoff_coordinates_dict"]
        {"a": ["0", "1/2", "1/2"], "c": ["3/8", "1/9", "8/9"]}
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Analyze symmetry
    analyzer = SpacegroupAnalyzer(structure)

    # Get the symmetrized structure which contains Wyckoff information
    symmetrized_structure = analyzer.get_symmetrized_structure()

    # Get Wyckoff symbols (e.g., "4a", "8c") and equivalent sites
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols
    equivalent_sites = symmetrized_structure.equivalent_sites

    # Initialize result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Process each unique Wyckoff position
    for wyckoff_symbol, site_group in zip(wyckoff_symbols, equivalent_sites):
        # Parse the Wyckoff symbol (e.g., "4a" -> multiplicity=4, letter="a")
        # The symbol format is "<multiplicity><letter>"
        multiplicity = int(''.join(c for c in wyckoff_symbol if c.isdigit()))
        letter = ''.join(c for c in wyckoff_symbol if c.isalpha())

        # Get the fractional coordinates of the first site in this group
        first_site = site_group[0]
        frac_coords = first_site.frac_coords

        # Convert coordinates to rational strings
        coord_strings = [
            float_to_rational_string(coord)
            for coord in frac_coords
        ]

        # Store in dictionaries
        wyckoff_multiplicity_dict[letter] = multiplicity
        wyckoff_coordinates_dict[letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
