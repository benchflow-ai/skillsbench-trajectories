"""
Wyckoff Position Multiplicity and Coordinate Analyzer for CIF Files.

This module provides functionality to analyze crystal structures from CIF files,
extracting Wyckoff position multiplicities and representative coordinates for
each Wyckoff position.
"""

from fractions import Fraction
from typing import Any
import warnings

from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def _to_simple_fraction(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to the closest fraction string with denominator <= max_denominator.

    Args:
        value: The floating point value to convert (typically a fractional coordinate).
        max_denominator: Maximum allowed denominator for the simplified fraction.

    Returns:
        A string representation of the fraction (e.g., "1/2", "3/8", "0").
    """
    # Normalize to [0, 1) range for fractional coordinates
    value = value % 1.0

    # Handle very small values that should be 0
    if abs(value) < 1e-9:
        return "0"
    if abs(value - 1.0) < 1e-9:
        return "0"

    frac = Fraction(value).limit_denominator(max_denominator)

    if frac.denominator == 1:
        return str(frac.numerator % 1) if frac.numerator >= 1 else str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def _coords_to_fraction_strings(coords: list, max_denominator: int = 12) -> list[str]:
    """
    Convert a list of coordinate values to fraction strings.

    Args:
        coords: List of fractional coordinate values [x, y, z].
        max_denominator: Maximum allowed denominator for fractions.

    Returns:
        List of fraction strings for each coordinate.
    """
    return [_to_simple_fraction(c, max_denominator) for c in coords]


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    This function reads a CIF file, determines the true space group symmetry,
    identifies equivalent atomic sites by their Wyckoff positions, and returns
    the multiplicity and representative coordinates for each unique Wyckoff position.

    Args:
        filepath: Path to the CIF file to analyze.

    Returns:
        A dictionary containing:
            - "wyckoff_multiplicity_dict": Dict mapping Wyckoff letters to multiplicities
            - "wyckoff_coordinates_dict": Dict mapping Wyckoff letters to coordinate lists
              where each coordinate is expressed as a simplified fraction string

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Suppress deprecation and user warnings from pymatgen
    with warnings.catch_warnings():
        warnings.simplefilter("ignore", FutureWarning)
        warnings.simplefilter("ignore", UserWarning)

        # Parse the CIF file
        parser = CifParser(filepath)
        structure = parser.get_structures()[0]

    # Analyze the space group symmetry
    sga = SpacegroupAnalyzer(structure)

    # Get symmetrized structure with Wyckoff information
    sym_struct = sga.get_symmetrized_structure()

    # Extract Wyckoff symbols (e.g., '4a', '8c')
    wyckoff_symbols = sym_struct.wyckoff_symbols
    equivalent_sites = sym_struct.equivalent_sites

    # Build the result dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for i, (wyckoff_symbol, sites) in enumerate(zip(wyckoff_symbols, equivalent_sites)):
        # Parse Wyckoff symbol to extract multiplicity and letter
        # Wyckoff symbols are like "4a", "8c", "16e", etc.
        # The number is the multiplicity, the letter is the Wyckoff position
        multiplicity = len(sites)
        wyckoff_letter = wyckoff_symbol[-1]  # Last character is the letter

        # Get the first site's fractional coordinates
        first_site_coords = sites[0].frac_coords.tolist()

        # Convert coordinates to simplified fraction strings
        coord_strings = _coords_to_fraction_strings(first_site_coords)

        # Store in dictionaries (using just the letter as key)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
