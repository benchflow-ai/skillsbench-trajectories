"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.

This module uses pymatgen for CIF file parsing and spglib for symmetry analysis
to determine Wyckoff positions and their multiplicities.
"""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating point number to a fraction string.

    Args:
        value: The floating point value to convert (assumed to be in range [0, 1))
        max_denominator: Maximum denominator for the fraction (default 12)

    Returns:
        A string representation of the fraction (e.g., "1/2", "3/8", "0")
    """
    # Normalize value to [0, 1) range
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    # Return as string
    if frac.numerator == 0:
        return "0"
    elif frac.denominator == 1:
        # For fractional coordinates, integer values should be 0
        return str(frac.numerator % 1) if frac.numerator != 0 else "0"
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position in a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - wyckoff_multiplicity_dict: Dict mapping Wyckoff letters to their multiplicities
        - wyckoff_coordinates_dict: Dict mapping Wyckoff letters to coordinates as fraction strings
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Create SpacegroupAnalyzer with a reasonable tolerance
    analyzer = SpacegroupAnalyzer(structure, symprec=0.1)

    # Get the symmetry dataset which contains Wyckoff letters
    symmetry_dataset = analyzer.get_symmetry_dataset()

    # Extract Wyckoff letters for each atom using attribute interface
    # spglib returns integer codes where 0='a', 1='b', etc.
    wyckoff_codes = symmetry_dataset.wyckoffs

    # Map integer codes to letters
    wyckoff_letters = [chr(ord('a') + code) if isinstance(code, int) else code
                       for code in wyckoff_codes]

    # Get the standardized structure to work with conventional coordinates
    # The equivalent_atoms array tells us which atoms are equivalent
    equivalent_atoms = symmetry_dataset.equivalent_atoms

    # Get fractional coordinates from the original structure
    frac_coords = structure.frac_coords

    # Build dictionaries for multiplicities and coordinates
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Group atoms by Wyckoff letter and track first occurrence
    wyckoff_first_atom = {}  # Maps Wyckoff letter to first atom index
    wyckoff_count = {}  # Maps Wyckoff letter to count

    # Track unique Wyckoff positions by their equivalent atom index
    # Each unique Wyckoff position is identified by its representative atom
    seen_representatives = {}  # Maps (wyckoff_letter, equivalent_atom_index) to first occurrence

    for i, (letter, eq_atom) in enumerate(zip(wyckoff_letters, equivalent_atoms)):
        key = (letter, eq_atom)
        if key not in seen_representatives:
            seen_representatives[key] = i

    # Now count multiplicities and get coordinates for each unique Wyckoff position
    # We need to group by Wyckoff letter only (not by equivalent atom)
    for i, letter in enumerate(wyckoff_letters):
        if letter not in wyckoff_first_atom:
            wyckoff_first_atom[letter] = i
            wyckoff_count[letter] = 1
        else:
            wyckoff_count[letter] += 1

    # Build the result dictionaries
    for letter in sorted(wyckoff_first_atom.keys()):
        first_idx = wyckoff_first_atom[letter]
        coords = frac_coords[first_idx]

        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = wyckoff_count[letter]

        # Convert coordinates to fraction strings
        coord_strings = [float_to_fraction_string(c) for c in coords]
        wyckoff_coordinates_dict[letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
