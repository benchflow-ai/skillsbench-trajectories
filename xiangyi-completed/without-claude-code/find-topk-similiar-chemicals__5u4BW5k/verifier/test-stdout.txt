"""
Solution for finding top-k similar chemicals using Tanimoto similarity with Morgan fingerprints.
"""

import subprocess
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def extract_molecules_from_pdf(pdf_path: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        pdf_path: Path to the PDF file containing molecule names.

    Returns:
        List of molecule names extracted from the PDF.
    """
    result = subprocess.run(
        ["pdftotext", pdf_path, "-"],
        capture_output=True,
        text=True,
        check=True
    )
    lines = result.stdout.strip().split('\n')
    molecules = [line.strip() for line in lines if line.strip()]
    return molecules


def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Look up a chemical name using PubChem and return its SMILES representation.

    Args:
        chemical_name: The common or IUPAC name of the chemical.

    Returns:
        The canonical SMILES string, or None if not found.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute the Morgan fingerprint for a molecule given its SMILES.

    Args:
        smiles: SMILES representation of the molecule.
        radius: Radius for the Morgan fingerprint (default: 2).
        use_chirality: Whether to include chirality information (default: True).

    Returns:
        Morgan fingerprint as a bit vector, or None if SMILES is invalid.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    generator = rdFingerprintGenerator.GetMorganGenerator(radius=radius, includeChirality=use_chirality)
    fp = generator.GetFingerprint(mol)
    return fp


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute the Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint.
        fp2: Second fingerprint.

    Returns:
        Tanimoto similarity score between 0 and 1.
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules from a pool to a target molecule.

    Similarity is computed using Morgan fingerprints with Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical
    ordering for ties.

    Args:
        target_molecule_name: The name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing the pool of molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score) for the top-k similar molecules.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities for each molecule in the pool
    similarities = []
    for mol_name in pool_molecules:
        smiles = get_smiles_from_name(mol_name)
        if smiles is None:
            continue

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = compute_tanimoto_similarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (alphabetically ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k results
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        print("Example: python solution.py Aspirin /root/molecules.pdf 5")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} molecules similar to '{target}':")
    print("-" * 50)
    for i, (name, score) in enumerate(results, 1):
        print(f"{i}. {name}: {score:.4f}")
