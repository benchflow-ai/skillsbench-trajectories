package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.eventtime.WatermarkStrategy;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.state.ListState;
import org.apache.flink.api.common.state.ListStateDescriptor;
import org.apache.flink.api.common.state.ValueState;
import org.apache.flink.api.common.state.ValueStateDescriptor;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.api.java.utils.ParameterTool;
import org.apache.flink.configuration.Configuration;
import org.apache.flink.connector.file.src.FileSource;
import org.apache.flink.connector.file.src.reader.TextLineInputFormat;
import org.apache.flink.core.fs.Path;
import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.co.CoProcessFunction;
import org.apache.flink.streaming.api.functions.sink.filesystem.StreamingFileSink;
import org.apache.flink.streaming.api.functions.sink.filesystem.rollingpolicies.OnCheckpointRollingPolicy;
import org.apache.flink.util.Collector;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

public class LongestSessionPerJob extends AppBase {

    // Event type constants
    private static final int EVENT_SUBMIT = 0;
    private static final int EVENT_FINISH = 4;
    private static final int EVENT_FAIL = 3;
    private static final int EVENT_KILL = 5;

    // 10 minutes in microseconds
    private static final long SESSION_GAP_MICROS = 10L * 60 * 1000 * 1000;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events
        FileSource<String> taskFileSource = FileSource
                .forRecordStreamFormat(new TextLineInputFormat(), new Path(taskInput))
                .build();

        DataStream<String> taskLines = env.fromSource(
                taskFileSource,
                WatermarkStrategy.noWatermarks(),
                "Task Events Source"
        );

        DataStream<TaskEvent> taskEvents = taskLines.map(new MapFunction<String, TaskEvent>() {
            @Override
            public TaskEvent map(String line) throws Exception {
                String[] fields = line.split(",", -1);
                TaskEvent event = new TaskEvent();
                event.timestamp = Long.parseLong(fields[0]);
                event.missingInfo = fields[1];
                event.jobId = Long.parseLong(fields[2]);
                event.taskIndex = Long.parseLong(fields[3]);
                event.machineId = fields[4].isEmpty() ? null : Long.parseLong(fields[4]);
                event.eventType = Integer.parseInt(fields[5]);
                event.userName = fields[6];
                event.schedulingClass = Integer.parseInt(fields[7]);
                event.priority = Integer.parseInt(fields[8]);
                event.cpuRequest = Double.parseDouble(fields[9]);
                event.memoryRequest = Double.parseDouble(fields[10]);
                event.diskRequest = Double.parseDouble(fields[11]);
                event.differentMachineConstraint = Integer.parseInt(fields[12]);
                return event;
            }
        });

        // Filter for SUBMIT events only
        DataStream<TaskEvent> submitEvents = taskEvents
                .filter(event -> event.eventType == EVENT_SUBMIT);

        // Read job events
        FileSource<String> jobFileSource = FileSource
                .forRecordStreamFormat(new TextLineInputFormat(), new Path(jobInput))
                .build();

        DataStream<String> jobLines = env.fromSource(
                jobFileSource,
                WatermarkStrategy.noWatermarks(),
                "Job Events Source"
        );

        DataStream<JobEvent> jobEvents = jobLines.map(new MapFunction<String, JobEvent>() {
            @Override
            public JobEvent map(String line) throws Exception {
                String[] fields = line.split(",", -1);
                JobEvent event = new JobEvent();
                event.timestamp = Long.parseLong(fields[0]);
                event.missingInfo = fields[1];
                event.jobId = Long.parseLong(fields[2]);
                event.eventType = Integer.parseInt(fields[3]);
                event.userName = fields[4];
                event.schedulingClass = Integer.parseInt(fields[5]);
                event.jobName = fields[6];
                event.logicalJobName = fields[7];
                return event;
            }
        });

        // Filter for job termination events (FINISH, FAIL, KILL)
        DataStream<JobEvent> jobTerminations = jobEvents
                .filter(event -> event.eventType == EVENT_FINISH ||
                               event.eventType == EVENT_FAIL ||
                               event.eventType == EVENT_KILL);

        // Connect submit events with job terminations
        DataStream<String> results = submitEvents
                .keyBy(event -> event.jobId)
                .connect(jobTerminations.keyBy(event -> event.jobId))
                .process(new ComputeLongestSession());

        // Write output to file
        if (out == null) {
            results.addSink(
                StreamingFileSink.<String>forRowFormat(
                    new Path(outputPath),
                    new org.apache.flink.api.common.serialization.SimpleStringEncoder<String>("UTF-8")
                )
                .withRollingPolicy(OnCheckpointRollingPolicy.build())
                .build()
            );
        } else {
            results.addSink((org.apache.flink.streaming.api.functions.sink.SinkFunction<String>) sinkOrTest(null));
        }

        env.execute("LongestSessionPerJob");
    }

    // Compute longest session when job terminates
    public static class ComputeLongestSession extends CoProcessFunction<TaskEvent, JobEvent, String> {

        private transient ListState<Long> submitTimestamps;
        private transient ValueState<Boolean> jobTerminated;
        private transient ValueState<Boolean> resultEmitted;

        @Override
        public void open(Configuration parameters) {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Long.class));
            jobTerminated = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("jobTerminated", Boolean.class));
            resultEmitted = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("resultEmitted", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<String> out) throws Exception {
            // Collect submit event timestamps
            submitTimestamps.add(event.timestamp);

            // If job already terminated, compute and emit result
            Boolean terminated = jobTerminated.value();
            Boolean emitted = resultEmitted.value();
            if (terminated != null && terminated && (emitted == null || !emitted)) {
                emitResult(event.jobId, out);
            }
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<String> out) throws Exception {
            // Mark job as terminated
            jobTerminated.update(true);

            // Compute and emit result if not already done
            Boolean emitted = resultEmitted.value();
            if (emitted == null || !emitted) {
                emitResult(event.jobId, out);
            }
        }

        private void emitResult(long jobId, Collector<String> out) throws Exception {
            // Get all submit timestamps and sort them
            List<Long> timestamps = new ArrayList<>();
            for (Long ts : submitTimestamps.get()) {
                timestamps.add(ts);
            }

            if (timestamps.isEmpty()) {
                // No submit events, no output needed
                resultEmitted.update(true);
                return;
            }

            Collections.sort(timestamps);

            // Find stages and count tasks in each
            int maxStageCount = 0;
            int currentStageCount = 1;

            for (int i = 1; i < timestamps.size(); i++) {
                long gap = timestamps.get(i) - timestamps.get(i - 1);

                if (gap > SESSION_GAP_MICROS) {
                    // Stage ended, check if it's the max
                    maxStageCount = Math.max(maxStageCount, currentStageCount);
                    currentStageCount = 1;
                } else {
                    // Same stage
                    currentStageCount++;
                }
            }

            // Don't forget the last stage
            maxStageCount = Math.max(maxStageCount, currentStageCount);

            // Emit result
            out.collect("(" + jobId + "," + maxStageCount + ")");
            resultEmitted.update(true);
        }
    }
}
