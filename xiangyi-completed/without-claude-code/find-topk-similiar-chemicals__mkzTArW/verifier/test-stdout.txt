"""
Solution for finding top k similar chemicals using Tanimoto similarity.
Uses Morgan fingerprints with radius=2 and chirality enabled.
"""

import PyPDF2
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from rdkit import RDLogger
import pubchempy as pcp
import time
import warnings

# Suppress deprecation warnings from RDKit and PubChemPy
warnings.filterwarnings('ignore', category=DeprecationWarning)
# Suppress RDKit warnings
RDLogger.DisableLog('rdApp.*')


def get_smiles_from_name(chemical_name, max_retries=3, delay=1.0):
    """
    Convert chemical name to SMILES representation using PubChem.

    Args:
        chemical_name: Name of the chemical
        max_retries: Maximum number of retry attempts
        delay: Delay between retries in seconds

    Returns:
        SMILES string or None if not found
    """
    for attempt in range(max_retries):
        try:
            # Search PubChem for the compound
            compounds = pcp.get_compounds(chemical_name, 'name')
            if compounds:
                # Get SMILES representation
                # Try to get isomeric SMILES first (includes stereochemistry)
                try:
                    smiles = compounds[0].isomeric_smiles
                    if smiles:
                        return smiles
                except (AttributeError, KeyError):
                    pass
                # Fallback to canonical SMILES
                try:
                    return compounds[0].canonical_smiles
                except (AttributeError, KeyError):
                    pass
        except Exception as e:
            if "503" in str(e) or "ServerBusy" in str(e):
                # Rate limiting - wait and retry
                if attempt < max_retries - 1:
                    time.sleep(delay * (attempt + 1))  # Exponential backoff
                    continue
            print(f"Error fetching SMILES for {chemical_name}: {e}")
    return None


def get_morgan_fingerprint(smiles, radius=2, use_chirality=True):
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES representation of molecule
        radius: Morgan fingerprint radius (default=2)
        use_chirality: Include chirality information (default=True)

    Returns:
        Morgan fingerprint or None if conversion fails
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        # Generate Morgan fingerprint with radius=2 and chirality
        fp = AllChem.GetMorganFingerprintAsBitVect(
            mol,
            radius=radius,
            nBits=2048,
            useChirality=use_chirality
        )
        return fp
    except Exception as e:
        print(f"Error generating fingerprint for {smiles}: {e}")
        return None


def calculate_tanimoto_similarity(fp1, fp2):
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0 to 1)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def extract_molecules_from_pdf(filepath):
    """
    Extract chemical names from PDF file.

    Args:
        filepath: Path to PDF file

    Returns:
        List of chemical names
    """
    molecules = []
    try:
        with open(filepath, 'rb') as file:
            pdf_reader = PyPDF2.PdfReader(file)
            all_text = ""
            for page in pdf_reader.pages:
                all_text += page.extract_text()

        # Split by lines and filter out empty lines
        lines = [line.strip() for line in all_text.split('\n') if line.strip()]

        # Some lines might have multiple molecules separated by spaces
        # We'll keep them as-is since the PDF seems to mostly have one per line
        molecules = lines

    except Exception as e:
        print(f"Error reading PDF: {e}")

    return molecules


def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Find the top k most similar molecules to a target molecule.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of molecule names sorted by similarity (descending), with alphabetical
        ordering for ties. The list contains up to top_k molecules.
    """
    # Extract molecules from PDF
    molecule_pool = extract_molecules_from_pdf(molecule_pool_filepath)

    if not molecule_pool:
        print("No molecules found in PDF")
        return []

    # Get SMILES for target molecule
    print(f"Fetching SMILES for target: {target_molecule_name}")
    target_smiles = get_smiles_from_name(target_molecule_name)

    if target_smiles is None:
        print(f"Could not find SMILES for target molecule: {target_molecule_name}")
        return []

    # Get fingerprint for target molecule
    target_fp = get_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)

    if target_fp is None:
        print(f"Could not generate fingerprint for target molecule: {target_molecule_name}")
        return []

    # Calculate similarities for all molecules in the pool
    similarities = []

    for mol_name in molecule_pool:
        # Skip the target molecule itself if it's in the pool
        if mol_name.lower() == target_molecule_name.lower():
            continue

        print(f"Processing: {mol_name}")

        # Get SMILES for pool molecule
        mol_smiles = get_smiles_from_name(mol_name)

        # Small delay to avoid rate limiting
        time.sleep(0.2)

        if mol_smiles is None:
            print(f"  -> Could not find SMILES for {mol_name}")
            continue

        # Get fingerprint for pool molecule
        mol_fp = get_morgan_fingerprint(mol_smiles, radius=2, use_chirality=True)

        if mol_fp is None:
            print(f"  -> Could not generate fingerprint for {mol_name}")
            continue

        # Calculate Tanimoto similarity
        similarity = calculate_tanimoto_similarity(target_fp, mol_fp)
        similarities.append((mol_name, similarity))
        print(f"  -> Similarity: {similarity:.4f}")

    # Sort by similarity (descending), then by name (alphabetically) for ties
    # Using negative similarity for descending order, then molecule name for alphabetical
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecules (just the names, not the scores)
    top_k_molecules = [mol_name for mol_name, _ in similarities[:top_k]]

    return top_k_molecules


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5

    print(f"Finding top {k} molecules similar to {target}...")
    print("=" * 60)

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print("\n" + "=" * 60)
    print(f"Top {k} similar molecules to {target}:")
    for i, mol in enumerate(results, 1):
        print(f"{i}. {mol}")
