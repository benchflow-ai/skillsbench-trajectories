package clusterdata.query;

import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;
import org.apache.flink.api.common.functions.FilterFunction;
import org.apache.flink.api.common.functions.FlatMapFunction;
import org.apache.flink.api.common.functions.MapFunction;
import org.apache.flink.api.common.functions.ReduceFunction;
import org.apache.flink.api.java.functions.KeySelector;
import org.apache.flink.api.java.tuple.Tuple2;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;
import org.apache.flink.streaming.api.functions.sink.SinkFunction;
import org.apache.flink.streaming.api.functions.source.SourceFunction;
import org.apache.flink.util.Collector;
import org.apache.flink.api.java.utils.ParameterTool;
import java.io.BufferedWriter;
import java.io.FileWriter;
import java.util.*;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        // Filter out empty strings from args
        java.util.List<String> filteredArgs = new java.util.ArrayList<>();
        for (String arg : args) {
            if (arg != null && !arg.isEmpty()) {
                filteredArgs.add(arg);
            }
        }
        String[] cleanArgs = filteredArgs.toArray(new String[0]);

        ParameterTool params = ParameterTool.fromArgs(cleanArgs);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  "+taskInput);
        System.out.println("job_input  "+jobInput);
        final long sesssize = params.getLong("sesssize", 600L) * 1000000L; // Convert seconds to microseconds

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1); // Single parallelism for deterministic output

        // Read task events from file and parse them
        SourceFunction<String> taskSourceFunc = new FileSourceFunction(taskInput);

        // Parse task events and filter for SUBMIT events (eventType == 0)
        env.addSource(taskSourceFunc)
                .map(new MapFunction<String, TaskEvent>() {
                    @Override
                    public TaskEvent map(String line) throws Exception {
                        return TaskEvent.fromCsvLine(line);
                    }
                })
                .filter(new FilterFunction<TaskEvent>() {
                    @Override
                    public boolean filter(TaskEvent event) throws Exception {
                        return event != null && event.eventType == 0; // Only SUBMIT events
                    }
                })
                .keyBy(new KeySelector<TaskEvent, Long>() {
                    @Override
                    public Long getKey(TaskEvent event) throws Exception {
                        return event.jobId;
                    }
                })
                .flatMap(new StageDetector(sesssize))
                // Now find the maximum count per job
                .keyBy(new KeySelector<Tuple2<Long, Integer>, Long>() {
                    @Override
                    public Long getKey(Tuple2<Long, Integer> value) throws Exception {
                        return value.f0;
                    }
                })
                .reduce(new ReduceFunction<Tuple2<Long, Integer>>() {
                    @Override
                    public Tuple2<Long, Integer> reduce(Tuple2<Long, Integer> v1, Tuple2<Long, Integer> v2) throws Exception {
                        // Keep the one with the larger count
                        if (v2.f1 > v1.f1) {
                            return v2;
                        }
                        return v1;
                    }
                })
                .addSink(new FileSink(outputPath));

        env.execute("LongestSessionPerJob");
    }

    // Detects stages by tracking gaps in event time
    static class StageDetector implements FlatMapFunction<TaskEvent, Tuple2<Long, Integer>> {
        private final long sessionGapMicros;
        private Map<Long, SessionState> jobSessions = new HashMap<>();

        StageDetector(long sessionGapMicros) {
            this.sessionGapMicros = sessionGapMicros;
        }

        @Override
        public void flatMap(TaskEvent event, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Long jobId = event.jobId;
            SessionState state = jobSessions.computeIfAbsent(jobId, k -> new SessionState());

            // If first event or gap is too large, start a new session
            if (state.lastEventTime == -1 ||
                (event.timestamp - state.lastEventTime > sessionGapMicros)) {
                // Output previous session if it exists
                if (state.currentSessionCount > 0) {
                    out.collect(new Tuple2<>(jobId, state.currentSessionCount));
                }
                // Start new session
                state.currentSessionCount = 1;
            } else {
                state.currentSessionCount++;
            }
            state.lastEventTime = event.timestamp;
        }

        private static class SessionState {
            long lastEventTime = -1;
            int currentSessionCount = 0;
        }
    }

    // Simple file source that reads from a gzipped CSV file
    static class FileSourceFunction implements SourceFunction<String> {
        private String filePath;
        private volatile boolean isRunning = true;

        FileSourceFunction(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<String> ctx) throws Exception {
            java.util.zip.GZIPInputStream gzipInputStream =
                new java.util.zip.GZIPInputStream(new java.io.FileInputStream(filePath));
            java.io.BufferedReader reader = new java.io.BufferedReader(
                new java.io.InputStreamReader(gzipInputStream));

            String line;
            while (isRunning && (line = reader.readLine()) != null) {
                ctx.collect(line);
            }

            reader.close();
            gzipInputStream.close();
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }

    // Custom sink to write to file
    static class FileSink implements SinkFunction<Tuple2<Long, Integer>> {
        private String outputPath;
        private transient BufferedWriter writer;

        FileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void invoke(Tuple2<Long, Integer> value, Context context) throws Exception {
            if (writer == null) {
                writer = new BufferedWriter(new FileWriter(outputPath, true));
            }
            writer.write("(" + value.f0 + "," + value.f1 + ")");
            writer.newLine();
            writer.flush();
        }

        @Override
        public void finish() throws Exception {
            if (writer != null) {
                writer.close();
            }
        }
    }

}
