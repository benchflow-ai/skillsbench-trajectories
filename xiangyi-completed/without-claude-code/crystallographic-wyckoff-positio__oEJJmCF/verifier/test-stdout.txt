"""
Module for analyzing Wyckoff position multiplicities and coordinates from CIF files.
"""

from typing import Any
from fractions import Fraction
import warnings

from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_string(value: float, max_denominator: int = 12) -> str:
    """
    Convert a floating point coordinate to a fraction string with limited denominator.

    Args:
        value: The fractional coordinate value
        max_denominator: Maximum allowed denominator for the fraction

    Returns:
        String representation of the fraction (e.g., "3/8", "1/2", "0")
    """
    # Wrap to [0, 1) range for periodic coordinates
    value = value % 1.0

    # Handle values very close to 0 or 1
    if abs(value) < 1e-6 or abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction with limited denominator
    frac = Fraction(value).limit_denominator(max_denominator)

    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the
    first atom of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
            - wyckoff_multiplicity_dict: Maps Wyckoff letter to multiplicity
            - wyckoff_coordinates_dict: Maps Wyckoff letter to coordinates
              of the first atom as list of fraction strings
    """
    # Parse the CIF file (suppress pymatgen precision warnings)
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        parser = CifParser(filepath)
        structure = parser.parse_structures(primitive=True)[0]

    # Analyze symmetry to get space group and Wyckoff information
    analyzer = SpacegroupAnalyzer(structure, symprec=0.01)

    # Get symmetrized structure with Wyckoff positions
    sym_struct = analyzer.get_symmetrized_structure()

    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    # Iterate through equivalent sites and their Wyckoff symbols
    for equiv_sites, wyckoff_symbol in zip(
        sym_struct.equivalent_sites,
        sym_struct.wyckoff_symbols
    ):
        # Extract Wyckoff letter (last character of symbol like "4a" -> "a")
        wyckoff_letter = wyckoff_symbol[-1]

        # Multiplicity is the number of equivalent sites
        multiplicity = len(equiv_sites)

        # Get coordinates of the first atom in this Wyckoff position
        first_site = equiv_sites[0]
        frac_coords = first_site.frac_coords

        # Convert to fraction strings
        coord_strings = [
            float_to_fraction_string(coord) for coord in frac_coords
        ]

        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity
        wyckoff_coordinates_dict[wyckoff_letter] = coord_strings

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
