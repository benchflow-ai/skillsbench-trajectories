"""
Find top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
import requests
from rdkit import Chem
from rdkit.Chem import AllChem
from typing import List, Tuple
import time


def get_smiles_from_pubchem(molecule_name: str) -> str:
    """
    Convert a chemical name to SMILES using PubChem API.

    Args:
        molecule_name: Chemical name

    Returns:
        SMILES string or None if not found
    """
    try:
        # PubChem API endpoint for name to structure conversion
        url = f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{molecule_name}/property/CanonicalSMILES/JSON"

        response = requests.get(url, timeout=30)

        if response.status_code == 200:
            data = response.json()
            # Handle the PropertyTable response format
            if "PropertyTable" in data:
                properties = data["PropertyTable"].get("Properties", [])
                if properties:
                    # Try CanonicalSMILES first, then ConnectivitySMILES
                    smiles = properties[0].get("CanonicalSMILES") or properties[0].get("ConnectivitySMILES")
                    if smiles:
                        return smiles
            # Handle alternative response format
            elif "properties" in data and len(data["properties"]) > 0:
                smiles = data["properties"][0].get("CanonicalSMILES")
                return smiles
    except requests.exceptions.RequestException:
        # Fallback: try alternative approach
        try:
            url = f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{molecule_name}/JSON"
            response = requests.get(url, timeout=30)

            if response.status_code == 200:
                data = response.json()
                compounds = data.get("PC_Compounds", [])
                if compounds:
                    compound = compounds[0]
                    props = compound.get("props", [])
                    for prop in props:
                        if prop.get("urn", {}).get("label") == "SMILES":
                            smiles = prop.get("value", {}).get("sval")
                            if smiles:
                                return smiles
        except requests.exceptions.RequestException:
            pass

    return None


def extract_molecules_from_pdf(pdf_path: str) -> List[str]:
    """
    Extract all chemical names from the PDF file.

    Args:
        pdf_path: Path to the molecules.pdf file

    Returns:
        List of chemical names
    """
    molecules = []

    try:
        with pdfplumber.open(pdf_path) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.strip().split('\n')
                    for line in lines:
                        molecule = line.strip()
                        if molecule:
                            molecules.append(molecule)
    except Exception as e:
        raise ValueError(f"Error reading PDF: {e}")

    return molecules


def calculate_morgan_fingerprint(smiles: str, radius: int = 2) -> object:
    """
    Calculate Morgan fingerprint for a SMILES string.

    Args:
        smiles: SMILES representation of the molecule
        radius: Radius for Morgan fingerprint (default 2)

    Returns:
        Morgan fingerprint object
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None

        # Use useCounts=False for standard fingerprint (bit vector)
        # Include chirality
        fingerprint = AllChem.GetMorganFingerprintAsBitVect(
            mol,
            radius=radius,
            nBits=2048,
            useChirality=True
        )
        return fingerprint
    except Exception as e:
        return None


def tanimoto_similarity(fp1: object, fp2: object) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First Morgan fingerprint
        fp2: Second Morgan fingerprint

    Returns:
        Tanimoto similarity score (0-1)
    """
    if fp1 is None or fp2 is None:
        return 0.0

    from rdkit import DataStructs
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k most similar molecules to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to the PDF file containing molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity (descending)
        and then alphabetically by name when similarities are equal
    """
    # Extract all molecules from PDF
    molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    print(f"Converting target molecule '{target_molecule_name}' to SMILES...")
    target_smiles = get_smiles_from_pubchem(target_molecule_name)

    if target_smiles is None:
        raise ValueError(f"Could not convert '{target_molecule_name}' to SMILES")

    # Calculate fingerprint for target molecule
    target_fp = calculate_morgan_fingerprint(target_smiles, radius=2)

    if target_fp is None:
        raise ValueError(f"Could not calculate fingerprint for '{target_molecule_name}'")

    # Calculate similarity for all molecules in the pool
    similarities = []

    for i, molecule in enumerate(molecules):
        # Rate limiting to avoid overwhelming the API
        if i > 0 and i % 10 == 0:
            time.sleep(0.5)

        print(f"Processing {i+1}/{len(molecules)}: {molecule}")

        smiles = get_smiles_from_pubchem(molecule)

        if smiles is not None:
            fp = calculate_morgan_fingerprint(smiles, radius=2)

            if fp is not None:
                similarity = tanimoto_similarity(target_fp, fp)
                similarities.append((molecule, similarity))

    # Sort by similarity (descending), then alphabetically by name
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    pdf_path = "/root/molecules.pdf"

    # Test with a molecule
    target = "Caffeine"
    k = 5

    try:
        results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

        print(f"\nTop {k} molecules similar to '{target}':")
        for i, (name, similarity) in enumerate(results, 1):
            print(f"{i}. {name}: {similarity:.4f}")
    except Exception as e:
        print(f"Error: {e}")
