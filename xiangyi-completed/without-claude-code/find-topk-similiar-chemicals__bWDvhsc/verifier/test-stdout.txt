"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
import requests
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
from typing import List, Tuple
import time
from functools import lru_cache


@lru_cache(maxsize=256)
def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert chemical name to SMILES representation using PubChem API.
    Results are cached to avoid repeated API calls.

    Args:
        chemical_name: Name of the chemical compound

    Returns:
        SMILES string representation of the molecule

    Raises:
        ValueError: If the chemical name cannot be found in PubChem
    """
    # PubChem API endpoint for compound search by name
    url = f"https://pubchem.ncbi.nlm.nih.gov/rest/pug/compound/name/{chemical_name}/property/CanonicalSMILES/JSON"

    max_retries = 3
    retry_delay = 2.0

    for attempt in range(max_retries):
        try:
            response = requests.get(url, timeout=10)
            response.raise_for_status()
            data = response.json()

            # Extract SMILES from response (try CanonicalSMILES first, then ConnectivitySMILES)
            properties = data['PropertyTable']['Properties'][0]
            smiles = properties.get('CanonicalSMILES') or properties.get('ConnectivitySMILES')
            if not smiles:
                raise ValueError("No SMILES found in response")
            return smiles
        except requests.exceptions.HTTPError as e:
            if e.response.status_code == 503 and attempt < max_retries - 1:
                # Server busy, wait and retry
                time.sleep(retry_delay * (attempt + 1))
                continue
            raise ValueError(f"Could not retrieve SMILES for '{chemical_name}': {str(e)}")
        except Exception as e:
            raise ValueError(f"Could not retrieve SMILES for '{chemical_name}': {str(e)}")


def calculate_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality information (default: True)

    Returns:
        Morgan fingerprint object

    Raises:
        ValueError: If SMILES string is invalid
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        raise ValueError(f"Invalid SMILES string: {smiles}")

    # Generate Morgan fingerprint with specified radius and chirality
    # Use MorganGenerator to avoid deprecation warnings
    try:
        from rdkit.Chem import MorganGenerator
        mgen = MorganGenerator(radius=radius, includeChirality=use_chirality)
        fingerprint = mgen.GetSparseCountFingerprint(mol)
    except ImportError:
        # Fallback to old method if MorganGenerator is not available
        fingerprint = AllChem.GetMorganFingerprint(mol, radius, useChirality=use_chirality)

    return fingerprint


def calculate_tanimoto_similarity(fp1, fp2) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint
        fp2: Second fingerprint

    Returns:
        Tanimoto similarity score (0.0 to 1.0)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def extract_chemicals_from_pdf(filepath: str) -> List[str]:
    """
    Extract chemical names from PDF file.

    Args:
        filepath: Path to the PDF file

    Returns:
        List of chemical names
    """
    pdf = pdfplumber.open(filepath)
    all_text = ""
    for page in pdf.pages:
        all_text += page.extract_text() + "\n"
    pdf.close()

    # Split into lines and filter out empty lines
    chemicals = [line.strip() for line in all_text.split('\n') if line.strip()]
    return chemicals


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k most similar molecules to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target chemical compound
        molecule_pool_filepath: Path to PDF file containing pool of molecule names
        top_k: Number of top similar molecules to return

    Returns:
        List of top k most similar molecule names, sorted by similarity (descending),
        with alphabetical ordering for ties
    """
    # Extract molecules from PDF
    molecule_pool = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Get SMILES for target molecule
    print(f"Fetching SMILES for target molecule: {target_molecule_name}")
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    target_fp = calculate_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)

    # Calculate similarities
    similarities = []

    for molecule_name in molecule_pool:
        # Skip the target molecule itself if it appears in the pool
        if molecule_name.lower() == target_molecule_name.lower():
            continue

        try:
            # Get SMILES and fingerprint for pool molecule
            smiles = get_smiles_from_pubchem(molecule_name)
            fp = calculate_morgan_fingerprint(smiles, radius=2, use_chirality=True)

            # Calculate Tanimoto similarity
            similarity = calculate_tanimoto_similarity(target_fp, fp)
            similarities.append((molecule_name, similarity))

            # Small delay to avoid overwhelming PubChem API
            time.sleep(0.3)

        except Exception as e:
            print(f"Warning: Could not process '{molecule_name}': {str(e)}")
            continue

    # Sort by similarity (descending), then alphabetically for ties
    # Using negative similarity for descending order, then molecule name for alphabetical
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k molecule names
    top_k_molecules = [name for name, similarity in similarities[:top_k]]

    return top_k_molecules
