"""
Materials science tool for analyzing Wyckoff position multiplicities and coordinates from CIF files.
"""

from fractions import Fraction
from typing import Any
from pymatgen.core import Structure
import spglib


def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze a CIF file to extract Wyckoff position multiplicities and calculate approximate
    coordinates for the first atom of each Wyckoff position.

    Args:
        filepath: Path to the CIF file

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Dictionary mapping Wyckoff letters to their multiplicities
        - "wyckoff_coordinates_dict": Dictionary mapping Wyckoff letters to coordinate lists
          where coordinates are expressed as fractions with denominators <= 12
    """

    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Prepare data for spglib
    lattice = structure.lattice.matrix
    positions = structure.frac_coords
    numbers = [site.species.elements[0].Z for site in structure]

    # Get symmetry dataset from spglib
    dataset = spglib.get_symmetry_dataset((lattice, positions, numbers), symprec=0.1)

    if dataset is None:
        raise ValueError(f"Could not determine symmetry for structure at {filepath}")

    # Extract Wyckoff information
    wyckoff_letters = dataset.wyckoffs

    # Group atoms by Wyckoff position
    wyckoff_groups = {}
    for atom_idx, wyckoff_letter in enumerate(wyckoff_letters):
        if wyckoff_letter not in wyckoff_groups:
            wyckoff_groups[wyckoff_letter] = []
        wyckoff_groups[wyckoff_letter].append(atom_idx)

    # Build output dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for wyckoff_letter in sorted(wyckoff_groups.keys()):
        atom_indices = wyckoff_groups[wyckoff_letter]

        # Calculate multiplicity (number of atoms in this Wyckoff position)
        multiplicity = len(atom_indices)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get coordinates of the first atom in this Wyckoff position
        first_atom_idx = atom_indices[0]
        first_atom_coords = positions[first_atom_idx]

        # Convert coordinates to fractions with denominators <= 12
        fractional_coords = []
        for coord in first_atom_coords:
            # Round to a reasonable precision to avoid floating point artifacts
            coord = round(coord, 10)
            # Find the best rational approximation with denominator <= 12
            frac = Fraction(coord).limit_denominator(12)
            fractional_coords.append(str(frac))

        wyckoff_coordinates_dict[wyckoff_letter] = fractional_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }


if __name__ == "__main__":
    # Example usage
    result = analyze_wyckoff_position_multiplicities_and_coordinates("/root/cif_files/FeS2_mp-226.cif")
    print("Result:")
    print(result)
