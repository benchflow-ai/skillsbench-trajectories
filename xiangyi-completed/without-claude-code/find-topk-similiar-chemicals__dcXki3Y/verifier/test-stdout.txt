"""
Solution for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import subprocess
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem.rdFingerprintGenerator import GetMorganGenerator
from rdkit import DataStructs


def extract_molecules_from_pdf(pdf_filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file containing molecule names.

    Returns:
        List of molecule names extracted from the PDF.
    """
    txt_filepath = pdf_filepath.replace('.pdf', '.txt')
    subprocess.run(['pdftotext', pdf_filepath, txt_filepath], check=True)

    with open(txt_filepath, 'r') as f:
        content = f.read()

    molecules = [line.strip() for line in content.split('\n') if line.strip()]
    return molecules


def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to its SMILES representation using PubChem.

    Args:
        chemical_name: The name of the chemical compound.

    Returns:
        The canonical SMILES string, or None if not found.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a given SMILES string.

    Args:
        smiles: SMILES representation of the molecule.
        radius: Radius for Morgan fingerprint (default: 2).
        use_chirality: Whether to include chirality information (default: True).

    Returns:
        Morgan fingerprint object, or None if computation fails.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None
    generator = GetMorganGenerator(radius=radius, includeChirality=use_chirality)
    return generator.GetFingerprint(mol)


def compute_tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint.
        fp2: Second fingerprint.

    Returns:
        Tanimoto similarity score between 0 and 1.
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a pool of molecules.

    Uses Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical ordering
    for ties.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to PDF file containing pool of molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score) for top-k most similar molecules.
    """
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    similarities = []
    for mol_name in pool_molecules:
        smiles = get_smiles_from_name(mol_name)
        if smiles is None:
            continue

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = compute_tanimoto_similarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity descending, then by name alphabetically for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    result = topk_tanimoto_similarity_molecules(
        "Aspirin",
        "/root/molecules.pdf",
        5
    )
    print("Top 5 similar molecules to Aspirin:")
    for mol_name, similarity in result:
        print(f"  {mol_name}: {similarity:.4f}")
