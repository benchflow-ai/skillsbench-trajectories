package clusterdata.query;

import clusterdata.datatypes.TaskEvent;

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);

        // 10 minutes in microseconds
        final long sessionGapMicros = 10L * 60L * 1000000L;

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Read task events
        DataStream<TaskEvent> taskEvents = env
                .addSource(taskSourceOrTest(new GzipCsvSourceFunction<TaskEvent>(taskInput) {
                    @Override
                    protected TaskEvent parse(String line) {
                        return TaskEvent.fromString(line);
                    }
                }))
                .filter(e -> e != null && e.getEventType() == TaskEvent.SUBMIT);

        // Assign watermarks based on timestamp (in microseconds, convert to millis)
        SingleOutputStreamOperator<TaskEvent> taskEventsWithWatermark = taskEvents
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(60))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000)
                );

        // Read job events (filter for finish/fail/kill)
        DataStream<JobEvent> jobEvents = env
                .addSource(jobSourceOrTest(new GzipCsvSourceFunction<JobEvent>(jobInput) {
                    @Override
                    protected JobEvent parse(String line) {
                        return JobEvent.fromString(line);
                    }
                }))
                .filter(e -> e != null && e.isFinished());

        SingleOutputStreamOperator<JobEvent> jobEventsWithWatermark = jobEvents
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofSeconds(60))
                                .withTimestampAssigner((event, timestamp) -> event.getTimestamp() / 1000)
                );

        // Connect task events and job events, keyed by job ID
        DataStream<Tuple2<Long, Long>> result = taskEventsWithWatermark
                .keyBy(TaskEvent::getJobId)
                .connect(jobEventsWithWatermark.keyBy(JobEvent::getJobId))
                .process(new SessionProcessFunction(sessionGapMicros));

        // Output result
        result.map(new MapFunction<Tuple2<Long, Long>, String>() {
            @Override
            public String map(Tuple2<Long, Long> value) {
                return "(" + value.f0 + "," + value.f1 + ")";
            }
        }).writeAsText(outputPath, FileSystem.WriteMode.OVERWRITE).setParallelism(1);

        env.execute("LongestSessionPerJob");
    }

    /**
     * Process function that tracks task SUBMIT events and job finish events.
     * Buffers all SUBMIT timestamps and computes sessions when job finishes.
     */
    public static class SessionProcessFunction
            extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Long>> {

        private final long sessionGapMicros;

        // State: list of all SUBMIT event timestamps for this job
        private transient ListState<Long> submitTimestamps;
        // State: whether job has finished
        private transient ValueState<Boolean> jobFinished;
        // State: whether result has been emitted
        private transient ValueState<Boolean> emitted;

        public SessionProcessFunction(long sessionGapMicros) {
            this.sessionGapMicros = sessionGapMicros;
        }

        @Override
        public void open(Configuration parameters) {
            submitTimestamps = getRuntimeContext().getListState(
                    new ListStateDescriptor<>("submitTimestamps", Types.LONG));
            jobFinished = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("jobFinished", Types.BOOLEAN));
            emitted = getRuntimeContext().getState(
                    new ValueStateDescriptor<>("emitted", Types.BOOLEAN));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            // Skip if already emitted
            if (Boolean.TRUE.equals(emitted.value())) {
                return;
            }

            // Buffer the timestamp
            submitTimestamps.add(event.getTimestamp());

            // Check if we can emit
            tryEmit(ctx, out);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            // Skip if already emitted
            if (Boolean.TRUE.equals(emitted.value())) {
                return;
            }

            jobFinished.update(true);
            tryEmit(ctx, out);
        }

        private void tryEmit(Context ctx, Collector<Tuple2<Long, Long>> out) throws Exception {
            if (Boolean.TRUE.equals(jobFinished.value()) && !Boolean.TRUE.equals(emitted.value())) {
                // Collect all timestamps and sort them
                List<Long> timestamps = new ArrayList<>();
                for (Long ts : submitTimestamps.get()) {
                    timestamps.add(ts);
                }

                if (timestamps.isEmpty()) {
                    // No task submits for this job, skip output
                    emitted.update(true);
                    return;
                }

                Collections.sort(timestamps);

                // Compute sessions and find the maximum
                long maxSessionCount = 0;
                long currentSessionCount = 1;

                for (int i = 1; i < timestamps.size(); i++) {
                    if (timestamps.get(i) - timestamps.get(i - 1) < sessionGapMicros) {
                        currentSessionCount++;
                    } else {
                        // Session ended
                        if (currentSessionCount > maxSessionCount) {
                            maxSessionCount = currentSessionCount;
                        }
                        currentSessionCount = 1;
                    }
                }
                // Don't forget the last session
                if (currentSessionCount > maxSessionCount) {
                    maxSessionCount = currentSessionCount;
                }

                out.collect(new Tuple2<>(ctx.getCurrentKey(), maxSessionCount));
                emitted.update(true);
                // Clean up state
                submitTimestamps.clear();
            }
        }
    }

    /**
     * Source function to read gzipped CSV files.
     */
    public static abstract class GzipCsvSourceFunction<T> implements SourceFunction<T> {
        private final String filePath;
        private volatile boolean isRunning = true;

        public GzipCsvSourceFunction(String filePath) {
            this.filePath = filePath;
        }

        protected abstract T parse(String line);

        @Override
        public void run(SourceContext<T> ctx) throws Exception {
            org.apache.flink.core.fs.Path path = new org.apache.flink.core.fs.Path(filePath);
            org.apache.flink.core.fs.FileSystem fs = path.getFileSystem();
            org.apache.flink.core.fs.FSDataInputStream inputStream = fs.open(path);

            try (BufferedReader reader = new BufferedReader(
                    new InputStreamReader(new GZIPInputStream(inputStream)))) {
                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    T event = parse(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            isRunning = false;
        }
    }
}
