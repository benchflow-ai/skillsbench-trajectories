"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import fitz  # PyMuPDF
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import DataStructs
from rdkit.Chem.rdFingerprintGenerator import GetMorganGenerator


def extract_molecules_from_pdf(filepath: str) -> list[str]:
    """
    Extract molecule names from a PDF file.

    Args:
        filepath: Path to the PDF file containing molecule names.

    Returns:
        List of molecule names extracted from the PDF.
    """
    doc = fitz.open(filepath)
    molecules = []

    for page in doc:
        text = page.get_text()
        # Split by newlines and filter empty lines
        lines = [line.strip() for line in text.split('\n') if line.strip()]
        molecules.extend(lines)

    doc.close()
    return molecules


def get_smiles_from_pubchem(molecule_name: str) -> str | None:
    """
    Get SMILES representation of a molecule from PubChem.

    Args:
        molecule_name: Common name of the molecule.

    Returns:
        SMILES string if found, None otherwise.
    """
    try:
        compounds = pcp.get_compounds(molecule_name, 'name')
        if compounds:
            # Use 'smiles' which returns the isomeric SMILES (preserves chirality)
            return compounds[0].smiles
    except Exception:
        pass
    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES representation of the molecule.
        radius: Radius for Morgan fingerprint (default: 2).
        use_chirality: Whether to include chirality in fingerprint (default: True).

    Returns:
        Morgan fingerprint object or None if molecule cannot be parsed.
    """
    mol = Chem.MolFromSmiles(smiles)
    if mol is None:
        return None

    generator = GetMorganGenerator(
        radius=radius,
        includeChirality=use_chirality,
        fpSize=2048
    )
    return generator.GetFingerprint(mol)


def tanimoto_similarity(fp1, fp2) -> float:
    """
    Compute Tanimoto similarity between two fingerprints.

    Args:
        fp1: First fingerprint.
        fp2: Second fingerprint.

    Returns:
        Tanimoto similarity score (0.0 to 1.0).
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top k most similar molecules to a target molecule.

    Args:
        target_molecule_name: Name of the target molecule to compare against.
        molecule_pool_filepath: Path to PDF file containing pool of molecule names.
        top_k: Number of top similar molecules to return.

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity
        in descending order, with alphabetical ordering for ties.
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target molecule
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for mol_name in pool_molecules:
        smiles = get_smiles_from_pubchem(mol_name)
        if smiles is None:
            continue

        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        similarity = tanimoto_similarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) < 4:
        print("Usage: python solution.py <target_molecule> <pdf_path> <top_k>")
        sys.exit(1)

    target = sys.argv[1]
    pdf_path = sys.argv[2]
    k = int(sys.argv[3])

    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules to {target}:")
    for i, (name, score) in enumerate(results, 1):
        print(f"{i}. {name}: {score:.4f}")
