                            int schedClass = parts[7].isEmpty() ? 0 : Integer.parseInt(parts[7]);
                            int priority = parts[8].isEmpty() ? 0 : Integer.parseInt(parts[8]);
                            double cpuRequested = parts[9].isEmpty() ? 0.0 : Double.parseDouble(parts[9]);
                            double memRequested = parts[10].isEmpty() ? 0.0 : Double.parseDouble(parts[10]);
                            double disk = parts[11].isEmpty() ? 0.0 : Double.parseDouble(parts[11]);
                            int restriction = parts[12].isEmpty() ? 0 : Integer.parseInt(parts[12]);

                            TaskEvent event = new TaskEvent(time, missing, jobId, taskIdx, machineId,

    public static class JobEventSource implements SourceFunction<JobEvent> {
        private String filePath;
        private volatile boolean isRunning = true;

        public JobEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceContext<JobEvent> ctx) throws Exception {
            try (java.util.zip.GZIPInputStream gzip = new java.util.zip.GZIPInputStream(
                    new java.io.FileInputStream(filePath));
                 java.io.BufferedReader reader = new java.io.BufferedReader(
                         new java.io.InputStreamReader(gzip))) {

                String line;
                while (isRunning && (line = reader.readLine()) != null) {
                    String[] parts = line.split(",");
                    if (parts.length >= 8) {
                        try {
                            long time = Long.parseLong(parts[0]);
                            String missing = parts[1].isEmpty() ? "" : parts[1];
                            long jobId = Long.parseLong(parts[2]);
                            int eventType = Integer.parseInt(parts[3]);
                            String user = parts[4].isEmpty() ? "" : parts[4];
                            int schedClass = parts[5].isEmpty() ? 0 : Integer.parseInt(parts[5]);
                            String jobName = parts[6].isEmpty() ? "" : parts[6];
                            String logicalJobName = parts[7].isEmpty() ? "" : parts[7];

                            JobEvent event = new JobEvent(time, missing, jobId, eventType, user,
                                    schedClass, jobName, logicalJobName);
                            ctx.collectWithTimestamp(event, time);
                        } catch (NumberFormatException e) {
                            // Skip malformed lines
                        }
                    }
                }
            }
        }

        @Override

        }

    public static class JobState implements Serializable {
        public List<Long> submitTimes = new ArrayList<>();
        public boolean finished = false;
        public int maxTasksInLongestStage = 0;

        public void calculateLongestStage(long sesssize) {
            if (submitTimes.isEmpty()) {
                maxTasksInLongestStage = 0;
                return;
            }

            Collections.sort(submitTimes);

            int maxTasksInStage = 0;
            int currentStageCount = 1;

            for (int i = 1; i < submitTimes.size(); i++) {
                long timeDiff = submitTimes.get(i) - submitTimes.get(i - 1);
                if (timeDiff > sesssize) {
                    // New stage starts
                    maxTasksInStage = Math.max(maxTasksInStage, currentStageCount);
                    currentStageCount = 1;
                } else {
                    currentStageCount++;
                }
            }
            maxTasksInStage = Math.max(maxTasksInStage, currentStageCount);

            maxTasksInLongestStage = maxTasksInStage;
        }
    }

    public static class StageProcessor extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {
        private long sesssize;
        private transient org.apache.flink.api.common.state.ValueState<JobState> jobStateState;

        public StageProcessor(long sesssize) {
            this.sesssize = sesssize;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            super.open(parameters);
            org.apache.flink.api.common.state.ValueStateDescriptor<JobState> descriptor =
                    new org.apache.flink.api.common.state.ValueStateDescriptor<>(
                            "job-state",
                            JobState.class);
            jobStateState = getRuntimeContext().getState(descriptor);
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Process task events - only SUBMIT events (type 0)
            if (event.eventType != 0) {
                return;
            }

            JobState state = jobStateState.value();
            if (state == null) {
                state = new JobState();
            }

            state.submitTimes.add(event.time);
            jobStateState.update(state);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Process job events - only FINISH events (type 4)
            if (event.eventType != 4) {
                return;
            }

            JobState state = jobStateState.value();
            if (state == null) {
                state = new JobState();
            }

            // Calculate the longest stage when job finishes
            state.calculateLongestStage(sesssize);
            out.collect(Tuple2.of(event.jobId, state.maxTasksInLongestStage));

            // Clean up state
            jobStateState.clear();
        }
    }

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        System.out.println("task_input  " + taskInput);
        System.out.println("job_input  " + jobInput);
        final long sesssize = params.getLong("sesssize", 600) * 1_000_000L; // Convert to microseconds

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        // Read task events
        SourceFunction<TaskEvent> taskSource = new TaskEventSource(taskInput);
        SourceFunction<TaskEvent> finalTaskSource = taskSourceOrTest(taskSource);

        // Read job events
        SourceFunction<JobEvent> jobSource = new JobEventSource(jobInput);
        SourceFunction<JobEvent> finalJobSource = jobSourceOrTest(jobSource);

        // Create task stream
        var taskStream = env.addSource(finalTaskSource)
                .keyBy(event -> event.jobId);

        // Create job stream
        var jobStream = env.addSource(finalJobSource)
                .keyBy(event -> event.jobId);

        // Join the two streams using co-process function
        var results = taskStream.connect(jobStream)
                .process(new StageProcessor(sesssize));

        // Write results to file
        SinkFunction<Tuple2<Long, Integer>> sink = new TupleSink(outputPath);
        SinkFunction finalSink = sinkOrTest(sink);

        results.addSink((SinkFunction<Tuple2<Long, Integer>>) finalSink);

        env.execute("LongestSessionPerJob");
    }
