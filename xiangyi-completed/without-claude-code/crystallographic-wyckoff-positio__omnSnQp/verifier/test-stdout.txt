"""
Analyze Wyckoff position multiplicities and coordinates from CIF files.
"""

from fractions import Fraction
from typing import Any
from collections import defaultdict

import spglib
from pymatgen.core import Structure


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Parameters
    ----------
    filepath : str
        Path to the CIF file to analyze

    Returns
    -------
    dict
        Dictionary containing:
        - "wyckoff_multiplicity_dict": dict mapping Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": dict mapping Wyckoff letter to coordinates
          of the first atom at that position (as fractions with denominators ≤ 12)

    Example
    -------
    >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
    ...     "/root/cif_files/FeS2_mp-226.cif"
    ... )
    >>> print(result)
    {
        "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
        "wyckoff_coordinates_dict": {
            "a": ["0", "1/2", "1/2"],
            "c": ["3/8", "1/9", "8/9"]
        }
    }
    """
    # Read the CIF file using pymatgen
    structure = Structure.from_file(filepath)

    # Prepare data for spglib
    lattice = structure.lattice.matrix
    positions = structure.frac_coords
    numbers = [site.species.elements[0].Z for site in structure]

    # Get space group and Wyckoff information from spglib
    dataset = spglib.get_symmetry_dataset((lattice, positions, numbers))

    # Get Wyckoff letters and standardized positions
    wyckoff_letters = dataset.wyckoffs
    std_positions = dataset.std_positions

    # Count multiplicities for each Wyckoff letter
    multiplicity_count = defaultdict(int)
    for letter in wyckoff_letters:
        multiplicity_count[letter] += 1

    wyckoff_multiplicity_dict = dict(multiplicity_count)

    # Get first coordinates for each Wyckoff letter
    wyckoff_first_coords = {}
    for letter, position in zip(wyckoff_letters, std_positions):
        if letter not in wyckoff_first_coords:
            wyckoff_first_coords[letter] = position

    # Convert coordinates to rational fractions with denominator ≤ 12
    wyckoff_coordinates_dict = {}
    for wyckoff_letter, coords in wyckoff_first_coords.items():
        frac_coords = []
        for coord in coords:
            # Convert to fraction with max denominator of 12
            frac = Fraction(coord).limit_denominator(12)
            # Format as string
            if frac.denominator == 1:
                frac_str = str(frac.numerator)
            else:
                frac_str = f"{frac.numerator}/{frac.denominator}"
            frac_coords.append(frac_str)
        wyckoff_coordinates_dict[wyckoff_letter] = frac_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }
