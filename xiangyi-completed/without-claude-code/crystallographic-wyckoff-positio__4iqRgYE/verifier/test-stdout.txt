"""
Solution for analyzing Wyckoff position multiplicities and coordinates from CIF files.

This module uses pymatgen to:
1. Read CIF files and parse crystal structure information
2. Analyze space group symmetry using SpacegroupAnalyzer
3. Identify Wyckoff positions and their multiplicities
4. Extract coordinates for the first atom of each Wyckoff position
5. Convert coordinates to simplified fraction format (max denominator = 12)

Example usage:
    result = analyze_wyckoff_position_multiplicities_and_coordinates("path/to/file.cif")
    # Returns:
    # {
    #     "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
    #     "wyckoff_coordinates_dict": {
    #         "a": ["0", "1/2", "1/2"],
    #         "c": ["3/8", "1/9", "8/9"]
    #     }
    # }
"""

from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def simplify_coordinate(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float coordinate to a simplified fraction string.

    Args:
        value: The coordinate value (should be between 0 and 1)
        max_denominator: Maximum denominator for the fraction

    Returns:
        String representation of the coordinate as a fraction or integer
    """
    # Normalize to [0, 1)
    value = value % 1.0

    # Round very small values to 0
    if abs(value) < 1e-6:
        return "0"

    # Round values very close to 1 to 0
    if abs(value - 1.0) < 1e-6:
        return "0"

    # Convert to fraction
    frac = Fraction(value).limit_denominator(max_denominator)

    # Handle the case where fraction is 1 (should be 0 in periodic coordinates)
    if frac.numerator == frac.denominator:
        return "0"

    if frac.denominator == 1:
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate coordinates for the first atom
    of each Wyckoff position from a CIF file.

    Args:
        filepath: Path to the CIF file

    Returns:
        Dictionary containing:
        - "wyckoff_multiplicity_dict": Dict mapping Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": Dict mapping Wyckoff letter to coordinates [x, y, z]
    """
    # Read the structure from CIF file
    structure = Structure.from_file(filepath)

    # Get the symmetrized structure to find Wyckoff positions
    sga = SpacegroupAnalyzer(structure)
    symmetrized_structure = sga.get_symmetrized_structure()

    # Get Wyckoff symbols and equivalent sites
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols
    equivalent_sites = symmetrized_structure.equivalent_sites

    # Dictionary to store multiplicity and first coordinates for each Wyckoff position
    wyckoff_data = {}

    # Iterate through Wyckoff positions
    for wyckoff_symbol, equiv_sites_list in zip(wyckoff_symbols, equivalent_sites):
        # Parse the Wyckoff symbol to extract letter (e.g., "4a" -> "a")
        # The format is typically: <multiplicity><letter>
        wyckoff_letter = ''.join(c for c in wyckoff_symbol if c.isalpha())

        # Get multiplicity from the Wyckoff symbol
        multiplicity = int(''.join(c for c in wyckoff_symbol if c.isdigit()))

        # Get the first site's coordinates
        first_site = equiv_sites_list[0]
        coords = first_site.frac_coords

        # Store data if not already present (only keep first occurrence)
        if wyckoff_letter not in wyckoff_data:
            wyckoff_data[wyckoff_letter] = {
                'multiplicity': multiplicity,
                'coordinates': coords
            }

    # Build output dictionaries
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for wyckoff_letter, data in wyckoff_data.items():
        wyckoff_multiplicity_dict[wyckoff_letter] = data['multiplicity']

        # Convert coordinates to simplified fraction strings
        coords_str = [
            simplify_coordinate(data['coordinates'][0]),
            simplify_coordinate(data['coordinates'][1]),
            simplify_coordinate(data['coordinates'][2])
        ]
        wyckoff_coordinates_dict[wyckoff_letter] = coords_str

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
