def load_map(db_path):
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    cursor = conn.cursor()

    cursor.execute("SELECT * FROM Map LIMIT 1")
    row = cursor.fetchone()
    width = row['Width']
    height = row['Height']

    tiles_data = {}
    cursor.execute("SELECT * FROM Plots")
    for row in cursor.fetchall():
        id = row['ID']
        x = id % width
        y = id // width
        
        terrain_raw = row['TerrainType']
        is_mountain = 'MOUNTAIN' in terrain_raw
        terrain = terrain_raw.replace('TERRAIN_', '').replace('_MOUNTAIN', '').replace('_HILLS', '')
        if terrain == 'GRASS': terrain = 'GRASS' # Keep it consistent
        
        tiles_data[(x, y)] = {
            'x': x, 'y': y,
            'terrain': terrain,
            'is_mountain': is_mountain,
            'is_hills': 'HILLS' in terrain_raw,
            'feature': None,
            'is_floodplains': False,
            'resource': None,
            'resource_type': None,
            'river_edges': [],
            'improvement': None
        }

    cursor.execute("SELECT * FROM PlotFeatures")
    for row in cursor.fetchall():
        id = row['ID']
        x = id % width
        y = id // width
        feature = row['FeatureType']
        tiles_data[(x, y)]['feature'] = feature
        if 'FLOODPLAINS' in feature:
            tiles_data[(x, y)]['is_floodplains'] = True

    cursor.execute("SELECT * FROM PlotRivers")
    for row in cursor.fetchall():
        id = row['ID']
        x = id % width
        y = id // width
        edges = []
        if row['IsNEOfRiver']: edges.append(1)
        if row['IsNWOfRiver']: edges.append(2)
        if row['IsWOfRiver']: edges.append(3)
        tiles_data[(x, y)]['river_edges'].extend(edges)
        
        # Add shared edges to neighbors
        # Edge 0 (E): Neighbor(direction 0).IsWOfRiver
        # Edge 5 (SE): Neighbor(direction 5).IsNWOfRiver
        # Edge 4 (SW): Neighbor(direction 4).IsNEOfRiver
        # We handle this during Tile object creation to avoid neighbor lookup issues here

    conn.close()

    # Finalize river edges and create Tile objects
    tiles = {}
    for (x, y), data in tiles_data.items():
        # Complete river edges from neighbors
        edges = set(data['river_edges'])
        
        # Neighbor E (0)
        nx, ny = x + 1, y # Simple East check, but should use get_neighbor_at_direction
        # Actually let's use the proper hex_utils
        
    # Re-do river edges properly
    from hex_utils import DIRECTIONS_EVEN_ROW, DIRECTIONS_ODD_ROW
    
    for (x, y), data in tiles_data.items():
        edges = set(data['river_edges'])
        
        # Check neighbors for shared river edges
        # Edge 0 (E): Neighbor(direction 0) is W (3)
        # Edge 5 (SE): Neighbor(direction 5) is NW (2)
        # Edge 4 (SW): Neighbor(direction 4) is NE (1)
        
        directions = DIRECTIONS_ODD_ROW if y % 2 == 1 else DIRECTIONS_EVEN_ROW
        
        # East (0)
        nx, ny = x + directions[0][0], y + directions[0][1]
        if (nx, ny) in tiles_data and 3 in tiles_data[(nx, ny)]['river_edges']:
            edges.add(0)
            
        # Southeast (5)
        nx, ny = x + directions[5][0], y + directions[5][1]
        if (nx, ny) in tiles_data and 2 in tiles_data[(nx, ny)]['river_edges']:
            edges.add(5)
            
        # Southwest (4)
        nx, ny = x + directions[4][0], y + directions[4][1]
        if (nx, ny) in tiles_data and 1 in tiles_data[(nx, ny)]['river_edges']:
            edges.add(4)
            
        data['river_edges'] = list(edges)
        
        # Create Tile object
        # Note: Tile class doesn't have is_mountain directly in __init__ but as property?
        # Looking at placement_rules.py:
        # Tile(x, y, terrain, feature, is_hills, is_floodplains, river_edges, river_names, resource, resource_type, improvement)
        # Wait, the Tile class I saw earlier:
        # @dataclass
        # class Tile:
        #     x: int
        #     y: int
        #     terrain: str
        #     feature: Optional[str] = None
        #     is_hills: bool = False
        #     is_floodplains: bool = False
        #     river_edges: List[int] = None
        #     river_names: List[str] = None
        #     resource: Optional[str] = None
        #     resource_type: Optional[str] = None
        #     improvement: Optional[str] = None
        
        # And is_mountain is a property:
        # @property
        # def is_mountain(self) -> bool:
        #     return self.terrain == "MOUNTAIN"
        
        terrain_obj = data['terrain']
        if data['is_mountain']:
            terrain_obj = 'MOUNTAIN'
            
        tiles[(x, y)] = Tile(
            x=x, y=y,
            terrain=terrain_obj,
            feature=data['feature'],
            is_hills=data['is_hills'],
            is_floodplains=data['is_floodplains'],
            river_edges=data['river_edges'],
            resource=data['resource'],
            resource_type=data['resource_type'],
            improvement=data['improvement']
        )
        
    return tiles, width, height

tiles, width, height = load_map('/data/maps/e2e_test_case_0.Civ6Map')

# Strategy:
# Pop 9 allows 3 specialty districts.
# We also want Aqueduct (+2 to IZ) and maybe Dam (+2 to IZ).
# IZ should be adjacent to Aqueduct and Dam.
# Campus should be adjacent to Mountains/Geothermal.
# Commercial Hub on a river.

best_total = -1
best_solution = None

# Sample some promising tiles (non-water, non-mountain)
promising_spots = []
for (x, y), tile in tiles.items():
    if not tile.is_water and not tile.is_mountain and tile.feature != 'FEATURE_ICE':
        promising_spots.append((x, y))

# To speed up, we can't try EVERY settlement and EVERY combination.
# Let's focus on areas with many mountains or geothermal for Campus.
def score_location_for_campus(x, y):
    score = 0
    for nx, ny in get_neighbors(x, y):
        ntile = tiles.get((nx, ny))
        if ntile:
            if ntile.is_mountain: score += 1
            if ntile.feature == 'FEATURE_GEOTHERMAL_FISSURE': score += 2
            if ntile.feature == 'FEATURE_REEF': score += 2
    return score

top_campus_spots = sorted(promising_spots, key=lambda p: score_location_for_campus(p[0], p[1]), reverse=True)[:50]

for cc_pos in top_campus_spots:
    # Try placing districts around this CC
    # Candidates: CAMPUS, INDUSTRIAL_ZONE, COMMERCIAL_HUB, AQUEDUCT, DAM
    rules = get_placement_rules(tiles, cc_pos, 9)
    
    # Simple greedy placement for a few districts
    potential_placements = {cc_pos: DistrictType.CITY_CENTER}
    
    # 1. Place Aqueduct (if possible) to boost IZ
    best_aq = None
    for nx, ny in get_neighbors(cc_pos[0], cc_pos[1]):
        res = rules.validate_placement(DistrictType.AQUEDUCT, nx, ny, potential_placements)
        if res.valid:
            best_aq = (nx, ny)
            break
    
    if best_aq:
        potential_placements[best_aq] = DistrictType.AQUEDUCT
        
    # 2. Place Dam (if possible)
    best_dam = None
    # Check all tiles within range 3 of CC
    for dx in range(-3, 4):
        for dy in range(-3, 4):
            x, y = cc_pos[0] + dx, cc_pos[1] + dy
            if hex_distance(cc_pos[0], cc_pos[1], x, y) <= 3:
                res = rules.validate_placement(DistrictType.DAM, x, y, potential_placements)
                if res.valid:
                    best_dam = (x, y)
                    break
        if best_dam: break
    if best_dam:
        potential_placements[best_dam] = DistrictType.DAM

    # 3. Place Industrial Zone (adjacent to Aqueduct and Dam)
    best_iz = None
    best_iz_score = -1
    for dx in range(-3, 4):
        for dy in range(-3, 4):
            x, y = cc_pos[0] + dx, cc_pos[1] + dy
            if hex_distance(cc_pos[0], cc_pos[1], x, y) <= 3:
                res = rules.validate_placement(DistrictType.INDUSTRIAL_ZONE, x, y, potential_placements)
                if res.valid:
                    # Score based on potential adjacency
                    score = 0
                    for nx, ny in get_neighbors(x, y):
                        if (nx, ny) == best_aq: score += 2
                        if (nx, ny) == best_dam: score += 2
                    if score > best_iz_score:
                        best_iz_score = score
                        best_iz = (x, y)
    if best_iz:
        potential_placements[best_iz] = DistrictType.INDUSTRIAL_ZONE

    # 4. Place Campus (near mountains)
    best_campus = None
    best_campus_score = -1
    for dx in range(-3, 4):
        for dy in range(-3, 4):
            x, y = cc_pos[0] + dx, cc_pos[1] + dy
            if hex_distance(cc_pos[0], cc_pos[1], x, y) <= 3:
                res = rules.validate_placement(DistrictType.CAMPUS, x, y, potential_placements)
                if res.valid:
                    score = score_location_for_campus(x, y)
                    if score > best_campus_score:
                        best_campus_score = score
                        best_campus = (x, y)
    if best_campus:
        potential_placements[best_campus] = DistrictType.CAMPUS

    # 5. Place Commercial Hub (near river and Harbor/CC)
    best_ch = None
    best_ch_score = -1
    for dx in range(-3, 4):
        for dy in range(-3, 4):
            x, y = cc_pos[0] + dx, cc_pos[1] + dy
            if hex_distance(cc_pos[0], cc_pos[1], x, y) <= 3:
                res = rules.validate_placement(DistrictType.COMMERCIAL_HUB, x, y, potential_placements)
                if res.valid:
                    score = 0
                    tile = tiles[(x, y)]
                    if tile.has_river: score += 2
                    # Adjacent to CC?
                    if hex_distance(x, y, cc_pos[0], cc_pos[1]) == 1:
                        # Minor bonus from CC is 0.5, needs 2 districts.
                        pass
                    if score > best_ch_score:
                        best_ch_score = score
                        best_ch = (x, y)
    if best_ch:
        potential_placements[best_ch] = DistrictType.COMMERCIAL_HUB

    # Calculate total adjacency
    calc = get_adjacency_calculator(tiles)
    total, per_district = calc.calculate_total_adjacency(potential_placements)
    
    if total > best_total:
        best_total = total
        best_solution = {
            "city_center": cc_pos,
            "placements": {d.name: [coord[0], coord[1]] for coord, d in potential_placements.items() if d != DistrictType.CITY_CENTER},
            "adjacency_bonuses": {k.split('@')[0]: v.total_bonus for k, v in per_district.items()},
            "total_adjacency": total
        }

# Output solution
print(json.dumps(best_solution, indent=2))
