import re
from fractions import Fraction
from typing import Any, Dict

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    # Load structure from CIF
    s = Structure.from_file(filepath)
    
    # Analyze symmetry
    # Default tolerance symprec=0.01 is usually sufficient for MP data
    sga = SpacegroupAnalyzer(s)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # sym_struct.equivalent_indices is a list of lists.
    # Each sublist contains indices of atoms that are symmetry equivalent.
    # wyckoff_symbols is a list of symbols corresponding to each group in equivalent_indices.
    
    for i, indices in enumerate(sym_struct.equivalent_indices):
        # Take the first index as representative
        idx = indices[0]
        
        # Get the Wyckoff symbol (e.g., '4a')
        # wyckoff_symbols corresponds to the groups in equivalent_indices
        symbol = sym_struct.wyckoff_symbols[i]
        
        # Parse the multiplicity and letter
        # Regex to capture number and letter(s)
        match = re.match(r"(\d+)([a-zA-Z]+)", symbol)
        if not match:
            continue
            
        multiplicity = int(match.group(1))
        letter = match.group(2)
        
        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get fractional coordinates of the representative atom
        frac_coords = sym_struct[idx].frac_coords
        
        # Convert coordinates to rational approximations (denom <= 12)
        rational_coords = []
        for coord in frac_coords:
            # We want to represent the coordinate as a simple fraction or integer.
            # Using limit_denominator(12) as requested.
            
            # Note: The coordinate might be slightly off due to numerical precision.
            f = Fraction(coord).limit_denominator(12)
            
            # Convert Fraction to string format
            if f.denominator == 1:
                rational_coords.append(str(f.numerator))
            else:
                rational_coords.append(f"{f.numerator}/{f.denominator}")
        
        wyckoff_coordinates_dict[letter] = rational_coords
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
