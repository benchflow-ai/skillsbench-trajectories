import pandas as pd

def get_turn_scores(rolls):
    scores = [0] * 6
    
    # 1. High and Often
    m = max(rolls)
    scores[0] = m * rolls.count(m)
    
    # 2. Summation
    scores[1] = sum(rolls)
    
    # 3. Highs and Lows
    hi = max(rolls)
    lo = min(rolls)
    scores[2] = hi * lo * (hi - lo)
    
    # 4. Only two numbers
    if len(set(rolls)) <= 2:
        scores[3] = 30
    
    # 5. All the numbers
    if sorted(rolls) == [1, 2, 3, 4, 5, 6]:
        scores[4] = 40
        
    # 6. Ordered subset of four
    has_subset = False
    for i in range(len(rolls) - 3):
        sub = rolls[i:i+4]
        # Check increasing
        if sub[1] == sub[0] + 1 and sub[2] == sub[1] + 1 and sub[3] == sub[2] + 1:
            has_subset = True
            break
        # Check decreasing
        if sub[1] == sub[0] - 1 and sub[2] == sub[1] - 1 and sub[3] == sub[2] - 1:
            has_subset = True
            break
    if has_subset:
        scores[5] = 50
        
    return scores

            break
        # Check decreasing
        if sub[1] == sub[0] - 1 and sub[2] == sub[1] - 1 and sub[3] == sub[2] - 1:
            has_subset = True
            break
