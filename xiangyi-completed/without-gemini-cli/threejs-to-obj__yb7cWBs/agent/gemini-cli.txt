I have created a script, `convert.js`, that exports the Three.js object from `/root/data/object.js` to `/root/output/object.obj`. This script handles the conversion of `InstancedMesh` to regular meshes to ensure all parts are exported, and it applies a -90 degree rotation around the X-axis to align the object with Blender's Z-up coordinate system while preserving the relative 3D positions of the components.

```javascript
import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';

// OBJExporter doesn't natively support InstancedMesh in many versions of Three.js.
// We convert InstancedMesh to regular Meshes to ensure they are exported.
function convertInstancedMeshes(root) {
  const instancedMeshes = [];
  root.traverse((child) => {
    if (child.isInstancedMesh) {
      instancedMeshes.push(child);
    }
  });

  instancedMeshes.forEach((imesh) => {
    const group = new THREE.Group();
    group.name = imesh.name + '_converted';
    group.copy(imesh, false); // Copy transform but not children/geometry

    for (let i = 0; i < imesh.count; i++) {
      const matrix = new THREE.Matrix4();
      imesh.getMatrixAt(i, matrix);
      const mesh = new THREE.Mesh(imesh.geometry, imesh.material);
      mesh.name = imesh.name + '_' + i;
      mesh.applyMatrix4(matrix);
      group.add(mesh);
    }

    if (imesh.parent) {
      imesh.parent.add(group);
      imesh.parent.remove(imesh);
    }
  });
}

const scene = createScene();

// Handle InstancedMesh
convertInstancedMeshes(scene);

// Apply -90 degrees X-rotation to convert to Blender Z-up space
scene.rotation.x = -Math.PI / 2;
scene.updateMatrixWorld(true);

const exporter = new OBJExporter();
const result = exporter.parse(scene);

// Ensure output directory exists
if (!fs.existsSync('/root/output')) {
  fs.mkdirSync('/root/output', { recursive: true });
}

fs.writeFileSync('/root/output/object.obj', result);
```
