import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs
import time
import sys

def get_molecules_from_pdf(filepath):
    molecules = []
    try:
        reader = pypdf.PdfReader(filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                lines = text.split('\n')
                for line in lines:
                    line = line.strip()
                    if line:
                        molecules.append(line)
    except Exception as e:
        print(f"Error reading PDF: {e}", file=sys.stderr)
    return molecules

def get_smiles(name):
    retries = 5
    for i in range(retries):
        try:
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                return compounds[0].isomeric_smiles
            return None
        except Exception as e:
            # Check for 503 or "ServerBusy" in the string representation
            if "503" in str(e) or "ServerBusy" in str(e):
                time.sleep(1.0 * (i + 1)) # Exponential-ish backoff
                continue
            else:
                print(f"Error fetching SMILES for {name}: {e}", file=sys.stderr)
                break
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # 1. Get Target SMILES
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Invalid SMILES for target molecule: {target_molecule_name}")

    # 2. Get Molecule Pool
    pool_names = get_molecules_from_pdf(molecule_pool_filepath)
    
    # 3. Compute Similarities
    # Use rdFingerprintGenerator to avoid deprecation warnings
    # Morgan fingerprint (radius=2) equivalent to ECFP4
    mf_gen = rdFingerprintGenerator.GetMorganGenerator(radius=2, includeChirality=True)
    target_fp = mf_gen.GetFingerprint(target_mol)
    
    results = []
    
    for name in pool_names:
        if not name:
            continue
            
        # Small delay to be polite to PubChem API
        time.sleep(0.25)
        
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = mf_gen.GetFingerprint(mol)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results.append((name, similarity))
        
    # 4. Sort
    # Descending order of similarity (-x[1])
    # Alphabetical ordering when ties happen (x[0])
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Top K
    top_results = [x[0] for x in results[:top_k]]
    
    return top_results
