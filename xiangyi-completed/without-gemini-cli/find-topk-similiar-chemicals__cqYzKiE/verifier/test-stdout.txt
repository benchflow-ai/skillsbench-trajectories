import os
import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs
from rdkit import RDLogger
from concurrent.futures import ThreadPoolExecutor

# Suppress RDKit warnings
RDLogger.DisableLog('rdApp.*')

def get_smiles(name):
    """
    Retrieves the SMILES string for a given chemical name using PubChemPy.
    """
    try:
        # Search for the compound by name
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            # Return the canonical SMILES
            return compounds[0].smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF molecule pool to a target molecule.
    Similarity is calculated using Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Returns a list of chemical names.
    """
    if not os.path.exists(molecule_pool_filepath):
        return []

    # Extract names from PDF file
    pool_names = []
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Assume each line is a chemical name
                    lines = [line.strip() for line in text.split('\n') if line.strip()]
                    pool_names.extend(lines)
    except Exception:
        return []
    
    if not pool_names:
        return []

    # Get SMILES for the target molecule
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Calculate Morgan fingerprint for target (radius=2, include chirality)
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # Get unique names from the pool to minimize PubChem API calls
    unique_pool_names = sorted(list(set(pool_names)))
    name_to_smiles = {}

    def fetch_smiles(name):
        return name, get_smiles(name)

    # Use ThreadPoolExecutor for faster retrieval of SMILES strings from PubChem
    with ThreadPoolExecutor(max_workers=5) as executor:
        for name, smiles in executor.map(fetch_smiles, unique_pool_names):
            if smiles:
                name_to_smiles[name] = smiles
            
    results_map = {}
    for name in unique_pool_names:
        smiles = name_to_smiles.get(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        # Calculate Morgan fingerprint (radius=2, include chirality)
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        # Calculate Tanimoto similarity
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        results_map[name] = similarity
    
    # Create results list based on original names in the pool (if they have SMILES)
    # The prompt says "top k similar chemicals in molecules.pdf". 
    # If a name appeared multiple times, we should probably return it only once in the unique ranking,
    # or follow the occurrences. Usually "top k chemicals" refers to unique entities.
    # Given the sorting requirement, we'll use the unique names that were successfully resolved.
    
    final_results = []
    for name in results_map:
        final_results.append((name, results_map[name]))
    
    # Sort by similarity descending, then by name alphabetical (ascending)
    final_results.sort(key=lambda x: (-x[1], x[0]))
    
    # Return the top k chemical names
    return [item[0] for item in final_results[:top_k]]
