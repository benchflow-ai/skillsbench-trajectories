import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem
import os

def extract_names_from_pdf(pdf_path):
    names = []
    if not os.path.exists(pdf_path):
        return names
    try:
        with pdfplumber.open(pdf_path) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    lines = text.split('\n')
                    for line in lines:
                        name = line.strip()
                        if name:
                            names.append(name)
    except Exception:
        pass
    return names

_smiles_cache = {}

def get_smiles(name):
    if name in _smiles_cache:
        return _smiles_cache[name]
    
    try:
        results = pcp.get_compounds(name, 'name')
        if results:
            smiles = results[0].smiles
            _smiles_cache[name] = smiles
            return smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k) -> list:
    """
    Finds the top k similar chemicals in a PDF pool to a target molecule.
    Returns a list of chemical names.
    """
    # 1. Get target SMILES
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    # Morgan fingerprints with radius 2, including chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, nBits=2048, useChirality=True)
    
    # 2. Extract names from PDF
    pool_names = extract_names_from_pdf(molecule_pool_filepath)
    unique_pool_names = []
    seen = set()
    for n in pool_names:
        if n not in seen:
            unique_pool_names.append(n)
            seen.add(n)
    
    # 3. Get SMILES and compute similarities
    results = []
    for name in unique_pool_names:
        smiles = get_smiles(name)
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, nBits=2048, useChirality=True)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                results.append((name, similarity))
    
    # 4. Sort results: similarity descending, then name ascending
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k names
    return [name for name, sim in results[:top_k]]

if __name__ == "__main__":
    import sys
    if len(sys.argv) > 1:
        target = sys.argv[1]
        k = int(sys.argv[2]) if len(sys.argv) > 2 else 5
        res = topk_tanimoto_similarity_molecules(target, "molecules.pdf", k)
        for name in res:
            print(name)
