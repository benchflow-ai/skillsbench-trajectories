import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from rdkit import RDLogger
import time

# Suppress RDKit warnings
RDLogger.DisableLog('rdApp.*')

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF to a target chemical.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing the list of molecules.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: List of top k similar molecule names.
    """
    
    # Read PDF
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        text = ""
        for page in reader.pages:
            text += page.extract_text() + "\n"
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []
    
    # Extract names (assume one per line, trim whitespace)
    # Filter out empty lines
    molecule_names = [line.strip() for line in text.splitlines() if line.strip()]
    # Remove duplicates
    molecule_names = list(set(molecule_names))

    # Get SMILES for target
    try:
        target_compounds = pcp.get_compounds(target_molecule_name, 'name')
        if not target_compounds:
            # Try to handle case insensitivity or small variations if needed, but strict for now
            print(f"Could not find target molecule: {target_molecule_name}")
            return []
        
        target_smiles = target_compounds[0].smiles
        target_mol = Chem.MolFromSmiles(target_smiles)
        
        if target_mol is None:
             print(f"Could not parse SMILES for target: {target_molecule_name}")
             return []
             
        # Radius 2, include chirality
        target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
        
    except Exception as e:
        print(f"Error processing target molecule {target_molecule_name}: {e}")
        return []

    results = []

    # Get SMILES for pool
    for name in molecule_names:
        # Rate limiting sleep
        time.sleep(0.2)
        try:
            compounds = pcp.get_compounds(name, 'name')
            if not compounds:
                # Sometimes names in PDF might have OCR errors or slight format diffs.
                # We skip if not found.
                # print(f"Warning: Could not find molecule: {name}")
                continue
            
            smiles = compounds[0].smiles
            mol = Chem.MolFromSmiles(smiles)
            if mol is None:
                # print(f"Warning: Could not parse SMILES for: {name}")
                continue
            
            fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
            similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
            results.append((name, similarity))
            
        except Exception as e:
            # print(f"Error processing {name}: {e}")
            continue
    
    # Sort: descending similarity (-x[1]), then alphabetical name (x[0])
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # Extract top k names
    top_k_results = [x[0] for x in results[:top_k]]
    
    return top_k_results
