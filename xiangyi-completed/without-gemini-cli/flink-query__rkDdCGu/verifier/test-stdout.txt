public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setRuntimeMode(RuntimeExecutionMode.BATCH);
        env.setParallelism(1);

        DataStream<TaskEvent> tasks = env.addSource(taskSourceOrTest(new FileTaskSource(taskInput)))
                .filter(e -> e != null && e.eventType == 0) // SUBMIT
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forMonotonousTimestamps()
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        SingleOutputStreamOperator<Tuple2<Long, Integer>> sessionCounts = tasks
                .keyBy(e -> e.jobId)
                .window(SessionWindows.withGap(Time.seconds(sesssize)))
                .apply(new WindowFunction<TaskEvent, Tuple2<Long, Integer>, Long, TimeWindow>() {
                    @Override
                    public void apply(Long jobId, TimeWindow window, Iterable<TaskEvent> input, Collector<Tuple2<Long, Integer>> out) {
                        int count = 0;
                        for (TaskEvent e : input) {
                            count++;
                        }
                        out.collect(new Tuple2<>(jobId, count));
                    }
                });

        DataStream<Tuple2<Long, Integer>> maxSessions = sessionCounts
                .keyBy(t -> t.f0)
                .reduce(new ReduceFunction<Tuple2<Long, Integer>>() {
                    @Override
                    public Tuple2<Long, Integer> reduce(Tuple2<Long, Integer> t1, Tuple2<Long, Integer> t2) {
                        return t1.f1 >= t2.f1 ? t1 : t2;
                    }
                });

        if (AppBase.out != null) {
            maxSessions.addSink(AppBase.out);
        } else if (outputPath != null) {
            maxSessions.writeAsText(outputPath).setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

    public static class FileTaskSource implements SourceFunction<TaskEvent> {
        private final String path;

        public FileTaskSource(String path) {
            this.path = path;
        }

        @Override
        public void run(SourceContext<TaskEvent> ctx) throws Exception {
            if (path == null) return;
            try (InputStream is = new FileInputStream(path);
                 InputStream gzis = new GZIPInputStream(is);
                 BufferedReader reader = new BufferedReader(new InputStreamReader(gzis, StandardCharsets.UTF_8))) {
                String line;
                while ((line = reader.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
        }
    }
}
