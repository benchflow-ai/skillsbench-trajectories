import struct
import collections
import sys
import json

def calculate_signed_volume(p1, p2, p3):
    # Triple product p1 . (p2 x p3) / 6
    return (p1[0] * (p2[1] * p3[2] - p2[2] * p3[1]) +
            p1[1] * (p2[2] * p3[0] - p2[0] * p3[2]) +
            p1[2] * (p2[0] * p3[1] - p2[1] * p3[0])) / 6.0

def main():
    try:
        with open('scan_data.stl', 'rb') as f:
            f.seek(80) # Skip header
            count_bytes = f.read(4)
            if not count_bytes: return
            num_triangles = struct.unpack('<I', count_bytes)[0]
            
            triangles = []
            for _ in range(num_triangles):
                data = f.read(50)
                if len(data) < 50: break
                parts = struct.unpack('<ffffffffffffH', data)
                v1 = (round(parts[3], 7), round(parts[4], 7), round(parts[5], 7))
                v2 = (round(parts[6], 7), round(parts[7], 7), round(parts[8], 7))
                v3 = (round(parts[9], 7), round(parts[10], 7), round(parts[11], 7))
                attr = parts[12]
                triangles.append((v1, v2, v3, attr))
    except Exception as e:
        print(f"Error reading STL: {e}", file=sys.stderr)
        return

    # Group by shared vertices
    vertex_to_triangles = collections.defaultdict(list)
    for i, (v1, v2, v3, _) in enumerate(triangles):
        for v in (v1, v2, v3):
            vertex_to_triangles[v].append(i)
    
    adj = collections.defaultdict(set)
    for triangle_indices in vertex_to_triangles.values():
        for i in range(len(triangle_indices)):
            for j in range(i + 1, len(triangle_indices)):
                adj[triangle_indices[i]].add(triangle_indices[j])
                adj[triangle_indices[j]].add(triangle_indices[i])
    
    components = []
    visited = [False] * len(triangles)
    for i in range(len(triangles)):
        if not visited[i]:
            comp = []
            queue = collections.deque([i])
            visited[i] = True
            while queue:
                idx = queue.popleft()
                comp.append(idx)
                for neighbor in adj[idx]:
                    if not visited[neighbor]:
                        visited[neighbor] = True
                        queue.append(neighbor)
            components.append(comp)

    # Densities
    densities = {1: 0.10, 10: 7.85, 25: 2.70, 42: 5.55, 99: 11.34}
    
    component_results = []
    for comp in components:
        vol = 0
        attrs = []
        for idx in comp:
            v1, v2, v3, attr = triangles[idx]
            vol += calculate_signed_volume(v1, v2, v3)
            attrs.append(attr)
        
        common_attr = collections.Counter(attrs).most_common(1)[0][0]
        component_results.append({
            'volume': abs(vol),
            'material_id': common_attr,
            'num_triangles': len(comp)
        })

    # Main part is the largest component
    main_part = max(component_results, key=lambda x: x['volume'])
    
    density = densities.get(main_part['material_id'], 0)
    mass = main_part['volume'] * density
    
    result = {
        "main_part_mass": round(mass, 2),
        "material_id": main_part['material_id']
    }
    
    with open('mass_report.json', 'w') as f:
        json.dump(result, f, indent=1)
    
    # Final check output
    print(json.dumps(result, indent=1))

if __name__ == "__main__":
    main()
