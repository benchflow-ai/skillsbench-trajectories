    s = s.replace('O', '0').replace('o', '0').replace('u', '0').replace('U', '0')
    s = s.replace('l', '1').replace('I', '1').replace('|', '1')
    s = s.replace('S', '5').replace('s', '5')
    s = s.replace('B', '8')

        elif ',' in s:
            if len(s.split(',')[-1]) == 2: s = s.replace(',', '.')
            else: s = s.replace(',', '')

    except:
        return None

                # Check exclusion
                if any(ex in line_upper for ex in EXCLUSION_KEYWORDS):
                    continue
                
                if keyword in line_upper:
                    # Capture potential number strings including typo chars
                    # We match digits, ., ,, and typo chars
                    # pattern: (digits/typos)+ [.,] (digits/typos){2}
                    # Simplified: just grab the last token that looks like a number
                    
                    # Remove the keyword to avoid matching "TOTAL" as a number? No.
                    
                    # Regex to find something that looks like an amount:
                    # \d+[.,]\d{2} but with typos allowed in digits
                    # [0-9OouUlISBZ]+[.,][0-9OouUlISBZ]{2}
                    
                    amount_chars = r'[0-9OouUlISBZ]'
                    pattern = f'({amount_chars}+(?:[.,]{amount_chars}{{2}}))'
                    
                    # Look in current line
                    matches = re.findall(pattern, line)
                    if matches:
                        val = clean_amount_str(matches[-1])
                        if val: return val
                        
                    # Look in next line
                    if i + 1 < len(lines):
                        next_line = lines[i+1]
                        if any(ex in next_line.upper() for ex in EXCLUSION_KEYWORDS):
                            continue
                        matches_next = re.findall(pattern, next_line)
                        if matches_next:
                            val = clean_amount_str(matches_next[-1])
                            if val: return val

    return None
