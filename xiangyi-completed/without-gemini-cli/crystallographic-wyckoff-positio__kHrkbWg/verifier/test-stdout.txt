from typing import Any, Dict, List, Union
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates 
    for the first atom of each Wyckoff position from a CIF file.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing wyckoff multiplicities and coordinates.
    """
    try:
        structure = Structure.from_file(filepath)
    except Exception as e:
        raise ValueError(f"Could not read CIF file at {filepath}: {e}")
    
    # Use SpacegroupAnalyzer to get Wyckoff positions
    # Default symprec is 0.01, which works for most Materials Project structures
    sga = SpacegroupAnalyzer(structure)
    dataset = sga.get_symmetry_dataset()
    
    if dataset is None:
        raise ValueError("Could not determine symmetry dataset for the structure.")

    # Access wyckoff letters (handle both attribute and dict access for compatibility)
    if hasattr(dataset, "wyckoffs"):
        wyckoffs = dataset.wyckoffs
    else:
        wyckoffs = dataset["wyckoffs"]

    wyckoff_multiplicity_dict: Dict[str, int] = {}
    wyckoff_coordinates_dict: Dict[str, List[str]] = {}
    
    seen_letters = set()
    
    for i, letter in enumerate(wyckoffs):
        # 1. Calculate Multiplicity
        wyckoff_multiplicity_dict[letter] = wyckoff_multiplicity_dict.get(letter, 0) + 1
        
        # 2. Calculate Coordinates for the first atom of each Wyckoff position
        if letter not in seen_letters:
            seen_letters.add(letter)
            
            # Get the fractional coordinates of the atom
            coords = structure[i].frac_coords
            
            rational_coords = []
            for c in coords:
                # Convert to fraction, limit denominator to 12
                # Also ensure the coordinate is in [0, 1) range using % 1
                val = Fraction(c).limit_denominator(12) % 1
                rational_coords.append(str(val))
            
            wyckoff_coordinates_dict[letter] = rational_coords
            
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }

if __name__ == "__main__":
