    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        
        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();

        // Task Events
        DataStream<TaskEvent> taskStream;
        if (AppBase.taskEvents != null) {
            taskStream = env.addSource(AppBase.taskEvents);
        } else {
            taskStream = env.readTextFile(taskInput)
                    .map(TaskEvent::fromString)
                    .filter(Objects::nonNull);
        }

        // Job Events
        DataStream<JobEvent> jobStream;
        if (AppBase.jobEvents != null) {
            jobStream = env.addSource(AppBase.jobEvents);
        } else {
            jobStream = env.readTextFile(jobInput)
                    .map(JobEvent::fromString)
                    .filter(Objects::nonNull);
        }

        // Assign Timestamps and Watermarks
        // Timestamps are in microseconds, converting to milliseconds for Flink
        taskStream = taskStream.assignTimestampsAndWatermarks(
                WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
        );

        jobStream = jobStream.assignTimestampsAndWatermarks(
                WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofMinutes(1))
                        .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
        );

        // Filter valid events
        // Task: Only SUBMIT (0)
        DataStream<TaskEvent> submits = taskStream.filter(t -> t.eventType == 0);

        // Job: FINISH (4), FAIL (3), KILL (5), LOST (6)
        // Note: prompt says "once the job has finished".
        DataStream<JobEvent> finishes = jobStream.filter(j ->
                j.eventType == 4 || j.eventType == 3 || j.eventType == 5 || j.eventType == 6);

        // Process
        DataStream<Tuple2<Long, Integer>> result = submits.connect(finishes)
                .keyBy(t -> t.jobId, j -> j.jobId)
                .process(new LongestSessionFunction());

        // Output
        if (AppBase.out != null) {
            result.addSink(AppBase.out);
        } else {
            result.writeAsText(outputPath, org.apache.flink.core.fs.FileSystem.WriteMode.OVERWRITE).setParallelism(1);
        }

        env.execute("LongestSessionPerJob");
    }

    public static class LongestSessionFunction extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {

        // State
        private ValueState<Integer> maxStageCount;
        private ValueState<Integer> currentStageCount;
        private ValueState<Long> lastSubmitTime;
        private ValueState<Boolean> isJobFinished;

        private static final long TEN_MINUTES_MS = 10 * 60 * 1000L;

        @Override
        public void open(Configuration parameters) throws Exception {
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class));
            lastSubmitTime = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTime", Long.class));
            isJobFinished = getRuntimeContext().getState(new ValueStateDescriptor<>("isJobFinished", Boolean.class));
        }

        @Override
        public void processElement1(TaskEvent value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Task SUBMIT
            long currentTime = value.timestamp / 1000;
            Long lastTime = lastSubmitTime.value();
            Integer current = currentStageCount.value();

            if (lastTime != null && (currentTime - lastTime) > TEN_MINUTES_MS) {
                // Gap detected, close previous stage
                updateMax(current);
                current = 0; // Reset for new stage
            }

            // Add to current stage
            int newCount = (current == null ? 0 : current) + 1;
            currentStageCount.update(newCount);
            lastSubmitTime.update(currentTime);

            // Register timer to close this stage
            ctx.timerService().registerEventTimeTimer(currentTime + TEN_MINUTES_MS);
        }

        @Override
        public void processElement2(JobEvent value, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Job Finished
            isJobFinished.update(true);

            // If we have no active stage (timer not pending or cleaned up), we might need to emit now.
            // However, if we have an active stage, the timer is set and will fire.
            // If currentStageCount is null, no active stage.
            if (currentStageCount.value() == null) {
                emitResult(out, ctx.getCurrentKey());
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            // Check if this timer corresponds to the end of the current stage
            Long lastTime = lastSubmitTime.value();
            if (lastTime != null && timestamp >= lastTime + TEN_MINUTES_MS) {
                // End of stage
                updateMax(currentStageCount.value());
                
                // Clear active stage
                currentStageCount.clear();
                lastSubmitTime.clear();

                // If job is finished, emit result
                if (Boolean.TRUE.equals(isJobFinished.value())) {
                    emitResult(out, ctx.getCurrentKey());
                }
            }
        }

        private void updateMax(Integer count) throws Exception {
            if (count == null) return;
            Integer max = maxStageCount.value();
            if (max == null || count > max) {
                maxStageCount.update(count);
            }
        }

        private void emitResult(Collector<Tuple2<Long, Integer>> out, Long jobId) throws Exception {
            Integer max = maxStageCount.value();
            if (max == null) max = 0;
            out.collect(new Tuple2<>(jobId, max));
            
            // Clean up all state
            maxStageCount.clear();
            currentStageCount.clear();
            lastSubmitTime.clear();
            isJobFinished.clear();
        }
    }
