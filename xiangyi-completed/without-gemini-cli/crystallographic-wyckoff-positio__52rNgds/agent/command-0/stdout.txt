```python
import re
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def to_rational_string(val: float) -> str:
    """
    Converts a float to a rational string with denominator <= 12.
    """
    # Ensure value is in [0, 1)
    val = val % 1
    if abs(val - 1) < 1e-6:
        val = 0.0
    if abs(val) < 1e-6:
        val = 0.0
        
    f = Fraction(val).limit_denominator(12)
    if f.numerator == 0 or f == 1:
        return "0"
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict]:
    """
    Analyzes Wyckoff position multiplicities and calculates approximate coordinates 
    for the first atom of each Wyckoff position from a CIF file.
    """
    structure = Structure.from_file(filepath)
    sga = SpacegroupAnalyzer(structure)
    symm_structure = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track counts of each Wyckoff letter to handle duplicates
    letter_counts = {}
    
    # wyckoff_symbols and equivalent_indices in SymmetrizedStructure are parallel lists
    # corresponding to each unique Wyckoff position (orbit) in the structure.
    for i, symbol in enumerate(symm_structure.wyckoff_symbols):
        # Extract multiplicity and letter from symbol (e.g., '4a', '8c', '16h')
        match = re.match(r"(\d+)([a-z]+)", symbol)
        if not match:
            continue
            
        multiplicity = int(match.group(1))
        letter = match.group(2)
        
        # Determine the key for the dictionaries
        if letter not in letter_counts:
            key = letter
            letter_counts[letter] = 1
        else:
            key = f"{letter}_{letter_counts[letter]}"
            letter_counts[letter] += 1
            
        # Get the first atom's coordinates for this Wyckoff position
        first_atom_idx = symm_structure.equivalent_indices[i][0]
        frac_coords = symm_structure[first_atom_idx].frac_coords
        
        wyckoff_multiplicity_dict[key] = multiplicity
        wyckoff_coordinates_dict[key] = [to_rational_string(c) for c in frac_coords]
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
```
