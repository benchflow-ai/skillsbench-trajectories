import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem

from rdkit import RDLogger
RDLogger.DisableLog('rdApp.*')

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF pool to a target chemical name.
    
    Args:
        target_molecule_name (str): The name of the target chemical.
        molecule_pool_filepath (str): Path to the PDF file containing chemical names.
        top_k (int): Number of top similar chemicals to return.
        
    Returns:
        list: A list of the top k chemical names, sorted by similarity (descending) 
              and then alphabetically.
    """
    # 1. Extract names from PDF
    pool_names = []
    try:
        with pdfplumber.open(molecule_pool_filepath) as pdf:
            for page in pdf.pages:
                text = page.extract_text()
                if text:
                    # Assuming one chemical name per line as seen in the PDF content
                    pool_names.extend([line.strip() for line in text.split('\n') if line.strip()])
    except Exception:
        return []

    if not pool_names:
        return []

    # 2. Get SMILES for target chemical
    try:
        target_compounds = pcp.get_compounds(target_molecule_name, 'name')
        if not target_compounds:
            return []
        target_smiles = target_compounds[0].smiles
    except Exception:
        return []

    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    
    # Morgan fingerprints with radius = 2, include chirality
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True, nBits=2048)

    # 3. Get SMILES for unique pool chemicals and calculate similarity
    # Use a set to avoid redundant API calls for the same name
    unique_pool_names = sorted(list(set(pool_names)))
    scored_results = []
    
    for name in unique_pool_names:
        try:
            # Fetch compound from PubChem
            comps = pcp.get_compounds(name, 'name')
            if not comps:
                continue
            
            smiles = comps[0].smiles
            mol = Chem.MolFromSmiles(smiles)
            if not mol:
                continue
            
            fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True, nBits=2048)
            similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
            scored_results.append((name, similarity))
        except Exception:
            # Skip names that cause errors in fetching or processing
            continue

    # 4. Sort results: similarity descending, then name ascending
    scored_results.sort(key=lambda x: (-x[1], x[0]))

    # 5. Return the top k names
    return [item[0] for item in scored_results[:top_k]]
