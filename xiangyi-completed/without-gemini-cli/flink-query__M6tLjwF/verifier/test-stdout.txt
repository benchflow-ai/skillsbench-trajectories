    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskStream;
        if (AppBase.taskEvents != null) {
            taskStream = env.addSource(AppBase.taskEvents);
        } else {
            taskStream = env.readTextFile(taskInput)
                    .map(TaskEvent::fromString);
        }

        taskStream = taskStream.filter(Objects::nonNull)
                .filter(e -> e.eventType == 0) // SUBMIT
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<TaskEvent>forBoundedOutOfOrderness(Duration.ofSeconds(10))
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        DataStream<JobEvent> jobStream;
        if (AppBase.jobEvents != null) {
            jobStream = env.addSource(AppBase.jobEvents);
        } else {
            jobStream = env.readTextFile(jobInput)
                    .map(JobEvent::fromString);
        }

        jobStream = jobStream.filter(Objects::nonNull)
                .filter(e -> e.eventType == 4) // FINISH
                .assignTimestampsAndWatermarks(
                        WatermarkStrategy.<JobEvent>forBoundedOutOfOrderness(Duration.ofSeconds(10))
                                .withTimestampAssigner((event, timestamp) -> event.timestamp / 1000)
                );

        DataStream<Tuple2<Long, Integer>> result = taskStream.keyBy(e -> e.jobId)
                .connect(jobStream.keyBy(e -> e.jobId))
                .process(new LongestStageProcessFunction(sesssize));

        if (outputPath != null) {
            result.writeAsText(outputPath).setParallelism(1);
        } else {
            result.addSink((SinkFunction<Tuple2<Long, Integer>>) sinkOrTest(new org.apache.flink.streaming.api.functions.sink.PrintSinkFunction<>()));
        }

        env.execute("LongestSessionPerJob");
    }

    public static class LongestStageProcessFunction extends KeyedCoProcessFunction<Long, TaskEvent, JobEvent, Tuple2<Long, Integer>> {
        private final long gapMicros;

        private ValueState<Integer> currentStageCount;
        private ValueState<Integer> maxStageCount;
        private ValueState<Long> lastTaskTimestamp; // in microseconds
        private ValueState<Boolean> jobFinished;
        private ValueState<Boolean> resultEmitted;

        public LongestStageProcessFunction(int sessSizeSeconds) {
            this.gapMicros = (long) sessSizeSeconds * 1000 * 1000;
        }

        @Override
        public void open(Configuration parameters) {
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Integer.class, 0));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Integer.class, 0));
            lastTaskTimestamp = getRuntimeContext().getState(new ValueStateDescriptor<>("lastTaskTimestamp", Long.class));
            jobFinished = getRuntimeContext().getState(new ValueStateDescriptor<>("jobFinished", Boolean.class, false));
            resultEmitted = getRuntimeContext().getState(new ValueStateDescriptor<>("resultEmitted", Boolean.class, false));
        }

        @Override
        public void processElement1(TaskEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Integer count = currentStageCount.value();
            Long lastTs = lastTaskTimestamp.value();

            if (lastTs != null && (event.timestamp - lastTs) > gapMicros) {
                // New stage started
                maxStageCount.update(Math.max(maxStageCount.value(), count));
                count = 1;
            } else {
                count++;
            }

            currentStageCount.update(count);
            lastTaskTimestamp.update(event.timestamp);

            // Register timer for stage end
            ctx.timerService().registerEventTimeTimer((event.timestamp + gapMicros) / 1000);
        }

        @Override
        public void processElement2(JobEvent event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            jobFinished.update(true);
            // Register timer to emit result
            ctx.timerService().registerEventTimeTimer(event.timestamp / 1000);
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            if (resultEmitted.value()) return;

            Long lastTs = lastTaskTimestamp.value();

            // Check if current stage has ended
            if (lastTs == null || (timestamp >= (lastTs + gapMicros) / 1000)) {
                maxStageCount.update(Math.max(maxStageCount.value(), currentStageCount.value()));
                currentStageCount.update(0);
            }

            if (jobFinished.value() && (lastTs == null || (timestamp >= (lastTs + gapMicros) / 1000))) {
                out.collect(new Tuple2<>(ctx.getCurrentKey(), maxStageCount.value()));
                resultEmitted.update(true);
            }
        }
    }
}
