I have created and executed a Node.js script, `export_obj.js`, which imports the Three.js object from `/root/data/object.js`, handles the conversion of `InstancedMesh` components into standard meshes, applies a -90 degree X-rotation to align with Blender's Z-up coordinate system, and exports the result to `/root/output/object.obj`.

```javascript
import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

// Create the scene root
const root = createScene();

// Handle InstancedMesh: OBJExporter doesn't support them directly.
// We convert them to individual Mesh objects to ensure they are exported.
const instancedMeshes = [];
root.traverse((child) => {
  if (child.isInstancedMesh) {
    instancedMeshes.push(child);
  }
});

instancedMeshes.forEach((instancedMesh) => {
  const parent = instancedMesh.parent;
  const count = instancedMesh.count;
  const geometry = instancedMesh.geometry;
  const material = instancedMesh.material;

  for (let i = 0; i < count; i++) {
    const mesh = new THREE.Mesh(geometry, material);
    const matrix = new THREE.Matrix4();
    instancedMesh.getMatrixAt(i, matrix);
    mesh.applyMatrix4(matrix);
    mesh.name = `${instancedMesh.name}_instance_${i}`;
    parent.add(mesh);
  }
  parent.remove(instancedMesh);
});

// Create a wrapper to apply the rotation for Blender Z-up space
const wrapper = new THREE.Group();
wrapper.add(root);

// Apply -90 degrees X-rotation as requested
wrapper.rotation.x = -Math.PI / 2;
wrapper.updateMatrixWorld(true);

const exporter = new OBJExporter();
const result = exporter.parse(wrapper);

const outputPath = '/root/output/object.obj';
const outputDir = path.dirname(outputPath);

if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
}

fs.writeFileSync(outputPath, result);
console.log(`Successfully exported to ${outputPath}`);
```
