```javascript
import * as THREE from 'three';
import { OBJExporter } from 'three/examples/jsm/exporters/OBJExporter.js';
import { createScene } from './data/object.js';
import fs from 'fs';
import path from 'path';

async function run() {
    const root = createScene();
    
    // Update world matrices of the original scene first
    root.updateMatrixWorld(true);
    
    const exportGroup = new THREE.Group();
    
    // Conversion matrix: -90 degrees around X axis to convert Y-up to Z-up
    // As specified in instructions and skill-book.
    const conversionMatrix = new THREE.Matrix4().makeRotationX(-Math.PI / 2);
    
    root.traverse((obj) => {
        if (obj.isMesh) {
            if (obj.isInstancedMesh) {
                // Expand InstancedMesh into individual meshes
                for (let i = 0; i < obj.count; i++) {
                    const instanceMatrix = new THREE.Matrix4();
                    obj.getMatrixAt(i, instanceMatrix);
                    
                    const worldMatrix = new THREE.Matrix4();
                    worldMatrix.copy(obj.matrixWorld).multiply(instanceMatrix);
                    
                    // Apply Z-up conversion
                    worldMatrix.premultiply(conversionMatrix);
                    
                    const geom = obj.geometry.clone();
                    geom.applyMatrix4(worldMatrix);
                    
                    const mesh = new THREE.Mesh(geom);
                    mesh.name = `${obj.name}_${i}`;
                    exportGroup.add(mesh);
                }
            } else {
                // Bake world transform into mesh geometry
                const worldMatrix = new THREE.Matrix4();
                worldMatrix.copy(obj.matrixWorld);
                
                // Apply Z-up conversion
                worldMatrix.premultiply(conversionMatrix);
                
                const geom = obj.geometry.clone();
                geom.applyMatrix4(worldMatrix);
                
                const mesh = new THREE.Mesh(geom);
                mesh.name = obj.name;
                exportGroup.add(mesh);
            }
        }
    });

    const exporter = new OBJExporter();
    const result = exporter.parse(exportGroup);

    const outputPath = '/root/output/object.obj';
    const outputDir = path.dirname(outputPath);
    if (!fs.existsSync(outputDir)) {
        fs.mkdirSync(outputDir, { recursive: true });
    }
    fs.writeFileSync(outputPath, result);
    console.log(`Exported to ${outputPath}`);
}

run().catch((err) => {
    console.error(err);
    process.exit(1);
});
```
