import numpy as np
from astropy.timeseries import BoxLeastSquares
from scipy.signal import savgol_filter

# 1. Load data
data = np.loadtxt('/root/data/tess_lc.txt')
time = data[:, 0]
flux = data[:, 1]
flag = data[:, 2]
err = data[:, 3]

# 2. Filter data
mask = (flag == 0)
time = time[mask]
flux = flux[mask]
err = err[mask]

# Remove NaNs
mask = ~np.isnan(flux)
time = time[mask]
flux = flux[mask]
err = err[mask]

# Simple sigma clipping for outliers
# Use a more robust median for clipping
median_flux = np.median(flux)
std_flux = np.std(flux)
mask = np.abs(flux - median_flux) < 5 * std_flux
time = time[mask]
flux = flux[mask]
err = err[mask]

# 3. Remove stellar variability
# Window size should be larger than the transit duration but smaller than stellar variability period
dt = np.median(np.diff(time))
# Let's try ~0.5 days window for detrending.
window_length = int(0.5 / dt)
if window_length % 2 == 0:
    window_length += 1

trend = savgol_filter(flux, window_length, polyorder=2)
flux_detrended = flux / trend

# 4. Identify the period using BLS
# Period range: 0.5 to 15 days
model = BoxLeastSquares(time, flux_detrended, dy=err)
periodogram = model.autopower(0.15, minimum_period=0.5, maximum_period=15.0)

best_period = periodogram.period[np.argmax(periodogram.power)]

# Refine the period search around the best period
periods = np.linspace(best_period - 0.1, best_period + 0.1, 20000)
# Filter periods within [0.5, 15]
periods = periods[(periods >= 0.5) & (periods <= 15.0)]
periodogram_refined = model.power(periods, 0.15)
best_period_refined = periodogram_refined.period[np.argmax(periodogram_refined.power)]

print(f"{best_period_refined:.5f}")

# Save the result
with open('/root/period.txt', 'w') as f:
    f.write(f"{best_period_refined:.5f}\n")
