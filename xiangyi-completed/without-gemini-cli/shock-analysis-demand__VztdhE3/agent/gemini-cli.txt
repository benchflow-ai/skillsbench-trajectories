# Replicate the table for Scenario 2 (r=35) and Scenario 3 (r=70)

def replicate_table(start_row, multiplier=None, import_share=None):
    # Copy headers
    for r in range(1, 3):
        for c in range(2, 12):
            ws_na.cell(row=start_row+r-1, column=c, value=ws_na.cell(row=r, column=c).value)
    
    # Table rows
    for i in range(23): # 2020-2042
        r_orig = i + 3
        r_new = start_row + i + 2
        ws_na.cell(row=r_new, column=2, value=ws_na.cell(row=r_orig, column=2).value)
        ws_na.cell(row=r_new, column=3, value=f"=WEO_Data!C{i+2}")
        
        # Allocation%
        ws_na.cell(row=r_new, column=4, value=ws_na.cell(row=r_orig, column=4).value)
        
        # Investment
        ws_na.cell(row=r_new, column=5, value=f"=$D${start_row+27}*D{r_new}*2.746")
        
        # Import Content Share
        ws_na.cell(row=r_new, column=6, value=f"=E{r_new}*$D${start_row+28}")
        
        # GDP impact (I-M)
        ws_na.cell(row=r_new, column=7, value=f"=E{r_new}-F{r_new}")
        
        # Multiplier impact
        ws_na.cell(row=r_new, column=8, value=f"=G{r_new}*$D${start_row+29}")
        
        # % of GDP
        ws_na.cell(row=r_new, column=9, value=f"=H{r_new}/C{r_new}")
        
        # Baseline Growth
        ws_na.cell(row=r_new, column=10, value=f"=WEO_Data!F{i+2}/100")
        
        # Projected Growth
        if i == 0:
            ws_na.cell(row=r_new, column=11, value=f"=J{r_new}")
        else:
            ws_na.cell(row=r_new, column=11, value=f"=((C{r_new}+H{r_new})/(C{r_new-1}+H{r_new-1}))-1")

    # Assumptions for the scenario
    ws_na.cell(row=start_row+27, column=3, value="total investment")
    ws_na.cell(row=start_row+27, column=4, value=6500)
    ws_na.cell(row=start_row+28, column=3, value="import content share")
    ws_na.cell(row=start_row+28, column=4, value=import_share if import_share else "='SUT Calc'!C46")
    ws_na.cell(row=start_row+29, column=3, value="demand multiplier")
    ws_na.cell(row=start_row+29, column=4, value=multiplier if multiplier else 0.8)

# Replicate for Scenario 2
replicate_table(30, multiplier=1.0)
# Replicate for Scenario 3
replicate_table(65, import_share=0.5)
