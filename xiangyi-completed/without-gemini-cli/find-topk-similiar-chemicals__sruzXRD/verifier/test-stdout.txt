import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from rdkit import RDLogger
import warnings

# Suppress PubChemPy deprecation warnings
warnings.filterwarnings("ignore", category=DeprecationWarning, module="pubchempy")

# Suppress RDKit warnings/info
RDLogger.DisableLog('rdApp.*')

def get_smiles(name):
    """
    Retrieves the isomeric SMILES for a given chemical name using PubChemPy.
    Returns None if retrieval fails or no compound is found.
    """
    try:
        # We query by name. 
        comps = pcp.get_compounds(name, 'name')
        if comps:
            # We prefer isomeric_smiles to capture chirality
            if hasattr(comps[0], 'isomeric_smiles'):
                return comps[0].isomeric_smiles
            elif hasattr(comps[0], 'smiles'):
                 return comps[0].smiles
    except Exception:
        return None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF pool to a target molecule.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing the pool of molecules.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: A list of chemical names from the pool, sorted by similarity (desc) then name (asc).
    """
    
    # 1. Get Target Molecule SMILES and Fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        # If we can't find the target, we can't compute similarity. 
        # Returning empty list seems appropriate.
        return []
        
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
        
    # Morgan fingerprint, radius=2, include chirality
    # Note: RDKit might warn to use MorganGenerator, but this is the standard legacy function.
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 2. Extract Molecule Names from PDF
    pool_names = []
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                # Split by lines and clean whitespace
                lines = text.split('\n')
                for line in lines:
                    cleaned_name = line.strip()
                    if cleaned_name:
                        pool_names.append(cleaned_name)
    except Exception as e:
        print(f"Error reading PDF file: {e}")
        return []
    
    # 3. Compute Similarities
    scored_molecules = []
    
    # Cache lookup to avoid redundant API calls
    name_to_smiles_cache = {}
    
    for name in pool_names:
        # Check cache
        if name in name_to_smiles_cache:
            smiles = name_to_smiles_cache[name]
        else:
            smiles = get_smiles(name)
            name_to_smiles_cache[name] = smiles
            
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        
        scored_molecules.append((name, similarity))
        
    # 4. Sort Results
    # Primary key: Similarity (Descending)
    # Secondary key: Name (Ascending) (Alphabetical)
    
    # We sort by name first (Ascending), then by similarity (Descending).
    # Since Python's sort is stable, this achieves the desired effect.
    scored_molecules.sort(key=lambda x: x[0]) 
    scored_molecules.sort(key=lambda x: x[1], reverse=True)
    
    # 5. Extract top k names
    top_molecules = [x[0] for x in scored_molecules[:top_k]]
    
    return top_molecules
