package clusterdata.query;

import clusterdata.datatypes.JobEvent;
import clusterdata.datatypes.TaskEvent;
import clusterdata.utils.AppBase;

import org.apache.flink.api.java.utils.ParameterTool;

import org.apache.flink.streaming.api.datastream.DataStream;
import org.apache.flink.streaming.api.environment.StreamExecutionEnvironment;

import org.apache.flink.streaming.api.functions.sink.SinkFunction;

import org.apache.flink.util.Collector;

import java.io.*;
import java.util.zip.GZIPInputStream;

public class LongestSessionPerJob extends AppBase {

    public static void main(String[] args) throws Exception {

        ParameterTool params = ParameterTool.fromArgs(args);
        String taskInput = params.get("task_input", null);
        String jobInput = params.get("job_input", null);
        String outputPath = params.get("output", null);
        final int sesssize = params.getInt("sesssize", 600);

        // set up streaming execution environment
        StreamExecutionEnvironment env = StreamExecutionEnvironment.getExecutionEnvironment();
        env.setParallelism(1);

        DataStream<TaskEvent> taskStream = env.addSource(AppBase.taskSourceOrTest(new TaskEventSource(taskInput)));
        DataStream<JobEvent> jobStream = env.addSource(AppBase.jobSourceOrTest(new JobEventSource(jobInput)));

        DataStream<Event> combined = taskStream
                .filter(e -> e.eventType == 0) // SUBMIT
                .map(e -> new Event(e.jobId, e.timestamp / 1000, true))
                .returns(Types.POJO(Event.class))
                .union(jobStream
                        .filter(e -> e.eventType >= 2 && e.eventType <= 6) // Terminal
                        .map(e -> new Event(e.jobId, e.timestamp / 1000, false))
                        .returns(Types.POJO(Event.class)));

        combined.assignTimestampsAndWatermarks(
                        WatermarkStrategy.<Event>forMonotonousTimestamps()
                                .withTimestampAssigner((SerializableTimestampAssigner<Event>) (event, timestamp) -> event.timestamp)
                )
                .keyBy(e -> e.jobId)
                .process(new LongestStageProcessFunction(sesssize * 1000L))
                .map(Tuple2::toString)
                .addSink((SinkFunction<String>) AppBase.sinkOrTest(new WriteToFileSink(outputPath)));

        env.execute("LongestSessionPerJob");
    }

    public static class Event implements Serializable {
        public long jobId;
        public long timestamp;
        public boolean isTaskSubmit;

        public Event() {}
        public Event(long jobId, long timestamp, boolean isTaskSubmit) {
            this.jobId = jobId;
            this.timestamp = timestamp;
            this.isTaskSubmit = isTaskSubmit;
        }
    }

    public static class LongestStageProcessFunction extends KeyedProcessFunction<Long, Event, Tuple2<Long, Integer>> {
        private final long gap;
        private transient ValueState<Long> lastSubmitTimestamp;
        private transient ValueState<Integer> currentStageCount;
        private transient ValueState<Integer> maxStageCount;
        private transient ValueState<Boolean> hasFinished;

        public LongestStageProcessFunction(long gap) {
            this.gap = gap;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            lastSubmitTimestamp = getRuntimeContext().getState(new ValueStateDescriptor<>("lastSubmitTimestamp", Types.LONG));
            currentStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("currentStageCount", Types.INT));
            maxStageCount = getRuntimeContext().getState(new ValueStateDescriptor<>("maxStageCount", Types.INT));
            hasFinished = getRuntimeContext().getState(new ValueStateDescriptor<>("hasFinished", Types.BOOLEAN));
        }

        @Override
        public void processElement(Event event, Context ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            if (event.isTaskSubmit) {
                Long last = lastSubmitTimestamp.value();
                Integer current = currentStageCount.value();
                Integer max = maxStageCount.value();

                if (last == null) {
                    current = 1;
                    max = 0;
                } else if (event.timestamp - last > gap) {
                    max = Math.max(max == null ? 0 : max, current);
                    current = 1;
                } else {
                    current = (current == null ? 0 : current) + 1;
                }
                lastSubmitTimestamp.update(event.timestamp);
                currentStageCount.update(current);
                maxStageCount.update(max);
            } else {
                // Job Terminal event
                Boolean finished = hasFinished.value();
                if (finished == null || !finished) {
                    hasFinished.update(true);
                    ctx.timerService().registerEventTimeTimer(event.timestamp);
                }
            }
        }

        @Override
        public void onTimer(long timestamp, OnTimerContext ctx, Collector<Tuple2<Long, Integer>> out) throws Exception {
            Integer current = currentStageCount.value();
            Integer max = maxStageCount.value();
            if (current != null) {
                max = Math.max(max == null ? 0 : max, current);
            }
            if (max != null) {
                out.collect(new Tuple2<>(ctx.getCurrentKey(), max));
            }
            // Do not clear hasFinished to avoid multiple outputs if multiple terminal events exist
        }
    }

    public static class TaskEventSource implements SourceFunction<TaskEvent> {
        private final String filePath;
        private volatile boolean running = true;

        public TaskEventSource(String filePath) {
            this.filePath = filePath;
        }

        @Override
        public void run(SourceFunction.SourceContext<TaskEvent> ctx) throws Exception {
            if (filePath == null) return;
            try (InputStream fileStream = new FileInputStream(filePath);
                 InputStream gzipStream = new GZIPInputStream(fileStream);
                 BufferedReader reader = new BufferedReader(new InputStreamReader(gzipStream))) {
                String line;
                while (running && (line = reader.readLine()) != null) {
                    TaskEvent event = TaskEvent.fromString(line);
                    if (event != null) {
                        ctx.collect(event);
                    }
                }
            }
        }

        @Override
        public void cancel() {
            running = false;
        }
    }

    }

    public static class WriteToFileSink extends RichSinkFunction<String> {
        private final String outputPath;
        private transient PrintWriter writer;

        public WriteToFileSink(String outputPath) {
            this.outputPath = outputPath;
        }

        @Override
        public void open(Configuration parameters) throws Exception {
            if (outputPath != null) {
                File file = new File(outputPath);
                if (file.getParentFile() != null) {
                    file.getParentFile().mkdirs();
                }
                writer = new PrintWriter(new BufferedWriter(new FileWriter(file)));
            }
        }

        @Override
        public void invoke(String value, SinkFunction.Context context) throws Exception {
            if (writer != null) {
                writer.println(value);
            }
        }

        @Override
        public void close() throws Exception {
            if (writer != null) {
                writer.close();
            }
        }
    }
