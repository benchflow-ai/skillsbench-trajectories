import os
import sys
import time
import pubchempy as pcp
from rdkit import Chem
from rdkit import DataStructs
from rdkit.Chem import AllChem
from pypdf import PdfReader

from rdkit import RDLogger
RDLogger.DisableLog('rdApp.*')

def get_smiles(name):
    """
    Fetches the SMILES for a given chemical name using PubChemPy.
    """
    try:
        # Adding a small delay to respect PubChem's rate limit
        # (5 requests per second)
        time.sleep(0.2)
        results = pcp.get_compounds(name, 'name')
        if results:
            # PubChemPy's 'smiles' property returns the isomeric SMILES if available
            return results[0].smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF pool to a target chemical.
    Similarity is computed using Morgan fingerprints and Tanimoto coefficient.
    """
    # 1. Extract names from PDF
    if not os.path.exists(molecule_pool_filepath):
        return []
    
    reader = PdfReader(molecule_pool_filepath)
    pool_names = []
    for page in reader.pages:
        text = page.extract_text()
        if text:
            lines = [line.strip() for line in text.split('\n') if line.strip()]
            pool_names.extend(lines)
    
    # Remove duplicates while preserving order
    unique_pool_names = []
    seen = set()
    for name in pool_names:
        if name not in seen:
            unique_pool_names.append(name)
            seen.add(name)
    
    # 2. Get SMILES and Fingerprint for target
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        return []
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True, nBits=2048)
    
    # 3. Get SMILES and similarity for each name in the pool
    scored_results = []
    for name in unique_pool_names:
        smiles = get_smiles(name)
        if smiles:
            mol = Chem.MolFromSmiles(smiles)
            if mol:
                fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True, nBits=2048)
                similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
                scored_results.append((name, similarity))
    
    # 4. Sort results: similarity descending, then name alphabetically
    # In Python, sort is stable. Sorting by name ascending then similarity descending
    # achieves the desired order.
    scored_results.sort(key=lambda x: x[0])  # Alphabetical ascending
    scored_results.sort(key=lambda x: x[1], reverse=True)  # Similarity descending
    
    # Return only the names of the top k results
    return [item[0] for item in scored_results[:top_k]]

if __name__ == "__main__":
    # Example usage or CLI interface
    if len(sys.argv) > 3:
        target = sys.argv[1]
        pool_path = sys.argv[2]
        k = int(sys.argv[3])
        results = topk_tanimoto_similarity_molecules(target, pool_path, k)
        for res in results:
            print(res)
