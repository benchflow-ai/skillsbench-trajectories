import pymatgen.core as mg
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any
import re

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    structure = mg.Structure.from_file(filepath)
    # symprec 0.01 is generally good for SHELX-derived CIFs
    sga = SpacegroupAnalyzer(structure, symprec=0.01)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # In SymmetrizedStructure, wyckoff_symbols is a list corresponding to equivalent_indices
    for i, indices in enumerate(sym_struct.equivalent_indices):
        wyckoff_label = sym_struct.wyckoff_symbols[i]
        
        # Extract the letter from the Wyckoff symbol (e.g., "4a" -> "a")
        match = re.search(r'[a-zA-Z]+', wyckoff_label)
        letter = match.group() if match else wyckoff_label
            
        multiplicity = len(indices)
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get coordinates of the first atom in the group
        first_idx = indices[0]
        first_site = sym_struct[first_idx]
        coords = first_site.frac_coords
        
        rational_coords = []
        for c in coords:
            # Wrap to [0, 1) range
            c_wrapped = c % 1
            # Rounding to closest rational number with denominator <= 12
            f = Fraction(float(c_wrapped)).limit_denominator(12)
            
            # Handle potential rounding to 1/1
            num = f.numerator % f.denominator
            den = f.denominator
            
            # Format as string
            if den == 1 or num == 0:
                rational_coords.append(str(num))
            else:
                rational_coords.append(f"{num}/{den}")
        
        wyckoff_coordinates_dict[letter] = rational_coords
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
