from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any

def to_rational(x):
    # Wrap to [0, 1) and handle floating point noise
    x_wrapped = x % 1
    # limit_denominator(12) as per requirement
    f = Fraction(x_wrapped).limit_denominator(12)
    
    # Handle the case where rounding leads to 1 (e.g., 0.99999 -> 1/1)
    if f == 1 or f == 0:
        return "0"
    
    if f.denominator == 1:
        return str(f.numerator)
    return f"{f.numerator}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict[str, Any]]:
    # Load the structure from the CIF file
    struct = Structure.from_file(filepath)
    
    # Use SpacegroupAnalyzer to get symmetry information
    # symprec is set to a reasonable default for CIF files
    sga = SpacegroupAnalyzer(struct, symprec=0.01)
    symm_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # Track the order of encountered Wyckoff letters to keep only the first one's coordinates
    # while summing multiplicities if multiple orbits have the same letter.
    # However, given the example, we should return one entry per Wyckoff letter.
    
    # symm_struct.equivalent_indices groups sites by symmetry
    # symm_struct.wyckoff_symbols gives the symbol for each group (e.g., '4a')
    
    for i, group in enumerate(symm_struct.equivalent_indices):
        symbol = symm_struct.wyckoff_symbols[i]
        
        # Multiplicity of this orbit
        multiplicity = len(group)
        
        # Wyckoff letter is the alphabetic part of the symbol
        letter = "".join(filter(str.isalpha, symbol))
        
        if letter not in wyckoff_multiplicity_dict:
            # First time encountering this Wyckoff position type
            wyckoff_multiplicity_dict[letter] = multiplicity
            
            # Get coordinates for the first atom of this Wyckoff position (orbit)
            # The indices in 'group' refer to the sites in symm_struct
            first_atom_idx = group[0]
            coords = symm_struct[first_atom_idx].frac_coords
            wyckoff_coordinates_dict[letter] = [to_rational(c) for c in coords]
        else:
            # Subsequent encounters for the same Wyckoff letter
            # Sum the multiplicity
            wyckoff_multiplicity_dict[letter] += multiplicity
            # We do not update coordinates_dict because we want the coordinates 
            # of the first atom of each Wyckoff position (meaning the first orbit of that type).
            
    # Sort the dicts by Wyckoff letter (a, b, c...) for consistent output if needed,
    # but pymatgen usually returns them in a sensible order.
    # The example shows 'a' then 'c', which is alphabetical.
    
    # Final result structure
    result = {
        "wyckoff_multiplicity_dict": dict(sorted(wyckoff_multiplicity_dict.items())),
        "wyckoff_coordinates_dict": dict(sorted(wyckoff_coordinates_dict.items()))
    }
    
    return result

if __name__ == "__main__":
    import json
