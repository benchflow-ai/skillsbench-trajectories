from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any, Dict

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    """
    Analyzes a CIF file to extract Wyckoff position multiplicities and coordinates.
    
    Args:
        filepath: Path to the CIF file.
        
    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": mapping Wyckoff letter to multiplicity (int).
        - "wyckoff_coordinates_dict": mapping Wyckoff letter to a list of coordinate strings (fractions).
    """
    # Load structure from CIF file
    structure = Structure.from_file(filepath)
    
    # Analyze symmetry
    # Default symprec=0.01 is usually sufficient for Materials Project CIFs
    sga = SpacegroupAnalyzer(structure)
    
    # Get symmetrized structure which groups equivalent atoms
    symmetrized_structure = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # wyckoff_symbols corresponds to the groups in equivalent_sites
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols
    equivalent_sites = symmetrized_structure.equivalent_sites
    
    for i, sites in enumerate(equivalent_sites):
        symbol = wyckoff_symbols[i]
        
        # Extract the Wyckoff letter. 
        # Standard format is like "4a", "8c". Letter is the last character.
        letter = symbol[-1]
        
        # Multiplicity is the number of equivalent sites in the group
        multiplicity = len(sites)
        
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get coordinates of the first atom in the group
        # Using the first site's fractional coordinates
        first_site = sites[0]
        coords = first_site.frac_coords
        
        formatted_coords = []
        for c in coords:
            # Normalize to [0, 1) to handle periodic boundary conditions standardly
            # We treat coordinates as modulo 1.
            c_mod = c % 1.0
            
            # Convert to fraction with limited denominator
            # This handles floating point approximations
            f = Fraction(c_mod).limit_denominator(12)
            
            # Ensure the fraction is also modulo 1 (e.g. if limit_denominator gave 1/1)
            f = f % 1
            
            formatted_coords.append(str(f))
            
        wyckoff_coordinates_dict[letter] = formatted_coords
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
