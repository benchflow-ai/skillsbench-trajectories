import time
import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem, DataStructs

def get_smiles(name, retries=3):
    for i in range(retries):
        try:
            compounds = pcp.get_compounds(name, 'name')
            if compounds:
                # Use isomeric_smiles to ensure chirality is preserved
                return compounds[0].isomeric_smiles
            return None
        except Exception as e:
            if "503" in str(e): # Server busy
                time.sleep(1 * (i + 1)) 
                continue
            print(f"Error fetching {name}: {e}")
            return None
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    # Read molecules from PDF
    molecule_names = []
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                molecule_names.extend(lines)
    except Exception as e:
        print(f"Error reading PDF: {e}")
        return []

    # Get target molecule info
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        # print(f"Could not find SMILES for target: {target_molecule_name}")
        return []

    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        return []

    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)

    results = []
    
    # Use a set to handle duplicates if any in the PDF
    unique_names = sorted(list(set(molecule_names)))

    for name in unique_names:
        # Rate limiting
        time.sleep(0.25)
        
        # Resolve SMILES
        smiles = get_smiles(name)
        if not smiles:
            continue
        
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
        
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        
        results.append((name, similarity))

    # Sort results: Descending similarity, then alphabetical name
    # We sort by similarity (desc) -> -x[1]
    # Then by name (asc) -> x[0]
    results.sort(key=lambda x: (-x[1], x[0]))

    return results[:top_k]
