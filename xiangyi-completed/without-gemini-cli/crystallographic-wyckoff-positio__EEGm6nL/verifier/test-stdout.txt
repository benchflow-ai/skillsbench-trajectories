import re
from fractions import Fraction
from typing import Any, Dict
from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    # Parse the CIF file
    # primitive=False is generally preferred to maintain the conventional unit cell setting
    # which usually corresponds to standard Wyckoff position listings.
    try:
        parser = CifParser(filepath)
        structure = parser.parse_structures(primitive=False)[0]
    except Exception as e:
        # Fallback or re-raise if parsing fails
        raise ValueError(f"Failed to parse CIF file: {e}")

    # Analyze symmetry
    sga = SpacegroupAnalyzer(structure)
    sym_struct = sga.get_symmetrized_structure()

    multiplicity_dict = {}
    coordinates_dict = {}

    # Helper to find a unique key if letters collide
    def get_unique_key(base_key, existing_keys):
        if base_key not in existing_keys:
            return base_key
        i = 1
        while f"{base_key}_{i}" in existing_keys:
            i += 1
        return f"{base_key}_{i}"

    for i, sites in enumerate(sym_struct.equivalent_sites):
        symbol = sym_struct.wyckoff_symbols[i]  # e.g., "4a", "8c"
        
        # Extract the letter part (e.g., "a" from "4a")
        # Standard format is number + letter.
        match = re.search(r'([a-z]+)$', symbol)
        if match:
            letter = match.group(1)
        else:
            letter = symbol # Should not happen typically
        
        # Determine unique key for the dictionary
        key = get_unique_key(letter, multiplicity_dict.keys())
        
        # Store multiplicity
        multiplicity_dict[key] = len(sites)
        
        # Get coordinates of the first site in the group
        first_site = sites[0]
        # Normalize to [0, 1)
        coords = [c % 1.0 for c in first_site.frac_coords]
        
        # Convert to fractional strings with denominator constraint
        coords_strs = []
        for c in coords:
            # Handle float precision issues near integer boundaries if necessary
            # Fraction(c).limit_denominator(12) handles rounding.
            # But 0.999999 might become 1/1 -> 0 if we mod first?
            # If c was -1e-10 -> 1.0 -> 0.0. Correct.
            # If c is 0.99999999 -> 1.0 -> 0.0. Correct.
            
            # Using limit_denominator directly on the float
            f = Fraction(c).limit_denominator(12)
            
            # If f is 1, it should be 0 (due to mod 1, but Fraction might round up to 1)
            # e.g. 0.99 -> 1/1.
            # If we want standard coords, 0 is preferred over 1.
            if f == 1:
                f = Fraction(0, 1)
                
            coords_strs.append(str(f))
            
        coordinates_dict[key] = coords_strs

    return {
        "wyckoff_multiplicity_dict": multiplicity_dict,
        "wyckoff_coordinates_dict": coordinates_dict
    }
