from pymatgen.io.cif import CifParser
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
import warnings
from typing import Any, Dict

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> Dict[str, Any]:
    # Suppress warnings to keep output clean
    warnings.filterwarnings("ignore")
    
    try:
        # Read the CIF file
        # primitive=False ensures we respect the cell defined in the CIF 
        # (usually conventional for these files)
        parser = CifParser(filepath)
        structure = parser.get_structures(primitive=False)[0]
    except Exception as e:
        # In case of read failure, re-raise or handle. 
        # Raising ensures the caller knows something went wrong.
        raise ValueError(f"Failed to process CIF file {filepath}: {e}")

    # Analyze symmetry
    # symprec=0.1 is used to handle potential slight numerical noise in coordinates,
    # though MP files are usually clean.
    sga = SpacegroupAnalyzer(structure, symprec=0.1)
    
    # Get the symmetrized structure. 
    # This structure groups equivalent atoms and assigns Wyckoff symbols.
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # equivalent_indices is a list of lists, where each inner list contains 
    # indices of atoms belonging to the same Wyckoff position.
    # sym_struct.wyckoff_symbols corresponds to these groups (one symbol per group).
    
    for i, group_indices in enumerate(sym_struct.equivalent_indices):
        # We take the first atom in the group as the representative
        first_idx = group_indices[0]
        
        # Get the Wyckoff symbol (e.g., "4a", "8c")
        # Access by group index 'i'
        symbol = sym_struct.wyckoff_symbols[i]
        
        # Extract the Wyckoff letter. 
        # Typically the symbol is [Multiplicity][Letter], e.g., "4a".
        # We strip digits to get the letter.
        letter = "".join([c for c in symbol if c.isalpha()])
        
        # The multiplicity is the number of atoms in this equivalent group
        # in the current unit cell.
        multiplicity = len(group_indices)
        
        # Store multiplicity
        wyckoff_multiplicity_dict[letter] = multiplicity
        
        # Get coordinates of the representative atom
        atom = sym_struct.sites[first_idx]
        coords = atom.frac_coords
        
        # Convert coordinates to rational approximations
        formatted_coords = []
        for val in coords:
            # Limit denominator to 12 as requested
            f = Fraction(val).limit_denominator(12)
            
            # Format as string
            if f.denominator == 1:
                formatted_coords.append(str(f.numerator))
            else:
                formatted_coords.append(f"{f.numerator}/{f.denominator}")
        
        wyckoff_coordinates_dict[letter] = formatted_coords
        
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
