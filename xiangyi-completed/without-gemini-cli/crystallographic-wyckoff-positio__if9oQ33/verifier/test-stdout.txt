import os
from typing import Any
from fractions import Fraction
from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer

def float_to_fraction(x: float, max_denom: int = 12) -> str:
    """
    Converts a float coordinate to a rational string with a denominator limit.
    Normalizes the coordinate to the range [0, 1).
    """
    # Normalize input to [0, 1)
    val = x % 1
    
    # Find the closest rational number with denominator <= max_denom
    f = Fraction(val).limit_denominator(max_denom)
    
    # Normalize the resulting fraction to [0, 1)
    # e.g., if f is 1/1, num becomes 0.
    num = f.numerator % f.denominator
    
    if num == 0:
        return "0"
    
    return f"{num}/{f.denominator}"

def analyze_wyckoff_position_multiplicities_and_coordinates(filepath: str) -> dict[str, dict] | dict[str, Any]:
    """
    Analyzes a CIF file to determine Wyckoff position multiplicities and 
    approximate rational coordinates for the first atom of each position.
    """
    if not os.path.exists(filepath):
        return {}

    try:
        # Load the structure from the CIF file
        structure = Structure.from_file(filepath)
    except Exception:
        return {}

    # Analyze symmetry to get the symmetrized structure
    # A symprec of 0.01 is standard for identifying symmetry in many crystal databases
    sga = SpacegroupAnalyzer(structure, symprec=0.01)
    sym_struct = sga.get_symmetrized_structure()
    
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}
    
    # symmetrized_structure.equivalent_indices groups sites that are symmetry-equivalent.
    # We iterate over these groups to get one set of data for each Wyckoff position.
    for i, indices in enumerate(sym_struct.equivalent_indices):
        # The Wyckoff letter for the sites in this group
        letter = sym_struct.wyckoff_letters[indices[0]]
        
        # The multiplicity is the number of atoms in this equivalent group
        multiplicity = len(indices)
        
        # Coordinates for the first atom in this equivalent group
        first_atom_coords = sym_struct[indices[0]].frac_coords
        
        # Convert fractional coordinates to rational strings with denominator <= 12
        coord_strings = [float_to_fraction(c) for c in first_atom_coords]
        
        # Use the Wyckoff letter as the key. 
        # If multiple sites share the same letter, we keep the first one found.
        if letter not in wyckoff_multiplicity_dict:
            wyckoff_multiplicity_dict[letter] = multiplicity
            wyckoff_coordinates_dict[letter] = coord_strings
            
    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
