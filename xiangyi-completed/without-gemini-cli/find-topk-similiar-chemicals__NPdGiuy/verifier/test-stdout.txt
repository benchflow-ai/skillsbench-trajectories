import pypdf
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit import DataStructs
from rdkit import RDLogger

# Suppress RDKit warnings
RDLogger.DisableLog('rdApp.*')

def get_smiles(name):
    """
    Retrieves the isomeric SMILES for a given chemical name using PubChemPy.
    Returns None if not found or on error.
    """
    try:
        compounds = pcp.get_compounds(name, 'name')
        if compounds:
            c = compounds[0]
            # Prioritize isomeric smiles
            if hasattr(c, 'isomeric_smiles') and c.isomeric_smiles:
                return c.isomeric_smiles
            elif hasattr(c, 'canonical_smiles') and c.canonical_smiles:
                return c.canonical_smiles
            elif hasattr(c, 'smiles') and c.smiles:
                return c.smiles
    except Exception:
        pass
    return None

def topk_tanimoto_similarity_molecules(target_molecule_name, molecule_pool_filepath, top_k):
    """
    Finds the top k similar chemicals in a PDF file to the target molecule.
    
    Args:
        target_molecule_name (str): Name of the target molecule.
        molecule_pool_filepath (str): Path to the PDF file containing molecule names.
        top_k (int): Number of top similar molecules to return.
        
    Returns:
        list: List of top k similar molecule names.
    """
    
    # 1. Get target molecule SMILES and Fingerprint
    target_smiles = get_smiles(target_molecule_name)
    if not target_smiles:
        raise ValueError(f"Could not find SMILES for target: {target_molecule_name}")
    
    target_mol = Chem.MolFromSmiles(target_smiles)
    if not target_mol:
        raise ValueError(f"Could not parse SMILES for target: {target_molecule_name}")
        
    # Morgan fingerprints with Tanimoto similarity (radius = 2, include chirality)
    target_fp = AllChem.GetMorganFingerprintAsBitVect(target_mol, 2, useChirality=True)
    
    # 2. Read molecules from PDF
    pool_names = []
    try:
        reader = pypdf.PdfReader(molecule_pool_filepath)
        for page in reader.pages:
            text = page.extract_text()
            if text:
                for line in text.splitlines():
                    clean_line = line.strip()
                    if clean_line:
                        pool_names.append(clean_line)
    except Exception as e:
        raise ValueError(f"Error reading PDF: {e}")
        
    # Remove duplicates and sort for deterministic processing
    pool_names = sorted(list(set(pool_names)))
    
    results = []
    
    # 3. Process pool
    for name in pool_names:
        # Optimization: if the name is the target, we know the score is 1.0, 
        # but to be safe and consistent with "chemical to structure", we process it.
        smiles = get_smiles(name)
        if not smiles:
            continue
            
        mol = Chem.MolFromSmiles(smiles)
        if not mol:
            continue
            
        fp = AllChem.GetMorganFingerprintAsBitVect(mol, 2, useChirality=True)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        
        results.append((name, similarity))
        
    # 4. Sort
    # Descending order of similarity, with alphabetical ordering when ties happen.
    results.sort(key=lambda x: (-x[1], x[0]))
    
    # 5. Return top k
    top_results = [x[0] for x in results[:top_k]]
    return top_results
