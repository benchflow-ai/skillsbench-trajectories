"""
Solution for finding top k similar chemicals using Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to molecular representations using PubChem
3. Compute Morgan fingerprints with chirality
4. Calculate Tanimoto similarity
5. Return top k most similar molecules
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs
from typing import List, Tuple
import warnings

# Suppress deprecation warnings
warnings.filterwarnings('ignore', category=DeprecationWarning)
warnings.filterwarnings('ignore', message='DEPRECATION WARNING')


def extract_chemicals_from_pdf(pdf_path: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_path: Path to the PDF file containing chemical names

    Returns:
        List of chemical names (strings)
    """
    chemicals = []
    with pdfplumber.open(pdf_path) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                chemicals.extend(lines)
    return chemicals


def get_smiles_from_name(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES representation using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string representation of the molecule, or None if not found
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            # Get SMILES representation with stereochemistry
            compound = compounds[0]
            # Try to get the smiles attribute, with fallback for older versions
            if hasattr(compound, 'isomeric_smiles'):
                return compound.isomeric_smiles or compound.canonical_smiles
            else:
                return compound.canonical_smiles
    except Exception:
        pass
    return None


def get_morgan_fingerprint(smiles: str, radius: int = 2, use_chirality: bool = True):
    """
    Generate Morgan fingerprint from SMILES string.

    Args:
        smiles: SMILES representation of molecule
        radius: Radius for Morgan fingerprint (default: 2)
        use_chirality: Whether to include chirality information (default: True)

    Returns:
        Morgan fingerprint object, or None if molecule cannot be parsed
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is not None:
            # Use the newer MorganGenerator API
            generator = rdFingerprintGenerator.GetMorganGenerator(radius=radius, includeChirality=use_chirality)
            fp = generator.GetSparseCountFingerprint(mol)
            return fp
    except Exception:
        pass
    return None


def calculate_tanimoto_similarity(fp1, fp2) -> float:
    """
    Calculate Tanimoto similarity between two fingerprints.

    Args:
        fp1: First Morgan fingerprint
        fp2: Second Morgan fingerprint

    Returns:
        Tanimoto similarity score (0.0 to 1.0)
    """
    return DataStructs.TanimotoSimilarity(fp1, fp2)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[str]:
    """
    Find the top k most similar molecules to a target molecule.

    Uses Morgan fingerprints (radius=2, with chirality) and Tanimoto similarity.
    Results are sorted in descending order of similarity, with alphabetical
    ordering when ties occur.

    Args:
        target_molecule_name: Name of the target molecule
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of chemical names (strings) of the top k most similar molecules
    """
    # Get target molecule SMILES and fingerprint
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES representation for target molecule: {target_molecule_name}")

    target_fp = get_morgan_fingerprint(target_smiles, radius=2, use_chirality=True)
    if target_fp is None:
        raise ValueError(f"Could not generate fingerprint for target molecule: {target_molecule_name}")

    # Extract chemicals from PDF
    pool_chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Calculate similarities
    similarities = []
    for chemical_name in pool_chemicals:
        # Get SMILES for pool molecule
        smiles = get_smiles_from_name(chemical_name)
        if smiles is None:
            continue

        # Generate fingerprint
        fp = get_morgan_fingerprint(smiles, radius=2, use_chirality=True)
        if fp is None:
            continue

        # Calculate Tanimoto similarity
        similarity = calculate_tanimoto_similarity(target_fp, fp)
        similarities.append((chemical_name, similarity))

    # Sort by similarity (descending), then by name (alphabetically) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k chemical names
    top_k_results = [name for name, sim in similarities[:top_k]]

    return top_k_results


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    k = 5

    print(f"Finding top {k} molecules similar to {target}...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for i, molecule in enumerate(results, 1):
        print(f"{i}. {molecule}")
