"""
Materials science crystal structure analyzer for Wyckoff positions.
Analyzes CIF files to extract Wyckoff position multiplicities and coordinates.
"""

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer
from fractions import Fraction
from typing import Any


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str,
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    Parameters
    ----------
    filepath : str
        Path to the CIF file to analyze.

    Returns
    -------
    dict[str, dict] | dict[str, Any]
        A dictionary containing:
        - "wyckoff_multiplicity_dict": Dict[str, int]
            Maps Wyckoff position letter to its multiplicity.
        - "wyckoff_coordinates_dict": Dict[str, List[str]]
            Maps Wyckoff position letter to fractional coordinates (as strings
            in rational format with denominators ≤ 12).

    Examples
    --------
    >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
    ...     "/root/cif_files/FeS2_mp-226.cif"
    ... )
    >>> print(result["wyckoff_multiplicity_dict"])
    {'a': 4, 'c': 8}
    >>> print(result["wyckoff_coordinates_dict"])
    {'a': ['0', '1/2', '1/2'], 'c': ['3/8', '1/9', '8/9']}
    """

    # Load structure from CIF file
    structure = Structure.from_file(filepath)

    # Get symmetry information
    analyzer = SpacegroupAnalyzer(structure)
    symmetrized_structure = analyzer.get_symmetrized_structure()

    # Extract Wyckoff symbols and equivalent sites
    wyckoff_symbols = symmetrized_structure.wyckoff_symbols
    equivalent_sites = symmetrized_structure.equivalent_sites

    # Build multiplicity dictionary
    wyckoff_multiplicity_dict = {}
    wyckoff_coordinates_dict = {}

    for wyckoff_symbol, equiv_sites in zip(wyckoff_symbols, equivalent_sites):
        # Extract letter from Wyckoff symbol (e.g., '4a' -> 'a', '8c' -> 'c')
        wyckoff_letter = wyckoff_symbol[-1]

        # Get multiplicity (number of equivalent sites)
        multiplicity = len(equiv_sites)
        wyckoff_multiplicity_dict[wyckoff_letter] = multiplicity

        # Get the fractional coordinates of the first atom in this Wyckoff position
        first_atom = equiv_sites[0]
        frac_coords = first_atom.frac_coords

        # Convert fractional coordinates to rational format
        rational_coords = _convert_to_rational_strings(frac_coords)
        wyckoff_coordinates_dict[wyckoff_letter] = rational_coords

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict,
    }


def _convert_to_rational_strings(
    frac_coords: list[float], max_denominator: int = 12
) -> list[str]:
    """
    Convert fractional coordinates to rational number strings.

    Parameters
    ----------
    frac_coords : list[float]
        List of fractional coordinates (typically 3 for x, y, z).
    max_denominator : int, optional
        Maximum denominator to use when approximating fractions (default: 12).

    Returns
    -------
    list[str]
        List of coordinate strings in rational format (e.g., ['0', '1/2', '3/8']).
    """
    rational_strings = []

    for coord in frac_coords:
        # Convert to Fraction with max denominator constraint
        frac = Fraction(coord).limit_denominator(max_denominator)

        # Format as string
        if frac.denominator == 1:
            # Whole number
            coord_str = str(frac.numerator)
        else:
            # Fraction
            coord_str = f"{frac.numerator}/{frac.denominator}"

        rational_strings.append(coord_str)

    return rational_strings
