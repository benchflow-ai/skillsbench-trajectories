"""
Solution for finding top k similar chemicals using Morgan fingerprints and Tanimoto similarity.
"""

import pdfplumber
from rdkit import Chem
from rdkit.Chem import AllChem
from rdkit.DataStructs import TanimotoSimilarity
import pubchempy as pcp
from typing import List, Tuple


def extract_molecules_from_pdf(pdf_filepath: str) -> List[str]:
    """
    Extract chemical names from a PDF file.

    Args:
        pdf_filepath: Path to the PDF file containing chemical names

    Returns:
        List of chemical names
    """
    molecules = []
    with pdfplumber.open(pdf_filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                # Split by newlines and filter out empty strings
                names = [line.strip() for line in text.split('\n') if line.strip()]
                molecules.extend(names)
    return molecules


def get_smiles_from_pubchem(chemical_name: str) -> str:
    """
    Convert a chemical name to SMILES format using PubChem.

    Args:
        chemical_name: Name of the chemical

    Returns:
        SMILES string, or None if not found
    """
    try:
        # Search for the compound by name
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def compute_morgan_fingerprint(smiles: str, radius: int = 2, include_chirality: bool = True):
    """
    Compute Morgan fingerprint for a molecule.

    Args:
        smiles: SMILES string representation of molecule
        radius: Radius for Morgan fingerprint (default 2)
        include_chirality: Whether to include chirality (default True)

    Returns:
        Morgan fingerprint object, or None if SMILES is invalid
    """
    try:
        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            return None
        # Use 2048 bits for Morgan fingerprint
        fp = AllChem.GetMorganFingerprintAsBitVect(
            mol,
            radius=radius,
            nBits=2048,
            useChirality=include_chirality
        )
        return fp
    except Exception:
        return None


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> List[Tuple[str, float]]:
    """
    Find the top k similar chemicals to a target molecule using Tanimoto similarity.

    Args:
        target_molecule_name: Name of the target chemical
        molecule_pool_filepath: Path to PDF file containing pool of molecules
        top_k: Number of top similar molecules to return

    Returns:
        List of tuples (molecule_name, similarity_score) sorted by similarity descending,
        with alphabetical ordering for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_pubchem(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Compute fingerprint for target
    target_fp = compute_morgan_fingerprint(target_smiles)
    if target_fp is None:
        raise ValueError(f"Could not compute fingerprint for target molecule: {target_molecule_name}")

    # Extract all molecules from PDF
    pool_molecules = extract_molecules_from_pdf(molecule_pool_filepath)

    # Compute similarities
    similarities = []
    for mol_name in pool_molecules:
        if mol_name == target_molecule_name:
            # Skip the target molecule itself
            continue

        # Get SMILES for pool molecule
        smiles = get_smiles_from_pubchem(mol_name)
        if smiles is None:
            continue

        # Compute fingerprint
        fp = compute_morgan_fingerprint(smiles)
        if fp is None:
            continue

        # Compute Tanimoto similarity
        similarity = TanimotoSimilarity(target_fp, fp)
        similarities.append((mol_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top k
    return similarities[:top_k]


if __name__ == "__main__":
    # Example usage
    target = "Aspirin"
    pdf_path = "/root/molecules.pdf"
    top_k = 10

    results = topk_tanimoto_similarity_molecules(target, pdf_path, top_k)

    print(f"Top {top_k} molecules similar to {target}:")
    for mol_name, similarity in results:
        print(f"  {mol_name}: {similarity:.4f}")
