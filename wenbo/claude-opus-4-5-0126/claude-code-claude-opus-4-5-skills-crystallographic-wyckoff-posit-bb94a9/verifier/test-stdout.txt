"""
Wyckoff position analysis module for CIF crystal structure files.

This module provides functionality to analyze Wyckoff position multiplicities
and calculate approximate coordinates for the first atom of each Wyckoff position
from CIF files generated by SHELX through X-ray Diffraction.
"""

from typing import Any
from fractions import Fraction

from pymatgen.core import Structure
from pymatgen.symmetry.analyzer import SpacegroupAnalyzer


def float_to_fraction_str(value: float, max_denominator: int = 12) -> str:
    """
    Convert a float to a fraction string representation.

    Args:
        value: The floating point value to convert (fractional coordinate)
        max_denominator: Maximum denominator for the fraction (default: 12)

    Returns:
        String representation of the fraction (e.g., "3/8", "1/2", "0")
    """
    # Normalize to [0, 1) range
    value = value % 1.0

    # Handle very small values close to 0 or 1
    if abs(value) < 1e-6:
        return "0"
    if abs(value - 1.0) < 1e-6:
        return "0"  # 1 is equivalent to 0 in fractional coordinates

    frac = Fraction(value).limit_denominator(max_denominator)

    if frac.denominator == 1:
        # Handle case where fraction rounds to 1 (equivalent to 0)
        if frac.numerator == 1:
            return "0"
        return str(frac.numerator)
    else:
        return f"{frac.numerator}/{frac.denominator}"


def analyze_wyckoff_position_multiplicities_and_coordinates(
    filepath: str
) -> dict[str, dict] | dict[str, Any]:
    """
    Analyze Wyckoff position multiplicities and calculate approximate coordinates
    for the first atom of each Wyckoff position from a CIF file.

    This function reads a CIF file, performs symmetry analysis to identify
    Wyckoff positions, calculates the multiplicity of each position, and
    returns the fractional coordinates of the first atom in each Wyckoff orbit.

    Args:
        filepath: Path to the CIF file to analyze

    Returns:
        A dictionary containing:
        - "wyckoff_multiplicity_dict": dict mapping Wyckoff letter to multiplicity
        - "wyckoff_coordinates_dict": dict mapping Wyckoff letter to coordinates
          as a list of 3 fraction strings

    Example:
        >>> result = analyze_wyckoff_position_multiplicities_and_coordinates(
        ...     "/root/cif_files/FeS2_mp-226.cif"
        ... )
        >>> result
        {
            "wyckoff_multiplicity_dict": {"a": 4, "c": 8},
            "wyckoff_coordinates_dict": {
                "a": ["0", "1/2", "1/2"],
                "c": ["3/8", "1/9", "8/9"]
            }
        }
    """
    # Load the structure from CIF file
    structure = Structure.from_file(filepath)

    # Create SpacegroupAnalyzer to perform symmetry analysis
    sga = SpacegroupAnalyzer(structure)

    # Get symmetry dataset which includes Wyckoff letters
    sym_dataset = sga.get_symmetry_dataset()

    # Extract Wyckoff letters and equivalent atom indices
    wyckoff_letters = sym_dataset.wyckoffs
    equiv_atoms = sym_dataset.equivalent_atoms

    # Get fractional coordinates
    frac_coords = structure.frac_coords

    # Group atoms by Wyckoff position
    # We use the order in which Wyckoff letters first appear to maintain consistency
    wyckoff_groups: dict[str, dict] = {}
    wyckoff_order: list[str] = []

    for i, (wyck, equiv_idx) in enumerate(zip(wyckoff_letters, equiv_atoms)):
        if wyck not in wyckoff_groups:
            wyckoff_groups[wyck] = {
                'indices': [],
                'first_index': i
            }
            wyckoff_order.append(wyck)
        wyckoff_groups[wyck]['indices'].append(i)

    # Build the result dictionaries
    wyckoff_multiplicity_dict: dict[str, int] = {}
    wyckoff_coordinates_dict: dict[str, list[str]] = {}

    for wyck in wyckoff_order:
        group = wyckoff_groups[wyck]

        # Multiplicity is the number of atoms in this Wyckoff orbit
        multiplicity = len(group['indices'])
        wyckoff_multiplicity_dict[wyck] = multiplicity

        # Get coordinates of the first atom in this orbit
        first_atom_idx = group['first_index']
        coords = frac_coords[first_atom_idx]

        # Convert to fraction strings
        coord_strs = [
            float_to_fraction_str(coord)
            for coord in coords
        ]
        wyckoff_coordinates_dict[wyck] = coord_strs

    return {
        "wyckoff_multiplicity_dict": wyckoff_multiplicity_dict,
        "wyckoff_coordinates_dict": wyckoff_coordinates_dict
    }
