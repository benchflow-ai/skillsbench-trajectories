"""
Module for finding top-k similar chemicals using Morgan fingerprints and Tanimoto similarity.

This module provides functionality to:
1. Extract chemical names from a PDF file
2. Convert chemical names to SMILES using PubChem
3. Compute Morgan fingerprints (radius=2, with chirality)
4. Calculate Tanimoto similarity between molecules
5. Return the top-k most similar molecules
"""

import pdfplumber
import pubchempy as pcp
from rdkit import Chem
from rdkit.Chem import rdFingerprintGenerator
from rdkit import DataStructs


def get_smiles_from_name(chemical_name: str) -> str | None:
    """
    Convert a chemical name to its SMILES representation using PubChem.

    Args:
        chemical_name: The name of the chemical compound.

    Returns:
        The SMILES string, or None if lookup fails.
    """
    try:
        compounds = pcp.get_compounds(chemical_name, 'name')
        if compounds:
            return compounds[0].smiles
    except Exception:
        pass
    return None


def extract_chemicals_from_pdf(filepath: str) -> list[str]:
    """
    Extract chemical names from a PDF file.

    Assumes the PDF contains chemical names, one per line.

    Args:
        filepath: Path to the PDF file.

    Returns:
        List of chemical names extracted from the PDF.
    """
    chemicals = []
    with pdfplumber.open(filepath) as pdf:
        for page in pdf.pages:
            text = page.extract_text()
            if text:
                lines = [line.strip() for line in text.split('\n') if line.strip()]
                chemicals.extend(lines)
    return chemicals


def compute_morgan_fingerprint(mol):
    """
    Compute Morgan fingerprint for a molecule.

    Uses radius=2 and includes chirality as specified.

    Args:
        mol: RDKit Mol object.

    Returns:
        Morgan fingerprint as ExplicitBitVect.
    """
    fpgen = rdFingerprintGenerator.GetMorganGenerator(radius=2, includeChirality=True)
    return fpgen.GetFingerprint(mol)


def topk_tanimoto_similarity_molecules(
    target_molecule_name: str,
    molecule_pool_filepath: str,
    top_k: int
) -> list:
    """
    Find the top-k most similar molecules to a target molecule from a pool of molecules in a PDF.

    This function:
    1. Converts the target molecule name to SMILES using PubChem
    2. Extracts chemical names from the PDF file
    3. Converts each chemical name to SMILES using PubChem
    4. Computes Morgan fingerprints (radius=2, with chirality)
    5. Calculates Tanimoto similarity between target and each pool molecule
    6. Returns top-k results sorted by similarity (descending), with alphabetical ordering for ties

    Args:
        target_molecule_name: The name of the target molecule to compare against.
        molecule_pool_filepath: Path to a PDF file containing chemical names (one per line).
        top_k: Number of top similar molecules to return.

    Returns:
        A list of the top-k most similar molecule names, sorted by:
        - Tanimoto similarity (descending)
        - Alphabetical order (ascending) for ties
    """
    # Get SMILES for target molecule
    target_smiles = get_smiles_from_name(target_molecule_name)
    if target_smiles is None:
        raise ValueError(f"Could not find SMILES for target molecule: {target_molecule_name}")

    # Create RDKit mol object for target
    target_mol = Chem.MolFromSmiles(target_smiles)
    if target_mol is None:
        raise ValueError(f"Could not parse SMILES for target molecule: {target_smiles}")

    # Compute fingerprint for target
    target_fp = compute_morgan_fingerprint(target_mol)

    # Extract chemical names from PDF
    pool_chemicals = extract_chemicals_from_pdf(molecule_pool_filepath)

    # Calculate similarities for all pool molecules
    similarities = []
    for chemical_name in pool_chemicals:
        smiles = get_smiles_from_name(chemical_name)
        if smiles is None:
            continue

        mol = Chem.MolFromSmiles(smiles)
        if mol is None:
            continue

        fp = compute_morgan_fingerprint(mol)
        similarity = DataStructs.TanimotoSimilarity(target_fp, fp)
        similarities.append((chemical_name, similarity))

    # Sort by similarity (descending), then by name (ascending) for ties
    similarities.sort(key=lambda x: (-x[1], x[0]))

    # Return top-k molecule names
    return [name for name, _ in similarities[:top_k]]


if __name__ == "__main__":
    # Example usage
    import sys

    if len(sys.argv) >= 4:
        target = sys.argv[1]
        pdf_path = sys.argv[2]
        k = int(sys.argv[3])
    else:
        target = "Aspirin"
        pdf_path = "/root/molecules.pdf"
        k = 5

    print(f"Finding top {k} molecules similar to '{target}'...")
    results = topk_tanimoto_similarity_molecules(target, pdf_path, k)

    print(f"\nTop {k} similar molecules:")
    for i, name in enumerate(results, 1):
        print(f"  {i}. {name}")
